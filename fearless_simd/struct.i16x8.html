Struct i16x8 

Source 
#[repr(C, align(16))]
pub struct i16x8<S: Simd> {
    pub(crate) val: S::i16x8,
    pub simd: S,
}
Expand description

A SIMD vector of 8 i16 elements.

You may construct this vector type using the Self::splat, Self::from_slice, Self::simd_from, Self::from_fn, and Self::block_splat methods.

fn construct_simd<S: Simd>(simd: S) {
    // From a single scalar value:
    let a = i16x8::splat(simd, 1);
    let b = i16x8::simd_from(simd, 1);

    // From a slice:
    let c = i16x8::from_slice(simd, &[1, 2, 3, 4, 5, 6, 7, 8]);

    // From an array:
    let d = i16x8::simd_from(simd, [1, 2, 3, 4, 5, 6, 7, 8]);

    // From an element-wise function:
    let e = i16x8::from_fn(simd, |i| i as i16);
}

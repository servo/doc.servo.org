Converts a Stream of data chunks into a reply.

§Example

use warp::Filter;
use futures_util::stream;
use bytes::Bytes;

let route = warp::any().map(|| {
    let chunks = vec![Ok::<_, std::io::Error>(Bytes::from("hello"))];
    warp::reply::stream(stream::iter(chunks))
});

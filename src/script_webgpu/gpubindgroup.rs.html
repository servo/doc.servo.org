28#[derive(JSTraceable, MallocSizeOf)]
29struct DroppableGPUBindGroup {
30    #[no_trace]
31    channel: WebGPU,
32    #[no_trace]
33    bind_group: WebGPUBindGroup,
34}
35
36impl Drop for DroppableGPUBindGroup {
37    fn drop(&mut self) {
38        if let Err(e) = self
39            .channel
40            .0
41            .send(WebGPURequest::DropBindGroup(self.bind_group.0))
42        {
43            warn!(
44                "Failed to send WebGPURequest::DropBindGroup({:?}) ({})",
45                self.bind_group.0, e
46            );
47        };
48    }
49}

51#[dom_struct]
52pub struct GPUBindGroup<D: DomTypes> {
53    reflector_: Reflector,
54    label: DomRefCell<USVString>,
55    #[no_trace]
56    device: WebGPUDevice,
57    layout: Dom<GPUBindGroupLayout<D>>,
58    droppable: DroppableGPUBindGroup,
59}
60
61impl<D: Equivalence> GPUBindGroup<D> {
62    fn new_inherited(
63        channel: WebGPU,
64        bind_group: WebGPUBindGroup,
65        device: WebGPUDevice,
66        layout: &GPUBindGroupLayout<D>,
67        label: USVString,
68    ) -> Self {
69        Self {
70            reflector_: Reflector::new(),
71            label: DomRefCell::new(label),
72            device,
73            layout: Dom::from_ref(layout),
74            droppable: DroppableGPUBindGroup {
75                channel,
76                bind_group,
77            },
78        }
79    }
80
81    pub(crate) fn new(
82        cx: &mut JSContext,
83        global: &D::GlobalScope,
84        channel: WebGPU,
85        bind_group: WebGPUBindGroup,
86        device: WebGPUDevice,
87        layout: &GPUBindGroupLayout<D>,
88        label: USVString,
89    ) -> DomRoot<Self> {
90        reflect_dom_object_with_wrap::<D, _, _>(
91            Box::new(GPUBindGroup::new_inherited(
92                channel, bind_group, device, layout, label,
93            )),
94            global,
95            cx,
96            GPUBindGroupWrap::<D>,
97        )
98    }
99}

101impl<D> GPUBindGroup<D>
102where
103    D: Equivalence,
104    D::GPUDevice: GPUDeviceTrait<D>,
105    D::GPUExternalTexture: GPUExternalTextureTrait<D>,
106    D::GlobalScope: WebGPUGlobalTrait,
107    D::Promise: PromiseHelpers<D>,
108{
109    pub fn id(&self) -> &WebGPUBindGroup {
110        &self.droppable.bind_group
111    }
112
113    /// <https://gpuweb.github.io/gpuweb/#dom-gpudevice-createbindgroup>
114    pub fn create(
115        cx: &mut JSContext,
116        device: &D::GPUDevice,
117        descriptor: &GPUBindGroupDescriptor<D>,
118    ) -> DomRoot<GPUBindGroup<D>> {
119        let entries = descriptor
120            .entries
121            .iter()
122            .map(|bind| convert_bind_group_entry(cx, bind))
123            .collect::<Vec<_>>();
124
125        let desc = BindGroupDescriptor {
126            label: (&descriptor.parent).convert(),
127            layout: descriptor.layout.id().0,
128            entries: Cow::Owned(entries),
129        };
130
131        let bind_group_id = <D::GPUDevice as DomGlobalGeneric<D>>::global_from_reflector(device)
132            .global_wgpu_id_hub()
133            .create_bind_group_id();
134        device
135            .channel()
136            .0
137            .send(WebGPURequest::CreateBindGroup {
138                device_id: device.id().0,
139                bind_group_id,
140                descriptor: desc,
141            })
142            .expect("Failed to create WebGPU BindGroup");
143
144        let bind_group = WebGPUBindGroup(bind_group_id);
145
146        let global = <D::GPUDevice as DomGlobalGeneric<D>>::global_from_reflector(device);
147        GPUBindGroup::new(
148            cx,
149            &*global,
150            device.channel(),
151            bind_group,
152            device.id(),
153            &descriptor.layout,
154            descriptor.parent.label.clone(),
155        )
156    }
157}

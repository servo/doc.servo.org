11//! Fork of Arc for Servo. This has the following advantages over std::sync::Arc:
12//!
13//! * We don't waste storage on the weak reference count.
14//! * We don't do extra RMU operations to handle the possibility of weak references.
15//! * We can experiment with arena allocation (todo).
16//! * We can add methods to support our custom use cases [1].
17//! * We have support for dynamically-sized types (see from_header_and_iter).
18//! * We have support for thin arcs to unsized types (see ThinArc).
19//! * We have support for references to static data, which don't do any
20//!   refcounting.
21//!
22//! [1]: https://bugzilla.mozilla.org/show_bug.cgi?id=1360883
23

25// duplicate those here.

46/// A soft limit on the amount of references that may be made to an `Arc`.
47///
48/// Going above this limit will abort your program (although not
49/// necessarily) at _exactly_ `MAX_REFCOUNT + 1` references.
50const MAX_REFCOUNT: usize = (isize::MAX) as usize;
51
52/// Special refcount value that means the data is not reference counted,
53/// and that the `Arc` is really acting as a read-only static reference.
54const STATIC_REFCOUNT: usize = usize::MAX;

56/// An atomically reference counted shared pointer
57///
58/// See the documentation for [`Arc`] in the standard library. Unlike the
59/// standard library `Arc`, this `Arc` does not support weak reference counting.
60///
61/// See the discussion in https://github.com/rust-lang/rust/pull/60594 for the
62/// usage of PhantomData.
63///
64/// [`Arc`]: https://doc.rust-lang.org/stable/std/sync/struct.Arc.html
65///
66/// cbindgen:derive-eq=false
67/// cbindgen:derive-neq=false
68#[repr(C)]
69pub struct Arc<T: ?Sized> {
70    p: ptr::NonNull<ArcInner<T>>,
71    phantom: PhantomData<T>,
72}

74/// An `Arc` that is known to be uniquely owned
75///
76/// When `Arc`s are constructed, they are known to be
77/// uniquely owned. In such a case it is safe to mutate
78/// the contents of the `Arc`. Normally, one would just handle
79/// this by mutating the data on the stack before allocating the
80/// `Arc`, however it's possible the data is large or unsized
81/// and you need to heap-allocate it earlier in such a way
82/// that it can be freely converted into a regular `Arc` once you're
83/// done.
84///
85/// `UniqueArc` exists for this purpose, when constructed it performs
86/// the same allocations necessary for an `Arc`, however it allows mutable access.
87/// Once the mutation is finished, you can call `.shareable()` and get a regular `Arc`
88/// out of it.
89///
90/// Ignore the doctest below there's no way to skip building with refcount
91/// logging during doc tests (see rust-lang/rust#45599).
92///
93/// ```rust,ignore
94/// # use servo_arc::UniqueArc;
95/// let data = [1, 2, 3, 4, 5];
96/// let mut x = UniqueArc::new(data);
97/// x[4] = 7; // mutate!
98/// let y = x.shareable(); // y is an Arc<T>
99/// ```
100pub struct UniqueArc<T: ?Sized>(Arc<T>);
101
102impl<T> UniqueArc<T> {
103    #[inline]
104    /// Construct a new UniqueArc
105    pub fn new(data: T) -> Self {
106        UniqueArc(Arc::new(data))
107    }
108
109    /// Construct an uninitialized arc
110    #[inline]
111    pub fn new_uninit() -> UniqueArc<mem::MaybeUninit<T>> {
112        unsafe {
113            let layout = Layout::new::<ArcInner<mem::MaybeUninit<T>>>();
114            let ptr = alloc::alloc(layout);
115            let mut p = ptr::NonNull::new(ptr)
116                .unwrap_or_else(|| alloc::handle_alloc_error(layout))
117                .cast::<ArcInner<mem::MaybeUninit<T>>>();
118            ptr::write(&mut p.as_mut().count, atomic::AtomicUsize::new(1));
119            #[cfg(feature = "track_alloc_size")]
120            ptr::write(&mut p.as_mut().alloc_size, layout.size());
121
122            #[cfg(feature = "gecko_refcount_logging")]
123            {
124                NS_LogCtor(p.as_ptr() as *mut _, b"ServoArc\0".as_ptr() as *const _, 8)
125            }
126
127            UniqueArc(Arc {
128                p,
129                phantom: PhantomData,
130            })
131        }
132    }
133
134    #[inline]
135    /// Convert to a shareable Arc<T> once we're done mutating it
136    pub fn shareable(self) -> Arc<T> {
137        self.0
138    }
139}

169/// The object allocated by an Arc<T>
170///
171/// See https://github.com/mozilla/cbindgen/issues/937 for the derive-{eq,neq}=false. But we don't
172/// use those anyways so we can just disable them.
173/// cbindgen:derive-eq=false
174/// cbindgen:derive-neq=false
175#[repr(C)]
176struct ArcInner<T: ?Sized> {
177    count: atomic::AtomicUsize,
178    // NOTE(emilio): This needs to be here so that HeaderSlice<> is deallocated properly if the
179    // allocator relies on getting the right Layout. We don't need to track the right alignment,
180    // since we know that statically.
181    //
182    // This member could be completely avoided once min_specialization feature is stable (by
183    // implementing a trait for HeaderSlice that gives you the right layout). For now, servo-only
184    // since Gecko doesn't need it (its allocator doesn't need the size for the alignments we care
185    // about). See https://github.com/rust-lang/rust/issues/31844.
186    #[cfg(feature = "track_alloc_size")]
187    alloc_size: usize,
188    data: T,
189}
190
191unsafe impl<T: ?Sized + Sync + Send> Send for ArcInner<T> {}
192unsafe impl<T: ?Sized + Sync + Send> Sync for ArcInner<T> {}

194/// Computes the offset of the data field within ArcInner.
195fn data_offset<T>() -> usize {
196    let size = size_of::<ArcInner<()>>();
197    let align = align_of::<T>();
198    // https://github.com/rust-lang/rust/blob/1.36.0/src/libcore/alloc.rs#L187-L207
199    size.wrapping_add(align).wrapping_sub(1) & !align.wrapping_sub(1)
200}

202impl<T> Arc<T> {
203    /// Construct an `Arc<T>`
204    #[inline]
205    pub fn new(data: T) -> Self {
206        let layout = Layout::new::<ArcInner<T>>();
207        let p = unsafe {
208            let ptr = ptr::NonNull::new(alloc::alloc(layout))
209                .unwrap_or_else(|| alloc::handle_alloc_error(layout))
210                .cast::<ArcInner<T>>();
211            ptr::write(
212                ptr.as_ptr(),
213                ArcInner {
214                    count: atomic::AtomicUsize::new(1),
215                    #[cfg(feature = "track_alloc_size")]
216                    alloc_size: layout.size(),
217                    data,
218                },
219            );
220            ptr
221        };
222
223        #[cfg(feature = "gecko_refcount_logging")]
224        unsafe {
225            // FIXME(emilio): Would be so amazing to have
226            // std::intrinsics::type_name() around, so that we could also report
227            // a real size.
228            NS_LogCtor(p.as_ptr() as *mut _, b"ServoArc\0".as_ptr() as *const _, 8);
229        }
230
231        Arc {
232            p,
233            phantom: PhantomData,
234        }
235    }
236
237    /// Construct an intentionally-leaked arc.
238    #[inline]
239    pub fn new_leaked(data: T) -> Self {
240        let arc = Self::new(data);
241        arc.mark_as_intentionally_leaked();
242        arc
243    }
244
245    /// Convert the Arc<T> to a raw pointer, suitable for use across FFI
246    ///
247    /// Note: This returns a pointer to the data T, which is offset in the allocation.
248    #[inline]
249    pub fn into_raw(this: Self) -> *const T {
250        let ptr = unsafe { &((*this.ptr()).data) as *const _ };
251        mem::forget(this);
252        ptr
253    }
254
255    /// Reconstruct the Arc<T> from a raw pointer obtained from into_raw()
256    ///
257    /// Note: This raw pointer will be offset in the allocation and must be preceded
258    /// by the atomic count.
259    #[inline]
260    pub unsafe fn from_raw(ptr: *const T) -> Self {
261        // To find the corresponding pointer to the `ArcInner` we need
262        // to subtract the offset of the `data` field from the pointer.
263        let ptr = (ptr as *const u8).sub(data_offset::<T>());
264        Arc {
265            p: ptr::NonNull::new_unchecked(ptr as *mut ArcInner<T>),
266            phantom: PhantomData,
267        }
268    }
269
270    /// Like from_raw, but returns an addrefed arc instead.
271    #[inline]
272    pub unsafe fn from_raw_addrefed(ptr: *const T) -> Self {
273        let arc = Self::from_raw(ptr);
274        mem::forget(arc.clone());
275        arc
276    }

278    /// Create a new static Arc<T> (one that won't reference count the object)
279    /// and place it in the allocation provided by the specified `alloc`
280    /// function.
281    ///
282    /// `alloc` must return a pointer into a static allocation suitable for
283    /// storing data with the `Layout` passed into it. The pointer returned by
284    /// `alloc` will not be freed.
285    #[inline]
286    pub unsafe fn new_static<F>(alloc: F, data: T) -> Arc<T>
287    where
288        F: FnOnce(Layout) -> *mut u8,
289    {
290        let layout = Layout::new::<ArcInner<T>>();
291        let ptr = alloc(layout) as *mut ArcInner<T>;
292
293        let x = ArcInner {
294            count: atomic::AtomicUsize::new(STATIC_REFCOUNT),
295            #[cfg(feature = "track_alloc_size")]
296            alloc_size: layout.size(),
297            data,
298        };
299
300        ptr::write(ptr, x);
301
302        Arc {
303            p: ptr::NonNull::new_unchecked(ptr),
304            phantom: PhantomData,
305        }
306    }
307
308    /// Produce a pointer to the data that can be converted back
309    /// to an Arc. This is basically an `&Arc<T>`, without the extra indirection.
310    /// It has the benefits of an `&T` but also knows about the underlying refcount
311    /// and can be converted into more `Arc<T>`s if necessary.
312    #[inline]
313    pub fn borrow_arc<'a>(&'a self) -> ArcBorrow<'a, T> {
314        ArcBorrow(&**self)
315    }
316
317    /// Returns the address on the heap of the Arc itself -- not the T within it -- for memory
318    /// reporting.
319    ///
320    /// If this is a static reference, this returns null.
321    pub fn heap_ptr(&self) -> *const c_void {
322        if self.inner().count.load(Relaxed) == STATIC_REFCOUNT {
323            ptr::null()
324        } else {
325            self.p.as_ptr() as *const ArcInner<T> as *const c_void
326        }
327    }
328}

330impl<T: ?Sized> Arc<T> {
331    #[inline]
332    fn inner(&self) -> &ArcInner<T> {
333        // This unsafety is ok because while this arc is alive we're guaranteed
334        // that the inner pointer is valid. Furthermore, we know that the
335        // `ArcInner` structure itself is `Sync` because the inner data is
336        // `Sync` as well, so we're ok loaning out an immutable pointer to these
337        // contents.
338        unsafe { &*self.ptr() }
339    }
340
341    #[inline(always)]
342    fn record_drop(&self) {
343        #[cfg(feature = "gecko_refcount_logging")]
344        unsafe {
345            NS_LogDtor(self.ptr() as *mut _, b"ServoArc\0".as_ptr() as *const _, 8);
346        }
347    }
348
349    /// Marks this `Arc` as intentionally leaked for the purposes of refcount
350    /// logging.
351    ///
352    /// It's a logic error to call this more than once, but it's not unsafe, as
353    /// it'd just report negative leaks.
354    ///
355    /// The allocation is expected to live for the rest of the process, so this
356    /// also marks it static: clone()/drop() then skip the atomic refcount
357    /// updates.
358    #[inline(always)]
359    pub fn mark_as_intentionally_leaked(&self) {
360        self.record_drop();
361        self.inner().count.store(STATIC_REFCOUNT, Relaxed);
362    }
363
364    // Non-inlined part of `drop`. Just invokes the destructor and calls the
365    // refcount logging machinery if enabled.
366    #[inline(never)]
367    unsafe fn drop_slow(&mut self) {
368        self.record_drop();
369        let inner = self.ptr();
370
371        let layout = Layout::for_value(&*inner);
372        #[cfg(feature = "track_alloc_size")]
373        let layout = Layout::from_size_align_unchecked((*inner).alloc_size, layout.align());
374
375        std::ptr::drop_in_place(inner);
376        alloc::dealloc(inner as *mut _, layout);
377    }
378
379    /// Test pointer equality between the two Arcs, i.e. they must be the _same_
380    /// allocation
381    #[inline]
382    pub fn ptr_eq(this: &Self, other: &Self) -> bool {
383        this.raw_ptr() == other.raw_ptr()
384    }
385
386    fn ptr(&self) -> *mut ArcInner<T> {
387        self.p.as_ptr()
388    }
389
390    /// Returns a raw ptr to the underlying allocation.
391    pub fn raw_ptr(&self) -> ptr::NonNull<()> {
392        self.p.cast()
393    }
394}

410impl<T: ?Sized> Clone for Arc<T> {
411    #[inline]
412    fn clone(&self) -> Self {
413        // NOTE(emilio): If you change anything here, make sure that the
414        // implementation in layout/style/ServoStyleConstsInlines.h matches!
415        //
416        // Using a relaxed ordering to check for STATIC_REFCOUNT is safe, since
417        // `count` never changes between STATIC_REFCOUNT and other values.
418        if self.inner().count.load(Relaxed) != STATIC_REFCOUNT {
419            // Using a relaxed ordering is alright here, as knowledge of the
420            // original reference prevents other threads from erroneously deleting
421            // the object.
422            //
423            // As explained in the [Boost documentation][1], Increasing the
424            // reference counter can always be done with memory_order_relaxed: New
425            // references to an object can only be formed from an existing
426            // reference, and passing an existing reference from one thread to
427            // another must already provide any required synchronization.
428            //
429            // [1]: (www.boost.org/doc/libs/1_55_0/doc/html/atomic/usage_examples.html)
430            let old_size = self.inner().count.fetch_add(1, Relaxed);
431
432            // However we need to guard against massive refcounts in case someone
433            // is `mem::forget`ing Arcs. If we don't do this the count can overflow
434            // and users will use-after free. We racily saturate to `isize::MAX` on
435            // the assumption that there aren't ~2 billion threads incrementing
436            // the reference count at once. This branch will never be taken in
437            // any realistic program.
438            //
439            // We abort because such a program is incredibly degenerate, and we
440            // don't care to support it.
441            if old_size > MAX_REFCOUNT {
442                process::abort();
443            }
444        }
445
446        unsafe {
447            Arc {
448                p: ptr::NonNull::new_unchecked(self.ptr()),
449                phantom: PhantomData,
450            }
451        }
452    }
453}

464impl<T: Clone> Arc<T> {
465    /// Makes a mutable reference to the `Arc`, cloning if necessary
466    ///
467    /// This is functionally equivalent to [`Arc::make_mut`][mm] from the standard library.
468    ///
469    /// If this `Arc` is uniquely owned, `make_mut()` will provide a mutable
470    /// reference to the contents. If not, `make_mut()` will create a _new_ `Arc`
471    /// with a copy of the contents, update `this` to point to it, and provide
472    /// a mutable reference to its contents.
473    ///
474    /// This is useful for implementing copy-on-write schemes where you wish to
475    /// avoid copying things if your `Arc` is not shared.
476    ///
477    /// [mm]: https://doc.rust-lang.org/stable/std/sync/struct.Arc.html#method.make_mut
478    #[inline]
479    pub fn make_mut(this: &mut Self) -> &mut T {
480        if !this.is_unique() {
481            // Another pointer exists; clone
482            *this = Arc::new((**this).clone());
483        }
484
485        unsafe {
486            // This unsafety is ok because we're guaranteed that the pointer
487            // returned is the *only* pointer that will ever be returned to T. Our
488            // reference count is guaranteed to be 1 at this point, and we required
489            // the Arc itself to be `mut`, so we're returning the only possible
490            // reference to the inner data.
491            &mut (*this.ptr()).data
492        }
493    }
494}

496impl<T: ?Sized> Arc<T> {
497    /// Provides mutable access to the contents _if_ the `Arc` is uniquely owned.
498    #[inline]
499    pub fn get_mut(this: &mut Self) -> Option<&mut T> {
500        if this.is_unique() {
501            unsafe {
502                // See make_mut() for documentation of the threadsafety here.
503                Some(&mut (*this.ptr()).data)
504            }
505        } else {
506            None
507        }
508    }
509
510    /// Whether or not the `Arc` is a static reference.
511    #[inline]
512    pub fn is_static(&self) -> bool {
513        // Using a relaxed ordering to check for STATIC_REFCOUNT is safe, since
514        // `count` never changes between STATIC_REFCOUNT and other values.
515        self.inner().count.load(Relaxed) == STATIC_REFCOUNT
516    }
517
518    /// Whether or not the `Arc` is uniquely owned (is the refcount 1?) and not
519    /// a static reference.
520    #[inline]
521    pub fn is_unique(&self) -> bool {
522        // See the extensive discussion in [1] for why this needs to be Acquire.
523        //
524        // [1] https://github.com/servo/servo/issues/21186
525        self.inner().count.load(Acquire) == 1
526    }
527}

529impl<T: ?Sized> Drop for Arc<T> {
530    #[inline]
531    fn drop(&mut self) {
532        // NOTE(emilio): If you change anything here, make sure that the
533        // implementation in layout/style/ServoStyleConstsInlines.h matches!
534        if self.is_static() {
535            return;
536        }
537
538        // Because `fetch_sub` is already atomic, we do not need to synchronize
539        // with other threads unless we are going to delete the object.
540        if self.inner().count.fetch_sub(1, Release) != 1 {
541            return;
542        }
543
544        // FIXME(bholley): Use the updated comment when [2] is merged.
545        //
546        // This load is needed to prevent reordering of use of the data and
547        // deletion of the data.  Because it is marked `Release`, the decreasing
548        // of the reference count synchronizes with this `Acquire` load. This
549        // means that use of the data happens before decreasing the reference
550        // count, which happens before this load, which happens before the
551        // deletion of the data.
552        //
553        // As explained in the [Boost documentation][1],
554        //
555        // > It is important to enforce any possible access to the object in one
556        // > thread (through an existing reference) to *happen before* deleting
557        // > the object in a different thread. This is achieved by a "release"
558        // > operation after dropping a reference (any access to the object
559        // > through this reference must obviously happened before), and an
560        // > "acquire" operation before deleting the object.
561        //
562        // [1]: (www.boost.org/doc/libs/1_55_0/doc/html/atomic/usage_examples.html)
563        // [2]: https://github.com/rust-lang/rust/pull/41714
564        self.inner().count.load(Acquire);
565
566        unsafe {
567            self.drop_slow();
568        }
569    }
570}

684/// Structure to allow Arc-managing some fixed-sized data and a variably-sized
685/// slice in a single allocation.
686///
687/// cbindgen:derive-eq=false
688/// cbindgen:derive-neq=false
689#[derive(Eq)]
690#[repr(C)]
691pub struct HeaderSlice<H, T> {
692    /// The fixed-sized data.
693    pub header: H,
694
695    /// The length of the slice at our end.
696    len: usize,
697
698    /// The dynamically-sized data.
699    data: [T; 0],
700}
701
702impl<H: PartialEq, T: PartialEq> PartialEq for HeaderSlice<H, T> {
703    fn eq(&self, other: &Self) -> bool {
704        self.header == other.header && self.slice() == other.slice()
705    }
706}

708impl<H, T> Drop for HeaderSlice<H, T> {
709    fn drop(&mut self) {
710        unsafe {
711            let mut ptr = self.data_mut();
712            for _ in 0..self.len {
713                std::ptr::drop_in_place(ptr);
714                ptr = ptr.offset(1);
715            }
716        }
717    }
718}

723            .field("header", &self.header)
724            .field("slice", &self.slice())
725            .finish()
726    }

734    }
735
736    #[inline(always)]
737    fn data(&self) -> *const T {
738        std::ptr::addr_of!(self.data) as *const _
739    }
740
741    #[inline(always)]
742    fn data_mut(&mut self) -> *mut T {

759impl<H, T> Arc<HeaderSlice<H, T>> {
760    /// Creates an Arc for a HeaderSlice using the given header struct and
761    /// iterator to generate the slice.
762    ///
763    /// `is_static` indicates whether to create a static Arc.
764    ///
765    /// `alloc` is used to get a pointer to the memory into which the
766    /// dynamically sized ArcInner<HeaderSlice<H, T>> value will be
767    /// written.  If `is_static` is true, then `alloc` must return a
768    /// pointer into some static memory allocation.  If it is false,
769    /// then `alloc` must return an allocation that can be dellocated
770    /// by calling Box::from_raw::<ArcInner<HeaderSlice<H, T>>> on it.
771    #[inline]
772    pub fn from_header_and_iter_alloc<F, I>(
773        alloc: F,
774        header: H,
775        mut items: I,
776        num_items: usize,
777        is_static: bool,
778    ) -> Self
779    where
780        F: FnOnce(Layout) -> *mut u8,
781        I: Iterator<Item = T>,
782    {
783        assert_ne!(size_of::<T>(), 0, "Need to think about ZST");
784
785        let layout = Layout::new::<ArcInner<HeaderSlice<H, T>>>();
786        debug_assert!(layout.align() >= align_of::<T>());
787        debug_assert!(layout.align() >= align_of::<usize>());
788        let array_layout = Layout::array::<T>(num_items).expect("Overflow");
789        let (layout, _offset) = layout.extend(array_layout).expect("Overflow");
790        let p = unsafe {
791            // Allocate the buffer.
792            let buffer = alloc(layout);
793            let mut p = ptr::NonNull::new(buffer)
794                .unwrap_or_else(|| alloc::handle_alloc_error(layout))
795                .cast::<ArcInner<HeaderSlice<H, T>>>();
796
797            // Write the data.
798            //
799            // Note that any panics here (i.e. from the iterator) are safe, since
800            // we'll just leak the uninitialized memory.
801            let count = if is_static {
802                atomic::AtomicUsize::new(STATIC_REFCOUNT)
803            } else {
804                atomic::AtomicUsize::new(1)
805            };
806            ptr::write(&mut p.as_mut().count, count);
807            #[cfg(feature = "track_alloc_size")]
808            ptr::write(&mut p.as_mut().alloc_size, layout.size());
809            ptr::write(&mut p.as_mut().data.header, header);
810            ptr::write(&mut p.as_mut().data.len, num_items);
811            if num_items != 0 {
812                let mut current = std::ptr::addr_of_mut!(p.as_mut().data.data) as *mut T;
813                for _ in 0..num_items {
814                    ptr::write(
815                        current,
816                        items
817                            .next()
818                            .expect("ExactSizeIterator over-reported length"),
819                    );
820                    current = current.offset(1);
821                }
822                // We should have consumed the buffer exactly, maybe accounting
823                // for some padding from the alignment.
824                debug_assert!(
825                    (buffer.add(layout.size()) as usize - current as *mut u8 as usize)
826                        < layout.align()
827                );
828            }
829            assert!(
830                items.next().is_none(),
831                "ExactSizeIterator under-reported length"
832            );
833            p
834        };
835        #[cfg(feature = "gecko_refcount_logging")]
836        unsafe {
837            if !is_static {
838                // FIXME(emilio): Would be so amazing to have
839                // std::intrinsics::type_name() around.
840                NS_LogCtor(p.as_ptr() as *mut _, b"ServoArc\0".as_ptr() as *const _, 8)
841            }
842        }
843
844        // Return the fat Arc.
845        assert_eq!(
846            size_of::<Self>(),
847            size_of::<usize>(),
848            "The Arc should be thin"
849        );
850
851        Arc {
852            p,
853            phantom: PhantomData,
854        }
855    }
856
857    /// Creates an Arc for a HeaderSlice using the given header struct and iterator to generate the
858    /// slice. Panics if num_items doesn't match the number of items.
859    #[inline]
860    pub fn from_header_and_iter_with_size<I>(header: H, items: I, num_items: usize) -> Self
861    where
862        I: Iterator<Item = T>,
863    {
864        Arc::from_header_and_iter_alloc(
865            |layout| unsafe { alloc::alloc(layout) },
866            header,
867            items,
868            num_items,
869            /* is_static = */ false,
870        )
871    }
872
873    /// Creates an Arc for a HeaderSlice using the given header struct and
874    /// iterator to generate the slice. The resulting Arc will be fat.
875    #[inline]
876    pub fn from_header_and_iter<I>(header: H, items: I) -> Self
877    where
878        I: Iterator<Item = T> + ExactSizeIterator,
879    {
880        let len = items.len();
881        Self::from_header_and_iter_with_size(header, items, len)
882    }
883}

885/// This is functionally equivalent to Arc<(H, [T])>
886///
887/// When you create an `Arc` containing a dynamically sized type like a slice, the `Arc` is
888/// represented on the stack as a "fat pointer", where the length of the slice is stored alongside
889/// the `Arc`'s pointer. In some situations you may wish to have a thin pointer instead, perhaps
890/// for FFI compatibility or space efficiency. `ThinArc` solves this by storing the length in the
891/// allocation itself, via `HeaderSlice`.

895pub type ThinArcUnion<H1, T1, H2, T2> = ArcUnion<HeaderSlice<H1, T1>, HeaderSlice<H2, T2>>;

929/// A "borrowed `Arc`". This is a pointer to
930/// a T that is known to have been allocated within an
931/// `Arc`.
932///
933/// This is equivalent in guarantees to `&Arc<T>`, however it is
934/// a bit more flexible. To obtain an `&Arc<T>` you must have
935/// an `Arc<T>` instance somewhere pinned down until we're done with it.
936/// It's also a direct pointer to `T`, so using this involves less pointer-chasing
937///
938/// However, C++ code may hand us refcounted things as pointers to T directly,
939/// so we have to conjure up a temporary `Arc` on the stack each time.
940///
941/// `ArcBorrow` lets us deal with borrows of known-refcounted objects
942/// without needing to worry about where the `Arc<T>` is.
943#[derive(Debug, Eq, PartialEq)]
944pub struct ArcBorrow<'a, T: 'a>(&'a T);
945
946impl<'a, T> Copy for ArcBorrow<'a, T> {}
947impl<'a, T> Clone for ArcBorrow<'a, T> {
948    #[inline]
949    fn clone(&self) -> Self {
950        *self
951    }
952}
953
954impl<'a, T> ArcBorrow<'a, T> {
955    /// Clone this as an `Arc<T>`. This bumps the refcount.
956    #[inline]
957    pub fn clone_arc(&self) -> Arc<T> {
958        let arc = unsafe { Arc::from_raw(self.0) };
959        // addref it!
960        mem::forget(arc.clone());
961        arc
962    }
963
964    /// For constructing from a reference known to be Arc-backed,
965    /// e.g. if we obtain such a reference over FFI
966    #[inline]
967    pub unsafe fn from_ref(r: &'a T) -> Self {
968        ArcBorrow(r)
969    }
970
971    /// Compare two `ArcBorrow`s via pointer equality. Will only return
972    /// true if they come from the same allocation
973    pub fn ptr_eq(this: &Self, other: &Self) -> bool {
974        this.0 as *const T == other.0 as *const T
975    }
976
977    /// Temporarily converts |self| into a bonafide Arc and exposes it to the
978    /// provided callback. The refcount is not modified.
979    #[inline]
980    pub fn with_arc<F, U>(&self, f: F) -> U
981    where
982        F: FnOnce(&Arc<T>) -> U,
983        T: 'static,
984    {
985        // Synthesize transient Arc, which never touches the refcount.
986        let transient = unsafe { mem::ManuallyDrop::new(Arc::from_raw(self.0)) };
987
988        // Expose the transient Arc to the callback, which may clone it if it wants.
989        let result = f(&transient);
990
991        // Forward the result.
992        result
993    }
994
995    /// Similar to deref, but uses the lifetime |a| rather than the lifetime of
996    /// self, which is incompatible with the signature of the Deref trait.
997    #[inline]
998    pub fn get(&self) -> &'a T {
999        self.0
1000    }
1001}

1004    type Target = T;
1005
1006    #[inline]
1007    fn deref(&self) -> &T {
1008        self.0
1009    }

1012/// A tagged union that can represent `Arc<A>` or `Arc<B>` while only consuming a
1013/// single word. The type is also `NonNull`, and thus can be stored in an Option
1014/// without increasing size.
1015///
1016/// This is functionally equivalent to
1017/// `enum ArcUnion<A, B> { First(Arc<A>), Second(Arc<B>)` but only takes up
1018/// up a single word of stack space.
1019///
1020/// This could probably be extended to support four types if necessary.
1021pub struct ArcUnion<A, B> {
1022    p: ptr::NonNull<()>,
1023    phantom_a: PhantomData<A>,
1024    phantom_b: PhantomData<B>,
1025}
1026
1027unsafe impl<A: Sync + Send, B: Send + Sync> Send for ArcUnion<A, B> {}
1028unsafe impl<A: Sync + Send, B: Send + Sync> Sync for ArcUnion<A, B> {}
1029
1030impl<A: PartialEq, B: PartialEq> PartialEq for ArcUnion<A, B> {
1031    fn eq(&self, other: &Self) -> bool {
1032        use crate::ArcUnionBorrow::*;
1033        match (self.borrow(), other.borrow()) {
1034            (First(x), First(y)) => x == y,
1035            (Second(x), Second(y)) => x == y,
1036            (_, _) => false,
1037        }
1038    }
1039}
1040
1041impl<A: Eq, B: Eq> Eq for ArcUnion<A, B> {}

1043/// This represents a borrow of an `ArcUnion`.
1044#[derive(Debug)]
1045pub enum ArcUnionBorrow<'a, A: 'a, B: 'a> {
1046    First(ArcBorrow<'a, A>),
1047    Second(ArcBorrow<'a, B>),
1048}
1049
1050impl<A, B> ArcUnion<A, B> {
1051    unsafe fn new(ptr: *mut ()) -> Self {
1052        ArcUnion {
1053            p: ptr::NonNull::new_unchecked(ptr),
1054            phantom_a: PhantomData,
1055            phantom_b: PhantomData,
1056        }
1057    }
1058
1059    /// Returns true if the two values are pointer-equal.
1060    #[inline]
1061    pub fn ptr_eq(this: &Self, other: &Self) -> bool {
1062        this.p == other.p
1063    }
1064
1065    #[inline]
1066    pub fn ptr(&self) -> ptr::NonNull<()> {
1067        self.p
1068    }
1069
1070    /// Returns an enum representing a borrow of either A or B.
1071    #[inline]
1072    pub fn borrow(&self) -> ArcUnionBorrow<'_, A, B> {
1073        if self.is_first() {
1074            let ptr = self.p.as_ptr() as *const ArcInner<A>;
1075            let borrow = unsafe { ArcBorrow::from_ref(&(*ptr).data) };
1076            ArcUnionBorrow::First(borrow)
1077        } else {
1078            let ptr = ((self.p.as_ptr() as usize) & !0x1) as *const ArcInner<B>;
1079            let borrow = unsafe { ArcBorrow::from_ref(&(*ptr).data) };
1080            ArcUnionBorrow::Second(borrow)
1081        }
1082    }
1083
1084    /// Creates an `ArcUnion` from an instance of the first type.
1085    pub fn from_first(other: Arc<A>) -> Self {
1086        let union = unsafe { Self::new(other.ptr() as *mut _) };
1087        mem::forget(other);
1088        union
1089    }
1090
1091    /// Creates an `ArcUnion` from an instance of the second type.
1092    pub fn from_second(other: Arc<B>) -> Self {
1093        let union = unsafe { Self::new(((other.ptr() as usize) | 0x1) as *mut _) };
1094        mem::forget(other);
1095        union
1096    }
1097
1098    /// Returns true if this `ArcUnion` contains the first type.
1099    pub fn is_first(&self) -> bool {
1100        self.p.as_ptr() as usize & 0x1 == 0
1101    }
1102
1103    /// Returns true if this `ArcUnion` contains the second type.
1104    pub fn is_second(&self) -> bool {
1105        !self.is_first()
1106    }
1107
1108    /// Returns a borrow of the first type if applicable, otherwise `None`.
1109    pub fn as_first(&self) -> Option<ArcBorrow<'_, A>> {
1110        match self.borrow() {
1111            ArcUnionBorrow::First(x) => Some(x),
1112            ArcUnionBorrow::Second(_) => None,
1113        }
1114    }
1115
1116    /// Returns a borrow of the second type if applicable, otherwise None.
1117    pub fn as_second(&self) -> Option<ArcBorrow<'_, B>> {
1118        match self.borrow() {
1119            ArcUnionBorrow::First(_) => None,
1120            ArcUnionBorrow::Second(x) => Some(x),
1121        }
1122    }
1123}

1126    fn clone(&self) -> Self {
1127        match self.borrow() {
1128            ArcUnionBorrow::First(x) => ArcUnion::from_first(x.clone_arc()),
1129            ArcUnionBorrow::Second(x) => ArcUnion::from_second(x.clone_arc()),
1130        }

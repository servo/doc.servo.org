7//! Font table tags
8
9#[derive(Copy, Clone, Debug, Eq, PartialEq, Hash, Ord, PartialOrd)]
10pub struct Tag([u8; 4]);
11
12impl Tag {
13    pub const fn new(bytes: &[u8; 4]) -> Self {
14        Self(*bytes)
15    }
16
17    /// Construct a new `Tag` from a big-endian `u32`
18    pub const fn from_u32(src: u32) -> Self {
19        Self::from_be_bytes(src.to_be_bytes())
20    }
21
22    /// Create a tag from raw big-endian bytes.
23    pub const fn from_be_bytes(bytes: [u8; 4]) -> Self {
24        Self(bytes)
25    }
26
27    /// Return the memory representation of this tag.
28    pub const fn to_be_bytes(self) -> [u8; 4] {
29        self.0
30    }
31}
32
33impl AsRef<[u8]> for Tag {
34    fn as_ref(&self) -> &[u8] {
35        &self.0
36    }
37}

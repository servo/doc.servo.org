21// Over 14k test fonts the max compression ratio seen to date was ~20.
22// >100 suggests you wrote a bad uncompressed size.
23const K_MAX_PLAUSIBLE_COMPRESSION_RATIO: f32 = 100.0;
24

26/// Decompress a WOFF2 file using a custom brotli decompressor passed as a closure
27pub fn decompress_woff2_with_custom_brotli(
28    raw_woff_data: &[u8],
29    decompress_brotli: &mut dyn FnMut(&[u8], usize) -> Result<Vec<u8>, Box<dyn Error>>,
30) -> Result<Vec<u8>, WuffErr> {
31    // Here we create a new view over the `raw_woff_data`. Because we pass `&mut input` to parsing functons,
32    // they will actually mutate the slice (not the data it points to) such that it only includes unparsed data.
33    //
34    // However `raw_woff_data` will still contain the full data for the WOFF.
35    let mut input = raw_woff_data;
36    let full_input_len = input.len();
37
38    // Parse header, table directory and collection directory
39    let header = WoffHeader::parse(&mut input)?;
40    bail_if!(header.woff_version != WoffVersion::Woff2);
41
42    let table_directory = TableDirectory::parse_woff2(&mut input, header.num_tables as usize)?;
43    let mut collection_directory = if header.is_collection() {
44        CollectionDirectory::parse(&mut input, &table_directory)?
45    } else {
46        CollectionDirectory::generate_for_single_font(header.flavor, &table_directory)
47    };
48
49    // Validate header (blocks do not overlap, and have at most 3 bytes padding between them)
50
51    let compressed_offset = full_input_len - input.len();
52    bail_if!(compressed_offset > u32::MAX as usize);
53
54    let mut src_offset = Round4!(compressed_offset + header.total_compressed_size as usize);
55    bail_if!(src_offset > full_input_len);
56
57    if header.meta_offset != 0 {
58        bail_if!(src_offset != header.meta_offset as usize);
59        src_offset = Round4!(header.meta_offset as usize + header.meta_length as usize);
60        bail_if!(src_offset > u32::MAX as usize);
61    }
62
63    if header.priv_offset != 0 {
64        bail_if!(src_offset != header.priv_offset as usize);
65        src_offset = Round4!(header.priv_offset as usize + header.priv_length as usize);
66        bail_if!(src_offset > u32::MAX as usize);
67    }
68
69    bail_if!(src_offset != Round4!(full_input_len));
70
71    // Re-order tables in output (OTSpec) order
72    collection_directory.sort_tables_within_each_font(&table_directory);
73    let num_fonts = collection_directory.fonts.len();
74
75    // Compute compression ratio using the trusted, table-directory-derived uncompressed size
76    // (not the untrusted `totalSfntSize` from the file header). Perform the plausibility check
77    // BEFORE decompressing so an implausible size never drives allocation/decompression.
78    let compression_ratio: f32 =
79        (table_directory.uncompressed_size as f32) / (raw_woff_data.len() as f32);
80
81    // Validate header (and compression ratio)
82    bail_if!(header.total_sfnt_size < 1);
83    bail_with_msg_if!(
84        compression_ratio > K_MAX_PLAUSIBLE_COMPRESSION_RATIO,
85        "Implausible compression ratio {:.1}",
86        compression_ratio
87    );
88
89    // Decompress data with brotli decoder. We pass the trusted `uncompressed_size` as the hard
90    // upper bound on the size of the decompressed data.
91    let compressed_data = &input[0..(header.total_compressed_size as usize)];
92    let decompressed_data = decompress_brotli(compressed_data, table_directory.uncompressed_size)
93        .map_err(|_| WuffErr::GenericError)?;
94
95    // The decompressed data block must be exactly the size of the tables it contains
96    // (tables are stored consecutively with no padding or extraneous data).
97    // <https://www.w3.org/TR/WOFF2/#conform-mustRejectExtraData>
98    bail_if!(decompressed_data.len() != table_directory.uncompressed_size);
99
100    let mut out: Vec<u8> = Vec::with_capacity(table_directory.uncompressed_size);
101
102    let mut out_header = generate_header(&header, &table_directory, &collection_directory);
103    out.extend_from_slice(&out_header.data);
104
105    // Metadata for tables that have been written. Index corresponds to the table's index within the tables Vec
106    let mut table_metadata: Vec<Option<TableMetadata>> = vec![None; header.num_tables as usize];
107    for i in 0..num_fonts {
108        reconstruct_font(
109            &decompressed_data,
110            &header,
111            &table_directory,
112            &collection_directory.fonts[i],
113            &mut out_header,
114            &mut table_metadata,
115            &mut out,
116            i,
117        )?;
118    }
119
120    // Update header
121    out[0..out_header.data.len()].copy_from_slice(&out_header.data);
122
123    Ok(out)
124}

126fn iter_tables_for_font<'a>(
127    font_entry: &'a CollectionDirectoryEntry,
128    tables: &'a TableDirectory,
129) -> impl Iterator<Item = (usize, &'a TableDirectoryEntry)> {
130    font_entry
131        .table_indices
132        .iter()
133        .map(|table_idx| (*table_idx as usize, &tables[*table_idx as usize]))
134}

136// Offset tables assumed to have been written in with 0's initially.
137// WOFF2Header isn't const so we can use [] instead of at() (which upsets FF)
138#[allow(clippy::too_many_arguments)]
139fn reconstruct_font(
140    woff_data: &[u8],
141    header: &WoffHeader,
142    tables: &TableDirectory,
143    font_entry: &CollectionDirectoryEntry,
144    out_header: &mut HeaderData,
145    table_metadata: &mut [Option<TableMetadata>],
146    out: &mut Vec<u8>,
147    font_idx: usize,
148) -> Result<(), WuffErr> {
149    let glyf_idx = font_entry.glyf_idx.map(|idx| idx as usize);
150    let loca_idx = font_entry.loca_idx.map(|idx| idx as usize);
151    let hhea_idx = font_entry.hhea_idx.map(|idx| idx as usize);
152
153    // Check the glyf and loca tables are compatible with each other
154    // 'glyf' without 'loca' doesn't make sense
155    match (glyf_idx, loca_idx) {
156        (Some(glyf_idx), Some(loca_idx)) => {
157            bail_with_msg_if!(
158                tables[glyf_idx].is_transformed() != tables[loca_idx].is_transformed(),
159                "Cannot transform just one of glyf/loca"
160            );
161        }
162        (Some(_), None) | (None, Some(_)) => {
163            bail_with_msg_if!(true, "Cannot have just one of glyf/loca")
164        }
165        (None, None) => {}
166    }
167
168    let mut font_checksum: u32 = if header.is_collection() {
169        out_header.font_infos[font_idx].header_checksum
170    } else {
171        out_header.checksum
172    };
173
174    // Read and store "num_hmetrics" from "hhea" table and then used to reconstruct "hmtx"
175    let num_hmetrics = match hhea_idx {
176        Some(hhea_idx) => {
177            let hhea_table = &tables[hhea_idx];
178            Some(read_num_hmetrics(hhea_table.data_as_slice(woff_data)?)?)
179        }
180        None => None,
181    };
182
183    // These are read from "glyf" and then used to reconstruct "hmtx"
184    let mut num_glyphs = None;
185    let mut x_mins = None;
186
187    // Iterate over the tables for this font.
188    // Note: tables within each font (what we are iterating over here) have already been sorted in alphabetical table tag order.
189    for (table_idx, table) in iter_tables_for_font(font_entry, tables) {
190        // TODO(user) a collection with optimized hmtx that reused glyf/loca
191        // would fail. We don't optimize hmtx for collections yet.
192        bail_if!(table.woff_offset as usize + table.woff_length as usize > woff_data.len());
193
194        // Check to see if we have already processed and saved metadata for this table.
195        // If we have then
196        // There are two cases when this occurs:
197        //   - When a table is reused between fonts in a collection (and this table has already been processed for an earlier font)
198        //   - For the "loca" table. This table gets processed as part of processing "glyf"
199        let metadata = if let Some(metadata) = table_metadata[table_idx] {
200            // Tables shouldn't be reused within a single font (they should be reused between different
201            // fonts in a collection). So if we encounter a table we have already computed metadata for in the first
202            // font unless the table is a "loca" table because we compute metadata for this table when processing the "glyf"
203            // table (so for "loca" encountering already-computed metadata doesn't necessarily indicate reuse).
204            bail_if!(font_idx == 0 && table.tag != LOCA);
205
206            metadata
207        }
208        // Any table which does not need to be transformed
209        else if !table.is_transformed() {
210            let check_sum_adjustment = if table.tag == HEAD {
211                bail_if!(table.woff_length < 12);
212                let checksum_slice =
213                    &woff_data[(table.woff_offset as usize + 8)..(table.woff_offset as usize + 12)];
214                let checksum_bytes: [u8; 4] = checksum_slice.try_into().unwrap();
215                u32::from_be_bytes(checksum_bytes)
216            } else {
217                0
218            };
219
220            let table_data = table.data_as_slice(woff_data)?;
221            let checksum = compute_checksum(table_data).wrapping_sub(check_sum_adjustment);
222
223            let metadata = TableMetadata {
224                dst_offset: out.len() as u32,
225                dst_length: table.woff_length,
226                checksum,
227            };
228            table_metadata[table_idx] = Some(metadata);
229
230            out.extend_from_slice(table_data);
231            out.resize(Round4!(out.len()), 0);
232
233            metadata
234        }
235        // glyf table (also process loca table)
236        else if table.tag == GLYF {
237            let loca_idx =
238                loca_idx.expect("We already returned an error if glyf is present but loca isn't");
239
240            // Generate transformed glyf and loca tables
241            let raw_glyf_table_data = table.data_as_slice(woff_data)?;
242            let glyf_and_loca_data = tranform_glyf_table(raw_glyf_table_data)?;
243
244            // The origLength of the loca table declared in the table directory must exactly
245            // match the size of the reconstructed loca table.
246            // <https://www.w3.org/TR/WOFF2/#conform-mustRejectLoca>
247            bail_with_msg_if!(
248                tables[loca_idx].orig_length as usize != glyf_and_loca_data.loca_table.len(),
249                "loca table origLength does not match reconstructed loca size"
250            );
251
252            // Store num_glyphs and x_mins
253            num_glyphs = Some(glyf_and_loca_data.num_glyphs);
254            x_mins = Some(glyf_and_loca_data.x_mins);
255
256            // Write glyf table
257            let glyf_dest_offset = out.len();
258            out.extend_from_slice(&glyf_and_loca_data.glyf_table);
259            out.resize(Round4!(out.len()), 0);
260            let glyf_metadata = TableMetadata {
261                checksum: glyf_and_loca_data.glyf_checksum,
262                dst_offset: glyf_dest_offset as u32,
263                dst_length: glyf_and_loca_data.glyf_table.len() as u32,
264            };
265            table_metadata[table_idx] = Some(glyf_metadata);
266
267            // Write loca table
268            let loca_dest_offset = out.len();
269            out.extend_from_slice(&glyf_and_loca_data.loca_table);
270            out.resize(Round4!(out.len()), 0);
271            let loca_metdata = TableMetadata {
272                checksum: glyf_and_loca_data.loca_checksum,
273                dst_offset: loca_dest_offset as u32,
274                dst_length: glyf_and_loca_data.loca_table.len() as u32,
275            };
276            table_metadata[loca_idx] = Some(loca_metdata);
277
278            // Return glyf metadata
279            glyf_metadata
280        } else if table.tag == LOCA {
281            unreachable!("loca table is computed when glyf table is processed");
282        }
283        // hmtx table
284        else if table.tag == HMTX {
285            // Tables are sorted so all the info we need has been gathered.
286            // TODO: better error_handling
287            let num_glyphs = num_glyphs.ok_or(WuffErr::GenericError)?;
288            let num_hmetrics = num_hmetrics.ok_or(WuffErr::GenericError)?;
289            let x_mins = x_mins.as_ref().ok_or(WuffErr::GenericError)?;
290
291            // Generate reconstructed hmtx table
292            let mut raw_hmtx_table_data = table.data_as_slice(woff_data)?;
293            let hmtx_data =
294                decode_hmtx_table(&mut raw_hmtx_table_data, num_glyphs, num_hmetrics, x_mins)?;
295            let hmtx_table = generate_hmtx_table(&hmtx_data)?;
296            let checksum = compute_checksum(&hmtx_table);
297
298            // Write table to output buffer
299            let dest_offset = out.len();
300            out.extend_from_slice(&hmtx_table);
301            out.resize(Round4!(out.len()), 0);
302            // Note: like the reference implementation, we record the origLength declared in
303            // the WOFF2 table directory (rather than the size of the reconstructed table)
304            // in the output table directory entry. The two may legitimately differ.
305            let hmtx_metadata = TableMetadata {
306                checksum,
307                dst_offset: dest_offset as u32,
308                dst_length: table.orig_length,
309            };
310            table_metadata[table_idx] = Some(hmtx_metadata);
311
312            hmtx_metadata
313        } else {
314            bail!()
315        };
316
317        // Update font checksum with the checksum for the table
318        font_checksum = font_checksum.wrapping_add(metadata.checksum);
319
320        // update the table entry with real values. We replaced 0's, so update  checksum.
321        out_header.update_table_entry(font_idx, table.tag, metadata);
322        font_checksum = font_checksum.wrapping_add(metadata.header_checksum_contribution());
323
324        // The table (as recorded in the output table directory) must not extend past the end
325        // of the data written (including padding) so far.
326        bail_if!(metadata.dst_offset as u64 + metadata.dst_length as u64 > out.len() as u64);
327    }
328
329    // Update 'head' checkSumAdjustment. We already set it to 0 and summed font.
330    //
331    // The 'head' table is a special case in checksum calculations, as it includes a checksumAdjustment field
332    // that is calculated and written after the table’s checksum is calculated and written into the table directory entry,
333    // necessarily invalidating that checksum value.
334    //
335    // When generating font data, to calculate and write the 'head' table checksum and checksumAdjustment field, do the following:
336    //
337    //   1. Set the checksumAdjustment field to 0.
338    //   2. Calculate the checksum for all tables including the 'head' table and enter the value
339    //      for each table into the corresponding record in the table directory.
340    //   3. Calculate the checksum for the entire font.
341    //   4. Subtract that value from 0xB1B0AFBA.
342    //   5. Store the result in the 'head' table checksumAdjustment field.
343    //
344    // <https://learn.microsoft.com/en-us/typography/opentype/spec/otff#calculating-checksums>
345    let checksum_adjustment = 0xB1B0AFBA_u32.wrapping_sub(font_checksum);
346    if let Some(head_table_idx) = font_entry.head_idx {
347        let head_table_metadata = &table_metadata[head_table_idx as usize]
348            .expect("Every table in the font should have metadata at this point");
349        let mut writer = &mut out[head_table_metadata.dst_offset as usize + 8..];
350        writer.put_u32(checksum_adjustment);
351    }
352
353    Ok(())
354}

356// Get numberOfHMetrics, https://www.microsoft.com/typography/otspec/hhea.htm
357fn read_num_hmetrics(mut hhea_data: &[u8]) -> Result<u16, WuffErr> {
358    bail_if!(hhea_data.remaining() < 34);
359    hhea_data.advance(34); // Skip 34 to reach 'hhea' numberOfHMetrics
360    Ok(hhea_data.try_get_u16()?)
361}

363struct HeaderData {
364    data: Vec<u8>,
365    checksum: u32,
366    font_infos: Vec<WOFF2FontInfo>,
367}
368
369#[derive(Clone, Copy, Default)]
370struct TableMetadata {
371    checksum: u32,
372    dst_offset: u32,
373    dst_length: u32,
374}
375
376impl TableMetadata {
377    pub fn is_already_computed(&self) -> bool {
378        self.dst_offset != 0
379    }
380
381    pub fn header_checksum_contribution(&self) -> u32 {
382        self.checksum
383            .wrapping_add(self.dst_offset)
384            .wrapping_add(self.dst_length)
385    }
386}
387
388impl HeaderData {
389    /// Update the table entry with real values.
390    fn update_table_entry(&mut self, font_idx: usize, tag: Tag, metadata: TableMetadata) {
391        // Write data
392        let table_entry_offset = self.font_infos[font_idx].table_entry_by_tag[&tag];
393
394        let mut out = &mut self.data[(table_entry_offset + 4)..(table_entry_offset + 16)];
395        out.put_u32(metadata.checksum);
396        out.put_u32(metadata.dst_offset);
397        out.put_u32(metadata.dst_length);
398
399        // Update checksum
400        let mut checksum = self.font_infos[font_idx].header_checksum;
401        checksum = checksum.wrapping_add(metadata.checksum);
402        checksum = checksum.wrapping_add(metadata.dst_offset);
403        checksum = checksum.wrapping_add(metadata.dst_length);
404        self.font_infos[font_idx].header_checksum = checksum;
405    }
406}

408// First table goes after all the headers, table directory, etc
409fn compute_header_size(collection_directory: &CollectionDirectory, is_collection: bool) -> usize {
410    if is_collection {
411        collection_directory.table_directories_required_size()
412            + collection_directory.collection_header_required_size()
413    } else {
414        collection_directory.table_directories_required_size()
415    }
416}
417
418fn generate_header(
419    header: &WoffHeader,
420    tables: &TableDirectory,
421    collection_directory: &CollectionDirectory,
422) -> HeaderData {
423    let num_fonts = collection_directory.fonts.len();
424    let size_of_header = compute_header_size(collection_directory, header.is_collection());
425    let mut output: Vec<u8> = Vec::with_capacity(size_of_header);
426    let mut font_infos: Vec<WOFF2FontInfo> = vec![WOFF2FontInfo::default(); num_fonts];
427
428    let mut checksum: u32 = 0;
429
430    // If TTC: write TTC header
431    if header.is_collection() {
432        // TTC header
433        output.put_u32(u32::from_be_bytes(header.flavor.to_be_bytes())); // TAG TTCTag
434        output.put_u32(collection_directory.version); // FIXED Version
435        output.put_u32(num_fonts as u32); // ULONG numFonts
436
437        // let mut offset_table_idx: usize = output.len(); // keep start of offset table for later
438
439        // Write tableDirectoryOffsets
440        let first_table_directory_offset = match collection_directory.version {
441            0x00010000 => 12 + (4 * num_fonts as u32),
442            0x00020000 => 12 + 12 + (4 * num_fonts as u32),
443            _ => unreachable!("Only 1.0 and 2.0 are supported versions"),
444        };
445        let mut table_directory_offset = first_table_directory_offset;
446        for font in collection_directory.fonts.iter() {
447            output.put_u32(table_directory_offset);
448            table_directory_offset += font.table_directory_size() as u32;
449        }
450
451        // space for DSIG fields for header v2
452        if collection_directory.version == 0x00020000 {
453            output.put_u32(0); // ULONG ulDsigTag
454            output.put_u32(0); // ULONG ulDsigLength
455            output.put_u32(0); // ULONG ulDsigOffset
456        }
457
458        checksum = checksum.wrapping_add(compute_checksum(&output));
459    }
460
461    // Write table directory(s)
462    // If file is a TTC: one per font. Else for a single font: one in total.
463    for (font, info) in collection_directory.fonts.iter().zip(font_infos.iter_mut()) {
464        // write the actual offset table so our header doesn't lie
465        // font.dst_offset = offset as u32;
466        let start_offset = output.len();
467        write_table_directory_header(&mut output, font.flavor, font.table_indices.len() as u16);
468
469        for &table_index in &font.table_indices {
470            let tag = tables[table_index as usize].tag;
471            info.table_entry_by_tag.insert(tag, output.len());
472            write_empty_offset_table_entry(&mut output, tag);
473        }
474
475        info.header_checksum = compute_checksum(&output[start_offset..]);
476        checksum = checksum.wrapping_add(info.header_checksum);
477    }
478
479    HeaderData {
480        data: output,
481        font_infos,
482        checksum,
483    }
484}

486// Writes a single Offset Table entry
487fn write_empty_offset_table_entry(output: &mut impl BufMut, tag: Tag) {
488    output.put_u32(u32::from_be_bytes(tag.to_be_bytes()));
489    output.put_u32(0);
490    output.put_u32(0);
491    output.put_u32(0);

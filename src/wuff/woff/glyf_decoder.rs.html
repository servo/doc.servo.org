11// simple glyph flags
12const GLYF_ON_CURVE: u8 = 1 << 0;
13const GLYF_X_SHORT: u8 = 1 << 1;
14const GLYF_Y_SHORT: u8 = 1 << 2;
15const GLYF_REPEAT: u8 = 1 << 3;
16const GLYF_THIS_X_IS_SAME: u8 = 1 << 4;
17const GLYF_THIS_Y_IS_SAME: u8 = 1 << 5;
18const OVERLAP_SIMPLE: u8 = 1 << 6;
19
20const NUM_SUB_STREAMS: usize = 7;
21const FLAG_OVERLAP_SIMPLE_BITMAP: u16 = 1 << 0;
22// 98% of Google Fonts have no glyph above 5k bytes. Largest glyph ever observed was 72k bytes
23const DEFAULT_GLYPH_BUF_SIZE: usize = 5120;
24
25const FLAG_ARG_1_AND_2_ARE_WORDS: u16 = 1 << 0;
26const FLAG_WE_HAVE_A_SCALE: u16 = 1 << 3;
27const FLAG_MORE_COMPONENTS: u16 = 1 << 5;
28const FLAG_WE_HAVE_AN_X_AND_Y_SCALE: u16 = 1 << 6;
29const FLAG_WE_HAVE_A_TWO_BY_TWO: u16 = 1 << 7;
30const FLAG_WE_HAVE_INSTRUCTIONS: u16 = 1 << 8;
31
32const END_PTS_OF_CONTOURS_OFFSET: usize = 10;
33const COMPOSITE_GLYPH_BEGIN: usize = 10;

62    n_points_stream: &'a [u8],

64    glyph_stream: &'a [u8],
65    composite_stream: &'a [u8],
66    bbox_bitmap: &'a [u8],
67    bbox_stream: &'a [u8],
68    instruction_stream: &'a [u8],
69    overlap_bitmap: Option<&'a [u8]>,

77impl GlyfDecoder<'_> {
78    pub fn new<'a>(data: &'a [u8]) -> Result<GlyfDecoder<'a>, WuffErr> {
79        let mut input = data;
80        let _: u16 = input.try_get_u16()?; // first 2 bytes are reserved
81        let flags: u16 = input.try_get_u16()?;
82        let has_overlap_bitmap: bool = (flags & FLAG_OVERLAP_SIMPLE_BITMAP) != 0;
83        let num_glyphs = input.try_get_u16()?;
84        let index_format = input.try_get_u16()?;
85
86        let mut offset: usize = (2 + NUM_SUB_STREAMS) * 4;
87        bail_if!(offset > data.len());
88
89        // Invariant from here on: data_size >= offset
90        let mut read_stream = || {
91            let substream_size: usize = input.try_get_u32()? as usize;
92            bail_if!(substream_size > data.len() - offset);
93            let substream_range = offset..(offset + substream_size);
94            offset += substream_size;
95
96            Ok(&data[substream_range])
97        };
98
99        let n_contour_stream = read_stream()?;
100        let n_points_stream = read_stream()?;
101        let flag_stream = read_stream()?;
102        let glyph_stream = read_stream()?;
103        let composite_stream = read_stream()?;
104        let unsplit_bbox_stream = read_stream()?;
105        let instruction_stream = read_stream()?;
106
107        // Safe because num_glyphs is bounded
108        let bitmap_length: usize = ((num_glyphs as usize + 31) >> 5) << 2;
109        bail_if!(bitmap_length > unsplit_bbox_stream.len());
110        let (bbox_bitmap, bbox_stream) = unsplit_bbox_stream.split_at(bitmap_length);
111
112        let mut overlap_bitmap: Option<&[u8]> = None;
113        if has_overlap_bitmap {
114            let overlap_bitmap_length = (num_glyphs as usize + 7) >> 3;
115            bail_if!(overlap_bitmap_length > data.len() - offset);
116            overlap_bitmap = Some(&data[offset..(offset + (overlap_bitmap_length))]);
117        }
118
119        // Scratch buffer to decode glyphs int.
120        let glyph_buf: Vec<u8> = Vec::with_capacity(DEFAULT_GLYPH_BUF_SIZE);
121
122        Ok(GlyfDecoder {
123            n_contour_stream,
124            n_points_stream,
125            flag_stream,
126            glyph_stream,
127            composite_stream,
128            bbox_bitmap,
129            bbox_stream,
130            instruction_stream,
131            overlap_bitmap,
132            glyph_buf,
133            num_glyphs,
134            index_format,
135        })
136    }

138    pub fn transform(mut self) -> Result<GlyfAndLocaData, WuffErr> {
139        // Setup state
140        let mut glyf_checksum: u32 = 0;
141        let mut glyf_table: Vec<u8> = Vec::with_capacity(self.num_glyphs as usize * 12);
142        let mut loca_values: Vec<u32> = Vec::with_capacity(self.num_glyphs as usize + 1);
143        let mut x_mins: Vec<i16> = vec![0; self.num_glyphs as usize];
144
145        // Iterate over each glyph
146        for i in 0..(self.num_glyphs as usize) {
147            loca_values.push(glyf_table.len() as u32);
148
149            let n_contours: u16 = self.n_contour_stream.try_get_u16()?;
150            let glyph_has_bbox = (self.bbox_bitmap[i >> 3] & (0x80 >> (i & 7))) != 0;
151
152            self.glyph_buf.clear();
153            if n_contours == 0xFFFF {
154                // composite glyphs must have an explicit bbox
155                bail_if!(!glyph_has_bbox);
156                self.parse_composite_glyph()?;
157            } else if n_contours > 0 {
158                // Note: while this look similar to the glyph_has_bbox code above, it's indexing into a different bitmap
159                let has_overlap_bit: bool = self
160                    .overlap_bitmap
161                    .is_some_and(|bitmap| (bitmap[i >> 3] & (0x80 >> (i & 7))) != 0);
162                self.parse_simple_glyph(n_contours, glyph_has_bbox, has_overlap_bit)?;
163            } else {
164                // n_contours == 0; empty glyph. Must NOT have a bbox.
165                bail_with_msg_if!(glyph_has_bbox, "Empty glyph has a bbox")
166            }
167
168            glyf_checksum = glyf_checksum.wrapping_add(compute_checksum(&self.glyph_buf));
169
170            // Write glyph to output table and pad output
171            //
172            // TODO(user) Old code aligned glyphs ... but do we actually need to?
173            // (definitely useful for loca)
174            glyf_table.extend_from_slice(&self.glyph_buf);
175            glyf_table.resize(Round4!(glyf_table.len()), 0);
176
177            // Read the x_min of the glyph in case we nede it to reconstruct 'hmtx'
178            // The x_min value an i16 stored as bytes 2-4 in the glyph header.
179            if n_contours > 0 {
180                let x_min = i16::from_be_bytes(self.glyph_buf[2..4].try_into().unwrap());
181                x_mins[i] = x_min;
182            }
183        }
184
185        // loca[n] will be equal the length of the glyph data ('glyf') table
186        loca_values.push(glyf_table.len() as u32);
187
188        // Generate loca table
189        let (loca_table, loca_checksum) = generate_loca_table(&loca_values, self.index_format)?;
190
191        Ok(GlyfAndLocaData {
192            num_glyphs: self.num_glyphs,
193            index_format: self.index_format,
194            x_mins,
195            loca_table,
196            loca_checksum,
197            glyf_table,
198            glyf_checksum,
199        })
200    }

202    /// Parse glyph data into `self.glyph_buf`
203    fn parse_composite_glyph(&mut self) -> Result<(), WuffErr> {
204        // Create a new iterator over the composite stream when computing the size so that we
205        // we can "rewind" and copy the bytes counted here below.
206        let mut ro_composite_stream = self.composite_stream;
207        let (composite_size, have_instructions) =
208            compute_size_of_composite(&mut ro_composite_stream)?;
209
210        let instruction_size: u16 = if have_instructions {
211            self.glyph_stream.try_get_variable_255_u16()?
212        } else {
213            0
214        };
215
216        let size_needed: usize = 12 + composite_size + (instruction_size as usize);
217        if size_needed > self.glyph_buf.capacity() {
218            self.glyph_buf
219                .reserve(size_needed - self.glyph_buf.capacity());
220        }
221
222        let n_contours: i16 = -1; // All composite glyphs has n_contours = -1
223        self.glyph_buf.put_i16(n_contours);
224
225        self.bbox_stream
226            .try_read_bytes_into(8, &mut self.glyph_buf)?;
227        self.composite_stream
228            .try_read_bytes_into(composite_size, &mut self.glyph_buf)?;
229
230        if have_instructions {
231            self.glyph_buf.put_u16(instruction_size);
232            self.instruction_stream
233                .try_read_bytes_into(instruction_size as usize, &mut self.glyph_buf)?;
234        }
235
236        Ok(())
237    }

239    fn parse_simple_glyph(
240        &mut self,
241        n_contours: u16,
242        glyph_has_bbox: bool,
243        has_overlap_bit: bool,
244    ) -> Result<(), WuffErr> {
245        let n_contours = n_contours as usize;
246
247        // simple glyph
248        let mut n_points_vec: Vec<u16> = Vec::with_capacity(n_contours);
249        let mut total_n_points: u32 = 0;
250        for _ in 0..n_contours {
251            let n_points_contour: u16 = self.n_points_stream.try_get_variable_255_u16()?;
252            n_points_vec.push(n_points_contour);
253            bail_if!(u32_will_overflow(total_n_points, n_points_contour as u32));
254            total_n_points += n_points_contour as u32;
255        }
256        let flag_size: usize = total_n_points as usize;
257        bail_if!(flag_size > self.flag_stream.len());
258
259        let flags_buf = self.flag_stream;
260        let triplet_buf = self.glyph_stream;
261
262        let mut triplet_bytes_consumed: usize = 0;
263
264        let mut points = Vec::with_capacity(total_n_points as usize);
265        triplet_bytes_consumed +=
266            decode_triplet(&flags_buf[0..flag_size], triplet_buf, &mut points)?;
267
268        self.flag_stream.advance(flag_size);
269        self.glyph_stream.advance(triplet_bytes_consumed); // FIXME: pass glyph_stream directly to decode_triplet instead?
270
271        let instruction_size: u16 = self.glyph_stream.try_get_variable_255_u16()?;
272        bail_if!(total_n_points >= (1 << 27) || instruction_size as u32 >= (1 << 30));
273
274        // Reserve needed size to reduce allocations
275        let size_needed: usize =
276            12 + 2 * n_contours + 5 * (total_n_points as usize) + (instruction_size as usize);
277        if self.glyph_buf.capacity() < size_needed {
278            self.glyph_buf
279                .reserve(size_needed - self.glyph_buf.capacity());
280        }
281
282        self.glyph_buf.put_i16(n_contours as i16);
283
284        if glyph_has_bbox {
285            self.bbox_stream
286                .try_read_bytes_into(8, &mut self.glyph_buf)?;
287        } else {
288            write_bbox(points.as_slice(), &mut self.glyph_buf);
289        }
290
291        // From this point, stop writing to the end of the glyph buffer and write to earlier in the buffer
292        // let mut writer = &mut÷ self.glyph_buf[END_PTS_OF_CONTOURS_OFFSET..];
293
294        let mut end_point: i32 = -1;
295        for countour in n_points_vec {
296            end_point += countour as i32;
297            bail_if!(end_point >= 65536);
298            self.glyph_buf.put_u16(end_point as u16);
299        }
300
301        self.glyph_buf.put_u16(instruction_size);
302        self.instruction_stream
303            .try_read_bytes_into(instruction_size as usize, &mut self.glyph_buf)?;
304
305        write_glyph_points(points.as_slice(), has_overlap_bit, &mut self.glyph_buf)?;
306
307        Ok(())
308    }
309}

311// This function stores just the point data. On entry, dst points to the
312// beginning of a simple glyph. Returns true on success.
313fn write_glyph_points(
314    points: &[Point],
315    has_overlap_bit: bool,
316    dst: &mut impl BufMut,
317) -> Result<(), WuffErr> {
318    // Write flags
319    let mut last_flag: u8 = u8::MAX; // not a valid flag so next flag will never be equal to it
320    let mut repeat_count: u8 = 0;
321    let mut last_x: i32 = 0;
322    let mut last_y: i32 = 0;
323    for (i, point) in points.iter().enumerate() {
324        // Compute flag value
325        let flag = {
326            let mut flag: u8 = 0;
327
328            if point.on_curve {
329                flag |= GLYF_ON_CURVE;
330            }
331            if has_overlap_bit && i == 0 {
332                flag |= OVERLAP_SIMPLE;
333            }
334
335            // Handle x
336            let dx: i32 = point.x - last_x;
337            if dx == 0 {
338                flag |= GLYF_THIS_X_IS_SAME;
339            } else if dx > -256 && dx < 256 {
340                flag |= GLYF_X_SHORT | (if dx > 0 { GLYF_THIS_X_IS_SAME } else { 0 });
341            } else {
342                // Do nothing
343            }
344
345            // Handle y
346            let dy: i32 = point.y - last_y;
347            if dy == 0 {
348                flag |= GLYF_THIS_Y_IS_SAME;
349            } else if dy > -256 && dy < 256 {
350                flag |= GLYF_Y_SHORT | (if dy > 0 { GLYF_THIS_Y_IS_SAME } else { 0 });
351            } else {
352                // Do nothing
353            }
354
355            flag
356        };
357
358        // Compare flag value with previous value and write previous value if appropriate.
359        //
360        // To keep writes to the output buffer strictly append-only we don't write flags immediately.
361        // Instead, we keep track of the "last_flag" along with an associated "repeat_count" until we
362        // know that this flag will not be repeated.
363
364        // If the current flag is the same as the previous one and we have not yet reached the repeat count
365        // limit (limit is 255 so it fits in one byte), then we increment the repeat count.
366        if flag == last_flag && repeat_count < 255 {
367            repeat_count += 1;
368        }
369        // Else if either:
370        //   - The current flag is not the same as the previous one.
371        //   - We have hit the repeat count limit
372        else {
373            // If the previous flag has a repeat count associated with it then
374            // set the GLYF_REPEAT flag on that flag and write 2 bytes:
375            //
376            //   1. The previous flag
377            //   2. The repeat count
378            //
379            // Note: it is important that we delay setting the GLYF_REPEAT until this point
380            // otherwise the "if flag == last_flag" will fail to detect repeats.
381            if last_flag != u8::MAX {
382                if repeat_count > 0 {
383                    dst.put_u8(last_flag | GLYF_REPEAT);
384                    dst.put_u8(repeat_count);
385                }
386                // If the repeat count is 0 then just write the previous flag
387                else {
388                    dst.put_u8(last_flag);
389                }
390            }
391
392            // Reset the repeat count to 0
393            repeat_count = 0;
394        }
395
396        // Store values from this iteration
397        last_x = point.x;
398        last_y = point.y;
399        last_flag = flag;
400    }
401
402    // Write final flag
403    if last_flag != u8::MAX {
404        if repeat_count > 0 {
405            dst.put_u8(last_flag | GLYF_REPEAT);
406            dst.put_u8(repeat_count);
407        } else {
408            dst.put_u8(last_flag);
409        };
410    }
411
412    // Write x coordinates
413    last_x = 0;
414    for point in points {
415        let dx: i32 = point.x - last_x;
416        if dx == 0 {
417            // do nothing
418        } else if dx > -256 && dx < 256 {
419            dst.put_u8(dx.unsigned_abs() as u8);
420        } else {
421            // will always fit for valid input, but overflow is harmless
422            dst.put_i16(dx as i16)
423        }
424        last_x += dx;
425    }
426
427    // Write y coordinates
428    last_y = 0;
429    for point in points {
430        let dy: i32 = point.y - last_y;
431        if dy == 0 {
432            // do nothing
433        } else if dy > -256 && dy < 256 {
434            dst.put_u8(dy.unsigned_abs() as u8);
435        } else {
436            dst.put_i16(dy as i16)
437        }
438        last_y += dy;
439    }
440
441    Ok(())
442}

444/// Compute the bounding box of the coordinates, and store into a glyf buffer.
445/// A precondition is that there are at least 10 bytes available.
446/// dst should point to the beginning of a 'glyf' record.
447fn write_bbox(points: &[Point], dst: &mut impl BufMut) {
448    let mut x_min: i32 = 0;
449    let mut y_min: i32 = 0;
450    let mut x_max: i32 = 0;
451    let mut y_max: i32 = 0;
452
453    if !points.is_empty() {
454        x_min = points[0].x;
455        x_max = points[0].x;
456        y_min = points[0].y;
457        y_max = points[0].y;
458    }
459    for &Point { x, y, .. } in points.iter().skip(1) {
460        x_min = x.min(x_min);
461        x_max = x.max(x_max);
462        y_min = y.min(y_min);
463        y_max = y.max(y_max);
464    }
465
466    dst.put_i16(x_min as i16);
467    dst.put_i16(y_min as i16);
468    dst.put_i16(x_max as i16);
469    dst.put_i16(y_max as i16);
470}

472fn compute_size_of_composite(composite_stream: &mut impl Buf) -> Result<(usize, bool), WuffErr> {
473    let mut bytes_read: usize = 0;
474    let mut we_have_instructions: bool = false;
475    let mut flags: u16 = FLAG_MORE_COMPONENTS;
476    while flags & FLAG_MORE_COMPONENTS != 0 {
477        flags = composite_stream.try_get_u16()?;
478        we_have_instructions |= (flags & FLAG_WE_HAVE_INSTRUCTIONS) != 0;
479        let mut arg_size: usize = 2; // glyph index
480        if flags & FLAG_ARG_1_AND_2_ARE_WORDS != 0 {
481            arg_size += 4;
482        } else {
483            arg_size += 2;
484        }
485        if flags & FLAG_WE_HAVE_A_SCALE != 0 {
486            arg_size += 2;
487        } else if flags & FLAG_WE_HAVE_AN_X_AND_Y_SCALE != 0 {
488            arg_size += 4;
489        } else if flags & FLAG_WE_HAVE_A_TWO_BY_TWO != 0 {
490            arg_size += 8;
491        }
492        bail_if!(composite_stream.remaining() < arg_size);
493        composite_stream.advance(arg_size);
494
495        // 2 bytes for the flags + arg_size
496        bytes_read += 2 + arg_size
497    }
498
499    Ok((bytes_read, we_have_instructions))
500}

502fn decode_triplet(flags_in: &[u8], in_: &[u8], result: &mut Vec<Point>) -> Result<usize, WuffErr> {
503    #[inline(always)]
504    fn with_sign(flag: i32, baseval: i32) -> i32 {
505        // Precondition: 0 <= baseval < 65536 (to avoid integer overflow)
506        if (flag & 1) != 0 { baseval } else { -baseval }
507    }
508
509    #[inline(always)]
510    fn safe_add(a: i32, b: i32) -> Result<i32, WuffErr> {
511        bail_if!(((a > 0) && (b > i32::MAX - a)) || ((a < 0) && (b < i32::MIN - a)));
512        Ok(a + b)
513    }
514
515    let mut x: i32 = 0;
516    let mut y: i32 = 0;
517
518    bail_if!(flags_in.len() > in_.len());
519
520    let mut triplet_index: usize = 0;
521
522    for &flag in flags_in {
523        let on_curve: bool = (flag >> 7) == 0;
524        let flag = (flag & 0x7f) as i32;
525
526        let n_data_bytes: usize = if flag < 84 {
527            1
528        } else if flag < 120 {
529            2
530        } else if flag < 124 {
531            3
532        } else {
533            4
534        };
535
536        // Second condition was "triplet_index + n_data_bytes < triplet_index" in C. Clippy detected as checking for overflow
537        // in a way that doesn't work in Rust (because Rust panics rather than wraps in debug mode)
538        bail_if!(
539            usize_will_overflow(triplet_index, n_data_bytes)
540                || (triplet_index + n_data_bytes) > in_.len()
541        );
542
543        let dx: i32;
544        let dy: i32;
545        if flag < 10 {
546            dx = 0;
547            dy = with_sign(flag, ((flag & 14) << 7) + in_[triplet_index] as i32);
548        } else if flag < 20 {
549            dx = with_sign(flag, (((flag - 10) & 14) << 7) + in_[triplet_index] as i32);
550            dy = 0;
551        } else if flag < 84 {
552            let b0: i32 = flag - 20;
553            let b1: i32 = in_[triplet_index] as i32;
554            dx = with_sign(flag, 1 + (b0 & 0x30) + (b1 >> 4));
555            dy = with_sign(flag >> 1, 1 + ((b0 & 0x0c) << 2) + (b1 & 0x0f));
556        } else if flag < 120 {
557            let b0: i32 = flag - 84;
558            dx = with_sign(flag, 1 + ((b0 / 12) << 8) + in_[triplet_index] as i32);
559            dy = with_sign(
560                flag >> 1,
561                1 + (((b0 % 12) >> 2) << 8) + in_[triplet_index + 1] as i32,
562            );
563        } else if flag < 124 {
564            let b2: i32 = in_[triplet_index + 1] as i32;
565            dx = with_sign(flag, ((in_[triplet_index] as i32) << 4) + (b2 >> 4));
566            dy = with_sign(
567                flag >> 1,
568                ((b2 & 0x0f) << 8) + in_[triplet_index + 2] as i32,
569            );
570        } else {
571            dx = with_sign(
572                flag,
573                ((in_[triplet_index] as i32) << 8) + in_[triplet_index + 1] as i32,
574            );
575            dy = with_sign(
576                flag >> 1,
577                ((in_[triplet_index + 2] as i32) << 8) + in_[triplet_index + 3] as i32,
578            );
579        }
580        triplet_index += n_data_bytes;
581        x = safe_add(x, dx)?;
582        y = safe_add(y, dy)?;
583
584        result.push(Point { x, y, on_curve }); // CHECK: was *result++
585    }
586
587    Ok(triplet_index)
588}

590/// Generate a loca table given a slice of loca offsets and an index format
591///
592/// See <https://developer.apple.com/fonts/TrueType-Reference-Manual/RM06/Chap6loca.html>
593pub(crate) fn generate_loca_table(
594    loca_values: &[u32],
595    index_format: u16,
596) -> Result<(Vec<u8>, u32), WuffErr> {
597    let loca_size = loca_values.len();
598    let offset_size: usize = if index_format != 0 { 4 } else { 2 };
599    bail_if!((loca_size << 2) >> 2 != loca_size);
600
601    let mut loca_content: Vec<u8> = Vec::with_capacity(loca_size * offset_size);
602    if index_format != 0 {
603        for &value in loca_values {
604            // loca long version. The actual local offset is stored.
605            loca_content.put_u32(value);
606        }
607    } else {
608        for &value in loca_values {
609            // loca short version. The actual local offset divided by 2 is stored.
610            // Right shift is a cheap divide by 2
611            loca_content.put_u16((value >> 1) as u16);
612        }
613    }
614
615    let checksum = compute_checksum(&loca_content);
616
617    Ok((loca_content, checksum))
618}

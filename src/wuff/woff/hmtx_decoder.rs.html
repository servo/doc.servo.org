7/// Data decoded from the WOFF2 hmtx table which can be used to reconstruct
8/// an open type hmtx table.
9pub struct HmtxData {
10    num_glyphs: u16,
11    num_hmetrics: u16,
12    advance_widths: Vec<u16>,
13    lsbs: Vec<i16>,
14}
15
16/// Decode a WOFF2 transformed hmtx table
17///
18/// <http://dev.w3.org/webfonts/WOFF2/spec/Overview.html#hmtx_table_format>
19pub(crate) fn decode_hmtx_table(
20    input: &mut impl Buf,
21    num_glyphs: u16,
22    num_hmetrics: u16,
23    x_mins: &[i16],
24) -> Result<HmtxData, WuffErr> {
25    // Decode flags
26    let hmtx_flags: u8 = input.try_get_u8()?;
27    let has_proportional_lsbs: bool = (hmtx_flags & 1) == 0;
28    let has_monospace_lsbs: bool = (hmtx_flags & 2) == 0;
29
30    // Bits 2-7 are reserved and MUST be zero.
31    bail_with_msg_if!(
32        (hmtx_flags & 0xFC) != 0,
33        "Illegal hmtx flags; bits 2-7 must be 0"
34    );
35
36    // you say you transformed but there is little evidence of it
37    bail_if!(has_proportional_lsbs && has_monospace_lsbs);
38
39    // Should always be true (*regardless* of input data) unless we've made a programming error.
40    // so we assert rather than bail.
41    bail_if!(x_mins.len() != num_glyphs as usize);
42
43    // num_glyphs 0 is OK if there is no 'glyf' but cannot then xform 'hmtx'.
44    bail_if!(num_hmetrics > num_glyphs);
45
46    // "...only one entry need be in the array, but that entry is required."
47    // <https://www.microsoft.com/typography/otspec/hmtx.htm>
48    bail_if!(num_hmetrics < 1);
49
50    // Read advance widths
51    let mut advance_widths: Vec<u16> = Vec::with_capacity(num_hmetrics as usize);
52    for _ in 0..num_hmetrics {
53        advance_widths.push(input.try_get_u16()?);
54    }
55
56    // Read lsb (proportional) and leftSideBearing (monospace) values into the same Vec
57    let mut lsbs: Vec<i16> = Vec::with_capacity(num_glyphs as usize);
58    for i in 0..num_hmetrics {
59        lsbs.push(match has_proportional_lsbs {
60            true => input.try_get_i16()?,
61            false => x_mins[i as usize],
62        });
63    }
64    for i in num_hmetrics..num_glyphs {
65        lsbs.push(match has_monospace_lsbs {
66            true => input.try_get_i16()?,
67            false => x_mins[i as usize],
68        });
69    }
70
71    Ok(HmtxData {
72        num_glyphs,
73        num_hmetrics,
74        advance_widths,
75        lsbs,
76    })
77}

79/// bake me a shiny new hmtx table
80pub(crate) fn generate_hmtx_table(hmtx_data: &HmtxData) -> Result<Vec<u8>, WuffErr> {
81    let num_glyphs = hmtx_data.num_glyphs as usize;
82    let num_hmetrics = hmtx_data.num_hmetrics as usize;
83
84    let hmtx_output_size: usize = 2 * num_glyphs + 2 * num_hmetrics;
85    let mut hmtx_table: Vec<u8> = Vec::with_capacity(hmtx_output_size);
86    for i in 0..num_glyphs {
87        if i < num_hmetrics {
88            hmtx_table.put_u16(hmtx_data.advance_widths[i]);
89        }
90        hmtx_table.put_i16(hmtx_data.lsbs[i]);
91    }
92
93    Ok(hmtx_table)
94}

5//! A generic, safe mechanism by which DOM objects can be pinned and transferred
6//! between threads (or intra-thread for asynchronous events). Akin to Gecko's
7//! nsMainThreadPtrHandle, this uses thread-safe reference counting and ensures
8//! that the actual SpiderMonkey GC integration occurs on the script thread via
9//! weak refcounts. Ownership of a `Trusted<T>` object means the DOM object of
10//! type T to which it points remains alive. Any other behaviour is undefined.
11//! To guarantee the lifetime of a DOM object when performing asynchronous operations,
12//! obtain a `Trusted<T>` from that object and pass it along with each operation.
13//! A usable pointer to the original DOM object can be obtained on the script thread
14//! from a `Trusted<T>` via the `root` method.
15//!
16//! The implementation of `Trusted<T>` is as follows:
17//! The `Trusted<T>` object contains an atomic reference counted pointer to the Rust DOM object.
18//! A hashtable resides in the script thread, keyed on the pointer.
19//! The values in this hashtable are weak reference counts. When a `Trusted<T>` object is
20//! created or cloned, the reference count is increased. When a `Trusted<T>` is dropped, the count
21//! decreases. If the count hits zero, the weak reference is emptied, and is removed from
22//! its hash table during the next GC. During GC, the entries of the hash table are counted
23//! as JS roots.
24

46/// A pointer to a Rust DOM object that needs to be destroyed.
47#[derive(MallocSizeOf)]
48struct TrustedReference(
49    #[ignore_malloc_size_of = "This is a shared reference."] *const libc::c_void,
50);
51unsafe impl Send for TrustedReference {}

53impl TrustedReference {
54    /// Creates a new TrustedReference from a pointer to a value that impements DOMObject.
55    /// This is not enforced by the type system to reduce duplicated generic code,
56    /// which is acceptable since this method is internal to this module.
57    unsafe fn new(ptr: *const libc::c_void) -> TrustedReference {
58        TrustedReference(ptr)
59    }
60}

62/// A safe wrapper around a raw pointer to a DOM object that can be
63/// shared among threads for use in asynchronous operations. The underlying
64/// DOM object is guaranteed to live at least as long as the last outstanding
65/// `Trusted<T>` instance.
66#[cfg_attr(crown, crown::unrooted_must_root_lint::allow_unrooted_interior)]
67#[derive(MallocSizeOf)]
68pub struct Trusted<T: DomObject> {
69    /// A pointer to the Rust DOM object of type T, but void to allow
70    /// sending `Trusted<T>` between threads, regardless of T's sendability.
71    #[conditional_malloc_size_of]
72    refcount: Arc<TrustedReference>,
73    #[ignore_malloc_size_of = "These are shared by all `Trusted` types."]
74    owner_thread: *const LiveDOMReferences,
75    phantom: PhantomData<T>,
76}

86impl<T: DomObject> Trusted<T> {
87    /// Create a new `Trusted<T>` instance from an existing DOM pointer. The DOM object will
88    /// be prevented from being GCed for the duration of the resulting `Trusted<T>` object's
89    /// lifetime.
90    pub fn new(ptr: &T) -> Trusted<T> {
91        fn add_live_reference(
92            ptr: *const libc::c_void,
93        ) -> (Arc<TrustedReference>, *const LiveDOMReferences) {
94            LIVE_REFERENCES.with(|r| {
95                let live_references = &*r.borrow();
96                let refcount = unsafe { live_references.addref(ptr) };
97                (refcount, live_references as *const _)
98            })
99        }
100
101        let (refcount, owner_thread) = add_live_reference(ptr as *const T as *const _);
102        Trusted {
103            refcount,
104            owner_thread,
105            phantom: PhantomData,
106        }
107    }
108
109    /// Obtain a usable DOM pointer from a pinned `Trusted<T>` value. Fails if used on
110    /// a different thread than the original value from which this `Trusted<T>` was
111    /// obtained.
112    pub fn root(&self) -> DomRoot<T> {
113        fn validate(owner_thread: *const LiveDOMReferences) {
114            assert!(LIVE_REFERENCES.with(|r| {
115                let r = r.borrow();
116                let live_references = &*r;
117                owner_thread == live_references
118            }));
119        }
120        validate(self.owner_thread);
121        unsafe { DomRoot::from_ref(&*(self.refcount.0 as *const T)) }
122    }
123}

125impl<T: DomObject> Clone for Trusted<T> {
126    fn clone(&self) -> Trusted<T> {
127        Trusted {
128            refcount: self.refcount.clone(),
129            owner_thread: self.owner_thread,
130            phantom: PhantomData,
131        }
132    }
133}

135/// The set of live, pinned DOM objects that are currently prevented
136/// from being garbage collected due to outstanding references.
137pub struct LiveDOMReferences {
138    // keyed on pointer to Rust DOM object
139    reflectable_table: RefCell<FxHashMap<*const libc::c_void, Weak<TrustedReference>>>,
140}
141
142/// Trace all the references that are in Trusted and liive.
143/// # Safety
144/// tracer must point to a valid, non-null JS tracer.
145pub unsafe fn trace_live_domreferences(tracer: *mut JSTracer) {
146    LIVE_REFERENCES.with(|r| {
147        let live_references = r.borrow();
148        let mut table = live_references.reflectable_table.borrow_mut();
149        remove_nulls(&mut table);
150        for obj in table.keys() {
151            unsafe {
152                trace_reflector(tracer, "refcounted", &*(*obj as *const Reflector));
153            }
154        }
155    })
156}

158impl LiveDOMReferences {
159    pub fn destruct() {
160        LIVE_REFERENCES.with(|r| {
161            let live_references = r.borrow_mut();
162            let _ = live_references.reflectable_table.take();
163        });
164    }
165
166    /// ptr must be a pointer to a type that implements DOMObject.
167    /// This is not enforced by the type system to reduce duplicated generic code,
168    /// which is acceptable since this method is internal to this module.
169    #[expect(clippy::arc_with_non_send_sync)]
170    unsafe fn addref(&self, ptr: *const libc::c_void) -> Arc<TrustedReference> {
171        let mut table = self.reflectable_table.borrow_mut();
172        let capacity = table.capacity();
173        let len = table.len();
174        if (0 < capacity) && (capacity <= len) {
175            trace!("growing refcounted references by {}", len);
176            remove_nulls(&mut table);
177            table.reserve(len);
178        }
179        match table.entry(ptr) {
180            Occupied(mut entry) => match entry.get().upgrade() {
181                Some(refcount) => refcount,
182                None => {
183                    let refcount = Arc::new(unsafe { TrustedReference::new(ptr) });
184                    entry.insert(Arc::downgrade(&refcount));
185                    refcount
186                },
187            },
188            Vacant(entry) => {
189                let refcount = Arc::new(unsafe { TrustedReference::new(ptr) });
190                entry.insert(Arc::downgrade(&refcount));
191                refcount
192            },
193        }
194    }
195}

197/// Remove null entries from the live references table
198fn remove_nulls<K: Eq + Hash + Clone, V>(table: &mut FxHashMap<K, Weak<V>>) {
199    let to_remove: Vec<K> = table
200        .iter()
201        .filter(|&(_, value)| Weak::upgrade(value).is_none())
202        .map(|(key, _)| key.clone())
203        .collect();
204    trace!("removing {} refcounted references", to_remove.len());
205    for key in to_remove {
206        table.remove(&key);
207    }
208}

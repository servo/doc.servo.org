6/// A trait declaration in an FFI module.
7#[derive(Clone, PartialEq, Eq, Hash, Serialize, Debug)]
8#[non_exhaustive]
9pub struct Trait {
10    pub name: Ident,
11    pub lifetimes: LifetimeEnv,
12    pub methods: Vec<TraitMethod>,
13    pub docs: Docs,
14    pub attrs: Attrs,
15    pub is_sync: bool,
16    pub is_send: bool,
17}
18
19#[derive(Clone, PartialEq, Eq, Hash, Serialize, Debug)]
20#[non_exhaustive]
21pub struct TraitMethod {
22    pub name: Ident,
23    pub abi_name: Ident,
24    pub self_param: Option<TraitSelfParam>,
25    // corresponds to the types in Function(Vec<Box<TypeName>>, Box<TypeName>)
26    // the callback type; except here the params aren't anonymous
27    pub params: Vec<Param>,
28    pub output_type: Option<TypeName>,
29    pub lifetimes: LifetimeEnv,
30    pub attrs: Attrs,
31    pub docs: Docs,
32}
33
34impl Trait {
35    /// Extract a [`Trait`] metadata value from an AST node.
36    pub fn new(trt: &syn::ItemTrait, parent_attrs: &Attrs) -> Self {
37        let mut attrs = parent_attrs.clone();
38        attrs.add_attrs(&trt.attrs);
39
40        let mut trait_fcts = Vec::new();
41
42        let self_ident = &trt.ident;
43        // TODO check this
44        let self_path_trait = PathType::from(&syn::TraitBound {
45            paren_token: None,
46            modifier: syn::TraitBoundModifier::None,
47            lifetimes: None, // todo this is an assumption
48            path: syn::PathSegment {
49                ident: self_ident.clone(),
50                arguments: syn::PathArguments::None,
51            }
52            .into(),
53        });
54        for trait_item in trt.items.iter() {
55            if let syn::TraitItem::Fn(fct) = trait_item {
56                let mut fct_attrs = attrs.clone();
57                fct_attrs.add_attrs(&fct.attrs);
58                // copied from the method parsing
59                let fct_ident = &fct.sig.ident;
60                let concat_fct_ident = format!("{self_ident}_{fct_ident}");
61                let extern_ident = syn::Ident::new(
62                    &attrs.abi_rename.apply(concat_fct_ident.into()),
63                    fct.sig.ident.span(),
64                );
65
66                let all_params = fct
67                    .sig
68                    .inputs
69                    .iter()
70                    .filter_map(|a| match a {
71                        syn::FnArg::Receiver(_) => None,
72                        syn::FnArg::Typed(ref t) => {
73                            Some(Param::from_syn(t, self_path_trait.clone()))
74                        }
75                    })
76                    .collect::<Vec<_>>();
77
78                let self_param = fct
79                    .sig
80                    .receiver()
81                    .map(|rec| TraitSelfParam::from_syn(rec, self_path_trait.clone()));
82
83                let output_type = match &fct.sig.output {
84                    syn::ReturnType::Type(_, return_typ) => Some(TypeName::from_syn(
85                        return_typ.as_ref(),
86                        Some(self_path_trait.clone()),
87                    )),
88                    syn::ReturnType::Default => None,
89                };
90
91                let lifetimes = LifetimeEnv::from_trait_item(
92                    trait_item,
93                    self_param.as_ref(),
94                    &all_params[..],
95                    output_type.as_ref(),
96                );
97
98                trait_fcts.push(TraitMethod {
99                    name: fct_ident.into(),
100                    abi_name: (&extern_ident).into(),
101                    self_param,
102                    params: all_params,
103                    output_type,
104                    lifetimes,
105                    attrs: fct_attrs,
106                    docs: Docs::from_attrs(&fct.attrs),
107                });
108            }
109        }
110        let (mut is_sync, mut is_send) = (false, false);
111        for supertrait in &trt.supertraits {
112            if let syn::TypeParamBound::Trait(syn::TraitBound {
113                path: syn::Path { segments, .. },
114                ..
115            }) = supertrait
116            {
117                let mut seg_iter = segments.iter();
118                if let Some(syn::PathSegment { ident, .. }) = seg_iter.next() {
119                    if *ident != "std" {
120                        continue;
121                    }
122                }
123                if let Some(syn::PathSegment { ident, .. }) = seg_iter.next() {
124                    if *ident != "marker" {
125                        continue;
126                    }
127                }
128                if let Some(syn::PathSegment { ident, .. }) = seg_iter.next() {
129                    if *ident == "Send" {
130                        is_send = true;
131                    } else if *ident == "Sync" {
132                        is_sync = true;
133                    }
134                }
135            }
136        }
137
138        Self {
139            name: (&trt.ident).into(),
140            methods: trait_fcts,
141            docs: Docs::from_attrs(&trt.attrs),
142            lifetimes: LifetimeEnv::from_trait(trt), // TODO
143            attrs,
144            is_send,
145            is_sync,
146        }
147    }
148}

7/// A method declared in the `impl` associated with an FFI struct.
8/// Includes both static and non-static methods, which can be distinguished
9/// by inspecting [`Method::self_param`].
10#[derive(Clone, PartialEq, Eq, Hash, Serialize, Debug)]
11#[non_exhaustive]
12pub struct Method {
13    /// The name of the method as initially declared.
14    pub name: Ident,
15
16    /// Lines of documentation for the method.
17    pub docs: Docs,
18
19    /// The name of the generated `extern "C"` function
20    pub abi_name: Ident,
21
22    /// The `self` param of the method, if any.
23    pub self_param: Option<SelfParam>,
24
25    // the 'Self' type of the method, if any.
26    pub self_type: Option<PathType>,
27
28    /// All non-`self` params taken by the method.
29    pub params: Vec<Param>,
30
31    /// The return type of the method, if any.
32    pub return_type: Option<TypeName>,
33
34    /// The lifetimes introduced in this method and surrounding impl block.
35    pub lifetime_env: LifetimeEnv,
36
37    /// The list of `cfg` attributes (if any).
38    ///
39    /// These are strings instead of `syn::Attribute` or `proc_macro2::TokenStream`
40    /// because those types are not `PartialEq`, `Hash`, `Serialize`, etc.
41    pub attrs: Attrs,
42}

44impl Method {
45    /// Extracts a [`Method`] from an AST node inside an `impl`.
46    pub fn from_syn(
47        m: &syn::ImplItemFn,
48        self_path_type: PathType,
49        impl_generics: Option<&syn::Generics>,
50        impl_attrs: &Attrs,
51    ) -> Method {
52        let mut attrs = impl_attrs.clone();
53        attrs.add_attrs(&m.attrs);
54
55        let self_ident = self_path_type.path.elements.last().unwrap();
56        let method_ident = &m.sig.ident;
57        let concat_method_ident = format!("{self_ident}_{method_ident}");
58        let extern_ident = syn::Ident::new(
59            &attrs.abi_rename.apply(concat_method_ident.into()),
60            m.sig.ident.span(),
61        );
62
63        let all_params = m
64            .sig
65            .inputs
66            .iter()
67            .filter_map(|a| match a {
68                syn::FnArg::Receiver(_) => None,
69                syn::FnArg::Typed(ref t) => Some(Param::from_syn(t, self_path_type.clone())),
70            })
71            .collect::<Vec<_>>();
72
73        let self_param = m
74            .sig
75            .receiver()
76            .map(|rec| SelfParam::from_syn(rec, self_path_type.clone()));
77
78        let return_ty = match &m.sig.output {
79            syn::ReturnType::Type(_, return_typ) => {
80                // When we allow lifetime elision, this is where we would want to
81                // support it so we can insert the expanded explicit lifetimes.
82                Some(TypeName::from_syn(
83                    return_typ.as_ref(),
84                    Some(self_path_type.clone()),
85                ))
86            }
87            syn::ReturnType::Default => None,
88        };
89
90        let lifetime_env = LifetimeEnv::from_method_item(
91            m,
92            impl_generics,
93            self_param.as_ref(),
94            &all_params[..],
95            return_ty.as_ref(),
96        );
97
98        Method {
99            name: Ident::from(method_ident),
100            docs: Docs::from_attrs(&m.attrs),
101            abi_name: Ident::from(&extern_ident),
102            self_param,
103            self_type: Some(self_path_type),
104            params: all_params,
105            return_type: return_ty,
106            lifetime_env,
107            attrs,
108        }
109    }
110
111    /// Returns the parameters that the output is lifetime-bound to.
112    ///
113    /// # Examples
114    ///
115    /// Given the following method:
116    /// ```ignore
117    /// fn foo<'a, 'b: 'a, 'c>(&'a self, bar: Bar<'b>, baz: Baz<'c>) -> FooBar<'a> { ... }
118    /// ```
119    /// Then this method would return the `&'a self` and `bar: Bar<'b>` params
120    /// because `'a` is in the return type, and `'b` must live at least as long
121    /// as `'a`. It wouldn't include `baz: Baz<'c>` though, because the return
122    /// type isn't bound by `'c` in any way.
123    ///
124    /// # Panics
125    ///
126    /// This method may panic if `TypeName::check_result_type_validity` (called by
127    /// `Method::check_validity`) doesn't pass first, since the result type may
128    /// contain elided lifetimes that we depend on for this method. The validity
129    /// checks ensure that the return type doesn't elide any lifetimes, ensuring
130    /// that this method will produce correct results.
131    pub fn borrowed_params(&self) -> BorrowedParams<'_> {
132        // To determine which params the return type is bound to, we just have to
133        // find the params that contain a lifetime that's also in the return type.
134        if let Some(ref return_type) = self.return_type {
135            // The lifetimes that must outlive the return type
136            let lifetimes = return_type.longer_lifetimes(&self.lifetime_env);
137
138            let held_self_param = self.self_param.as_ref().filter(|self_param| {
139                // Check if `self` is a reference with a lifetime in the return type.
140                if let Some((Lifetime::Named(ref name), _)) = self_param.reference {
141                    if lifetimes.contains(&name) {
142                        return true;
143                    }
144                }
145                self_param.path_type.lifetimes.iter().any(|lt| {
146                    if let Lifetime::Named(name) = lt {
147                        lifetimes.contains(&name)
148                    } else {
149                        false
150                    }
151                })
152            });
153
154            // Collect all the params that contain a named lifetime that's also
155            // in the return type.
156            let held_params = self
157                .params
158                .iter()
159                .filter_map(|param| {
160                    let mut lt_kind = LifetimeKind::ReturnValue;
161                    param
162                        .ty
163                        .visit_lifetimes(&mut |lt, _| {
164                            // Thanks to `TypeName::visit_lifetimes`, we can
165                            // traverse the lifetimes without allocations and
166                            // short-circuit if we find a match.
167                            match lt {
168                                Lifetime::Named(name) if lifetimes.contains(&name) => {
169                                    return ControlFlow::Break(());
170                                }
171                                Lifetime::Static => {
172                                    lt_kind = LifetimeKind::Static;
173                                    return ControlFlow::Break(());
174                                }
175                                _ => {}
176                            };
177                            ControlFlow::Continue(())
178                        })
179                        .is_break()
180                        .then_some((param, lt_kind))
181                })
182                .collect();
183
184            BorrowedParams(held_self_param, held_params)
185        } else {
186            BorrowedParams(None, vec![])
187        }
188    }

190    /// Checks whether the method qualifies for special write handling.
191    /// To qualify, a method must:
192    ///  - not return any value
193    ///  - have the last argument be an `&mut diplomat_runtime::DiplomatWrite`
194    ///
195    /// Typically, methods of this form will be transformed in the bindings to a
196    /// method that doesn't take the write as an argument but instead creates
197    /// one locally and just returns the final string.
198    pub fn is_write_out(&self) -> bool {
199        let return_compatible = self
200            .return_type
201            .as_ref()
202            .map(|return_type| match return_type {
203                TypeName::Unit => true,
204                TypeName::Result(ok, _, _) | TypeName::Option(ok, _) => {
205                    matches!(ok.as_ref(), TypeName::Unit)
206                }
207
208                _ => false,
209            })
210            .unwrap_or(true);
211
212        return_compatible && self.params.last().map(Param::is_write).unwrap_or(false)
213    }
214
215    /// Checks if any parameters are write (regardless of other compatibilities for write output)
216    pub fn has_write_param(&self) -> bool {
217        self.params.iter().any(|p| p.is_write())
218    }
219
220    /// Returns the documentation block
221    pub fn docs(&self) -> &Docs {
222        &self.docs
223    }
224}

226/// The `self` parameter taken by a [`Method`].
227#[derive(Clone, PartialEq, Eq, Hash, Serialize, Debug)]
228#[non_exhaustive]
229pub struct SelfParam {
230    /// The lifetime and mutability of the `self` param, if it's a reference.
231    pub reference: Option<(Lifetime, Mutability)>,
232
233    /// The type of the parameter, which will be a named reference to
234    /// the associated struct,
235    pub path_type: PathType,
236
237    /// Associated attributes with this self parameter. Used in Demo Generation, mostly.
238    pub attrs: Attrs,
239}
240
241impl SelfParam {
242    pub fn to_typename(&self) -> TypeName {
243        let typ = TypeName::Named(self.path_type.clone());
244        if let Some((ref lifetime, ref mutability)) = self.reference {
245            return TypeName::Reference(lifetime.clone(), *mutability, Box::new(typ));
246        }
247        typ
248    }
249
250    pub fn from_syn(rec: &syn::Receiver, path_type: PathType) -> Self {
251        SelfParam {
252            reference: rec
253                .reference
254                .as_ref()
255                .map(|(_, lt)| (lt.into(), Mutability::from_syn(&rec.mutability))),
256            path_type,
257            attrs: Attrs::from_attrs(&rec.attrs),
258        }
259    }
260}

262/// The `self` parameter taken by a [`TraitMethod`].

267    pub reference: Option<(Lifetime, Mutability)>,

274impl TraitSelfParam {

294/// A parameter taken by a [`Method`], not including `self`.
295#[derive(Clone, PartialEq, Eq, Hash, Serialize, Debug)]
296#[non_exhaustive]
297pub struct Param {
298    /// The name of the parameter in the original method declaration.
299    pub name: Ident,
300
301    /// The type of the parameter.
302    pub ty: TypeName,
303
304    /// Parameter attributes (like #[diplomat::demo(label = "Out")])
305    pub attrs: Attrs,
306}
307
308impl Param {
309    /// Check if this parameter is a Write
310    pub fn is_write(&self) -> bool {
311        match self.ty {
312            TypeName::Reference(_, Mutability::Mutable, ref w) => **w == TypeName::Write,
313            _ => false,
314        }
315    }
316
317    pub fn from_syn(t: &syn::PatType, self_path_type: PathType) -> Self {
318        let ident = match t.pat.as_ref() {
319            syn::Pat::Ident(ident) => ident,
320            _ => panic!("Unexpected param type"),
321        };
322
323        let attrs = Attrs::from_attrs(&t.attrs);
324
325        Param {
326            name: (&ident.ident).into(),
327            ty: TypeName::from_syn(&t.ty, Some(self_path_type)),
328            attrs,
329        }
330    }
331}

333/// The type of lifetime.
334#[derive(Debug, Copy, Clone, PartialEq, Eq)]
335pub enum LifetimeKind {
336    /// Param must live at least as long as the returned object.
337    ReturnValue,
338    /// Param must live for the duration of the program.
339    Static,
340}
341
342#[derive(Default, Debug)]
343/// Parameters in a method that might be borrowed in the return type.
344#[non_exhaustive]
345pub struct BorrowedParams<'a>(
346    pub Option<&'a SelfParam>,
347    pub Vec<(&'a Param, LifetimeKind)>,
348);
349
350impl BorrowedParams<'_> {
351    /// Returns an [`Iterator`] through the names of the parameters that are borrowed
352    /// for the lifetime of the return value, accepting an `Ident` that the `self`
353    /// param will be called if present.
354    pub fn return_names<'a>(&'a self, self_name: &'a Ident) -> impl Iterator<Item = &'a Ident> {
355        self.0.iter().map(move |_| self_name).chain(
356            self.1
357                .iter()
358                .filter(|(_, ltk)| *ltk == LifetimeKind::ReturnValue)
359                .map(|(param, _)| &param.name),
360        )
361    }
362
363    /// Returns an [`Iterator`] through the names of the parameters that are borrowed for a
364    /// static lifetime.
365    pub fn static_names(&self) -> impl Iterator<Item = &'_ Ident> {
366        self.1
367            .iter()
368            .filter(|(_, ltk)| *ltk == LifetimeKind::Static)
369            .map(|(param, _)| &param.name)
370    }
371
372    /// Returns `true` if a provided param name is included in the borrowed params,
373    /// otherwise `false`.
374    ///
375    /// This method doesn't check the `self` parameter. Use
376    /// [`BorrowedParams::borrows_self`] instead.
377    pub fn contains(&self, param_name: &Ident) -> bool {
378        self.1.iter().any(|(param, _)| &param.name == param_name)
379    }
380
381    /// Returns `true` if there are no borrowed parameters, otherwise `false`.
382    pub fn is_empty(&self) -> bool {
383        self.0.is_none() && self.1.is_empty()
384    }
385
386    /// Returns `true` if the `self` param is borrowed, otherwise `false`.
387    pub fn borrows_self(&self) -> bool {
388        self.0.is_some()
389    }
390
391    /// Returns `true` if there are any borrowed params, otherwise `false`.
392    pub fn borrows_params(&self) -> bool {
393        !self.1.is_empty()
394    }
395
396    /// Returns the number of borrowed params.
397    pub fn len(&self) -> usize {
398        self.1.len() + usize::from(self.0.is_some())
399    }
400}

485    macro_rules! assert_borrowed_params {
486        ([$($return_param:ident),*] $(, [$($static_param:ident),*])? => $($tokens:tt)* ) => {{
487            let method = Method::from_syn(
488                &syn::parse_quote! { $($tokens)* },
489                PathType::new(Path::empty().sub_path(Ident::from("MyStructContainingMethod"))),
490                None,
491                &Attrs::default()
492            );
493
494            let borrowed_params = method.borrowed_params();
495            // The ident parser in syn doesn't allow `self`, so we use "this" as a placeholder
496            // and then change it.
497            let mut actual_return: Vec<&str> = borrowed_params.return_names(&Ident::THIS).map(|ident| ident.as_str()).collect();
498            if borrowed_params.0.is_some() {
499                actual_return[0] = "self";
500            }
501            let expected_return: &[&str] = &[$(stringify!($return_param)),*];
502            assert_eq!(actual_return, expected_return);
503            let actual_static: Vec<&str> = borrowed_params.static_names().map(|ident| ident.as_str()).collect();
504            let expected_static: &[&str] = &[$($(stringify!($static_param)),*)?];
505            assert_eq!(actual_static, expected_static);
506        }};
507    }

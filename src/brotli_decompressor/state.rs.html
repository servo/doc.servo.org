157// Maximum number of compound dictionary chunks that may be attached to a
158// decoder instance, per the shared-brotli draft
159// (https://datatracker.ietf.org/doc/draft-vandevenne-shared-brotli-format/).
160pub const SHARED_BROTLI_MAX_COMPOUND_DICTS: usize = 15;
161// Practical limit for the total size of attached raw dictionaries; matches
162// SHARED_BROTLI_MAX_RAW_DICT_SIZE in the C implementation.
163#[cfg(target_pointer_width = "64")]
164pub const SHARED_BROTLI_MAX_RAW_DICT_SIZE: usize = 1usize << 31;
165#[cfg(not(target_pointer_width = "64"))]
166pub const SHARED_BROTLI_MAX_RAW_DICT_SIZE: usize = 1usize << 27;

168// Backing store for an attached dictionary: either memory this decoder owns
169// (allocated through AllocU8, freed on cleanup) or memory the caller owns and
170// has promised to keep alive and unmodified for as long as the decoder lives.
171//
172// The borrowed variant exists so that the FFI can honor the C API's contract
173// -- "Attached dictionary ownership is not transferred. Data provided to this
174// method should be kept accessible until decoding is finished and decoder
175// instance is destroyed." (c/include/brotli/decode.h) -- which lets many
176// decoder instances share one dictionary image at zero marginal cost instead
177// of each holding a private copy.
178#[allow(dead_code)]
179pub enum MaybeOwnedSlice<AllocU8: alloc::Allocator<u8>> {
180  Owned(AllocU8::AllocatedMemory),
181  // The 'static here is a lie maintained by the unsafe constructors below:
182  // the referent only has to outlive the BrotliState that holds it. It is
183  // spelled 'static rather than as a raw pointer so that BrotliState keeps
184  // its automatic Send/Sync.
185  Borrowed(&'static [u8]),
186  // A dictionary of this length whose bytes are lent to every
187  // BrotliDecompressStreamWithDictionary call instead.
188  External(usize),
189}
190
191impl<AllocU8: alloc::Allocator<u8>> Default for MaybeOwnedSlice<AllocU8> {
192  fn default() -> Self {
193    MaybeOwnedSlice::Owned(AllocU8::AllocatedMemory::default())
194  }
195}
196
197impl<AllocU8: alloc::Allocator<u8>> alloc::SliceWrapper<u8> for MaybeOwnedSlice<AllocU8> {
198  fn slice(&self) -> &[u8] {
199    match *self {
200      MaybeOwnedSlice::Owned(ref mem) => mem.slice(),
201      MaybeOwnedSlice::Borrowed(data) => data,
202      MaybeOwnedSlice::External(_) => &[],
203    }
204  }
205}

207// Sentinel stored in BrotliDecoderCompoundDictionary::block_bits while the
208// block map is stale, i.e. a marker rather than a shift amount. Real shift
209// amounts computed by EnsureCompoundDictionaryInitialized are in 0..=23
210// (total_size is capped at SHARED_BROTLI_MAX_RAW_DICT_SIZE), so this value can
211// never collide with one. The C implementation uses -1 for the same purpose;
212// the field is unsigned here.
213pub const COMPOUND_DICTIONARY_BLOCK_MAP_UNINITIALIZED: u32 = 255;

221// LZ77 prefix ("compound") dictionaries attached to the decoder. Unlike the
222// historical approach of copying the dictionary into the ring buffer, chunks
223// are kept in their own buffers and referenced by backward distances in
224// (max_distance, max_distance + total_size], so they stay addressable even
225// after the ring buffer wraps (issue #42).
226pub struct BrotliDecoderCompoundDictionary<AllocU8: alloc::Allocator<u8>> {
227  pub num_chunks: usize,
228  pub total_size: usize,
229  // Map from address >> block_bits to the first chunk that might contain that
230  // address; COMPOUND_DICTIONARY_BLOCK_MAP_UNINITIALIZED means "not yet
231  // computed".
232  pub block_bits: u32,
233  pub block_map: [u8; 256],
234  // chunk_offsets[i] is the address of the first byte of chunk i;
235  // chunk_offsets[num_chunks] == total_size.
236  pub chunk_offsets: [u32; SHARED_BROTLI_MAX_COMPOUND_DICTS + 1],
237  pub chunks: [MaybeOwnedSlice<AllocU8>; SHARED_BROTLI_MAX_COMPOUND_DICTS],
238  // Cursor for an in-progress copy from the dictionary into the ring buffer:
239  // a single copy command may be interrupted to flush the ring buffer.
240  pub br_index: usize,
241  pub br_offset: usize,
242  pub br_length: usize,
243  pub br_copied: usize,
244}
245
246impl<AllocU8: alloc::Allocator<u8>> Default for BrotliDecoderCompoundDictionary<AllocU8> {
247  fn default() -> Self {
248    BrotliDecoderCompoundDictionary::<AllocU8> {
249      num_chunks: 0,
250      total_size: 0,
251      block_bits: COMPOUND_DICTIONARY_BLOCK_MAP_UNINITIALIZED,
252      block_map: [0; 256],
253      chunk_offsets: [0; SHARED_BROTLI_MAX_COMPOUND_DICTS + 1],
254      chunks: Default::default(),
255      br_index: 0,
256      br_offset: 0,
257      br_length: 0,
258      br_copied: 0,
259    }
260  }
261}

527      AllocU32 : alloc::Allocator<u32>,
528      AllocHC : alloc::Allocator<HuffmanCode> > BrotliState<AllocU8, AllocU32, AllocHC> {
529    pub fn new(alloc_u8 : AllocU8,
530           alloc_u32 : AllocU32,
531           alloc_hc : AllocHC) -> Self{
532        let mut retval = make_brotli_state!(alloc_u8, alloc_u32, alloc_hc, AllocU8::AllocatedMemory::default());
533        retval.large_window = true;
534        BrotliInitBitReader(&mut retval.br);
535        retval
536    }
537    pub fn new_with_custom_dictionary(alloc_u8 : AllocU8,
538           alloc_u32 : AllocU32,
539           alloc_hc : AllocHC,
540           custom_dict: AllocU8::AllocatedMemory) -> Self{
541        let custom_dict_len = custom_dict.slice().len();
542        let mut retval = make_brotli_state!(alloc_u8, alloc_u32, alloc_hc, custom_dict);
543        retval.large_window =  true;
544        BrotliInitBitReader(&mut retval.br);
545        // The dictionary becomes the furthest compound dictionary chunk;
546        // any subsequently attach_dictionary'd ones are nearer in distance
547        // space.
548        if custom_dict_len != 0 && custom_dict_len <= SHARED_BROTLI_MAX_RAW_DICT_SIZE {
549            let dict = core::mem::replace(&mut retval.custom_dict,
550                                          AllocU8::AllocatedMemory::default());
551            // A fresh state has no chunks, so the size check above makes this
552            // infallible. Oversized dictionaries remain in `custom_dict` and
553            // are rejected through the normal decoder error path at startup.
554            let attached = retval.attach_compound_dictionary_chunk(MaybeOwnedSlice::Owned(dict));
555            debug_assert!(attached);
556        }
557        retval
558    }
559    pub fn new_strict(alloc_u8 : AllocU8,
560           alloc_u32 : AllocU32,
561           alloc_hc : AllocHC) -> Self{
562        let mut retval = make_brotli_state!(alloc_u8, alloc_u32, alloc_hc, AllocU8::AllocatedMemory::default());
563        retval.large_window =  false;
564        BrotliInitBitReader(&mut retval.br);
565        retval
566    }
567    // Sets a decoder parameter, the equivalent of the C API
568    // BrotliDecoderSetParameter: a nonzero value disables ring buffer
569    // reallocation (the whole window is allocated at once) or accepts
570    // large-window streams. Allowed only before any compressed data has been
571    // processed; returns false, changing nothing, afterwards.
572    pub fn set_parameter(self : &mut Self,
573                         param: BrotliDecoderParameter,
574                         value: u32) -> bool {
575        match self.state {
576            BrotliRunningState::BROTLI_STATE_UNINITED => {},
577            _ => return false,
578        }
579        match param {
580            BrotliDecoderParameter::BROTLI_DECODER_PARAM_DISABLE_RING_BUFFER_REALLOCATION =>
581                self.canny_ringbuffer_allocation = value == 0,
582            BrotliDecoderParameter::BROTLI_DECODER_PARAM_LARGE_WINDOW =>
583                self.large_window = value != 0,
584        }
585        true
586    }
587    // Sets the smallest ring buffer the decoder starts from, rounded up to a
588    // power of two of at least 32 bytes and capped at the stream's window,
589    // which is only known once decoding starts; 0 restores the default of
590    // 1 KiB. With u32::MAX the whole window is allocated at once and never
591    // regrown, so a u8 allocator that is a fixed pool that never merges freed
592    // blocks needs only the window from it, rather than up to twice that. If
593    // the stream's first metablock of data is its last, the ring buffer is
594    // still sized to just hold it, as it is for small or incompressible
595    // input, which encoders store as one uncompressed metablock and an
596    // empty last one, when that empty metablock is already in the input as
597    // the uncompressed one starts. Disabling ring buffer reallocation with
598    // set_parameter overrides this. Allowed only before any compressed data
599    // has been processed; returns false, changing nothing, afterwards.
600    pub fn set_initial_ring_buffer_size(self : &mut Self, size: u32) -> bool {
601        match self.state {
602            BrotliRunningState::BROTLI_STATE_UNINITED => {},
603            _ => return false,
604        }
605        self.initial_ringbuffer_size = size;
606        true
607    }
608    // Attaches a raw LZ77 prefix dictionary, the equivalent of the C API
609    // BrotliDecoderAttachDictionary with BROTLI_SHARED_DICTIONARY_RAW.
610    // Up to SHARED_BROTLI_MAX_COMPOUND_DICTS dictionaries may be attached,
611    // only before any compressed data has been processed. The most recently
612    // attached dictionary is the closest in backward-distance space.
613    // Returns false (and frees the dictionary) if it cannot be attached.
614    pub fn attach_dictionary(self : &mut Self,
615                             dict: AllocU8::AllocatedMemory) -> bool {
616        self.attach_dictionary_chunk(MaybeOwnedSlice::Owned(dict))
617    }
618    // As attach_dictionary, but the decoder only borrows `dict`: nothing is
619    // copied and nothing is freed on cleanup, so one dictionary image can back
620    // any number of concurrent decoder instances.
621    //
622    // The 'static bound is what makes this safe without a lifetime parameter
623    // on BrotliState: the data provably outlives the decoder. Callers holding
624    // a shorter-lived buffer can leak it (Box::leak, a leaked mmap) to obtain
625    // one, or use attach_dictionary and pay for the copy.
626    pub fn attach_dictionary_borrowed(self : &mut Self, dict: &'static [u8]) -> bool {
627        self.attach_dictionary_chunk(MaybeOwnedSlice::Borrowed(dict))
628    }
629    // As attach_dictionary_borrowed, for a dictionary of any lifetime: only
630    // its size is attached, and its bytes are lent to every
631    // BrotliDecompressStreamWithDictionary call instead, so they only have to
632    // outlive each call, not the decoder. At most one dictionary can be
633    // attached this way.
634    pub fn attach_external_dictionary(self : &mut Self, size: usize) -> bool {
635        if self.external_dictionary_len() != 0 {
636            return false;
637        }
638        self.attach_dictionary_chunk(MaybeOwnedSlice::External(size))
639    }
640    // The size of the external dictionary every call must lend, or 0.
641    pub(crate) fn external_dictionary_len(&self) -> usize {
642        let addon = &self.compound_dictionary;
643        for chunk in addon.chunks.iter().take(addon.num_chunks) {
644            if let MaybeOwnedSlice::External(len) = *chunk {
645                return len;
646            }
647        }
648        0
649    }
650    fn attach_dictionary_chunk(self : &mut Self,
651                               dict: MaybeOwnedSlice<AllocU8>) -> bool {
652        match self.state {
653            BrotliRunningState::BROTLI_STATE_UNINITED => {},
654            _ => {
655                self.free_chunk(dict);
656                return false;
657            },
658        }
659        self.attach_compound_dictionary_chunk(dict)
660    }
661    // Attaches a serialized shared dictionary (magic bytes 0x91 0x00), the
662    // equivalent of the C API BrotliDecoderAttachDictionary with
663    // BROTLI_SHARED_DICTIONARY_SERIALIZED. An embedded LZ77 prefix dictionary
664    // becomes a compound dictionary chunk; custom word lists and transform
665    // lists replace the built-in static dictionary. Only one attached
666    // dictionary may carry custom word/transform lists. Allowed only before
667    // any compressed data has been processed. Returns false (and frees the
668    // dictionary) on failure.
669    pub fn attach_serialized_dictionary(self : &mut Self,
670                                        dict_data: AllocU8::AllocatedMemory) -> bool {
671        self.attach_serialized_dictionary_chunk(MaybeOwnedSlice::Owned(dict_data))
672    }
673    // As attach_serialized_dictionary, but the decoder only borrows
674    // `dict_data`: the blob is referenced in place and an embedded LZ77
675    // prefix becomes a borrowed chunk rather than a copy. Note that custom
676    // word/transform lists keep referencing the blob for the whole decode,
677    // not just for parsing, which is why the 'static bound covers both.
678    pub fn attach_serialized_dictionary_borrowed(self : &mut Self,
679                                                 dict_data: &'static [u8]) -> bool {
680        self.attach_serialized_dictionary_chunk(MaybeOwnedSlice::Borrowed(dict_data))
681    }
682    fn attach_serialized_dictionary_chunk(self : &mut Self,
683                                          dict_data: MaybeOwnedSlice<AllocU8>) -> bool {
684        match self.state {
685            BrotliRunningState::BROTLI_STATE_UNINITED => {},
686            _ => {
687                self.free_chunk(dict_data);
688                return false;
689            },
690        }
691        let summary = match shared_dictionary::dry_parse_serialized_dictionary(
692            dict_data.slice()) {
693            Ok(summary) => summary,
694            Err(()) => {
695                self.free_chunk(dict_data);
696                return false;
697            },
698        };
699        let is_custom = summary.num_word_lists != 0 || summary.num_transform_lists != 0;
700        // Cannot combine different custom static dictionaries, only prefix
701        // dictionaries.
702        if is_custom && self.dictionary.is_custom() {
703            self.free_chunk(dict_data);
704            return false;
705        }
706        // If the blob itself is borrowed, an embedded prefix can be handed to
707        // the compound dictionary as a subslice of it; both are shared
708        // references to caller-owned memory, so no copy is needed.
709        let borrowed_blob: Option<&'static [u8]> = match dict_data {
710            MaybeOwnedSlice::Borrowed(data) => Some(data),
711            MaybeOwnedSlice::Owned(_) | MaybeOwnedSlice::External(_) => None,
712        };
713        let mut parsed = BrotliSharedDictionary::<AllocU8, AllocU32>::default();
714        if is_custom {
715            let arena_size = summary.num_word_lists as usize * WORD_LIST_STRIDE +
716                             summary.num_transform_lists as usize * TRANSFORM_LIST_STRIDE;
717            parsed.meta = self.alloc_u32.alloc_cell(arena_size);
718            if parsed.meta.slice().len() != arena_size ||
719               shared_dictionary::parse_serialized_dictionary_into(
720                   dict_data.slice(), &summary, &mut parsed).is_err() {
721                self.alloc_u32.free_cell(core::mem::replace(&mut parsed.meta,
722                                         AllocU32::AllocatedMemory::default()));
723                self.free_chunk(dict_data);
724                return false;
725            }
726        }
727        if let Some((prefix_offset, prefix_len)) = summary.prefix {
728            let chunk = match borrowed_blob {
729                Some(blob) => MaybeOwnedSlice::Borrowed(
730                    &blob[prefix_offset..prefix_offset + prefix_len]),
731                // An owned blob is not kept around unless it carries custom
732                // lists, so the prefix is copied into its own buffer; that
733                // also leaves the parsed word/transform offsets referencing
734                // the blob unchanged.
735                None => {
736                    let mut chunk = self.alloc_u8.alloc_cell(prefix_len);
737                    if chunk.slice().len() != prefix_len {
738                        self.alloc_u8.free_cell(chunk);
739                        self.alloc_u32.free_cell(core::mem::replace(&mut parsed.meta,
740                                                 AllocU32::AllocatedMemory::default()));
741                        self.free_chunk(dict_data);
742                        return false;
743                    }
744                    chunk.slice_mut().clone_from_slice(
745                        &dict_data.slice()[prefix_offset..prefix_offset + prefix_len]);
746                    MaybeOwnedSlice::Owned(chunk)
747                },
748            };
749            if !self.attach_compound_dictionary_chunk(chunk) {
750                self.alloc_u32.free_cell(core::mem::replace(&mut parsed.meta,
751                                         AllocU32::AllocatedMemory::default()));
752                self.free_chunk(dict_data);
753                return false;
754            }
755        }
756        if is_custom {
757            parsed.blob = dict_data;
758            let old = core::mem::replace(&mut self.dictionary, parsed);
759            self.free_chunk(old.blob);
760            self.alloc_u32.free_cell(old.meta);
761        } else {
762            // Nothing references the blob: a prefix chunk, if any, either
763            // borrows the caller's memory or was copied out.
764            self.free_chunk(dict_data);
765        }
766        true
767    }
768    // Attaches one LZ77 prefix ("compound") dictionary chunk. Chunks must be
769    // attached before any data is decompressed; the most recently attached
770    // chunk is the closest in backward-distance space. Returns false (and
771    // frees the chunk) if the chunk cannot be attached.
772    // Releases a dictionary backing store: owned memory goes back to the
773    // allocator, borrowed memory belongs to the caller and is left alone.
774    pub(crate) fn free_chunk(self : &mut Self, chunk: MaybeOwnedSlice<AllocU8>) {
775        if let MaybeOwnedSlice::Owned(mem) = chunk {
776            self.alloc_u8.free_cell(mem);
777        }
778    }
779    pub(crate) fn attach_compound_dictionary_chunk(self : &mut Self,
780                                            chunk: MaybeOwnedSlice<AllocU8>) -> bool {
781        let size = match chunk {
782            MaybeOwnedSlice::External(len) => len,
783            _ => chunk.slice().len(),
784        };
785        // A zero-length chunk is a no-op and not counted toward the limit.
786        if size == 0 {
787            self.free_chunk(chunk);
788            return true;
789        }
790        let addon = &mut self.compound_dictionary;
791        if addon.num_chunks == SHARED_BROTLI_MAX_COMPOUND_DICTS ||
792           size > SHARED_BROTLI_MAX_RAW_DICT_SIZE - addon.total_size {
793            self.free_chunk(chunk);
794            return false;
795        }
796        addon.chunks[addon.num_chunks] = chunk;
797        addon.num_chunks += 1;
798        addon.total_size += size;
799        addon.chunk_offsets[addon.num_chunks] = addon.total_size as u32;
800        // Invalidate the lazily-computed block map.
801        addon.block_bits = COMPOUND_DICTIONARY_BLOCK_MAP_UNINITIALIZED;
802        true
803    }
804    pub fn BrotliStateMetablockBegin(self : &mut Self) {
805        self.meta_block_remaining_len = 0;
806        self.block_type_length_state.block_length[0] = 1u32 << 24;
807        self.block_type_length_state.block_length[1] = 1u32 << 24;
808        self.block_type_length_state.block_length[2] = 1u32 << 24;
809        self.block_type_length_state.num_block_types[0] = 1;
810        self.block_type_length_state.num_block_types[1] = 1;
811        self.block_type_length_state.num_block_types[2] = 1;
812        self.block_type_length_state.block_type_rb[0] = 1;
813        self.block_type_length_state.block_type_rb[1] = 0;
814        self.block_type_length_state.block_type_rb[2] = 1;
815        self.block_type_length_state.block_type_rb[3] = 0;
816        self.block_type_length_state.block_type_rb[4] = 1;
817        self.block_type_length_state.block_type_rb[5] = 0;
818        self.context_map_slice_index = 0;
819        self.literal_htree_index = 0;
820        self.dist_context_map_slice_index = 0;
821        self.dist_htree_index = 0;
822        self.context_lookup = &kContextLookup[0];
823    }
824    pub fn BrotliStateCleanupAfterMetablock(self : &mut Self) {
825        self.alloc_u8.free_cell(core::mem::replace(&mut self.context_map,
826                                             AllocU8::AllocatedMemory::default()));
827        self.alloc_u8.free_cell(core::mem::replace(&mut self.context_modes,
828                                             AllocU8::AllocatedMemory::default()));
829        self.alloc_u8.free_cell(core::mem::replace(&mut self.dist_context_map,
830                                             AllocU8::AllocatedMemory::default()));
831
832
833        self.literal_hgroup.reset(&mut self.alloc_u32, &mut self.alloc_hc);
834        self.insert_copy_hgroup.reset(&mut self.alloc_u32, &mut self.alloc_hc);
835        self.distance_hgroup.reset(&mut self.alloc_u32, &mut self.alloc_hc);
836    }

838   fn BrotliStateCleanup(self : &mut Self) {
839      self.BrotliStateCleanupAfterMetablock();
840      self.alloc_u8.free_cell(core::mem::replace(&mut self.ringbuffer,
841                              AllocU8::AllocatedMemory::default()));
842      self.alloc_hc.free_cell(core::mem::replace(&mut self.block_type_length_state.block_type_trees,
843                              AllocHC::AllocatedMemory::default()));
844      self.alloc_hc.free_cell(core::mem::replace(&mut self.block_type_length_state.block_len_trees,
845                              AllocHC::AllocatedMemory::default()));
846      self.alloc_hc.free_cell(core::mem::replace(&mut self.context_map_table,
847                              AllocHC::AllocatedMemory::default()));
848      self.alloc_u8.free_cell(core::mem::replace(&mut self.custom_dict,
849                              AllocU8::AllocatedMemory::default()));
850      let num_chunks = self.compound_dictionary.num_chunks.min(self.compound_dictionary.chunks.len());
851      for chunk in self.compound_dictionary.chunks[..num_chunks].iter_mut() {
852        if let MaybeOwnedSlice::Owned(mem) =
853            core::mem::replace(chunk, MaybeOwnedSlice::default()) {
854          self.alloc_u8.free_cell(mem);
855        }
856      }
857      self.compound_dictionary.num_chunks = 0;
858      self.compound_dictionary.total_size = 0;
859      if let MaybeOwnedSlice::Owned(mem) =
860          core::mem::replace(&mut self.dictionary.blob, MaybeOwnedSlice::default()) {
861        self.alloc_u8.free_cell(mem);
862      }
863      self.alloc_u32.free_cell(core::mem::replace(&mut self.dictionary.meta,
864                               AllocU32::AllocatedMemory::default()));
865
866      //FIXME??  BROTLI_FREE(s, s->legacy_input_buffer);
867      //FIXME??  BROTLI_FREE(s, s->legacy_output_buffer);
868    }

884    pub fn BrotliHuffmanTreeGroupInit(self :&mut Self, group : WhichTreeGroup,

887            WhichTreeGroup::LITERAL => self.literal_hgroup.init(&mut self.alloc_u32,
888                                                                &mut self.alloc_hc,
889                                                                alphabet_size, max_symbol, ntrees),
890            WhichTreeGroup::INSERT_COPY => self.insert_copy_hgroup.init(&mut self.alloc_u32,
891                                                                        &mut self.alloc_hc,
892                                                                        alphabet_size, max_symbol, ntrees),
893            WhichTreeGroup::DISTANCE => self.distance_hgroup.init(&mut self.alloc_u32,
894                                                                  &mut self.alloc_hc,
895                                                                  alphabet_size, max_symbol, ntrees),
896        }
897    }
898    pub fn BrotliHuffmanTreeGroupRelease(self :&mut Self, group : WhichTreeGroup) {
899        match group {
900            WhichTreeGroup::LITERAL => self.literal_hgroup.reset(&mut self.alloc_u32,
901                                                                 &mut self.alloc_hc),
902            WhichTreeGroup::INSERT_COPY => self.insert_copy_hgroup.reset(&mut self.alloc_u32,
903                                                                         &mut self.alloc_hc),
904            WhichTreeGroup::DISTANCE => self.distance_hgroup.reset(&mut self.alloc_u32,
905                                                                   &mut self.alloc_hc),
906        }
907    }

119/// Spin up a devtools server that listens for connections on the specified port.
120pub fn start_server(
121    embedder: EmbedderProxy,
122    mem_profiler_chan: ProfilerChan,
123) -> Sender<DevtoolsControlMsg> {
124    let (sender, receiver) = unbounded();
125    {
126        let sender = sender.clone();
127        let sender2 = sender.clone();
128        thread::Builder::new()
129            .name("Devtools".to_owned())
130            .spawn(move || {
131                mem_profiler_chan.run_with_memory_reporting(
132                    || {
133                        if let Some(instance) = DevtoolsInstance::create(sender, receiver, embedder)
134                        {
135                            instance.run()
136                        }
137                    },
138                    String::from("devtools-reporter"),
139                    sender2,
140                    |chan| {
141                        DevtoolsControlMsg::FromChrome(
142                            ChromeToDevtoolsControlMsg::CollectMemoryReport(chan),
143                        )
144                    },
145                )
146            })
147            .expect("Thread spawning failed");
148    }
149    sender
150}

156struct DevtoolsInstance {
157    #[conditional_malloc_size_of]
158    registry: Arc<ActorRegistry>,
159    #[conditional_malloc_size_of]
160    id_map: Arc<Mutex<IdMap>>,
161    browsing_contexts: FxHashMap<BrowsingContextId, String>,
162    /// This is handed to clients so they can notify the devtools instance when
163    /// their connection closes.
164    sender: Sender<DevtoolsControlMsg>,
165    receiver: Receiver<DevtoolsControlMsg>,
166    pipelines: FxHashMap<PipelineId, BrowsingContextId>,
167    actor_workers: FxHashMap<WorkerId, String>,
168    actor_requests: HashMap<String, String>,

170    ///
171    /// Client threads remove their connection from here once they exit.

178    fn create(
179        sender: Sender<DevtoolsControlMsg>,
180        receiver: Receiver<DevtoolsControlMsg>,
181        embedder: EmbedderProxy,
182    ) -> Option<Self> {
183        let address = if pref!(devtools_server_listen_address).is_empty() {
184            SocketAddr::new(Ipv4Addr::new(127, 0, 0, 1).into(), 7000)
185        } else if let Ok(addr) = SocketAddr::from_str(&pref!(devtools_server_listen_address)) {
186            addr
187        } else if let Ok(port) = pref!(devtools_server_listen_address).parse() {
188            SocketAddr::new(Ipv4Addr::new(127, 0, 0, 1).into(), port)
189        } else {
190            SocketAddr::new(Ipv4Addr::new(127, 0, 0, 1).into(), 7000)
191        };
192        println!("Binding devtools to {address}");
193
194        let bound = TcpListener::bind(address).ok().and_then(|l| {
195            l.local_addr()
196                .map(|addr| addr.port())
197                .ok()
198                .map(|port| (l, port))
199        });
200
201        // A token shared with the embedder to bypass permission prompt.
202        let port = if bound.is_some() {
203            Ok(address.port())
204        } else {
205            Err(())
206        };
207        let token = format!("{:X}", rng().next_u32());
208        embedder.send(EmbedderMsg::OnDevtoolsStarted(port, token.clone()));
209
210        let listener = match bound {
211            Some((l, _)) => l,
212            None => {
213                return None;
214            },
215        };
216
217        // Create basic actors
218        let mut registry = ActorRegistry::default();
219        RootActor::register(&mut registry);
220
221        let instance = Self {
222            registry: Arc::new(registry),
223            id_map: Arc::new(Mutex::new(IdMap::default())),
224            browsing_contexts: FxHashMap::default(),
225            pipelines: FxHashMap::default(),
226            sender: sender.clone(),
227            receiver,
228            actor_requests: HashMap::new(),
229            actor_workers: FxHashMap::default(),
230            connections: Default::default(),
231            next_resource_id: 1,
232        };
233
234        thread::Builder::new()
235            .name("DevtoolsCliAcceptor".to_owned())
236            .spawn(move || {
237                // accept connections and process them, spawning a new thread for each one
238                for stream in listener.incoming() {
239                    let mut stream = stream.expect("Can't retrieve stream");
240                    if !allow_devtools_client(&mut stream, &embedder, &token) {
241                        continue;
242                    };
243                    // connection succeeded and accepted
244                    sender
245                        .send(DevtoolsControlMsg::FromChrome(
246                            ChromeToDevtoolsControlMsg::AddClient(stream),
247                        ))
248                        .unwrap();
249                }
250            })
251            .expect("Thread spawning failed");
252
253        Some(instance)
254    }

256    fn run(mut self) {
257        let mut next_id = StreamId(0);
258        while let Ok(msg) = self.receiver.recv() {
259            trace!("{:?}", msg);
260            match msg {
261                DevtoolsControlMsg::FromChrome(ChromeToDevtoolsControlMsg::AddClient(stream)) => {
262                    let id = next_id;
263                    next_id = StreamId(id.0 + 1);
264
265                    {
266                        let connections = self.connections.lock().unwrap();
267                        if connections.is_empty() {
268                            // We used to have no connection, now we have one.
269                            // Therefore, we need updates from script threads.
270                            for browsing_context_name in self.browsing_contexts.values() {
271                                let browsing_context_actor = self
272                                    .registry
273                                    .find::<BrowsingContextActor>(browsing_context_name);
274                                browsing_context_actor.instruct_script_to_send_live_updates(true);
275                            }
276                        }
277                    }
278
279                    let connection: DevtoolsConnection = stream.into();
280                    let registry = self.registry.clone();
281                    let connections = self.connections.clone();
282                    let sender_clone = self.sender.clone();
283                    thread::Builder::new()
284                        .name("DevtoolsClientHandler".to_owned())
285                        .spawn(move || {
286                            handle_client(registry, connection, id, connections, sender_clone)
287                        })
288                        .expect("Thread spawning failed");
289                },
290                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::FramerateTick(
291                    actor_name,
292                    tick,
293                )) => self.handle_framerate_tick(actor_name, tick),
294                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::TitleChanged(
295                    pipeline,
296                    title,
297                )) => self.handle_title_changed(pipeline, title),
298                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::NewGlobal(
299                    ids,
300                    script_sender,
301                    pageinfo,
302                )) => self.handle_new_global(ids, script_sender, pageinfo),
303                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::Navigate(
304                    browsing_context,
305                    state,
306                )) => self.handle_navigate(browsing_context, state),
307                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::ConsoleAPI(
308                    pipeline_id,
309                    console_message,
310                    worker_id,
311                )) => {
312                    let console_message =
313                        DevtoolsConsoleMessage::new(console_message, &self.registry);
314                    self.handle_console_resource(
315                        pipeline_id,
316                        worker_id,
317                        ConsoleResource::ConsoleMessage(console_message),
318                    );
319                },
320                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::ClearConsole(
321                    pipeline_id,
322                    worker_id,
323                )) => self.handle_clear_console(pipeline_id, worker_id),
324                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::CreateSourceActor(
325                    script_sender,
326                    pipeline_id,
327                    source_info,
328                )) => self.handle_create_source_actor(script_sender, pipeline_id, source_info),
329                DevtoolsControlMsg::FromScript(
330                    ScriptToDevtoolsControlMsg::UpdateSourceContent(pipeline_id, source_content),
331                ) => self.handle_update_source_content(pipeline_id, source_content),
332                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::ReportPageError(
333                    pipeline_id,
334                    page_error,
335                )) => self.handle_console_resource(
336                    pipeline_id,
337                    None,
338                    ConsoleResource::PageError(page_error.into()),
339                ),
340                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::ReportCSSError(
341                    pipeline_id,
342                    css_error,
343                )) => {
344                    let console_message = ConsoleMessage {
345                        fields: ConsoleMessageFields {
346                            level: ConsoleLogLevel::Warn,
347                            filename: css_error.filename,
348                            line_number: css_error.line,
349                            column_number: css_error.column,
350                            time_stamp: get_time_stamp(),
351                        },
352                        arguments: vec![DebuggerValue::StringValue(css_error.msg)],
353                        stacktrace: None,
354                    };
355                    let console_message =
356                        DevtoolsConsoleMessage::new(console_message, &self.registry);
357
358                    self.handle_console_resource(
359                        pipeline_id,
360                        None,
361                        ConsoleResource::ConsoleMessage(console_message),
362                    )
363                },
364                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::DomMutation(
365                    pipeline_id,
366                    dom_mutation,
367                )) => {
368                    self.handle_dom_mutation(pipeline_id, dom_mutation).unwrap();
369                },
370                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::DebuggerPause(
371                    pipeline_id,
372                    frame_offset,
373                    pause_reason,
374                )) => self.handle_debugger_pause(pipeline_id, frame_offset, pause_reason),
375                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::CreateFrameActor(
376                    result_sender,
377                    pipeline_id,
378                    frame_info,
379                )) => self.handle_create_frame_actor(result_sender, pipeline_id, frame_info),
380                DevtoolsControlMsg::FromScript(ScriptToDevtoolsControlMsg::CreateObjectActor(
381                    result_sender,
382                    value,
383                )) => self.handle_create_object_actor(result_sender, value),
384                DevtoolsControlMsg::FromScript(
385                    ScriptToDevtoolsControlMsg::CreateEnvironmentActor(
386                        result_sender,
387                        environment,
388                        parent,
389                        actor,
390                    ),
391                ) => {
392                    self.handle_create_environment_actor(result_sender, environment, parent, actor)
393                },
394                DevtoolsControlMsg::FromChrome(ChromeToDevtoolsControlMsg::NetworkEvent(
395                    request_id,
396                    network_event,
397                )) => {
398                    // copy the connections vector
399                    // FIXME: Why do we need to do this? Cloning the connections here is
400                    // almost certainly wrong and means that they might shut down without
401                    // us noticing.
402                    let mut connections = Vec::<DevtoolsConnection>::new();
403                    for connection in self.connections.lock().unwrap().values() {
404                        connections.push(connection.clone());
405                    }
406                    self.handle_network_event(connections, request_id, network_event);
407                },
408                DevtoolsControlMsg::FromChrome(ChromeToDevtoolsControlMsg::ServerExitMsg) => break,
409                DevtoolsControlMsg::FromChrome(
410                    ChromeToDevtoolsControlMsg::CollectMemoryReport(chan),
411                ) => {
412                    perform_memory_report(|ops| {
413                        let reports = vec![Report {
414                            path: path!["devtools"],
415                            kind: ReportKind::ExplicitSystemHeapSize,
416                            size: self.size_of(ops),
417                        }];
418                        chan.send(ProcessReports::new(reports));
419                    });
420                },
421                DevtoolsControlMsg::ClientExited => {
422                    if self.connections.lock().unwrap().is_empty() {
423                        // Tell every browsing context to stop sending us updates, because we have nowhere to
424                        // send them to.
425                        for browsing_context_name in self.browsing_contexts.values() {
426                            let browsing_context_actor = self
427                                .registry
428                                .find::<BrowsingContextActor>(browsing_context_name);
429                            browsing_context_actor.instruct_script_to_send_live_updates(false);
430                        }
431                    }
432                },
433            }
434        }
435
436        // Shut down all active connections
437        let mut connections = self.connections.lock().unwrap();
438        for connection in connections.values_mut() {
439            let _ = connection.shutdown(Shutdown::Both);
440        }
441        connections.clear();
442    }

449    fn handle_navigate(&mut self, browsing_context_id: BrowsingContextId, state: NavigationState) {
450        let Some(browsing_context_name) = self.browsing_contexts.get(&browsing_context_id) else {
451            return;
452        };
453        let browsing_context_name = browsing_context_name.clone();
454        let browsing_context_actor = self
455            .registry
456            .find::<BrowsingContextActor>(&browsing_context_name);
457        let watcher_actor = self
458            .registry
459            .find::<WatcherActor>(&browsing_context_actor.watcher_name);
460        let mut id_map = self.id_map.lock().unwrap();
461        let mut connections = self.connections.lock().unwrap();
462
463        match &state {
464            NavigationState::Start(url) => {
465                watcher_actor.emit_will_navigate(
466                    browsing_context_id,
467                    url.clone(),
468                    &mut connections.values_mut(),
469                    &mut id_map,
470                );
471            },
472            NavigationState::Stop(pipeline_id, page_info) => {
473                watcher_actor.emit_target_available_or_destroyed(
474                    &browsing_context_actor,
475                    &self.registry,
476                    connections.values_mut(),
477                    false,
478                );
479
480                let outer_window_id = id_map.outer_window_id(*pipeline_id);
481                browsing_context_actor.update_pipeline(
482                    *pipeline_id,
483                    outer_window_id,
484                    page_info.clone(),
485                );
486
487                watcher_actor.emit_target_available_or_destroyed(
488                    &browsing_context_actor,
489                    &self.registry,
490                    connections.values_mut(),
491                    true,
492                );
493
494                // TODO: Correctly destroy targets, we probably need to create new browsing context actors too.
495            },
496        }
497    }

499    // We need separate actor representations for each script global that exists;
500    // clients can theoretically connect to multiple globals simultaneously.

502    fn handle_new_global(
503        &mut self,
504        ids: (BrowsingContextId, PipelineId, Option<WorkerId>, WebViewId),
505        script_sender: GenericSender<DevtoolScriptControlMsg>,
506        page_info: DevtoolsPageInfo,
507    ) {
508        let (browsing_context_id, pipeline_id, worker_id, webview_id) = ids;
509        let id_map = &mut self.id_map.lock().unwrap();
510        let devtools_browser_id = id_map.browser_id(webview_id);
511        let devtools_browsing_context_id = id_map.browsing_context_id(browsing_context_id);
512        let devtools_outer_window_id = id_map.outer_window_id(pipeline_id);
513
514        let console_name = new_actor_name::<ConsoleActor>();
515
516        let parent_actor = if let Some(id) = worker_id {
517            let thread_actor = ThreadActor::register(&self.registry, script_sender.clone(), None);
518
519            let worker_type = if page_info.is_service_worker {
520                WorkerType::Service
521            } else {
522                WorkerType::Dedicated
523            };
524            let worker_actor = WorkerTargetActor::register(
525                &self.registry,
526                console_name.clone(),
527                thread_actor.name().into(),
528                id,
529                page_info.url,
530                worker_type,
531                script_sender,
532            );
533            let root_actor = self.registry.find::<RootActor>("root");
534            if page_info.is_service_worker {
535                root_actor
536                    .service_workers
537                    .borrow_mut()
538                    .push(worker_actor.name().into());
539            } else {
540                root_actor
541                    .workers
542                    .borrow_mut()
543                    .push(worker_actor.name().into());
544            }
545
546            self.actor_workers.insert(id, worker_actor.name().into());
547
548            Root::DedicatedWorker(worker_actor.name().into())
549        } else {
550            self.pipelines.insert(pipeline_id, browsing_context_id);
551            let browsing_context_name = self
552                .browsing_contexts
553                .entry(browsing_context_id)
554                .or_insert_with(|| {
555                    BrowsingContextActor::register(
556                        &self.registry,
557                        console_name.clone(),
558                        devtools_browser_id,
559                        devtools_browsing_context_id,
560                        page_info,
561                        pipeline_id,
562                        devtools_outer_window_id,
563                        script_sender.clone(),
564                    )
565                    .name()
566                    .into()
567                });
568            let browsing_context_actor = self
569                .registry
570                .find::<BrowsingContextActor>(browsing_context_name);
571            browsing_context_actor.handle_new_global(pipeline_id, script_sender);
572            Root::BrowsingContext(browsing_context_name.clone())
573        };
574
575        ConsoleActor::register(&self.registry, console_name, parent_actor);
576    }

593    fn handle_console_resource(
594        &mut self,

603        let console_actor = self.registry.find::<ConsoleActor>(&console_actor_name);
604        let id = worker_id.map_or(UniqueId::Pipeline(pipeline_id), UniqueId::Worker);
605
606        for connection in self.connections.lock().unwrap().values_mut() {
607            console_actor.handle_console_resource(
608                resource.clone(),
609                id.clone(),
610                &self.registry,
611                connection,
612            );
613        }

616    fn handle_dom_mutation(
617        &mut self,
618        pipeline_id: PipelineId,
619        dom_mutation: DomMutation,
620    ) -> Result<(), ActorError> {
621        let Some(browsing_context_id) = self.pipelines.get(&pipeline_id) else {
622            log::warn!("Devtools received notification for unknown pipeline {pipeline_id}");
623            return Err(ActorError::Internal);
624        };
625        let Some(browsing_context_name) = self.browsing_contexts.get(browsing_context_id) else {
626            return Err(ActorError::Internal);
627        };
628        let browsing_context_actor = self
629            .registry
630            .find::<BrowsingContextActor>(browsing_context_name);
631        let inspector_actor = self
632            .registry
633            .find::<InspectorActor>(&browsing_context_actor.inspector_name);
634        let walker_actor = self
635            .registry
636            .find::<WalkerActor>(&inspector_actor.walker_name);
637
638        for connection in self.connections.lock().unwrap().values_mut() {
639            walker_actor.handle_dom_mutation(dom_mutation.clone(), connection)?;
640        }
641
642        Ok(())
643    }

658    fn find_console_actor(
659        &self,
660        pipeline_id: PipelineId,
661        worker_id: Option<WorkerId>,
662    ) -> Option<String> {
663        if let Some(worker_id) = worker_id {
664            let worker_name = self.actor_workers.get(&worker_id)?;
665            Some(
666                self.registry

673            let browsing_context_name = self.browsing_contexts.get(browsing_context_id)?;
674            Some(
675                self.registry
676                    .find::<BrowsingContextActor>(browsing_context_name)
677                    .console_name
678                    .clone(),

684        &mut self,
685        connections: Vec<DevtoolsConnection>,
686        request_id: String,
687        network_event: NetworkEvent,
688    ) {
689        let browsing_context_id = match &network_event {
690            NetworkEvent::HttpRequest(req) => req.browsing_context_id,
691            NetworkEvent::HttpRequestUpdate(req) => req.browsing_context_id,
692            NetworkEvent::HttpResponse(resp) => resp.browsing_context_id,
693            NetworkEvent::SecurityInfo(update) => update.browsing_context_id,

695
696        let Some(browsing_context_name) = self.browsing_contexts.get(&browsing_context_id) else {
697            return;
698        };
699
700        let network_event_name = match self.actor_requests.get(&request_id) {
701            Some(name) => name.clone(),
702            None => self.create_network_event_actor(request_id, browsing_context_name.clone()),
703        };
704
705        handle_network_event(
706            Arc::clone(&self.registry),
707            network_event_name,
708            connections,
709            network_event,
710        )
711    }

731    fn handle_create_source_actor(
732        &mut self,
733        script_sender: GenericSender<DevtoolScriptControlMsg>,
734        pipeline_id: PipelineId,
735        source_info: SourceInfo,
736    ) {
737        let source_content = source_info
738            .content
739            .or_else(|| self.registry.inline_source_content(pipeline_id));
740        let source_actor = SourceActor::register(
741            &self.registry,
742            pipeline_id,
743            source_info.url,
744            source_content,
745            source_info.content_type,
746            source_info.spidermonkey_id,
747            source_info.introduction_type,
748            script_sender,
749        );
750        let source_form = self
751            .registry
752            .find::<SourceActor>(source_actor.name())
753            .source_form();
754
755        if let Some(worker_id) = source_info.worker_id {
756            let Some(worker_name) = self.actor_workers.get(&worker_id) else {
757                return;
758            };
759
760            let thread_actor_name = self
761                .registry
762                .find::<WorkerTargetActor>(worker_name)
763                .thread_name
764                .clone();
765            let thread_actor = self.registry.find::<ThreadActor>(&thread_actor_name);
766
767            thread_actor.source_manager.add_source(source_actor.name());
768
769            let worker_actor = self.registry.find::<WorkerTargetActor>(worker_name);
770
771            for stream in self.connections.lock().unwrap().values_mut() {
772                worker_actor.resource_array(
773                    &source_form,
774                    "source".into(),
775                    ResourceArrayType::Available,
776                    stream,
777                );
778            }
779        } else {
780            let Some(browsing_context_id) = self.pipelines.get(&pipeline_id) else {
781                return;
782            };
783            let Some(browsing_context_name) = self.browsing_contexts.get(browsing_context_id)
784            else {
785                return;
786            };
787
788            // Notify browsing context about the new source
789            let browsing_context_actor = self
790                .registry
791                .find::<BrowsingContextActor>(browsing_context_name);
792
793            let thread_actor_name = browsing_context_actor.thread_name.clone();
794            let thread_actor = self.registry.find::<ThreadActor>(&thread_actor_name);
795            thread_actor.source_manager.add_source(source_actor.name());
796
797            for stream in self.connections.lock().unwrap().values_mut() {
798                browsing_context_actor.resource_array(
799                    &source_form,
800                    "source".into(),
801                    ResourceArrayType::Available,
802                    stream,
803                );
804            }
805        }
806    }

808    fn handle_update_source_content(&mut self, pipeline_id: PipelineId, source_content: String) {
809        for source_name in self.registry.source_actor_names_for_pipeline(pipeline_id) {
810            let source_actor = self.registry.find::<SourceActor>(&source_name);
811            let mut content = source_actor.content.borrow_mut();
812            if content.is_none() {
813                *content = Some(source_content.clone());
814            }
815        }
816
817        // Store the source content separately for any future source actors that get created *after* we finish parsing
818        // the HTML. For example, adding an `import` to an inline module script can delay it until after parsing.
819        self.registry
820            .set_inline_source_content(pipeline_id, source_content);
821    }

823    fn handle_debugger_pause(
824        &mut self,
825        pipeline_id: PipelineId,
826        frame_offset: FrameOffset,
827        pause_reason: PauseReason,
828    ) {
829        let Some(browsing_context_name) = self
830            .pipelines
831            .get(&pipeline_id)
832            .and_then(|id| self.browsing_contexts.get(id))
833        else {
834            return;
835        };
836
837        let browsing_context_actor = self
838            .registry
839            .find::<BrowsingContextActor>(browsing_context_name);
840        let thread_actor = self
841            .registry
842            .find::<ThreadActor>(&browsing_context_actor.thread_name);
843
844        let pause_name = PauseActor::register(&self.registry).name().into();
845
846        let frame_actor = self.registry.find::<FrameActor>(&frame_offset.actor);
847        frame_actor.set_offset(frame_offset.column, frame_offset.line);
848
849        let msg = ThreadInterruptedReply {
850            from: thread_actor.name().into(),
851            type_: "paused".to_owned(),
852            actor: pause_name,
853            frame: frame_actor.encode(&self.registry),
854            why: pause_reason,
855        };
856
857        for stream in self.connections.lock().unwrap().values_mut() {
858            let _ = stream.write_json_packet(&msg);
859        }
860    }

935fn allow_devtools_client(stream: &mut TcpStream, embedder: &EmbedderProxy, token: &str) -> bool {
936    // By-pass prompt if we receive a valid token.
937    let token = format!("25:{{\"auth_token\":\"{}\"}}", token);
938    let mut buf = [0; 28];
939    let timeout = std::time::Duration::from_millis(500);
940    // This will read but not consume the bytes from the stream.
941    stream.set_read_timeout(Some(timeout)).unwrap();
942    let peek = stream.peek(&mut buf);
943    stream.set_read_timeout(None).unwrap();
944    if let Ok(len) = peek &&
945        len == buf.len() &&
946        let Ok(s) = std::str::from_utf8(&buf) &&
947        s == token
948    {
949        // Consume the message as it was relevant to us.
950        let _ = stream.read_exact(&mut buf);
951        return true;
952    };
953
954    // No token found. Prompt user
955    let (request_sender, request_receiver) =
956        generic_channel::channel().expect("Failed to create IPC channel!");
957    embedder.send(EmbedderMsg::RequestDevtoolsConnection(request_sender));
958    request_receiver.recv().unwrap() == AllowOrDeny::Allow
959}

961/// Process the input from a single devtools client until EOF.
962fn handle_client(
963    registry: Arc<ActorRegistry>,
964    mut stream: DevtoolsConnection,
965    stream_id: StreamId,
966    connections: Arc<Mutex<FxHashMap<StreamId, DevtoolsConnection>>>,
967    sender: Sender<DevtoolsControlMsg>,
968) {
969    connections
970        .lock()
971        .unwrap()
972        .insert(stream_id, stream.clone());
973
974    log::info!("Connection established to {}", stream.peer_addr().unwrap());
975    let msg = registry.encode::<RootActor, _>("root");
976    if let Err(error) = stream.write_json_packet(&msg) {
977        warn!("Failed to send initial packet from root actor: {error:?}");
978        return;
979    }
980
981    loop {
982        match stream.read_json_packet() {
983            Ok(Some(json_packet)) => {
984                if let Err(()) = registry.handle_message(
985                    json_packet.as_object().unwrap(),
986                    &mut stream,
987                    stream_id,
988                ) {
989                    log::error!("Devtools actor stopped responding");
990                    let _ = stream.shutdown(Shutdown::Both);
991                    break;
992                }
993            },
994            Ok(None) => {
995                log::info!("Devtools connection closed");
996                break;
997            },
998            Err(err_msg) => {
999                log::error!("Failed to read message from devtools client: {}", err_msg);
1000                break;
1001            },
1002        }
1003    }
1004
1005    connections.lock().unwrap().remove(&stream_id);
1006    let _ = sender.send(DevtoolsControlMsg::ClientExited);
1007
1008    registry.cleanup(stream_id);
1009}

1011/// <https://searchfox.org/mozilla-central/source/devtools/server/actors/object/utils.js#148>
1012pub(crate) fn debugger_value_to_json(registry: &ActorRegistry, value: DebuggerValue) -> Value {
1013    let mut v = Map::new();
1014    match value {
1015        DebuggerValue::VoidValue => {
1016            v.insert("type".to_owned(), Value::String("undefined".to_owned()));
1017            Value::Object(v)
1018        },
1019        DebuggerValue::NullValue(uninitialized) => {
1020            v.insert("type".to_owned(), Value::String("null".to_owned()));
1021            if uninitialized {
1022                v.insert("uninitialized".to_owned(), Value::Bool(true));
1023            }
1024            Value::Object(v)
1025        },
1026        DebuggerValue::BooleanValue(boolean) => Value::Bool(boolean),
1027        DebuggerValue::NumberValue(val) => {
1028            if val.is_nan() {
1029                v.insert("type".to_owned(), Value::String("NaN".to_owned()));
1030                Value::Object(v)
1031            } else if val.is_infinite() {
1032                if val < 0. {
1033                    v.insert("type".to_owned(), Value::String("-Infinity".to_owned()));
1034                } else {
1035                    v.insert("type".to_owned(), Value::String("Infinity".to_owned()));
1036                }
1037                Value::Object(v)
1038            } else if val == 0. && val.is_sign_negative() {
1039                v.insert("type".to_owned(), Value::String("-0".to_owned()));
1040                Value::Object(v)
1041            } else {
1042                Value::Number(Number::from_f64(val).unwrap())
1043            }
1044        },
1045        DebuggerValue::StringValue(str) => Value::String(str),
1046        DebuggerValue::ObjectValue {
1047            actor,
1048            class,
1049            own_property_length,
1050            preview,
1051        } => {
1052            let object_name = ObjectActor::register(
1053                registry,
1054                actor,
1055                class,
1056                own_property_length,
1057                preview.map(|preview| *preview),
1058            );
1059            let object_msg = registry.encode::<ObjectActor, _>(&object_name);
1060            let value = serde_json::to_value(object_msg).unwrap_or_default();
1061            Value::Object(value.as_object().cloned().unwrap_or_default())
1062        },
1063    }

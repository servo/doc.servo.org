164    /// Whether the item's block size depends on the size of its row(s), creating a cyclic
165    /// dependency with baseline alignment (which affects row sizing). Per
166    /// <https://www.w3.org/TR/css-grid-1/#row-align> such items do not participate in baseline
167    /// alignment and are aligned using their fallback alignment instead.
168    #[inline(always)]
169    pub fn has_cyclic_block_size_dependency(&self) -> bool {
170        self.size.height.0.uses_percentage() && (self.crosses_intrinsic_row || self.crosses_flexible_row)
171    }

173    /// Returns true if the item participates in baseline alignment: it has `align-self: baseline`
174    /// and neither of its block-axis margins are `auto`.
175    /// See <https://www.w3.org/TR/css-align-3/#baseline-align-self>
176    #[inline(always)]
177    pub fn participates_in_baseline_alignment(&self) -> bool {

243    /// For an item spanning multiple tracks, the upper limit used to calculate its limited min-/max-content contribution is the
244    /// sum of the fixed max track sizing functions of any tracks it spans, and is applied if it only spans such tracks.
245    pub fn spanned_track_limit(
246        &mut self,
247        axis: AbstractAxis,
248        axis_tracks: &[GridTrack],
249        axis_parent_size: Option<f32>,
250        resolve_calc_value: &dyn Fn(*const (), f32) -> f32,
251    ) -> Option<f32> {
252        let spanned_tracks = &axis_tracks[self.track_range_excluding_lines(axis)];
253        let tracks_all_fixed = spanned_tracks.iter().all(|track| {
254            track.max_track_sizing_function.definite_limit(axis_parent_size, resolve_calc_value).is_some()
255        });
256        if tracks_all_fixed {
257            let limit: f32 = spanned_tracks
258                .iter()
259                .map(|track| {
260                    track.max_track_sizing_function.definite_limit(axis_parent_size, resolve_calc_value).unwrap()
261                })
262                .sum();
263            Some(limit)
264        } else {
265            None
266        }
267    }

295    /// Retrieve the known_dimensions for the given grid area size from the cache or compute them.
296    /// The min-content and max-content contributions of an item are both computed from the same known_dimensions.
297    #[inline(always)]
298    fn known_dimensions_cached(
299        &mut self,
300        tree: &mut impl LayoutPartialTree,
301        grid_area_size: Size<Option<f32>>,
302    ) -> Size<Option<f32>> {
303        if let Some(entry) = self.known_dimensions_cache {
304            if entry.grid_area_size == grid_area_size {
305                return entry.known_dimensions;
306            }
307        }
308        let known_dimensions = self.known_dimensions(tree, grid_area_size);
309        self.known_dimensions_cache = Some(KnownDimensionsCacheEntry { grid_area_size, known_dimensions });
310        known_dimensions
311    }

323    /// Compute the known_dimensions to be passed to the child sizing functions
324    /// The key thing that is being done here is applying stretch alignment, which is necessary to
325    /// allow percentage sizes further down the tree to resolve properly in some cases
326    fn known_dimensions(
327        &self,
328        tree: &mut impl LayoutPartialTree,
329        grid_area_size: Size<Option<f32>>,
330    ) -> Size<Option<f32>> {
331        let margins = self.margins_axis_sums_with_baseline_shims(grid_area_size.width, tree);
332
333        let aspect_ratio = self.aspect_ratio;
334        // CSS resolves percentage padding and border against the inline size of the containing
335        // block. For a grid item under intrinsic measurement, that inline-size basis is the grid
336        // area's width when it is definite.
337        // Spec:
338        // https://www.w3.org/TR/css-grid-1/#item-margins
339        // https://www.w3.org/TR/CSS22/box.html#padding-properties
340        let padding = self.padding.resolve_or_zero(grid_area_size.width, |val, basis| tree.calc(val, basis));
341        let border = self.border.resolve_or_zero(grid_area_size.width, |val, basis| tree.calc(val, basis));
342        let padding_border_size = (padding + border).sum_axes();
343        let box_sizing_adjustment =
344            if self.box_sizing == BoxSizing::ContentBox { padding_border_size } else { Size::ZERO };
345        let inherent_size = self
346            .size
347            .maybe_resolve(grid_area_size, |val, basis| tree.calc(val, basis))
348            .maybe_apply_aspect_ratio(aspect_ratio)
349            .maybe_add(box_sizing_adjustment);
350        let min_size = self
351            .min_size
352            .maybe_resolve(grid_area_size, |val, basis| tree.calc(val, basis))
353            .maybe_apply_aspect_ratio(aspect_ratio)
354            .maybe_add(box_sizing_adjustment)
355            // The size of the item is floored by its padding and border
356            .or(padding_border_size.map(Some))
357            .maybe_max(padding_border_size);
358        let max_size = self
359            .max_size
360            .maybe_resolve(grid_area_size, |val, basis| tree.calc(val, basis))
361            .maybe_apply_aspect_ratio(aspect_ratio)
362            .maybe_add(box_sizing_adjustment);
363
364        let grid_area_minus_item_margins_size = grid_area_size.maybe_sub(margins).maybe_max(Size::ZERO);
365
366        // If node is absolutely positioned and width is not set explicitly, then deduce it
367        // from left, right and container_content_box if both are set.
368        let width = inherent_size.width.or_else(|| {
369            // A width that is a sizing keyword is not auto, so it does not stretch. The stretch
370            // keyword resolves to an exact width; the others resolve during content measurement.
371            if self.size.width.is_sizing_keyword() {
372                return match resolve_sizing_keyword(
373                    self.size.width,
374                    grid_area_minus_item_margins_size.width,
375                    grid_area_size.width,
376                    |val, basis| tree.calc(val, basis),
377                ) {
378                    Some(SizingKeywordResolution::Exact(width)) => Some(width),
379                    _ => None,
380                };
381            }
382
383            // Apply width based on stretch alignment if:
384            //  - Alignment style is "stretch" or "normal". Note that "normal" is treated as "stretch"
385            //    here regardless of whether the item has a preferred size or aspect ratio in this axis,
386            //    which differs from the rule used when the item is finally aligned (see `align_and_position_item`).
387            //    Replaced elements are not stretched by "normal".
388            //  - The node is not absolutely positioned
389            //  - The node does not have auto margins in this axis.
390            if !self.margin.left.is_auto() && !self.margin.right.is_auto() && self.is_stretch_aligned(self.justify_self)
391            {
392                return grid_area_minus_item_margins_size.width;
393            }
394
395            None
396        });
397        // Reapply aspect ratio after stretch and absolute position width adjustments
398        let Size { width, height } =
399            Size { width, height: inherent_size.height }.maybe_apply_aspect_ratio(aspect_ratio);
400
401        let height = height.or_else(|| {
402            // A height that is a sizing keyword is not auto, so it does not stretch. The stretch
403            // keyword resolves to an exact height; the others resolve during content measurement.
404            if self.size.height.is_sizing_keyword() {
405                return match resolve_sizing_keyword(
406                    self.size.height,
407                    grid_area_minus_item_margins_size.height,
408                    grid_area_size.height,
409                    |val, basis| tree.calc(val, basis),
410                ) {
411                    Some(SizingKeywordResolution::Exact(height)) => Some(height),
412                    _ => None,
413                };
414            }
415
416            // Apply height based on stretch alignment if:
417            //  - Alignment style is "stretch" or "normal" (see the note on width above)
418            //  - The node is not absolutely positioned
419            //  - The node does not have auto margins in this axis.
420            if !self.margin.top.is_auto() && !self.margin.bottom.is_auto() && self.is_stretch_aligned(self.align_self) {
421                return grid_area_minus_item_margins_size.height;
422            }
423
424            None
425        });
426        // Reapply aspect ratio after stretch and absolute position height adjustments
427        let Size { width, height } = Size { width, height }.maybe_apply_aspect_ratio(aspect_ratio);
428
429        // Clamp size by min and max width/height
430        let Size { width, height } = Size { width, height }.maybe_clamp(min_size, max_size);
431
432        Size { width, height }
433    }

435    /// Returns the grid area's size in the specified axis when every spanned track has a definite fixed size.
436    ///
437    /// During intrinsic sizing, percentages on grid items resolve against the size of the grid area,
438    /// not the grid container. If the spanned tracks in an axis are not all definite yet, the grid
439    /// area is still indefinite in that axis and percentage-dependent values must stay unresolved here.
440    ///
441    /// Spec:
442    /// https://www.w3.org/TR/css-grid-1/#grid-item-sizing
443    /// https://www.w3.org/TR/css-grid-1/#algo-overview
444    ///
445    /// Compute the available_space to be passed to the child sizing functions
446    /// These are estimates based on either the max track sizing function or the provisional base size in the opposite
447    /// axis to the one currently being sized.
448    /// https://www.w3.org/TR/css-grid-1/#algo-overview
449    pub fn grid_area_size(
450        &self,
451        axis: AbstractAxis,
452        axis_tracks: &[GridTrack],
453        other_axis_tracks: &[GridTrack],
454        available_space: Size<Option<f32>>,
455        get_track_size_estimate: impl Fn(&GridTrack, Option<f32>) -> Option<f32>,
456        resolve_calc_value: &impl Fn(*const (), f32) -> f32,
457    ) -> Size<Option<f32>> {
458        let mut size = Size::NONE;
459        size.set(
460            axis,
461            axis_tracks[self.track_range_excluding_lines(axis)]
462                .iter()
463                .map(|track| {
464                    let min_size = track
465                        .min_track_sizing_function
466                        .definite_value(available_space.get(axis), resolve_calc_value)?;
467                    let max_size = track
468                        .max_track_sizing_function
469                        .definite_value(available_space.get(axis), resolve_calc_value)?;
470
471                    if min_size == max_size {
472                        Some(track.base_size)
473                    } else {
474                        None
475                    }
476                })
477                .sum::<Option<f32>>(),
478        );
479
480        size.set(
481            axis.other(),
482            other_axis_tracks[self.track_range_excluding_lines(axis.other())]
483                .iter()
484                .map(|track| {
485                    get_track_size_estimate(track, available_space.get(axis.other()))
486                        .map(|size| size + track.content_alignment_adjustment)
487                })
488                .sum::<Option<f32>>(),
489        );
490
491        size
492    }

494    /// Retrieve the available_space from the cache or compute them using the passed parameters
495    pub fn grid_area_size_cached(
496        &mut self,
497        axis: AbstractAxis,
498        axis_tracks: &[GridTrack],
499        other_axis_tracks: &[GridTrack],
500        available_space: Size<Option<f32>>,
501        get_track_size_estimate: impl Fn(&GridTrack, Option<f32>) -> Option<f32>,
502        resolve_calc_value: &impl Fn(*const (), f32) -> f32,
503    ) -> Size<Option<f32>> {
504        self.grid_area_size_cache.unwrap_or_else(|| {
505            let grid_area_size = self.grid_area_size(
506                axis,
507                axis_tracks,
508                other_axis_tracks,
509                available_space,
510                get_track_size_estimate,
511                resolve_calc_value,
512            );
513            self.grid_area_size_cache = Some(grid_area_size);
514            grid_area_size
515        })
516    }

518    /// Compute the item's resolved margins for size contributions. Horizontal percentage margins always resolve
519    /// to zero if the container size is indefinite as otherwise this would introduce a cyclic dependency.
520    #[inline(always)]
521    pub fn margins_axis_sums_with_baseline_shims(
522        &self,
523        inner_node_width: Option<f32>,
524        tree: &impl LayoutPartialTree,
525    ) -> Size<f32> {
526        Rect {
527            left: self.margin.left.resolve_or_zero(Some(0.0), |val, basis| tree.calc(val, basis)),
528            right: self.margin.right.resolve_or_zero(Some(0.0), |val, basis| tree.calc(val, basis)),
529            top: self.margin.top.resolve_or_zero(inner_node_width, |val, basis| tree.calc(val, basis))
530                + self.baseline_shim,
531            bottom: self.margin.bottom.resolve_or_zero(inner_node_width, |val, basis| tree.calc(val, basis)),
532        }
533        .sum_axes()
534    }

536    /// Compute the item's min content contribution from the provided parameters
537    pub fn min_content_contribution(
538        &mut self,
539        axis: AbstractAxis,
540        tree: &mut impl LayoutPartialTree,
541        grid_area_size: Size<Option<f32>>,
542        available_space: Size<Option<f32>>,
543    ) -> f32 {
544        let known_dimensions = self.known_dimensions_cached(tree, grid_area_size);
545        // If the item's size in the axis being measured is already known then that size is its contribution,
546        // and we can avoid calling into the child entirely.
547        if let Some(size) = known_dimensions.get(axis) {
548            return size;
549        }
550        // The child sees the grid area as its containing block during intrinsic measurement, so
551        // percentage box properties resolve against the grid area when that size is definite.
552        // Spec:
553        // https://www.w3.org/TR/css-grid-1/#grid-item-sizing
554        // https://www.w3.org/TR/css-grid-1/#algo-overview
555        tree.measure_child_size(
556            self.node,
557            known_dimensions,
558            grid_area_size,
559            self.keyword_adjusted_available_space(
560                grid_area_size,
561                self.available_space_minus_margins(grid_area_size, available_space, tree).map(|opt| match opt {
562                    Some(size) => AvailableSpace::Definite(size),
563                    None => AvailableSpace::MinContent,
564                }),
565                tree,
566            ),
567            SizingMode::InherentSize,
568            axis.as_abs_naive(),
569            Line::FALSE,
570        )
571    }

589    /// Compute the item's max content contribution from the provided parameters
590    pub fn max_content_contribution(
591        &mut self,
592        axis: AbstractAxis,
593        tree: &mut impl LayoutPartialTree,
594        grid_area_size: Size<Option<f32>>,
595        available_space: Size<Option<f32>>,
596    ) -> f32 {
597        let known_dimensions = self.known_dimensions_cached(tree, grid_area_size);
598        // If the item's size in the axis being measured is already known then that size is its contribution,
599        // and we can avoid calling into the child entirely.
600        if let Some(size) = known_dimensions.get(axis) {
601            return size;
602        }
603        // See the min-content path above. Max-content measurement uses the same containing-block
604        // basis so percentage-dependent item geometry is measured from the grid area rather than
605        // from the container.
606        tree.measure_child_size(
607            self.node,
608            known_dimensions,
609            grid_area_size,
610            self.keyword_adjusted_available_space(
611                grid_area_size,
612                self.available_space_minus_margins(grid_area_size, available_space, tree).map(|opt| match opt {
613                    Some(size) => AvailableSpace::Definite(size),
614                    None => AvailableSpace::MaxContent,
615                }),
616                tree,
617            ),
618            SizingMode::InherentSize,
619            axis.as_abs_naive(),
620            Line::FALSE,
621        )
622    }

624    /// Subtract the item's margins (and baseline shim) from the definite axes of the space available
625    /// to the item's margin box, giving the available space that is passed to the item.
626    /// As when the item is laid out into its final position, margins resolve against the width
627    /// of the grid area and `auto` margins are treated as zero.
628    #[inline(always)]
629    fn available_space_minus_margins(
630        &self,
631        grid_area_size: Size<Option<f32>>,
632        available_space: Size<Option<f32>>,
633        tree: &impl LayoutPartialTree,
634    ) -> Size<Option<f32>> {
635        if available_space.width.is_none() && available_space.height.is_none() {
636            return available_space;
637        }
638        let margin = self.margin.resolve_or_zero(grid_area_size.width, |val, basis| tree.calc(val, basis));
639        Size {
640            width: available_space.width.map(|width| f32_max(width - margin.horizontal_axis_sum(), 0.0)),
641            height: available_space
642                .height
643                .map(|height| f32_max(height - margin.vertical_axis_sum() - self.baseline_shim, 0.0)),
644        }
645    }

647    /// Override the available space in each axis whose size style is a sizing keyword that
648    /// measures the item under a specific available space constraint
649    /// (min-content, max-content, fit-content, fit-content(...))
650    fn keyword_adjusted_available_space(
651        &self,
652        grid_area_size: Size<Option<f32>>,
653        available_space: Size<AvailableSpace>,
654        tree: &impl LayoutPartialTree,
655    ) -> Size<AvailableSpace> {
656        if !self.size.width.is_sizing_keyword() && !self.size.height.is_sizing_keyword() {
657            return available_space;
658        }
659        let margins = self.margins_axis_sums_with_baseline_shims(grid_area_size.width, tree);
660        let mut adjusted = available_space;
661        for axis in [AbstractAxis::Inline, AbstractAxis::Block] {
662            let size_style = self.size.get(axis);
663            if !size_style.is_sizing_keyword() {
664                continue;
665            }
666            let stretch_size = grid_area_size.get(axis).maybe_sub(margins.get(axis)).maybe_max(0.0);
667            if let Some(SizingKeywordResolution::Measure(available)) =
668                resolve_sizing_keyword(size_style, stretch_size, grid_area_size.get(axis), |val, basis| {
669                    tree.calc(val, basis)
670                })
671            {
672                adjusted.set(axis, available);
673            }
674        }
675        adjusted
676    }

694    /// The minimum contribution of an item is the smallest outer size it can have.
695    /// Specifically:
696    ///   - If the item’s computed preferred size behaves as auto or depends on the size of its containing block in the relevant axis:
697    ///     Its minimum contribution is the outer size that would result from assuming the item’s used minimum size as its preferred size;
698    ///   - Else the item’s minimum contribution is its min-content contribution.
699    ///
700    /// Because the minimum contribution often depends on the size of the item’s content, it is considered a type of intrinsic size contribution.
701    /// See: https://www.w3.org/TR/css-grid-1/#min-size-auto
702    pub fn minimum_contribution(
703        &mut self,
704        tree: &mut impl LayoutPartialTree,
705        axis: AbstractAxis,
706        axis_tracks: &[GridTrack],
707        grid_area_size: Size<Option<f32>>,
708        inner_node_size: Size<Option<f32>>,
709    ) -> f32 {
710        let padding = self.padding.resolve_or_zero(grid_area_size.width, |val, basis| tree.calc(val, basis));
711        let border = self.border.resolve_or_zero(grid_area_size.width, |val, basis| tree.calc(val, basis));
712        let padding_border_size = (padding + border).sum_axes();
713        let box_sizing_adjustment =
714            if self.box_sizing == BoxSizing::ContentBox { padding_border_size } else { Size::ZERO };
715        self.size
716            .maybe_resolve(grid_area_size, |val, basis| tree.calc(val, basis))
717            .maybe_apply_aspect_ratio(self.aspect_ratio)
718            .maybe_add(box_sizing_adjustment)
719            .get(axis)
720            .map(|size| {
721                // A definite preferred size is clamped by the minimum and maximum sizes. For a compressible replaced
722                // element, an indefinite percentage maximum size is resolved against zero.
723                let min_size = self.min_size.get(axis);
724                let max_size = self.max_size.get(axis);
725                if min_size.is_auto() && max_size.is_auto() {
726                    return size;
727                }
728                let basis = grid_area_size.get(axis);
729                let max_size_basis = if self.is_compressible_replaced { basis.or(Some(0.0)) } else { basis };
730                let adjustment = box_sizing_adjustment.get(axis);
731                let min_size = min_size.maybe_resolve(basis, |val, basis| tree.calc(val, basis));
732                let max_size = max_size.maybe_resolve(max_size_basis, |val, basis| tree.calc(val, basis));
733                size.maybe_clamp(min_size.maybe_add(adjustment), max_size.maybe_add(adjustment))
734            })
735            .or_else(|| {
736                // A preferred size that is a sizing keyword other than `stretch` does not behave as auto,
737                // so the minimum contribution is the min-content contribution.
738                let size = self.size.get(axis);
739                (size.is_sizing_keyword() && !size.is_stretch())
740                    .then(|| self.min_content_contribution_cached(axis, tree, grid_area_size, grid_area_size))
741            })
742            .or_else(|| {
743                self.min_size
744                    .maybe_resolve(grid_area_size.map(|size| size.unwrap_or(0.0)), |val, basis| tree.calc(val, basis))
745                    .maybe_apply_aspect_ratio(self.aspect_ratio)
746                    .maybe_add(box_sizing_adjustment)
747                    .get(axis)
748            })
749            .or_else(|| self.overflow.get(axis).maybe_into_automatic_min_size())
750            .unwrap_or_else(|| {
751                // Automatic minimum size. See https://www.w3.org/TR/css-grid-1/#min-size-auto
752
753                // To provide a more reasonable default minimum size for grid items, the used value of its automatic minimum size
754                // in a given axis is the content-based minimum size if all of the following are true:
755                let item_axis_tracks = &axis_tracks[self.track_range_excluding_lines(axis)];
756
757                // it is not a scroll container (handled above)
758
759                // it spans at least one track in that axis whose min track sizing function is auto
760                let spans_auto_min_track = item_axis_tracks
761                    .iter()
762                    .any(|track| track.min_track_sizing_function.behaves_as_auto(inner_node_size.get(axis)));
763
764                // if it spans more than one track in that axis, none of those tracks are flexible
765                let only_span_one_track = item_axis_tracks.len() == 1;
766                let use_content_based_minimum = spans_auto_min_track
767                    && (only_span_one_track
768                        || !item_axis_tracks.iter().any(|track| track.max_track_sizing_function.is_fr()));
769
770                // Otherwise, the automatic minimum size is zero, as usual.
771                if use_content_based_minimum {
772                    let mut minimum_contribution =
773                        self.min_content_contribution_cached(axis, tree, grid_area_size, grid_area_size);
774
775                    // If the item is a compressible replaced element, and has a definite preferred size or maximum size in the
776                    // relevant axis, the size suggestion is capped by those sizes; for this purpose, any indefinite percentages
777                    // in these sizes are resolved against zero (and considered definite).
778                    if self.is_compressible_replaced {
779                        let size = self.size.get(axis).maybe_resolve(Some(0.0), |val, basis| tree.calc(val, basis));
780                        let max_size =
781                            self.max_size.get(axis).maybe_resolve(Some(0.0), |val, basis| tree.calc(val, basis));
782                        minimum_contribution = minimum_contribution.maybe_min(size).maybe_min(max_size);
783                    }
784
785                    // The content-based minimum size is additionally clamped by the sum of any fixed max track sizing
786                    // functions of the tracks the item spans. Note that this clamp does not apply to explicitly specified
787                    // preferred or minimum sizes, and that the argument to fit-content() does not clamp the content-based
788                    // minimum size in the same way as a fixed max track sizing function. The clamp is a stretch fit into
789                    // the limit, so it does not shrink the item below its padding + border.
790                    let limit =
791                        self.spanned_fixed_track_limit(axis, axis_tracks, inner_node_size.get(axis), &|val, basis| {
792                            tree.resolve_calc_value(val, basis)
793                        });
794                    minimum_contribution.maybe_min(limit.map(|limit| f32_max(limit, padding_border_size.get(axis))))
795                } else {
796                    0.0
797                }
798            })
799    }

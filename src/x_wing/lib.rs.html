9//! # Usage
10//!
11//! This crate implements the X-Wing Key Encapsulation Method (X-Wing-KEM) algorithm.
12//! X-Wing-KEM is a KEM in the sense that it creates an (decapsulation key, encapsulation key) pair,
13//! such that anyone can use the encapsulation key to establish a shared key with the holder of the
14//! decapsulation key. X-Wing-KEM is a general-purpose hybrid post-quantum KEM, combining x25519 and ML-KEM-768.
15#![cfg_attr(feature = "getrandom", doc = "```")]
16#![cfg_attr(not(feature = "getrandom"), doc = "```ignore")]
17//! // NOTE: requires the `getrandom` feature is enabled
18//! use x_wing::{
19//!     XWingKem,
20//!     kem::{Decapsulate, Encapsulate, Kem}
21//! };
22//!
23//! let (sk, pk) = XWingKem::generate_keypair();
24//! let (ct, sk_sender) = pk.encapsulate();
25//! let sk_receiver = sk.decapsulate(&ct);
26//! assert_eq!(sk_sender, sk_receiver);
27//! ```

94/// X-Wing encapsulation or public key.
95#[derive(Clone, Debug, Eq, PartialEq)]
96pub struct EncapsulationKey {
97    pk_m: MlKem768EncapsulationKey,
98    pk_x: PublicKey,
99}
100
101impl EncapsulationKey {
102    /// Encapsulates with the given randomness. Uses the first 32 bytes for ML-KEM and the last 32
103    /// bytes for x25519. This is useful for testing against known vectors.
104    ///
105    /// # Warning
106    /// Do NOT use this function unless you know what you're doing. If you fail to use all uniform
107    /// random bytes even once, you can have catastrophic security failure.
108    #[doc(hidden)]
109    #[cfg_attr(not(feature = "hazmat"), doc(hidden))]
110    #[expect(clippy::must_use_candidate)]
111    pub fn encapsulate_deterministic(
112        &self,
113        randomness: &ArrayN<u8, ENCAPSULATION_RANDOMNESS_SIZE>,
114    ) -> (Ciphertext, SharedKey) {
115        // Split randomness into two 32-byte arrays
116        let (rand_m, rand_x) = randomness.split::<U32>();
117
118        // Encapsulate with ML-KEM first. This is infallible
119        let (ct_m, ss_m) = self.pk_m.encapsulate_deterministic(&rand_m);
120
121        let ek_x = StaticSecret::from(rand_x.0);
122        // Equal to ct_x = x25519(ek_x, BASE_POINT)
123        let ct_x = PublicKey::from(&ek_x);
124        // Equal to ss_x = x25519(ek_x, pk_x)
125        let ss_x = ek_x.diffie_hellman(&self.pk_x);
126
127        let ss = combiner(&ss_m, &ss_x, &ct_x, &self.pk_x);
128        let ct = CiphertextMessage { ct_m, ct_x };
129
130        (ct.into(), ss)
131    }
132}

134impl Encapsulate for EncapsulationKey {

137    fn encapsulate_with_rng<R>(&self, rng: &mut R) -> (Ciphertext, SharedKey)
138    where
139        R: CryptoRng + ?Sized,
140    {
141        #[allow(unused_mut)]
142        let mut randomness = Array::generate_from_rng(rng);
143        let res = self.encapsulate_deterministic(&randomness);
144
145        #[cfg(feature = "zeroize")]
146        randomness.zeroize();
147
148        res
149    }
150}

211impl DecapsulationKey {
212    fn decapsulate_inner(
213        &self,
214        ct: &CiphertextMessage,
215    ) -> (SharedKey, x25519_dalek::SharedSecret, PublicKey) {
216        let (sk_m, sk_x, _pk_m, pk_x) = expand_key(&self.sk);
217
218        let ss_m = sk_m.decapsulate(&ct.ct_m);
219
220        // equal to ss_x = x25519(sk_x, ct_x)
221        let ss_x = sk_x.diffie_hellman(&ct.ct_x);
222
223        (ss_m, ss_x, pk_x)
224    }
225}

227impl Decapsulate for DecapsulationKey {
228    #[allow(clippy::similar_names)] // So we can use the names as in the RFC
229    fn decapsulate(&self, ct: &Ciphertext) -> SharedKey {
230        let ct = CiphertextMessage::from(ct);
231        let (ss_m, ss_x, pk_x) = self.decapsulate_inner(&ct);
232
233        combiner(&ss_m, &ss_x, &ct.ct_x, &pk_x)
234    }
235}

288/// X-Wing decapsulation key or private key.
289///
290/// This decapsulation key rejects "non-contributory behaviour" in the X25519 component of
291/// X-Wing. To accept instead, use [`DecapsulationKey`].
292///
293/// # Backstory
294///
295/// [RFC 7748] defines the `X25519` function, and [specifies] that when used for ECDH
296/// (as it is inside X-Wing), the implementation **MAY** abort if the all-zero value
297/// is produced as a shared secret. X-Wing, as initially specified, used the X25519
298/// function without making any mention of this **MAY**, meaning that implementations
299/// inherited whatever behaviour their underlying X25519 ECDH implementation provided.
300///
301/// This crate initially did not check for non-contributory behaviour, which meant it
302/// was incompatible with other implementations that did (in that it would accept
303/// ciphertexts that other implementations reject).
304///
305/// [CFRG have decided] that they will pick a single behaviour for the IETF X-Wing
306/// standard. Until the corresponding RFC is published, this crate supports both
307/// behaviours: constructing a [`DecapsulationKey`] accepts non-contributory behaviour for
308/// backwards-compatibility with existing usages, and this struct can be used to instead
309/// reject non-contributory behaviour. Once the RFC is published, `DecapsulationKey` will
310/// be altered to match it.
311///
312/// [RFC 7748]: https://www.rfc-editor.org/info/rfc7748/#section-5
313/// [specifies]: https://www.rfc-editor.org/info/rfc7748/#section-6.1
314/// [CFRG have decided]: https://mailarchive.ietf.org/arch/msg/cfrg/v9fEHQj3QTUpdu72AzjyyrY4j2g/
315#[derive(Clone, Debug)]
316pub struct DecapsulationKeyRejectNonContrib(DecapsulationKey);
317
318impl TryDecapsulate for DecapsulationKeyRejectNonContrib {
319    type Error = DecapsulationError;
320
321    #[allow(clippy::similar_names)] // So we can use the names as in the RFC
322    fn try_decapsulate(&self, ct: &Ciphertext) -> Result<SharedKey, Self::Error> {
323        let ct = CiphertextMessage::from(ct);
324        let (ss_m, ss_x, pk_x) = self.0.decapsulate_inner(&ct);
325
326        if !ss_x.was_contributory() {
327            return Err(DecapsulationError);
328        }
329
330        Ok(combiner(&ss_m, &ss_x, &ct.ct_x, &pk_x))
331    }
332}

337    fn encapsulation_key(&self) -> &EncapsulationKey {
338        self.0.encapsulation_key()
339    }
340}

373fn expand_key(
374    sk: &[u8; DECAPSULATION_KEY_SIZE],
375) -> (
376    MlKem768DecapsulationKey,
377    StaticSecret,
378    MlKem768EncapsulationKey,
379    PublicKey,
380) {
381    use sha3::digest::Update;
382    let mut shaker = Shake256::default();
383    shaker.update(sk);
384    let mut expanded: Shake256Reader = shaker.finalize_xof();
385
386    let seed = read_from(&mut expanded).into();
387    let (sk_m, pk_m) = MlKem768::from_seed(&seed);
388
389    let sk_x = read_from(&mut expanded);
390    let sk_x = StaticSecret::from(sk_x);
391    let pk_x = PublicKey::from(&sk_x);
392
393    (sk_m, sk_x, pk_m, pk_x)
394}

396/// X-Wing ciphertext.
397#[derive(Clone, Debug, PartialEq, Eq)]
398pub struct CiphertextMessage {
399    ct_m: ArrayN<u8, 1088>,
400    ct_x: PublicKey,
401}
402
403impl CiphertextMessage {
404    /// Convert the ciphertext to the following format:
405    /// ML-KEM-768 ciphertext(1088 bytes) || X25519 ciphertext(32 bytes).
406    #[must_use]
407    pub fn to_bytes(&self) -> Ciphertext {
408        let mut buffer = Ciphertext::default();
409        buffer[0..1088].copy_from_slice(&self.ct_m);
410        buffer[1088..].copy_from_slice(self.ct_x.as_bytes());
411        buffer
412    }
413}
414
415impl From<&Ciphertext> for CiphertextMessage {
416    fn from(value: &Ciphertext) -> Self {
417        let mut ct_m = [0; 1088];
418        ct_m.copy_from_slice(&value[0..1088]);
419        let mut ct_x = [0; 32];
420        ct_x.copy_from_slice(&value[1088..]);
421
422        CiphertextMessage {
423            ct_m: ct_m.into(),
424            ct_x: ct_x.into(),
425        }
426    }
427}

443fn combiner(
444    ss_m: &ArrayN<u8, 32>,
445    ss_x: &x25519_dalek::SharedSecret,
446    ct_x: &PublicKey,
447    pk_x: &PublicKey,
448) -> SharedKey {
449    use sha3::Digest;
450
451    let mut hasher = Sha3_256::new();
452    hasher.update(ss_m);
453    hasher.update(ss_x);
454    hasher.update(ct_x);
455    hasher.update(pk_x.as_bytes());
456    hasher.update(X_WING_LABEL);
457    hasher.finalize()
458}

460fn read_from<const N: usize>(reader: &mut Shake256Reader) -> [u8; N] {
461    let mut data = [0; N];
462    reader.read(&mut data);
463    data
464}

489        let (sk, pk) = XWingKem::generate_keypair();
490
491        // Construct a ciphertext with non-contributory behaviour.
492        let ct = CiphertextMessage {
493            ct_m: pk.pk_m.encapsulate().0,
494            ct_x: PublicKey::from([0; 32]),

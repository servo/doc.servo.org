61pub struct MemCacheStorageEngine {
62    /// <https://w3c.github.io/ServiceWorker/#dfn-name-to-cache-map>
63    name_to_cache_map: HashMap<(ImmutableOrigin, String), RequestResponseList>,
64}
65
66impl CacheStorageEngine for MemCacheStorageEngine {
67    type Error = ();
68
69    /// <https://w3c.github.io/ServiceWorker/#cache-storage-has>
70    /// The parallel steps.
71    fn has_cache(
72        &mut self,
73        origin: ImmutableOrigin,
74        cache_name: &str,
75    ) -> Result<bool, CacheStorageError<Self::Error>> {
76        // Step 2.1:For each key → value of the relevant name to cache map:
77        // Step 2.1.1: If cacheName matches key, resolve promise with true and abort these steps.
78        // Step 2.2: Resolve promise with false.
79        // Note: promise resolved in the callback in CacheStorage.
80        Ok(self
81            .name_to_cache_map
82            .contains_key(&(origin, cache_name.to_string())))
83    }
84
85    /// <https://w3c.github.io/ServiceWorker/#cache-storage-open>
86    /// The parallel steps.
87    fn open_cache(
88        &mut self,
89        origin: ImmutableOrigin,
90        cache_name: String,
91        proxy_map: &StorageProxyMap,
92    ) -> Result<(), CacheStorageError<Self::Error>> {
93        // Step 2.1: For each key → value of the relevant name to cache map:
94        // Step 2.1.1: If cacheName matches key, then:
95        // Resolve promise with a new Cache object that represents value.
96        // Note: promise resolved in script.
97        if self
98            .name_to_cache_map
99            .contains_key(&(origin.clone(), cache_name.clone()))
100        {
101            // Step 2.1.2: Abort these steps.
102            return Ok(());
103        }
104
105        // Step 2.2: Let cache be a new request response list.
106        let cache = RequestResponseList { list: Vec::new() };
107
108        // Step 2.3: Set the relevant name to cache map[cacheName] to cache.
109        // If this cache write operation failed due to exceeding the granted quota limit,
110        // reject promise with a QuotaExceededError and abort these steps.
111        // Note: there are no quota checks storage side at this point.
112        let Ok(response) = proxy_map
113            .handle
114            .create_database(proxy_map.bottle_id, cache_name.clone())
115            .recv()
116        else {
117            return Err(CacheStorageError::Internal(()));
118        };
119        if response.is_err() {
120            return Err(CacheStorageError::Internal(()));
121        }
122        self.name_to_cache_map.insert((origin, cache_name), cache);
123
124        // Step 2.4: Resolve promise with a new Cache object that represents cache.
125        // Note: promise resolved in script.
126        Ok(())
127    }
128
129    /// <https://w3c.github.io/ServiceWorker/#cache-keys>
130    /// The parallel steps.
131    fn keys(
132        &mut self,
133        origin: ImmutableOrigin,
134        cache_name: &str,
135    ) -> Result<Vec<String>, CacheStorageError<Self::Error>> {
136        // Step 5.1: Let requests be an empty list.
137        let mut requests: Vec<String> = Vec::new();
138
139        // Step 5.2: If the optional argument request is omitted, then:
140        // Step 5.2.1: For each requestResponse of the relevant request response list:
141        let Some(relevant_cache) = self
142            .name_to_cache_map
143            .get(&(origin, cache_name.to_string()))
144        else {
145            return Err(CacheStorageError::Internal(()));
146        };
147        // Step 5.2.2: Add requestResponse’s request to requests.
148        for (request, _response) in &relevant_cache.list {
149            requests.push(request.url().to_string());
150        }
151
152        // Step 5.3: Else:
153        // Note: implementing this steps depends on Query Cache; todo.
154
155        Ok(requests)
156    }
157
158    /// <https://w3c.github.io/ServiceWorker/#dom-cachestorage-delete>
159    /// The "running the algorithm specified in has" part, and the parallel steps.
160    fn delete_cache(
161        &mut self,
162        origin: ImmutableOrigin,
163        cache_name: &str,
164        proxy: &StorageProxyMap,
165    ) -> Result<bool, CacheStorageError<Self::Error>> {
166        // Step 1: Let promise be the result of running the algorithm specified in has(cacheName) method with cacheName.
167        // Step 2: Return the result of reacting to promise with a fulfillment handler that,
168        // when called with argument cacheExists,
169        // performs the following substeps:
170        // Note: skipping th promise here.
171
172        // Step 2.1: If cacheExists is false, then:
173        // Step 2.1.1: Resolve promise with false.
174        // Note: done below using return value from `is_some`.
175
176        // Step 2.3: Run the following substeps in parallel:
177        // Step 2.3.1: Remove the relevant name to cache map[cacheName].
178        let has = self
179            .name_to_cache_map
180            .remove(&(origin, cache_name.to_string()))
181            .is_some();
182
183        if has {
184            let Ok(response) = proxy
185                .handle
186                .delete_database(proxy.bottle_id, cache_name.to_string())
187                .recv()
188            else {
189                return Err(CacheStorageError::Internal(()));
190            };
191            if response.is_err() {
192                return Err(CacheStorageError::Internal(()));
193            }
194        }
195
196        // Step 2.3.2: Resolve promise with true.
197        // Note: promise resolved in script.
198        Ok(has)
199    }
200}

202pub trait CacheStorageThreadFactory {
203    fn new(config_dir: Option<PathBuf>, temporary_storage: bool) -> Self;
204}
205
206impl CacheStorageThreadFactory for CacheStorageThreadHandle {
207    fn new(config_dir: Option<PathBuf>, temporary_storage: bool) -> CacheStorageThreadHandle {
208        let (generic_sender, generic_receiver) = generic_channel::channel().unwrap();
209        let mut temp_dir: Option<tempfile::TempDir> = None;
210        let base_dir = config_dir
211            .unwrap_or_else(|| {
212                let tmp_dir = tempfile::tempdir().unwrap();
213                let path = tmp_dir.path().to_path_buf();
214                temp_dir = Some(tmp_dir);
215                path
216            })
217            .join("cachestorage");
218        let storage_dir = if temporary_storage {
219            let unique_id = uuid::Uuid::new_v4().to_string();
220            base_dir.join("temporary").join(unique_id)
221        } else {
222            base_dir.join("default_v1")
223        };
224        std::fs::create_dir_all(&storage_dir)
225            .expect("Failed to create CacheStorage storage directory");
226        let sender_clone = generic_sender.clone();
227        thread::Builder::new()
228            .name("CacheStorageThread".to_owned())
229            .spawn(move || {
230                // Keep temp_dir alive while the thread runs.
231                let _ = temp_dir;
232                let engine = MemCacheStorageEngine {
233                    name_to_cache_map: Default::default(),
234                };
235                let mut cache_storage_thread =
236                    CacheStorageThread::new(sender_clone, generic_receiver, engine);
237                cache_storage_thread.start();
238            })
239            .expect("Thread spawning failed");
240
241        CacheStorageThreadHandle::new(generic_sender)
242    }
243}

245struct CacheStorageThread<E: CacheStorageEngine> {
246    receiver: GenericReceiver<CacheStorageThreadMessage>,
247    // Note: a sender to self might be required later for the storage engine.
248    _sender: GenericSender<CacheStorageThreadMessage>,
249    engine: E,
250}
251
252impl<E> CacheStorageThread<E>
253where
254    E: CacheStorageEngine,
255{
256    pub fn new(
257        _sender: GenericSender<CacheStorageThreadMessage>,
258        receiver: GenericReceiver<CacheStorageThreadMessage>,
259        engine: E,
260    ) -> CacheStorageThread<E> {
261        CacheStorageThread {
262            _sender,
263            receiver,
264            engine,
265        }
266    }
267
268    pub fn start(&mut self) {
269        while let Ok(message) = self.receiver.recv() {
270            match message {
271                CacheStorageThreadMessage::HasCache {
272                    cache_name,
273                    callback,
274                    proxy: _,
275                    origin,
276                } => {
277                    let result = self.engine.has_cache(origin.clone(), &cache_name);
278                    if callback
279                        .send(CacheStorageThreadResponse::HasCacheResult(
280                            result.map_err(|e| format!("{:?}", e)),
281                        ))
282                        .is_err()
283                    {
284                        error!("Failed to send response to script for HasCache message.");
285                    }
286                },
287                CacheStorageThreadMessage::OpenCache {
288                    cache_name,
289                    callback,
290                    proxy,
291                    origin,
292                } => {
293                    let result = self
294                        .engine
295                        .open_cache(origin.clone(), cache_name.clone(), &proxy);
296                    if callback
297                        .send(CacheStorageThreadResponse::OpenCacheResult {
298                            result: result.map_err(|e| format!("{:?}", e)),
299                            cache_name,
300                        })
301                        .is_err()
302                    {
303                        error!("Failed to send response to script for OpenCache message.");
304                    }
305                },
306                CacheStorageThreadMessage::Keys {
307                    cache_name,
308                    callback,
309                    origin,
310                } => {
311                    let result = self.engine.keys(origin.clone(), &cache_name);
312                    if callback
313                        .send(CacheStorageThreadResponse::KeysResult(
314                            result.map_err(|e| format!("{:?}", e)),
315                        ))
316                        .is_err()
317                    {
318                        error!("Failed to send response to script for Keys message.");
319                    }
320                },
321                CacheStorageThreadMessage::DeleteCache {
322                    cache_name,
323                    callback,
324                    proxy,
325                    origin,
326                } => {
327                    let result = self
328                        .engine
329                        .delete_cache(origin.clone(), &cache_name, &proxy);
330                    if callback
331                        .send(CacheStorageThreadResponse::DeleteCacheResult(
332                            result.map_err(|e| format!("{:?}", e)),
333                        ))
334                        .is_err()
335                    {
336                        error!("Failed to send response to script for DeleteCache message.");
337                    }
338                },
339                CacheStorageThreadMessage::Exit(sender) => {
340                    let _ = sender.send(());
341                    break;
342                },
343            }
344        }
345    }

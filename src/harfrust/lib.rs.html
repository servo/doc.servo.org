114bitflags::bitflags! {
115    /// Flags that control how a buffer is shaped.
116    #[derive(Default, Debug, Clone, Copy)]
117    pub struct BufferFlags: u32 {
118        /// Treat the buffer as the beginning of a paragraph.
119        ///
120        /// Set this when the buffer includes the beginning of the text, rather
121        /// than a segment shaped with surrounding context.
122        const BEGINNING_OF_TEXT             = 0x0000_0001;
123        /// Treat the buffer as the end of a paragraph.
124        ///
125        /// See [`BufferFlags::BEGINNING_OF_TEXT`].
126        const END_OF_TEXT                   = 0x0000_0002;
127        /// Keep glyphs for default-ignorable Unicode characters.
128        ///
129        /// Without this flag, shaping hides them by substituting a space glyph
130        /// with zero advance. This takes precedence over
131        /// [`BufferFlags::REMOVE_DEFAULT_IGNORABLES`].
132        const PRESERVE_DEFAULT_IGNORABLES   = 0x0000_0004;
133        /// Remove default-ignorable Unicode characters from the glyph output.
134        ///
135        /// [`BufferFlags::PRESERVE_DEFAULT_IGNORABLES`] takes precedence.
136        const REMOVE_DEFAULT_IGNORABLES     = 0x0000_0008;
137        /// Suppress dotted circles for broken character sequences.
138        const DO_NOT_INSERT_DOTTED_CIRCLE   = 0x0000_0010;
139        /// Reserved for HarfBuzz-compatible shaping verification.
140        ///
141        /// Verification is not currently implemented by this crate.
142        const VERIFY                        = 0x0000_0020;
143        /// Produce [`GlyphFlags::UNSAFE_TO_CONCAT`] on output glyphs.
144        ///
145        /// Disabled by default because it adds shaping work.
146        const PRODUCE_UNSAFE_TO_CONCAT      = 0x0000_0040;
147        /// Produce [`GlyphFlags::SAFE_TO_INSERT_TATWEEL`] on output glyphs.
148        const PRODUCE_SAFE_TO_INSERT_TATWEEL      = 0x0000_0080;
149        /// All currently defined flags.
150        const DEFINED = 0x0000_00FF;
151    }
152}

154/// How input text clusters are assigned to output glyphs.
155///
156/// Cluster values relate glyphs to positions in the input. Monotone levels
157/// preserve cluster order, which is useful for finding line break positions.
158#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
159pub enum ClusterLevel {
160    /// Keep grapheme clusters together and preserve cluster order.
161    MonotoneGraphemes,
162    /// Preserve cluster order, allowing characters to keep separate clusters.
163    MonotoneCharacters,
164    /// Allow separate character clusters without preserving cluster order.
165    Characters,
166    /// Keep grapheme clusters together without preserving cluster order.
167    Graphemes,
168}
169
170impl ClusterLevel {
171    #[inline]
172    fn new(level: u32) -> Self {
173        match level {
174            0 => Self::MonotoneGraphemes,
175            1 => Self::MonotoneCharacters,
176            2 => Self::Characters,
177            3 => Self::Graphemes,
178            _ => Self::MonotoneGraphemes,
179        }
180    }
181    #[inline]
182    fn is_monotone(self) -> bool {
183        matches!(self, Self::MonotoneGraphemes | Self::MonotoneCharacters)
184    }
185    #[inline]
186    fn is_graphemes(self) -> bool {
187        matches!(self, Self::MonotoneGraphemes | Self::Graphemes)
188    }
189    #[inline]
190    fn _is_characters(self) -> bool {
191        matches!(self, Self::MonotoneCharacters | Self::Characters)
192    }
193}
194
195impl Default for ClusterLevel {
196    #[inline]
197    fn default() -> Self {
198        ClusterLevel::MonotoneGraphemes
199    }
200}

205    pub struct SerializeFlags: u8 {
206        /// Omit glyph cluster values.
207        const NO_CLUSTERS       = 0b0000_0001;
208        /// Omit glyph position information.
209        const NO_POSITIONS      = 0b0000_0010;
210        /// Serialize glyph IDs instead of glyph names.
211        const NO_GLYPH_NAMES    = 0b0000_0100;
212        /// Serialize glyph extents.
213        const GLYPH_EXTENTS     = 0b0000_1000;
214        /// Serialize glyph flags.
215        const GLYPH_FLAGS       = 0b0001_0000;
216        /// Omit advances and report offsets as absolute glyph positions.
217        const NO_ADVANCES       = 0b0010_0000;
218        /// All currently defined flags.
219        const DEFINED = 0b0011_1111;
220    }

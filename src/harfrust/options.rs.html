1use crate::{Feature, ShapePlan};

3/// Optional settings for a shaping operation.
4///
5/// By default, shaping creates a plan, applies the font's default features,
6/// and has no point size for the `trak` table.
7#[derive(Default)]
8pub struct ShapeOptions<'a> {
9    pub(crate) plan: Option<&'a ShapePlan>,
10    pub(crate) point_size: Option<f32>,
11    pub(crate) features: &'a [Feature],
12}
13
14impl<'a> ShapeOptions<'a> {
15    /// Creates a default set of shape options ready for configuration.
16    pub fn new() -> Self {
17        Self::default()
18    }
19
20    /// Sets the plan to use for shaping.
21    ///
22    /// The shape plan must be compatible with the properties of the buffer
23    /// passed to shaping.
24    pub fn plan(mut self, plan: Option<&'a ShapePlan>) -> Self {
25        self.plan = plan;
26        self
27    }
28
29    /// Sets the point size used by the `trak` tracking table.
30    pub fn point_size(mut self, point_size: Option<f32>) -> Self {
31        self.point_size = point_size;
32        self
33    }
34
35    /// Sets user features to apply during shaping.
36    pub fn features(mut self, features: &'a [Feature]) -> Self {
37        self.features = features;
38        self
39    }
40}

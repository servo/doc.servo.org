22pub(crate) fn get_class<T: bytemuck::AnyBitPattern + FixedSize>(
23    machine: &ExtendedStateTable<'_, T>,
24    glyph_id: GlyphId,
25    cache: &ClassCache,
26) -> u16 {
27    if let Some(klass) = cache.get(glyph_id.to_u32()) {
28        return klass as u16;
29    }
30    let klass = machine
31        .class(glyph_id)
32        .unwrap_or(class::OUT_OF_BOUNDS as u16);
33    cache.set(glyph_id.to_u32(), klass as u32);
34    klass
35}

133    pub(crate) fn buffer_intersects_machine(&self) -> bool {
134        if let Some(first_set) = &self.first_set {
135            if self.using_buffer_digest && !self.buffer.digest.may_intersect(&self.first_digest) {
136                return false;
137            }
138            if self.using_buffer_glyph_set {
139                return self.buffer.glyph_set.intersects_set(first_set);
140            }
141            for info in &self.buffer.info {
142                if first_set.contains(info.glyph_id) {
143                    return true;
144                }
145            }
146            false
147        } else {
148            true
149        }
150    }

152    pub fn output_glyph(&mut self, glyph: u32) -> Option<()> {
153        self.buffer.digest.add(glyph);
154        if self.using_buffer_glyph_set {
155            self.buffer.glyph_set.insert(glyph);
156        }
157
158        // Insertion state machines can emit glyphs during the end-of-text
159        // transition, when there is no current input glyph. output_glyph()
160        // supports that by copying the previous output glyph's metadata.
161        // Apply the AAT metadata to that new output glyph instead of indexing
162        // past the end of the input buffer.
163        let at_end = self.buffer.idx == self.buffer.len;
164        if at_end {
165            let out_len = self.buffer.out_len;
166            self.buffer.output_glyph(glyph);
167            if self.buffer.out_len == out_len {
168                return Some(());
169            }
170        }
171
172        if glyph == DELETED_GLYPH {
173            self.buffer.scratch_flags |= HB_BUFFER_SCRATCH_FLAG_AAT_HAS_DELETED;
174            let info = if at_end {
175                self.buffer.prev_mut()?
176            } else {
177                self.buffer.cur_mut(0)?
178            };
179            info.set_aat_deleted();
180        } else {
181            if self.has_glyph_classes {
182                let glyph_props = self.layout.ot.glyph_props(glyph.into());
183                let info = if at_end {
184                    self.buffer.prev_mut()?
185                } else {
186                    self.buffer.cur_mut(0)?
187                };
188                info.set_glyph_props(glyph_props);
189            }
190        }
191        if !at_end {
192            self.buffer.output_glyph(glyph);
193        }
194        Some(())
195    }
196
197    pub fn replace_glyph(&mut self, glyph: u32) -> Option<()> {
198        if glyph == DELETED_GLYPH {
199            self.buffer.scratch_flags |= HB_BUFFER_SCRATCH_FLAG_AAT_HAS_DELETED;
200            self.buffer.cur_mut(0)?.set_aat_deleted();
201        }
202
203        self.buffer.digest.add(glyph);
204        if self.using_buffer_glyph_set {
205            self.buffer.glyph_set.insert(glyph);
206        }
207        if self.has_glyph_classes {
208            self.buffer
209                .cur_mut(0)?
210                .set_glyph_props(self.layout.ot.glyph_props(glyph.into()));
211        }
212        self.buffer.replace_glyph(glyph);
213        Some(())
214    }
215
216    pub fn delete_glyph(&mut self) -> Option<()> {
217        self.buffer.scratch_flags |= HB_BUFFER_SCRATCH_FLAG_AAT_HAS_DELETED;
218        self.buffer.cur_mut(0)?.set_aat_deleted();
219        self.buffer.digest.add(DELETED_GLYPH);
220        if self.using_buffer_glyph_set {
221            self.buffer.glyph_set.insert(DELETED_GLYPH);
222        }
223        self.buffer.replace_glyph(DELETED_GLYPH);
224        Some(())
225    }
226
227    #[inline(always)]
228    pub fn replace_glyph_inplace(&mut self, i: usize, glyph: u32) -> Option<()> {
229        let info = self.buffer.info.get_mut(i)?;
230        info.glyph_id = glyph;
231        if glyph == DELETED_GLYPH {
232            self.buffer.scratch_flags |= HB_BUFFER_SCRATCH_FLAG_AAT_HAS_DELETED;
233            info.set_aat_deleted();
234        }
235        self.buffer.digest.add(glyph);
236        if self.using_buffer_glyph_set {
237            self.buffer.glyph_set.insert(glyph);
238        }
239        if self.has_glyph_classes {
240            info.set_glyph_props(self.layout.ot.glyph_props(glyph.into()));
241        }
242        Some(())
243    }

258/// Per-table acceleration for the drive loops' safe-to-break computation.
259///
260/// The data for every state-machine subtable in one AAT table is stored in
261/// two packed vectors. Each top-level subtable cache carries a compact
262/// descriptor for its ranges. This avoids two `Vec` allocations in every
263/// subtable cache while retaining O(classes) + O(states) storage.
264#[derive(Default)]
265pub(crate) struct SafeToBreakAccel {
266    wouldbe: Vec<u16>,
267    eot_tail: Vec<u64>,
268}
269
270#[derive(Clone, Copy, Default)]
271pub(crate) struct SafeToBreakSubtable {
272    wouldbe_start: u32,
273    wouldbe_end: u32,
274    eot_tail_start: u32,
275    eot_tail_end: u32,
276}
277
278/// The slice of [`SafeToBreakAccel`] belonging to the active subtable.
279#[derive(Clone, Copy, Default)]
280pub(crate) struct SafeToBreak<'a> {
281    n_classes: u32,
282    /// Per class: `entry(START_OF_TEXT, class)` packed as 15 bits of
283    /// new state with the advance bit on top when present and
284    /// non-actionable, `!0` otherwise — so condition 2c is a single
285    /// compare. Real machines' state counts fit 15 bits.
286    wouldbe: &'a [u16],
287    /// Condition-3 bits for states 64 and up; states below 64 keep
288    /// using the `start_end_safe_to_break` word. Empty for machines
289    /// with at most 64 states.
290    eot_tail: &'a [u64],
291}
292
293pub(crate) const WOULDBE_NONE: u16 = !0;
294
295#[inline(always)]
296pub(crate) fn pack_wouldbe(new_state: u16, advance: bool) -> u16 {
297    // States fit 15 bits in any real font. A pathological new state
298    // that doesn't can alias the sentinel and misreport condition 2c,
299    // which only perturbs unsafe-to-break flags -- not worth runtime
300    // checks on this path.
301    new_state | ((advance as u16) << 15)
302}
303
304impl SafeToBreak<'_> {
305    /// Condition 2c: would starting from the start state on this class
306    /// take a non-actionable transition to the same state, advancing
307    /// the same way?
308    #[inline(always)]
309    pub(crate) fn wouldbe_matches(&self, class: u16, next_state: u16, advance: bool) -> bool {
310        let mut class = class as usize;
311        if class >= self.n_classes as usize {
312            class = class::OUT_OF_BOUNDS as usize;
313        }
314        self.wouldbe.get(class).copied() == Some(pack_wouldbe(next_state, advance))
315    }
316
317    /// Condition 3 for states 64 and up: no end-of-text action can fire
318    /// out of this state.
319    #[inline(always)]
320    pub(crate) fn eot_safe_high(&self, state: u16) -> bool {
321        let ix = (state - 64) as usize;
322        self.eot_tail
323            .get(ix / 64)
324            .is_some_and(|word| word & (1 << (ix % 64)) != 0)
325    }
326}

328impl SafeToBreakAccel {
329    /// Returns a view of the acceleration data described by a top-level
330    /// subtable cache.
331    pub(crate) fn subtable(&self, subtable: SafeToBreakSubtable) -> Option<SafeToBreak<'_>> {
332        let wouldbe_start = subtable.wouldbe_start as usize;
333        let wouldbe_end = subtable.wouldbe_end as usize;
334        let eot_tail_start = subtable.eot_tail_start as usize;
335        let eot_tail_end = subtable.eot_tail_end as usize;
336
337        Some(SafeToBreak {
338            n_classes: subtable.wouldbe_end - subtable.wouldbe_start,
339            wouldbe: self.wouldbe.get(wouldbe_start..wouldbe_end)?,
340            eot_tail: self.eot_tail.get(eot_tail_start..eot_tail_end)?,
341        })
342    }
343
344    /// Describes a subtable that does not contain a state machine.
345    pub(crate) fn empty_subtable(&self) -> SafeToBreakSubtable {
346        SafeToBreakSubtable {
347            wouldbe_start: self.wouldbe.len() as u32,
348            wouldbe_end: self.wouldbe.len() as u32,
349            eot_tail_start: self.eot_tail.len() as u32,
350            eot_tail_end: self.eot_tail.len() as u32,
351        }
352    }
353
354    /// Appends the data for an extended state table whose machine
355    /// starts at the beginning of `data`. The predicates mirror the
356    /// subtable kind's notion of actionable/advancing entries.
357    #[inline(never)]
358    pub(crate) fn build_extended<T: bytemuck::AnyBitPattern + FixedSize>(
359        &mut self,
360        machine: &ExtendedStateTable<T>,
361        data: &[u8],
362        is_actionable: &dyn Fn(&StateEntry<T>) -> bool,
363        can_advance: &dyn Fn(&StateEntry<T>) -> bool,
364    ) -> SafeToBreakSubtable {
365        let n_classes = machine.n_classes;
366        let wouldbe_start = self.wouldbe.len();
367        let eot_tail_start = self.eot_tail.len();
368
369        // Cover the OUT_OF_BOUNDS column the runtime clamp can select
370        // even on degenerate machines; entry() applies the same clamp
371        // internally, so the aliasing matches.
372        // Classes are u16 values, so columns past 0x10000 are
373        // unreachable and need no slots.
374        let wouldbe_len = n_classes
375            .max(class::OUT_OF_BOUNDS as usize + 1)
376            .min(1 << 16);
377        self.wouldbe.reserve(wouldbe_len);
378        for class in 0..wouldbe_len {
379            self.wouldbe
380                .push(match machine.entry(START_OF_TEXT, class as u16) {
381                    Ok(entry) if !is_actionable(&entry) => {
382                        pack_wouldbe(entry.new_state, can_advance(&entry))
383                    }
384                    _ => WOULDBE_NONE,
385                });
386        }
387
388        // The state array runs from its offset to the end of the
389        // subtable, matching how the table reader slices it; states
390        // beyond it fail entry() and stay unsafe, like the probes they
391        // replace.
392        if n_classes > 0 {
393            if let Ok(parts) = StateTableParts::read(FontData::new(data)) {
394                let n_cells = data.len().saturating_sub(parts.state_array_offset as usize)
395                    / u16::RAW_BYTE_LEN;
396                let n_rows = n_cells.div_ceil(n_classes).min(u16::MAX as usize + 1);
397                if n_rows > 64 {
398                    let eot_tail_start = self.eot_tail.len();
399                    self.eot_tail
400                        .resize(eot_tail_start + (n_rows - 64).div_ceil(64), 0);
401                    for state in 64..n_rows {
402                        if let Ok(entry) =
403                            machine.entry(state as u16, u16::from(class::END_OF_TEXT))
404                        {
405                            if !is_actionable(&entry) {
406                                let ix = state - 64;
407                                self.eot_tail[eot_tail_start + ix / 64] |= 1 << (ix % 64);
408                            }
409                        }
410                    }
411                }
412            }
413        }
414
415        SafeToBreakSubtable::new(
416            wouldbe_len,
417            wouldbe_start,
418            self.wouldbe.len(),
419            eot_tail_start,
420            self.eot_tail.len(),
421        )
422    }
423
424    /// The legacy (`kern` Format1) counterpart of [`Self::build_extended`]:
425    /// one-byte state cells and u8 classes.
426    #[inline(never)]
427    pub(crate) fn build_legacy<T: bytemuck::AnyBitPattern + FixedSize>(
428        &mut self,
429        machine: &StateTable<T>,
430        is_actionable: &dyn Fn(&StateEntry<T>) -> bool,
431        can_advance: &dyn Fn(&StateEntry<T>) -> bool,
432    ) -> SafeToBreakSubtable {
433        let n_classes = machine.header.state_size() as usize;
434        let wouldbe_start = self.wouldbe.len();
435        let eot_tail_start = self.eot_tail.len();
436
437        let wouldbe_len = n_classes.max(class::OUT_OF_BOUNDS as usize + 1).min(1 << 8);
438        self.wouldbe.reserve(wouldbe_len);
439        for class in 0..wouldbe_len {
440            self.wouldbe
441                .push(match machine.entry(START_OF_TEXT, class as u8) {
442                    Ok(entry) if !is_actionable(&entry) => {
443                        pack_wouldbe(entry.new_state, can_advance(&entry))
444                    }
445                    _ => WOULDBE_NONE,
446                });
447        }
448
449        if n_classes > 0 {
450            if let Ok(state_array) = machine.header.state_array() {
451                let n_cells = state_array.data().len();
452                let n_rows = n_cells.div_ceil(n_classes).min(u16::MAX as usize + 1);
453                if n_rows > 64 {
454                    let eot_tail_start = self.eot_tail.len();
455                    self.eot_tail
456                        .resize(eot_tail_start + (n_rows - 64).div_ceil(64), 0);
457                    for state in 64..n_rows {
458                        if let Ok(entry) = machine.entry(state as u16, class::END_OF_TEXT) {
459                            if !is_actionable(&entry) {
460                                let ix = state - 64;
461                                self.eot_tail[eot_tail_start + ix / 64] |= 1 << (ix % 64);
462                            }
463                        }
464                    }
465                }
466            }
467        }
468
469        SafeToBreakSubtable::new(
470            wouldbe_len,
471            wouldbe_start,
472            self.wouldbe.len(),
473            eot_tail_start,
474            self.eot_tail.len(),
475        )
476    }
477}

760    #[test]
761    fn safe_to_break_subtable_views_are_packed_and_bounded() {
762        assert_eq!(size_of::<SafeToBreakSubtable>(), 16);
763
764        let mut wouldbe = alloc::vec![WOULDBE_NONE; 8];
765        wouldbe[class::OUT_OF_BOUNDS as usize] = pack_wouldbe(7, true);
766        wouldbe[4 + class::OUT_OF_BOUNDS as usize] = pack_wouldbe(9, false);
767        let first_subtable = SafeToBreakSubtable::new(4, 0, 4, 0, 1);
768        let empty_subtable = SafeToBreakSubtable {
769            wouldbe_start: 4,
770            wouldbe_end: 4,
771            eot_tail_start: 1,
772            eot_tail_end: 1,
773        };
774        let last_subtable = SafeToBreakSubtable::new(4, 4, 8, 1, 2);
775        let accel = SafeToBreakAccel {
776            wouldbe,
777            eot_tail: alloc::vec![1, 2],
778        };
779
780        let first = accel.subtable(first_subtable).unwrap();
781        assert!(first.wouldbe_matches(99, 7, true));
782        assert!(first.eot_safe_high(64));
783        assert!(!first.eot_safe_high(65));
784        assert!(!first.eot_safe_high(128));
785
786        let empty = accel.subtable(empty_subtable).unwrap();
787        assert_eq!(empty.wouldbe, []);
788        assert_eq!(empty.eot_tail, []);
789
790        let last = accel.subtable(last_subtable).unwrap();
791        assert!(last.wouldbe_matches(99, 9, false));
792        assert!(!last.eot_safe_high(64));
793        assert!(last.eot_safe_high(65));
794    }

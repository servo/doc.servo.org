4// In harfbuzz, despite having `Script`, script can actually have any tag.
5// So we're doing the same.
6// The only difference is that `Script` cannot be set to `HB_SCRIPT_INVALID`.
7/// A text script identified by an ISO 15924 tag.
8#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Debug)]
9pub struct Script(pub(crate) Tag);
10
11impl Script {
12    #[inline]
13    pub(crate) const fn from_bytes(bytes: &[u8; 4]) -> Self {
14        Script(Tag::new(bytes))
15    }
16
17    /// Converts an ISO 15924 script tag to a corresponding `Script`.
18    pub const fn from_iso15924_tag(tag: Tag) -> Option<Script> {
19        let tag = u32::from_be_bytes(tag.to_be_bytes());
20
21        if tag == 0 {
22            return None;
23        }
24
25        // Be lenient, adjust case (one capital letter followed by three small letters).
26        let tag = (tag & 0xDFDF_DFDF) | 0x0020_2020;
27
28        if tag & 0xE0E0_E0E0 != 0x4060_6060 {
29            return Some(Script::UNKNOWN);
30        }
31
32        Some(match &tag.to_be_bytes() {
33            // Script variants from https://unicode.org/iso15924/
34            b"Aran" => Script::ARABIC,
35            b"Cyrs" => Script::CYRILLIC,
36            b"Geok" => Script::GEORGIAN,
37            b"Hans" | b"Hant" => Script::HAN,
38            b"Jamo" => Script::HANGUL,
39            b"Latf" | b"Latg" => Script::LATIN,
40            b"Syre" | b"Syrj" | b"Syrn" => Script::SYRIAC,
41            &t => Script(Tag::from_be_bytes(t)),
42        })
43    }
44
45    /// Returns script's tag.
46    #[inline]
47    pub fn tag(&self) -> Tag {
48        self.0
49    }
50}
51
52impl FromStr for Script {
53    type Err = &'static str;
54
55    fn from_str(s: &str) -> Result<Self, Self::Err> {
56        let tag = Tag::from_bytes_lossy(s.as_bytes());
57        Script::from_iso15924_tag(tag).ok_or("invalid script")
58    }
59}

21pub(crate) type Codepoint = u32;

23/// One step of canonical decomposition.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25pub enum Decomposed {
26    /// A single resulting code point.
27    Singleton(u32),
28    /// Two resulting code points.
29    Pair(u32, u32),
30}
31
32/// Compose two code points, if they have a canonical composition.
33#[inline]
34pub fn compose(a: u32, b: u32) -> Option<u32> {
35    compose_impl(a, b)
36}
37
38/// Return one step of canonical decomposition, if one exists.
39#[inline]
40pub fn decompose(codepoint: u32) -> Option<Decomposed> {
41    decompose_impl(codepoint).map(|(a, b)| {
42        if b == 0 {
43            Decomposed::Singleton(a)
44        } else {
45            Decomposed::Pair(a, b)
46        }
47    })
48}
49
50/// The canonical combining class of a code point.
51#[inline]
52pub fn combining_class(codepoint: u32) -> u8 {
53    combining_class_for(codepoint)
54}
55
56/// The Unicode script of a code point, using the same data as shaping.
57///
58/// Values outside the Unicode range have the unknown script.
59#[inline]
60pub fn script(codepoint: u32) -> Script {
61    if codepoint > 0x10_FFFF {
62        Script::UNKNOWN
63    } else {
64        script_for(codepoint)
65    }
66}

157impl GeneralCategory {
158    pub fn is_mark(&self) -> bool {
159        matches!(
160            *self,
161            Self::SPACING_MARK | Self::ENCLOSING_MARK | Self::NON_SPACING_MARK
162        )
163    }
164
165    pub fn is_letter(&self) -> bool {
166        matches!(
167            *self,
168            Self::LOWERCASE_LETTER
169                | Self::MODIFIER_LETTER
170                | Self::OTHER_LETTER
171                | Self::TITLECASE_LETTER
172                | Self::UPPERCASE_LETTER
173        )
174    }
175
176    #[inline(always)]
177    pub(crate) const fn flag(self) -> u32 {
178        1 << self.0
179    }
180
181    #[inline(always)]
182    pub(crate) const fn flag_unsafe(self) -> u32 {
183        if self.0 < 32 {
184            1 << self.0
185        } else {
186            0
187        }
188    }
189
190    #[inline(always)]
191    pub(crate) const fn in_range_inclusive(self, start: Self, end: Self) -> bool {
192        debug_assert!(start.0 < end.0);
193        let range_mask = (1 << (end.0 + 1)) - (1 << start.0);
194        self.flag_unsafe() & range_mask != 0
195    }
196}

497    fn space_fallback(self) -> space_fallback::SpaceCode {
498        use space_fallback::*;
499
500        // All GC=Zs chars that can use a fallback.
501        match self {
502            0x0020 => SPACE,             // SPACE
503            0x00A0 => SPACE,             // NO-BREAK SPACE
504            0x2000 => SPACE_EM_2,        // EN QUAD
505            0x2001 => SPACE_EM,          // EM QUAD
506            0x2002 => SPACE_EM_2,        // EN SPACE
507            0x2003 => SPACE_EM,          // EM SPACE
508            0x2004 => SPACE_EM_3,        // THREE-PER-EM SPACE
509            0x2005 => SPACE_EM_4,        // FOUR-PER-EM SPACE
510            0x2006 => SPACE_EM_6,        // SIX-PER-EM SPACE
511            0x2007 => SPACE_FIGURE,      // FIGURE SPACE
512            0x2008 => SPACE_PUNCTUATION, // PUNCTUATION SPACE
513            0x2009 => SPACE_EM_5,        // THIN SPACE
514            0x200A => SPACE_EM_16,       // HAIR SPACE
515            0x202F => SPACE_NARROW,      // NARROW NO-BREAK SPACE
516            0x205F => SPACE_4_EM_18,     // MEDIUM MATHEMATICAL SPACE
517            0x3000 => SPACE_EM,          // IDEOGRAPHIC SPACE
518            _ => NOT_SPACE,              // OGHAM SPACE MARK
519        }
520    }

522    fn modified_combining_class(self) -> u8 {
523        let u = self;
524
525        // Reorder SAKOT to ensure it comes after any tone marks.
526        if u == 0x1A60 {
527            return 254;
528        }
529
530        // Reorder PADMA to ensure it comes after any vowel marks.
531        if u == 0x0FC6 {
532            return 254;
533        }
534
535        // Reorder TSA -PHRU to reorder before U+0F74
536        if u == 0x0F39 {
537            return 127;
538        }
539
540        let k = u.combining_class();
541
542        MODIFIED_COMBINING_CLASS[k as usize]
543    }

549    /// Default_Ignorable codepoints:
550    ///
551    /// Note: While U+115F, U+1160, U+3164 and U+FFA0 are Default_Ignorable,
552    /// we do NOT want to hide them, as the way Uniscribe has implemented them
553    /// is with regular spacing glyphs, and that's the way fonts are made to work.
554    /// As such, we make exceptions for those four.
555    /// Also ignoring U+1BCA0..1BCA3. https://github.com/harfbuzz/harfbuzz/issues/503
556    ///
557    /// Unicode 14.0:
558    /// $ grep '; Default_Ignorable_Code_Point ' DerivedCoreProperties.txt | sed 's/;.*#/#/'
559    /// 00AD          # Cf       SOFT HYPHEN
560    /// 034F          # Mn       COMBINING GRAPHEME JOINER
561    /// 061C          # Cf       ARABIC LETTER MARK
562    /// 115F..1160    # Lo   [2] HANGUL CHOSEONG FILLER..HANGUL JUNGSEONG FILLER
563    /// 17B4..17B5    # Mn   [2] KHMER VOWEL INHERENT AQ..KHMER VOWEL INHERENT AA
564    /// 180B..180D    # Mn   [3] MONGOLIAN FREE VARIATION SELECTOR ONE..MONGOLIAN FREE VARIATION SELECTOR THREE
565    /// 180E          # Cf       MONGOLIAN VOWEL SEPARATOR
566    /// 180F          # Mn       MONGOLIAN FREE VARIATION SELECTOR FOUR
567    /// 200B..200F    # Cf   [5] ZERO WIDTH SPACE..RIGHT-TO-LEFT MARK
568    /// 202A..202E    # Cf   [5] LEFT-TO-RIGHT EMBEDDING..RIGHT-TO-LEFT OVERRIDE
569    /// 2060..2064    # Cf   [5] WORD JOINER..INVISIBLE PLUS
570    /// 2065          # Cn       <reserved-2065>
571    /// 2066..206F    # Cf  [10] LEFT-TO-RIGHT ISOLATE..NOMINAL DIGIT SHAPES
572    /// 3164          # Lo       HANGUL FILLER
573    /// FE00..FE0F    # Mn  [16] VARIATION SELECTOR-1..VARIATION SELECTOR-16
574    /// FEFF          # Cf       ZERO WIDTH NO-BREAK SPACE
575    /// FFA0          # Lo       HALFWIDTH HANGUL FILLER
576    /// FFF0..FFF8    # Cn   [9] <reserved-FFF0>..<reserved-FFF8>
577    /// 1BCA0..1BCA3  # Cf   [4] SHORTHAND FORMAT LETTER OVERLAP..SHORTHAND FORMAT UP STEP
578    /// 1D173..1D17A  # Cf   [8] MUSICAL SYMBOL BEGIN BEAM..MUSICAL SYMBOL END PHRASE
579    /// E0000         # Cn       <reserved-E0000>
580    /// E0001         # Cf       LANGUAGE TAG
581    /// E0002..E001F  # Cn  [30] <reserved-E0002>..<reserved-E001F>
582    /// E0020..E007F  # Cf  [96] TAG SPACE..CANCEL TAG
583    /// E0080..E00FF  # Cn [128] <reserved-E0080>..<reserved-E00FF>
584    /// E0100..E01EF  # Mn [240] VARIATION SELECTOR-17..VARIATION SELECTOR-256
585    /// E01F0..E0FFF  # Cn [3600] <reserved-E01F0>..<reserved-E0FFF>
586    fn is_default_ignorable(self) -> bool {
587        let ch = self;
588        let plane = ch >> 16;
589        if plane == 0 {
590            // BMP
591            let page = ch >> 8;
592            match page {
593                0x00 => ch == 0x00AD,
594                0x03 => ch == 0x034F,
595                0x06 => ch == 0x061C,
596                0x17 => (0x17B4..=0x17B5).contains(&ch),
597                0x18 => (0x180B..=0x180F).contains(&ch),
598                0x20 => {
599                    (0x200B..=0x200F).contains(&ch)
600                        || (0x202A..=0x202E).contains(&ch)
601                        || (0x2060..=0x206F).contains(&ch)
602                }
603                0xFE => (0xFE00..=0xFE0F).contains(&ch) || ch == 0xFEFF,
604                0xFF => (0xFFF0..=0xFFF8).contains(&ch),
605                _ => false,
606            }
607        } else {
608            // Other planes
609            match plane {
610                0x01 => (0x1D173..=0x1D17A).contains(&ch),
611                0x0E => (0xE0000..=0xE0FFF).contains(&ch),
612                _ => false,
613            }
614        }
615    }
616
617    fn is_variation_selector(self) -> bool {
618        // U+180B..180D, U+180F MONGOLIAN FREE VARIATION SELECTORs are handled in the
619        //Arabic shaper. No need to match them here.
620        (0x0FE00..=0x0FE0F).contains(&self) || // VARIATION SELECTOR - 1..16
621        (0xE0100..=0xE01EF).contains(&self) // VARIATION SELECTOR - 17..256
622    }

624    fn vertical(self) -> Option<Codepoint> {
625        Some(match self >> 8 {
626            0x20 => match self {
627                0x2013 => 0xfe32, // EN DASH
628                0x2014 => 0xfe31, // EM DASH
629                0x2025 => 0xfe30, // TWO DOT LEADER
630                0x2026 => 0xfe19, // HORIZONTAL ELLIPSIS
631                _ => return None,
632            },
633            0x30 => match self {
634                0x3001 => 0xfe11, // IDEOGRAPHIC COMMA
635                0x3002 => 0xfe12, // IDEOGRAPHIC FULL STOP
636                0x3008 => 0xfe3f, // LEFT ANGLE BRACKET
637                0x3009 => 0xfe40, // RIGHT ANGLE BRACKET
638                0x300a => 0xfe3d, // LEFT DOUBLE ANGLE BRACKET
639                0x300b => 0xfe3e, // RIGHT DOUBLE ANGLE BRACKET
640                0x300c => 0xfe41, // LEFT CORNER BRACKET
641                0x300d => 0xfe42, // RIGHT CORNER BRACKET
642                0x300e => 0xfe43, // LEFT WHITE CORNER BRACKET
643                0x300f => 0xfe44, // RIGHT WHITE CORNER BRACKET
644                0x3010 => 0xfe3b, // LEFT BLACK LENTICULAR BRACKET
645                0x3011 => 0xfe3c, // RIGHT BLACK LENTICULAR BRACKET
646                0x3014 => 0xfe39, // LEFT TORTOISE SHELL BRACKET
647                0x3015 => 0xfe3a, // RIGHT TORTOISE SHELL BRACKET
648                0x3016 => 0xfe17, // LEFT WHITE LENTICULAR BRACKET
649                0x3017 => 0xfe18, // RIGHT WHITE LENTICULAR BRACKET
650                _ => return None,
651            },
652            0xfe => match self {
653                0xfe4f => 0xfe34, // WAVY LOW LINE
654                _ => return None,
655            },
656            0xff => match self {
657                0xff01 => 0xfe15, // FULLWIDTH EXCLAMATION MARK
658                0xff08 => 0xfe35, // FULLWIDTH LEFT PARENTHESIS
659                0xff09 => 0xfe36, // FULLWIDTH RIGHT PARENTHESIS
660                0xff0c => 0xfe10, // FULLWIDTH COMMA
661                0xff1a => 0xfe13, // FULLWIDTH COLON
662                0xff1b => 0xfe14, // FULLWIDTH SEMICOLON
663                0xff1f => 0xfe16, // FULLWIDTH QUESTION MARK
664                0xff3b => 0xfe47, // FULLWIDTH LEFT SQUARE BRACKET
665                0xff3d => 0xfe48, // FULLWIDTH RIGHT SQUARE BRACKET
666                0xff3f => 0xfe33, // FULLWIDTH LOW LINE
667                0xff5b => 0xfe37, // FULLWIDTH LEFT CURLY BRACKET
668                0xff5d => 0xfe38, // FULLWIDTH RIGHT CURLY BRACKET
669                _ => return None,
670            },
671            _ => return None,
672        })
673    }

807    pub(crate) fn compose_impl(a: Codepoint, b: Codepoint) -> Option<Codepoint> {
808        // Hangul is handled algorithmically.
809        if let Some(ab) = compose_hangul(a, b) {
810            return Some(ab);
811        }
812
813        let u = if (a & 0xFFFF_F800) == 0x0000 && (b & 0xFFFF_FF80) == 0x0300 {
814            /* If "a" is small enough and "b" is in the U+0300 range,
815             * the composition data is encoded in a 32bit array sorted
816             * by "a,b" pair. */
817            let k = HB_CODEPOINT_ENCODE3_11_7_14(a, b, 0);
818            let v = ucd_dm2_u32_map
819                .binary_search_by(|probe| {
820                    let key = probe & HB_CODEPOINT_ENCODE3_11_7_14(0x001F_FFFF, 0x001F_FFFF, 0);
821                    key.cmp(&k)
822                })
823                .ok()
824                .map(|index| ucd_dm2_u32_map[index]);
825
826            HB_CODEPOINT_DECODE3_11_7_14_3(v?)
827        } else {
828            /* Otherwise it is stored in a 64bit array sorted by
829             * "a,b" pair. */
830            let k = HB_CODEPOINT_ENCODE3(a, b, 0);
831            let v = ucd_dm2_u64_map
832                .binary_search_by(|probe| {
833                    let key = probe & HB_CODEPOINT_ENCODE3(0x001F_FFFF, 0x001F_FFFF, 0);
834                    key.cmp(&k)
835                })
836                .ok()
837                .map(|index| ucd_dm2_u64_map[index]);
838
839            HB_CODEPOINT_DECODE3_3(v?)
840        };
841
842        if u == 0 {
843            None
844        } else {
845            Some(u)
846        }
847    }

849    pub(crate) fn decompose_impl(ab: Codepoint) -> Option<(Codepoint, Codepoint)> {
850        if let Some((a, b)) = decompose_hangul(ab) {
851            return Some((a, b));
852        }
853
854        let mut i = ucd_dm(ab as usize) as usize;
855
856        // If no data, there's no decomposition.
857        if i == 0 {
858            return None;
859        }
860        i -= 1;
861
862        if i < ucd_dm1_p0_map.len() + ucd_dm1_p2_map.len() {
863            let a = if i < ucd_dm1_p0_map.len() {
864                ucd_dm1_p0_map[i] as u32
865            } else {
866                let j = i - ucd_dm1_p0_map.len();
867                0x20000 | ucd_dm1_p2_map[j] as u32
868            };
869            return Some((a, 0));
870        }
871
872        i -= ucd_dm1_p0_map.len() + ucd_dm1_p2_map.len();
873
874        if i < ucd_dm2_u32_map.len() {
875            let v = ucd_dm2_u32_map[i];
876            let a = HB_CODEPOINT_DECODE3_11_7_14_1(v);
877            let b = HB_CODEPOINT_DECODE3_11_7_14_2(v);
878            return Some((a, b));
879        }
880
881        i -= ucd_dm2_u32_map.len();
882
883        let v = ucd_dm2_u64_map[i];
884        let a = HB_CODEPOINT_DECODE3_1(v);
885        let b = HB_CODEPOINT_DECODE3_2(v);
886        Some((a, b))
887    }
888
889    const S_BASE: u32 = 0xAC00;
890    const L_BASE: u32 = 0x1100;
891    const V_BASE: u32 = 0x1161;
892    const T_BASE: u32 = 0x11A7;
893    const L_COUNT: u32 = 19;
894    const V_COUNT: u32 = 21;
895    const T_COUNT: u32 = 28;
896    const N_COUNT: u32 = V_COUNT * T_COUNT;
897    const S_COUNT: u32 = L_COUNT * N_COUNT;
898
899    fn compose_hangul(a: Codepoint, b: Codepoint) -> Option<Codepoint> {
900        let l = a;
901        let v = b;
902        if L_BASE <= l && l < (L_BASE + L_COUNT) && V_BASE <= v && v < (V_BASE + V_COUNT) {
903            let r = S_BASE + (l - L_BASE) * N_COUNT + (v - V_BASE) * T_COUNT;
904            Some(r)
905        } else if S_BASE <= l
906            && l <= (S_BASE + S_COUNT - T_COUNT)
907            && T_BASE < v
908            && v < (T_BASE + T_COUNT)
909            && (l - S_BASE) % T_COUNT == 0
910        {
911            let r = l + (v - T_BASE);
912            Some(r)
913        } else {
914            None
915        }
916    }
917
918    fn decompose_hangul(ab: Codepoint) -> Option<(Codepoint, Codepoint)> {
919        let si = ab.wrapping_sub(S_BASE);
920        if si >= S_COUNT {
921            return None;
922        }
923        let (a, b) = if si % T_COUNT != 0 {
924            // LV,T
925            (S_BASE + (si / T_COUNT) * T_COUNT, T_BASE + (si % T_COUNT))
926        } else {
927            // L,V
928            (L_BASE + (si / N_COUNT), V_BASE + (si % N_COUNT) / T_COUNT)
929        };
930        Some((a, b))
931    }

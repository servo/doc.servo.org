56/// <https://drafts.csswg.org/css-counter-styles/#typedef-counter-style>
57///
58/// Note that 'none' is not a valid name, but we include this (along with String) for space
59/// efficiency when storing list-style-type.
60#[derive(
61    Clone, Debug, Eq, MallocSizeOf, PartialEq, ToComputedValue, ToCss, ToResolvedValue, ToShmem,
62)]
63#[repr(u8)]
64pub enum CounterStyle {
65    /// The 'none' value.
66    None,
67    /// `<counter-style-name>`
68    Name(CustomIdent),
69    /// `symbols()`
70    #[css(function)]
71    Symbols {
72        /// The <symbols-type>, or symbolic if not specified.
73        #[css(skip_if = "is_symbolic")]
74        ty: SymbolsType,
75        /// The actual symbols.
76        symbols: Symbols,
77    },
78    /// A single string value, useful for `<list-style-type>`.
79    String(AtomString),
80}
81
82#[inline]
83fn is_symbolic(symbols_type: &SymbolsType) -> bool {
84    *symbols_type == SymbolsType::Symbolic
85}

87impl CounterStyle {
88    /// disc value
89    pub fn disc() -> Self {
90        CounterStyle::Name(CustomIdent(atom!("disc")))
91    }
92
93    /// decimal value
94    pub fn decimal() -> Self {
95        CounterStyle::Name(CustomIdent(atom!("decimal")))
96    }
97
98    /// Is this a bullet? (i.e. `list-style-type: disc|circle|square|disclosure-closed|disclosure-open`)
99    #[inline]
100    pub fn is_bullet(&self) -> bool {
101        match self {
102            CounterStyle::Name(CustomIdent(ref name)) => {
103                name == &atom!("disc")
104                    || name == &atom!("circle")
105                    || name == &atom!("square")
106                    || name == &atom!("disclosure-closed")
107                    || name == &atom!("disclosure-open")
108            },
109            _ => false,
110        }
111    }
112
113    /// Returns whether this is a particular name.
114    pub fn is_name(&self, name: &Atom) -> bool {
115        match *self {
116            CounterStyle::Name(CustomIdent(ref n)) => n == name,
117            _ => false,
118        }
119    }

124    /// Flags to control parsing of counter styles.
125    pub struct CounterStyleParsingFlags: u8 {
126        /// Whether `none` is allowed.
127        const ALLOW_NONE = 1 << 0;
128        /// Whether a bare string is allowed.
129        const ALLOW_STRING = 1 << 1;
130    }

133impl CounterStyle {
134    /// Parse a counter style, and optionally none|string (for list-style-type).
135    pub fn parse<'i, 't>(
136        context: &ParserContext,
137        input: &mut Parser<'i, 't>,
138        flags: CounterStyleParsingFlags,
139    ) -> Result<Self, ParseError<'i>> {
140        use self::CounterStyleParsingFlags as Flags;
141        let location = input.current_source_location();
142        match input.next()? {
143            Token::QuotedString(ref string) if flags.intersects(Flags::ALLOW_STRING) => {
144                Ok(Self::String(AtomString::from(string.as_ref())))
145            },
146            Token::Ident(ref ident) => {
147                if flags.intersects(Flags::ALLOW_NONE) && ident.eq_ignore_ascii_case("none") {
148                    return Ok(Self::None);
149                }
150                Ok(Self::Name(counter_style_name_from_ident(ident, location)?))
151            },
152            Token::Function(ref name) if name.eq_ignore_ascii_case("symbols") => {
153                input.parse_nested_block(|input| {
154                    let symbols_type = input
155                        .try_parse(SymbolsType::parse)
156                        .unwrap_or(SymbolsType::Symbolic);
157                    let symbols = Symbols::parse(context, input)?;
158                    // There must be at least two symbols for alphabetic or
159                    // numeric system.
160                    if (symbols_type == SymbolsType::Alphabetic
161                        || symbols_type == SymbolsType::Numeric)
162                        && symbols.0.len() < 2
163                    {
164                        return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
165                    }
166                    // Identifier is not allowed in symbols() function.
167                    if symbols.0.iter().any(|sym| !sym.is_allowed_in_symbols()) {
168                        return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
169                    }
170                    Ok(Self::Symbols {
171                        ty: symbols_type,
172                        symbols,
173                    })
174                })
175            },
176            t => Err(location.new_unexpected_token_error(t.clone())),
177        }
178    }
179}

197fn parse_counter_style_name<'i>(input: &mut Parser<'i, '_>) -> Result<CustomIdent, ParseError<'i>> {
198    let location = input.current_source_location();
199    let ident = input.expect_ident()?;
200    counter_style_name_from_ident(ident, location)
201}

203/// This allows the reserved counter style names "decimal" and "disc".
204fn counter_style_name_from_ident<'i>(
205    ident: &CowRcStr<'i>,
206    location: SourceLocation,
207) -> Result<CustomIdent, ParseError<'i>> {
208    macro_rules! predefined {
209        ($($name: tt,)+) => {{
210            ascii_case_insensitive_phf_map! {
211                predefined -> Atom = {
212                    $(
213                        $name => atom!($name),
214                    )+
215                }
216            }
217
218            // This effectively performs case normalization only on predefined names.
219            if let Some(lower_case) = predefined::get(&ident) {
220                Ok(CustomIdent(lower_case.clone()))
221            } else {
222                // none is always an invalid <counter-style> value.
223                CustomIdent::from_ident(location, ident, &["none"])
224            }
225        }}
226    }
227    include!("predefined.rs")
228}

230fn is_valid_name_definition(ident: &CustomIdent) -> bool {
231    ident.0 != atom!("decimal")
232        && ident.0 != atom!("disc")
233        && ident.0 != atom!("circle")
234        && ident.0 != atom!("square")
235        && ident.0 != atom!("disclosure-closed")
236        && ident.0 != atom!("disclosure-open")
237}

239/// Parse the prelude of an @counter-style rule
240pub fn parse_counter_style_name_definition<'i, 't>(
241    input: &mut Parser<'i, 't>,
242) -> Result<CustomIdent, ParseError<'i>> {
243    parse_counter_style_name(input).and_then(|ident| {
244        if !is_valid_name_definition(&ident) {
245            Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError))
246        } else {
247            Ok(ident)
248        }
249    })

254pub struct CounterStyleRule {
255    name: CustomIdent,
256    generation: Wrapping<u32>,
257    descriptors: Descriptors,
258    /// The parser location of the @counter-style rule.
259    pub source_location: SourceLocation,
260}

262/// Parse the body (inside `{}`) of an @counter-style rule
263pub fn parse_counter_style_body<'i, 't>(
264    name: CustomIdent,
265    context: &ParserContext,
266    input: &mut Parser<'i, 't>,
267    location: SourceLocation,
268) -> Result<CounterStyleRule, ParseError<'i>> {
269    let start = input.current_source_location();
270    let mut rule = CounterStyleRule::empty(name, location);
271    {
272        let mut parser = DescriptorParser {
273            context,
274            descriptors: &mut rule.descriptors,
275        };
276        let mut iter = RuleBodyParser::new(input, &mut parser);
277        while let Some(declaration) = iter.next() {
278            if let Err((error, slice)) = declaration {
279                let location = error.location;
280                let error = ContextualParseError::UnsupportedCounterStyleDescriptorDeclaration(
281                    slice, error,
282                );
283                context.log_css_error(location, error)
284            }
285        }
286    }
287    let error = match *rule.resolved_system() {
288        ref system @ System::Cyclic
289        | ref system @ System::Fixed { .. }
290        | ref system @ System::Symbolic
291        | ref system @ System::Alphabetic
292        | ref system @ System::Numeric
293            if rule.descriptors.symbols.is_none() =>
294        {
295            let system = system.to_css_string();
296            Some(ContextualParseError::InvalidCounterStyleWithoutSymbols(
297                system,
298            ))
299        },
300        ref system @ System::Alphabetic | ref system @ System::Numeric
301            if rule.descriptors.symbols.as_ref().unwrap().0.len() < 2 =>
302        {
303            let system = system.to_css_string();
304            Some(ContextualParseError::InvalidCounterStyleNotEnoughSymbols(
305                system,
306            ))
307        },
308        System::Additive if rule.descriptors.additive_symbols.is_none() => {
309            Some(ContextualParseError::InvalidCounterStyleWithoutAdditiveSymbols)
310        },
311        System::Extends(_) if rule.descriptors.symbols.is_some() => {
312            Some(ContextualParseError::InvalidCounterStyleExtendsWithSymbols)
313        },
314        System::Extends(_) if rule.descriptors.additive_symbols.is_some() => {
315            Some(ContextualParseError::InvalidCounterStyleExtendsWithAdditiveSymbols)
316        },
317        _ => None,
318    };
319    if let Some(error) = error {
320        context.log_css_error(start, error);
321        Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError))
322    } else {
323        Ok(rule)
324    }
325}

337// Implements the special checkers for some setters.
338// See <https://drafts.csswg.org/css-counter-styles/#the-csscounterstylerule-interface>
339impl CounterStyleRule {
340    fn empty(name: CustomIdent, source_location: SourceLocation) -> Self {
341        Self {
342            name: name,
343            generation: Wrapping(0),
344            descriptors: Descriptors::default(),
345            source_location,
346        }
347    }
348
349    /// Expose the descriptors as read-only, since we rely on updating our generation when they
350    /// change, and some setters need extra validation.
351    pub fn descriptors(&self) -> &Descriptors {
352        &self.descriptors
353    }
354
355    /// Set a descriptor, with the relevant validation. returns whether the descriptor changed.
356    pub fn set_descriptor<'i>(
357        &mut self,
358        id: DescriptorId,
359        context: &ParserContext,
360        input: &mut Parser<'i, '_>,
361    ) -> Result<bool, ParseError<'i>> {
362        // Some descriptors need a couple extra checks, deal with them specially.
363        // TODO(emilio): Remove these, see https://github.com/w3c/csswg-drafts/issues/5717
364        if id == DescriptorId::AdditiveSymbols
365            && matches!(*self.resolved_system(), System::Extends(..))
366        {
367            // No additive symbols should be set for extends system.
368            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
369        }
370        let changed = match id {
371            DescriptorId::System => {
372                let system = input.parse_entirely(|i| System::parse(context, i))?;
373                if !self.check_system(&system) {
374                    return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
375                }
376                let new = Some(system);
377                if self.descriptors.system == new {
378                    return Ok(false);
379                }
380                self.descriptors.system = new;
381                true
382            },
383            DescriptorId::Symbols => {
384                let symbols = input.parse_entirely(|i| Symbols::parse(context, i))?;
385                if !self.check_symbols(&symbols) {
386                    return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
387                }
388                let new = Some(symbols);
389                if self.descriptors.symbols == new {
390                    return Ok(false);
391                }
392                self.descriptors.symbols = new;
393                true
394            },
395            _ => self.descriptors.set(id, context, input)?,
396        };
397        if changed {
398            self.generation += Wrapping(1);
399        }
400        Ok(changed)
401    }
402
403    /// Check that the system is effectively not changed. Only params of system descriptor is
404    /// changeable.
405    fn check_system(&self, value: &System) -> bool {
406        mem::discriminant(self.resolved_system()) == mem::discriminant(value)
407    }
408
409    fn check_symbols(&self, value: &Symbols) -> bool {
410        match *self.resolved_system() {
411            // These two systems require at least two symbols.
412            System::Numeric | System::Alphabetic => value.0.len() >= 2,
413            // No symbols should be set for extends system.
414            System::Extends(_) => false,
415            _ => true,
416        }
417    }
418
419    /// Get the name of the counter style rule.
420    pub fn name(&self) -> &CustomIdent {
421        &self.name
422    }
423
424    /// Set the name of the counter style rule. Caller must ensure that
425    /// the name is valid.
426    pub fn set_name(&mut self, name: CustomIdent) {
427        debug_assert!(is_valid_name_definition(&name));
428        self.name = name;
429    }
430
431    /// Get the current generation of the counter style rule.
432    pub fn generation(&self) -> u32 {
433        self.generation.0
434    }
435
436    /// Get the system of this counter style rule, default to
437    /// `symbolic` if not specified.
438    pub fn resolved_system(&self) -> &System {
439        match self.descriptors.system {
440            Some(ref system) => system,
441            None => &System::Symbolic,
442        }
443    }
444}

446/// <https://drafts.csswg.org/css-counter-styles/#counter-style-system>
447#[derive(Clone, Debug, MallocSizeOf, ToShmem, PartialEq)]
448pub enum System {
449    /// 'cyclic'
450    Cyclic,
451    /// 'numeric'
452    Numeric,
453    /// 'alphabetic'
454    Alphabetic,
455    /// 'symbolic'
456    Symbolic,
457    /// 'additive'
458    Additive,
459    /// 'fixed <integer>?'
460    Fixed {
461        /// '<integer>?'
462        first_symbol_value: Option<Integer>,
463    },
464    /// 'extends <counter-style-name>'
465    Extends(CustomIdent),
466}
467
468impl Parse for System {
469    fn parse<'i, 't>(
470        context: &ParserContext,
471        input: &mut Parser<'i, 't>,
472    ) -> Result<Self, ParseError<'i>> {
473        try_match_ident_ignore_ascii_case! { input,
474            "cyclic" => Ok(System::Cyclic),
475            "numeric" => Ok(System::Numeric),
476            "alphabetic" => Ok(System::Alphabetic),
477            "symbolic" => Ok(System::Symbolic),
478            "additive" => Ok(System::Additive),
479            "fixed" => {
480                let first_symbol_value = input.try_parse(|i| Integer::parse(context, i)).ok();
481                Ok(System::Fixed { first_symbol_value })
482            },
483            "extends" => {
484                let other = parse_counter_style_name(input)?;
485                Ok(System::Extends(other))
486            },
487        }
488    }
489}
490
491impl ToCss for System {
492    fn to_css<W>(&self, dest: &mut CssWriter<W>) -> fmt::Result
493    where
494        W: Write,
495    {
496        match self {
497            System::Cyclic => dest.write_str("cyclic"),
498            System::Numeric => dest.write_str("numeric"),
499            System::Alphabetic => dest.write_str("alphabetic"),
500            System::Symbolic => dest.write_str("symbolic"),
501            System::Additive => dest.write_str("additive"),
502            System::Fixed { first_symbol_value } => {
503                if let Some(value) = first_symbol_value {
504                    dest.write_str("fixed ")?;
505                    value.to_css(dest)
506                } else {
507                    dest.write_str("fixed")
508                }
509            },
510            System::Extends(ref other) => {
511                dest.write_str("extends ")?;
512                other.to_css(dest)
513            },
514        }
515    }
516}

518/// <https://drafts.csswg.org/css-counter-styles/#typedef-symbol>
519#[derive(
520    Clone, Debug, Eq, MallocSizeOf, PartialEq, ToComputedValue, ToResolvedValue, ToCss, ToShmem,
521)]
522#[repr(u8)]
523pub enum Symbol {
524    /// <string>
525    String(crate::OwnedStr),
526    /// <custom-ident>
527    Ident(CustomIdent),
528    // Not implemented:
529    // /// <image>
530    // Image(Image),
531}

533impl Parse for Symbol {
534    fn parse<'i, 't>(
535        _context: &ParserContext,
536        input: &mut Parser<'i, 't>,
537    ) -> Result<Self, ParseError<'i>> {
538        let location = input.current_source_location();
539        match *input.next()? {
540            Token::QuotedString(ref s) => Ok(Symbol::String(s.as_ref().to_owned().into())),
541            Token::Ident(ref s) => Ok(Symbol::Ident(CustomIdent::from_ident(location, s, &[])?)),
542            ref t => Err(location.new_unexpected_token_error(t.clone())),
543        }
544    }
545}

583/// <https://drafts.csswg.org/css-counter-styles/#counter-style-range>
584///
585/// Empty represents 'auto'
586#[derive(Clone, Debug, MallocSizeOf, ToCss, ToShmem, PartialEq)]
587#[css(comma)]
588pub struct CounterRanges(#[css(iterable, if_empty = "auto")] pub crate::OwnedSlice<CounterRange>);
589
590/// A bound found in `CounterRanges`.
591#[derive(Clone, Debug, MallocSizeOf, ToCss, ToShmem, PartialEq)]
592pub enum CounterBound {
593    /// An integer bound.
594    Integer(Integer),
595    /// The infinite bound.
596    Infinite,
597}
598
599impl Parse for CounterRanges {
600    fn parse<'i, 't>(
601        context: &ParserContext,
602        input: &mut Parser<'i, 't>,
603    ) -> Result<Self, ParseError<'i>> {
604        if input
605            .try_parse(|input| input.expect_ident_matching("auto"))
606            .is_ok()
607        {
608            return Ok(CounterRanges(Default::default()));
609        }
610
611        let ranges = input.parse_comma_separated(|input| {
612            let start = parse_bound(context, input)?;
613            let end = parse_bound(context, input)?;
614            if let (CounterBound::Integer(ref s), CounterBound::Integer(ref e)) = (&start, &end) {
615                // Rejects calc expressions that cannot be resolved at parse time,
616                // since @counter-style descriptors require concrete values.
617                let s = s.resolve();
618                let e = e.resolve();
619                if s.is_none() || e.is_none() || s.unwrap() > e.unwrap() {
620                    return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
621                }
622            }
623            Ok(CounterRange { start, end })
624        })?;
625
626        Ok(CounterRanges(ranges.into()))
627    }
628}
629
630fn parse_bound<'i, 't>(
631    context: &ParserContext,
632    input: &mut Parser<'i, 't>,
633) -> Result<CounterBound, ParseError<'i>> {
634    if let Ok(integer) = input.try_parse(|input| Integer::parse(context, input)) {
635        return Ok(CounterBound::Integer(integer));
636    }
637    input.expect_ident_matching("infinite")?;
638    Ok(CounterBound::Infinite)
639}

641/// <https://drafts.csswg.org/css-counter-styles/#counter-style-pad>
642#[derive(Clone, Debug, MallocSizeOf, ToCss, ToShmem, PartialEq)]
643pub struct Pad(pub Integer, pub Symbol);
644
645impl Parse for Pad {
646    fn parse<'i, 't>(
647        context: &ParserContext,
648        input: &mut Parser<'i, 't>,
649    ) -> Result<Self, ParseError<'i>> {
650        let pad_with = input.try_parse(|input| Symbol::parse(context, input));
651        let min_length = Integer::parse_non_negative(context, input)?;
652        // Rejects calc expressions that cannot be resolved at parse time,
653        // since @counter-style descriptors require concrete values.
654        if min_length.resolve().is_none() {
655            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
656        }
657        let pad_with = pad_with.or_else(|_| Symbol::parse(context, input))?;
658        Ok(Pad(min_length, pad_with))
659    }
660}

675/// <https://drafts.csswg.org/css-counter-styles/#descdef-counter-style-symbols>
676#[derive(
677    Clone, Debug, Eq, MallocSizeOf, PartialEq, ToComputedValue, ToResolvedValue, ToCss, ToShmem,
678)]
679#[repr(C)]
680pub struct Symbols(
681    #[css(iterable)]
682    #[ignore_malloc_size_of = "Arc"]
683    pub crate::ArcSlice<Symbol>,
684);
685
686impl Parse for Symbols {
687    fn parse<'i, 't>(
688        context: &ParserContext,
689        input: &mut Parser<'i, 't>,
690    ) -> Result<Self, ParseError<'i>> {
691        let mut symbols = smallvec::SmallVec::<[_; 5]>::new();
692        while let Ok(s) = input.try_parse(|input| Symbol::parse(context, input)) {
693            symbols.push(s);
694        }
695        if symbols.is_empty() {
696            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
697        }
698        Ok(Symbols(crate::ArcSlice::from_iter(symbols.drain(..))))
699    }
700}

702/// <https://drafts.csswg.org/css-counter-styles/#descdef-counter-style-additive-symbols>
703#[derive(Clone, Debug, MallocSizeOf, ToCss, ToShmem, PartialEq)]
704#[css(comma)]
705pub struct AdditiveSymbols(#[css(iterable)] pub crate::OwnedSlice<AdditiveTuple>);
706
707impl Parse for AdditiveSymbols {
708    fn parse<'i, 't>(
709        context: &ParserContext,
710        input: &mut Parser<'i, 't>,
711    ) -> Result<Self, ParseError<'i>> {
712        let tuples = Vec::<AdditiveTuple>::parse(context, input)?;
713        if tuples.iter().any(|t| t.weight.resolve().is_none()) {
714            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
715        }
716        // FIXME maybe? https://github.com/w3c/csswg-drafts/issues/1220
717        if tuples
718            .windows(2)
719            .any(|window| window[0].weight.get() <= window[1].weight.get())
720        {
721            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
722        }
723        Ok(AdditiveSymbols(tuples.into()))
724    }
725}

727/// <integer> && <symbol>
728#[derive(Clone, Debug, MallocSizeOf, ToCss, ToShmem, PartialEq)]
729pub struct AdditiveTuple {
730    /// <integer>
731    pub weight: Integer,
732    /// <symbol>
733    pub symbol: Symbol,
734}
735
736impl OneOrMoreSeparated for AdditiveTuple {
737    type S = Comma;
738}
739
740impl Parse for AdditiveTuple {
741    fn parse<'i, 't>(
742        context: &ParserContext,
743        input: &mut Parser<'i, 't>,
744    ) -> Result<Self, ParseError<'i>> {
745        let symbol = input.try_parse(|input| Symbol::parse(context, input));
746        let weight = Integer::parse_non_negative(context, input)?;
747        let symbol = symbol.or_else(|_| Symbol::parse(context, input))?;
748        Ok(Self { weight, symbol })
749    }
750}

752/// <https://drafts.csswg.org/css-counter-styles/#counter-style-speak-as>
753#[derive(Clone, Debug, MallocSizeOf, ToCss, PartialEq, ToShmem)]
754pub enum SpeakAs {
755    /// auto
756    Auto,
757    /// bullets
758    Bullets,
759    /// numbers
760    Numbers,
761    /// words
762    Words,
763    // /// spell-out, not supported, see bug 1024178
764    // SpellOut,
765    /// <counter-style-name>
766    Other(CustomIdent),
767}
768
769impl Parse for SpeakAs {
770    fn parse<'i, 't>(
771        _context: &ParserContext,
772        input: &mut Parser<'i, 't>,
773    ) -> Result<Self, ParseError<'i>> {
774        let mut is_spell_out = false;
775        let result = input.try_parse(|input| {
776            let ident = input.expect_ident().map_err(|_| ())?;
777            match_ignore_ascii_case! { &*ident,
778                "auto" => Ok(SpeakAs::Auto),
779                "bullets" => Ok(SpeakAs::Bullets),
780                "numbers" => Ok(SpeakAs::Numbers),
781                "words" => Ok(SpeakAs::Words),
782                "spell-out" => {
783                    is_spell_out = true;
784                    Err(())
785                },
786                _ => Err(()),
787            }
788        });
789        if is_spell_out {
790            // spell-out is not supported, but don’t parse it as a <counter-style-name>.
791            // See bug 1024178.
792            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
793        }
794        result.or_else(|_| Ok(SpeakAs::Other(parse_counter_style_name(input)?)))
795    }
796}

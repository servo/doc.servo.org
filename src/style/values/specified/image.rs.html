96impl Color {
97    fn has_modern_syntax(&self) -> bool {
98        match self {
99            Self::Absolute(absolute) => !absolute.color.is_legacy_syntax(),
100            Self::ColorMix(mix) => {
101                if mix.flags.contains(ColorMixFlags::RESULT_IN_MODERN_SYNTAX) {
102                    true
103                } else {
104                    mix.items.iter().any(|item| item.color.has_modern_syntax())
105                }
106            },
107            Self::LightDark(ld) => ld.light.has_modern_syntax() || ld.dark.has_modern_syntax(),
108
109            // The default is that this color doesn't have any modern syntax.
110            _ => false,
111        }
112    }
113}

115fn default_color_interpolation_method<T>(
116    items: &[generic::GradientItem<Color, T>],
117) -> ColorInterpolationMethod {
118    let has_modern_syntax_item = items.iter().any(|item| match item {
119        generic::GenericGradientItem::SimpleColorStop(color) => color.has_modern_syntax(),
120        generic::GenericGradientItem::ComplexColorStop { color, .. } => color.has_modern_syntax(),
121        generic::GenericGradientItem::InterpolationHint(_) => false,
122    });
123
124    if has_modern_syntax_item {
125        ColorInterpolationMethod::default()
126    } else {
127        ColorInterpolationMethod::srgb()
128    }
129}

220impl Image {
221    fn parse_with_cors_mode<'i, 't>(
222        context: &ParserContext,
223        input: &mut Parser<'i, 't>,
224        cors_mode: CorsMode,
225        flags: ParseImageFlags,
226    ) -> Result<Image, ParseError<'i>> {
227        if !flags.contains(ParseImageFlags::FORBID_NONE)
228            && input.try_parse(|i| i.expect_ident_matching("none")).is_ok()
229        {
230            return Ok(generic::Image::None);
231        }
232
233        if let Ok(url) =
234            input.try_parse(|input| SpecifiedUrl::parse_with_cors_mode(context, input, cors_mode))
235        {
236            return Ok(generic::Image::Url(url));
237        }
238
239        if !flags.contains(ParseImageFlags::FORBID_IMAGE_SET) {
240            if let Ok(is) =
241                input.try_parse(|input| ImageSet::parse(context, input, cors_mode, flags))
242            {
243                return Ok(generic::Image::ImageSet(Box::new(is)));
244            }
245        }
246
247        if flags.contains(ParseImageFlags::FORBID_NON_URL) {
248            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
249        }
250
251        if let Ok(gradient) = input.try_parse(|i| Gradient::parse(context, i)) {
252            return Ok(generic::Image::Gradient(Box::new(gradient)));
253        }
254
255        let function = input.expect_function()?.clone();
256        input.parse_nested_block(|input| Ok(match_ignore_ascii_case! { &function,
257            #[cfg(feature = "servo")]
258            "paint" => Self::PaintWorklet(Box::new(<PaintWorklet>::parse_args(context, input)?)),
259            "cross-fade" if cross_fade_enabled() => Self::CrossFade(Box::new(CrossFade::parse_args(context, input, cors_mode, flags)?)),
260            "image" => Self::Image(Box::new(Color::parse(context, input)?)),
261            "light-dark" if image_light_dark_enabled(context) => Self::LightDark(Box::new(GenericLightDark::parse_args_with(input, |input| {
262                // `none` in `light-dark()` has a special meaning.
263                Self::parse_with_cors_mode(context, input, cors_mode, flags & !ParseImageFlags::FORBID_NONE)
264            })?)),
265            #[cfg(feature = "gecko")]
266            "-moz-element" => Self::Element(Self::parse_element(input)?),
267            #[cfg(feature = "gecko")]
268            "-moz-symbolic-icon" if context.chrome_rules_enabled() => Self::MozSymbolicIcon(input.expect_ident()?.as_ref().into()),
269            _ => return Err(input.new_custom_error(StyleParseErrorKind::UnexpectedFunction(function))),
270        }))
271    }
272}

344impl CrossFadeElement {
345    fn parse_percentage<'i, 't>(
346        context: &ParserContext,
347        input: &mut Parser<'i, 't>,
348    ) -> Option<Percentage> {
349        // We clamp our values here as this is the way that Safari and Chrome's
350        // implementation handle out-of-bounds percentages but whether or not
351        // this behavior follows the specification is still being discussed.
352        // See: <https://github.com/w3c/csswg-drafts/issues/5333>
353        let mut p = input
354            .try_parse(|input| Percentage::parse_non_negative(context, input))
355            .ok()?;
356        p.clamp_to_hundred();
357        Some(p)
358    }
359
360    /// <cf-image> = <percentage>? && [ <image> | <color> ]
361    fn parse<'i, 't>(
362        context: &ParserContext,
363        input: &mut Parser<'i, 't>,
364        cors_mode: CorsMode,
365        flags: ParseImageFlags,
366    ) -> Result<Self, ParseError<'i>> {
367        // Try and parse a leading percent sign.
368        let mut percent = Self::parse_percentage(context, input);
369        // Parse the image
370        let image = CrossFadeImage::parse(context, input, cors_mode, flags)?;
371        // Try and parse a trailing percent sign.
372        if percent.is_none() {
373            percent = Self::parse_percentage(context, input);
374        }
375        Ok(Self {
376            percent: percent.into(),
377            image,
378        })
379    }
380}

430impl ImageSetItem {
431    fn parse_type<'i>(p: &mut Parser<'i, '_>) -> Result<crate::OwnedStr, ParseError<'i>> {
432        p.expect_function_matching("type")?;
433        p.parse_nested_block(|input| Ok(input.expect_string()?.as_ref().to_owned().into()))
434    }
435
436    fn parse<'i, 't>(
437        context: &ParserContext,
438        input: &mut Parser<'i, 't>,
439        cors_mode: CorsMode,
440        flags: ParseImageFlags,
441    ) -> Result<Self, ParseError<'i>> {
442        let start = input.position().byte_index();
443        let location = input.current_source_location();
444        let image = match input.try_parse(|i| i.expect_url_or_string()) {
445            Ok(url) => {
446                let end = input.position().byte_index();
447                Image::Url(SpecifiedUrl::parse_from_string(
448                    url.as_ref().into(),
449                    start,
450                    end,
451                    context,
452                    cors_mode,
453                    location,
454                )?)
455            },
456            Err(..) => Image::parse_with_cors_mode(
457                context,
458                input,
459                cors_mode,
460                flags | ParseImageFlags::FORBID_NONE | ParseImageFlags::FORBID_IMAGE_SET,
461            )?,
462        };
463
464        let mut resolution = input
465            .try_parse(|input| Resolution::parse(context, input))
466            .ok();
467        let mime_type = input.try_parse(Self::parse_type).ok();
468
469        // Try to parse resolution after type().
470        if mime_type.is_some() && resolution.is_none() {
471            resolution = input
472                .try_parse(|input| Resolution::parse(context, input))
473                .ok();
474        }
475
476        let resolution = resolution.unwrap_or_else(|| Resolution::from_x(1.0));
477        let has_mime_type = mime_type.is_some();
478        let mime_type = mime_type.unwrap_or_default();
479
480        Ok(Self {
481            image,
482            resolution,
483            has_mime_type,
484            mime_type,
485        })
486    }
487}

517            },

520                (Shape::Linear, true, GradientCompatMode::Moz)
521            },
522            "radial-gradient" => {
523                (Shape::Radial, false, GradientCompatMode::Modern)
524            },
525            "-webkit-radial-gradient" => {
526                (Shape::Radial, false, GradientCompatMode::WebKit)
527            },
528            #[cfg(feature = "gecko")]
529            "-moz-radial-gradient" => {
530                (Shape::Radial, false, GradientCompatMode::Moz)

533                (Shape::Radial, true, GradientCompatMode::Modern)
534            },
535            "-webkit-repeating-radial-gradient" => {
536                (Shape::Radial, true, GradientCompatMode::WebKit)
537            },
538            #[cfg(feature = "gecko")]
539            "-moz-repeating-radial-gradient" => {
540                (Shape::Radial, true, GradientCompatMode::Moz)
541            },
542            "conic-gradient" => {
543                (Shape::Conic, false, GradientCompatMode::Modern)
544            },
545            "repeating-conic-gradient" => {
546                (Shape::Conic, true, GradientCompatMode::Modern)
547            },
548            "-webkit-gradient" => {
549                return input.parse_nested_block(|i| {

551                });
552            },
553            _ => {

555                return Err(input.new_custom_error(StyleParseErrorKind::UnexpectedFunction(func)));
556            }
557        };

562                Shape::Radial => Self::parse_radial(context, i, repeating, compat_mode)?,
563                Shape::Conic => Self::parse_conic(context, i, repeating)?,
564            })
565        })?)
566    }

569impl Gradient {
570    fn parse_webkit_gradient_argument<'i, 't>(
571        context: &ParserContext,
572        input: &mut Parser<'i, 't>,
573    ) -> Result<Self, ParseError<'i>> {
574        use crate::values::specified::position::{
575            HorizontalPositionKeyword as X, VerticalPositionKeyword as Y,
576        };
577        type Point = GenericPosition<Component<X>, Component<Y>>;
578
579        #[derive(Clone, Parse)]
580        enum Component<S> {
581            Center,
582            Number(NumberOrPercentage),
583            Side(S),
584        }
585
586        fn line_direction_from_points(first: Point, second: Point) -> LineDirection {
587            let h_ord = first.horizontal.partial_cmp(&second.horizontal);
588            let v_ord = first.vertical.partial_cmp(&second.vertical);
589            let (h, v) = match (h_ord, v_ord) {
590                (Some(h), Some(v)) => (h, v),
591                _ => return LineDirection::Vertical(Y::Bottom),
592            };
593            match (h, v) {
594                (Ordering::Less, Ordering::Less) => LineDirection::Corner(X::Right, Y::Bottom),
595                (Ordering::Less, Ordering::Equal) => LineDirection::Horizontal(X::Right),
596                (Ordering::Less, Ordering::Greater) => LineDirection::Corner(X::Right, Y::Top),
597                (Ordering::Equal, Ordering::Greater) => LineDirection::Vertical(Y::Top),
598                (Ordering::Equal, Ordering::Equal) | (Ordering::Equal, Ordering::Less) => {
599                    LineDirection::Vertical(Y::Bottom)
600                },
601                (Ordering::Greater, Ordering::Less) => LineDirection::Corner(X::Left, Y::Bottom),
602                (Ordering::Greater, Ordering::Equal) => LineDirection::Horizontal(X::Left),
603                (Ordering::Greater, Ordering::Greater) => LineDirection::Corner(X::Left, Y::Top),
604            }
605        }
606
607        impl Parse for Point {
608            fn parse<'i, 't>(
609                context: &ParserContext,
610                input: &mut Parser<'i, 't>,
611            ) -> Result<Self, ParseError<'i>> {
612                input.try_parse(|i| {
613                    let x = Component::parse(context, i)?;
614                    let y = Component::parse(context, i)?;
615
616                    // TODO(Bug 2037751) - Enable calc()-expressions that can only be resolved at
617                    // computed value time (due to relative lengths, sibling-index(), etc.).
618                    if matches!(&x, Component::Number(NumberOrPercentage::Number(n)) if n.resolve().is_none()) ||
619                        matches!(&y, Component::Number(NumberOrPercentage::Number(n)) if n.resolve().is_none())
620                    {
621                        return Err(i.new_custom_error(StyleParseErrorKind::UnspecifiedError));
622                    }
623
624                    Ok(Self::new(x, y))
625                })
626            }
627        }
628
629        impl<S: Side> Into<NumberOrPercentage> for Component<S> {
630            fn into(self) -> NumberOrPercentage {
631                match self {
632                    Component::Center => NumberOrPercentage::Percentage(Percentage::new(0.5)),
633                    Component::Number(number) => number,
634                    Component::Side(side) => {
635                        let p = if side.is_start() {
636                            Percentage::zero()
637                        } else {
638                            Percentage::hundred()
639                        };
640                        NumberOrPercentage::Percentage(p)
641                    },
642                }
643            }
644        }
645
646        impl<S: Side> Into<PositionComponent<S>> for Component<S> {
647            fn into(self) -> PositionComponent<S> {
648                match self {
649                    Component::Center => PositionComponent::Center,
650                    Component::Number(NumberOrPercentage::Number(number)) => {
651                        // Unresolvable calc is rejected in Point::parse.
652                        PositionComponent::Length(Length::from_px(number.resolve().unwrap()).into())
653                    },
654                    Component::Number(NumberOrPercentage::Percentage(p)) => {
655                        PositionComponent::Length(p.to_length_percentage())
656                    },
657                    Component::Side(side) => PositionComponent::Side(side, None),
658                }
659            }
660        }
661
662        impl<S: Copy + Side> Component<S> {
663            fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
664                match (self.clone().into(), other.clone().into()) {
665                    (
666                        NumberOrPercentage::Percentage(ref a),
667                        NumberOrPercentage::Percentage(ref b),
668                    ) => a.resolve().partial_cmp(&b.resolve()),
669                    (NumberOrPercentage::Number(a), NumberOrPercentage::Number(b)) => {
670                        a.resolve().partial_cmp(&b.resolve())
671                    },
672                    (_, _) => None,
673                }
674            }
675        }
676
677        let ident = input.expect_ident_cloned()?;
678        input.expect_comma()?;
679
680        Ok(match_ignore_ascii_case! { &ident,
681            "linear" => {
682                let first = Point::parse(context, input)?;
683                input.expect_comma()?;
684                let second = Point::parse(context, input)?;
685
686                let direction = line_direction_from_points(first, second);
687                let items = Gradient::parse_webkit_gradient_stops(context, input, false)?;
688
689                generic::Gradient::Linear {
690                    direction,
691                    color_interpolation_method: ColorInterpolationMethod::srgb(),
692                    items,
693                    // Legacy gradients always use srgb as a default.
694                    flags: generic::GradientFlags::HAS_DEFAULT_COLOR_INTERPOLATION_METHOD,
695                    compat_mode: GradientCompatMode::Modern,
696                }
697            },
698            "radial" => {
699                let first_point = Point::parse(context, input)?;
700                input.expect_comma()?;
701                let first_radius = Number::parse_non_negative(context, input)?;
702                input.expect_comma()?;
703                let second_point = Point::parse(context, input)?;
704                input.expect_comma()?;
705                let second_radius = Number::parse_non_negative(context, input)?;
706
707                // TODO(Bug 2037751) - Enable calc()-expressions that can only be resolved at
708                // computed value time (due to relative lengths, sibling-index(), etc.).
709                if first_radius.resolve().is_none() || second_radius.resolve().is_none() {
710                    return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
711                }
712
713                let (reverse_stops, point, radius) = if second_radius.resolve() >= first_radius.resolve() {
714                    (false, second_point, second_radius)
715                } else {
716                    (true, first_point, first_radius)
717                };
718
719                // Unresolvable calc is rejected above.
720                let rad = Circle::Radius(NonNegative(Length::from_px(radius.resolve().unwrap())));
721                let shape = generic::EndingShape::Circle(rad);
722                let position = Position::new(point.horizontal.into(), point.vertical.into());
723                let items = Gradient::parse_webkit_gradient_stops(context, input, reverse_stops)?;
724
725                generic::Gradient::Radial {
726                    shape,
727                    position,
728                    color_interpolation_method: ColorInterpolationMethod::srgb(),
729                    items,
730                    // Legacy gradients always use srgb as a default.
731                    flags: generic::GradientFlags::HAS_DEFAULT_COLOR_INTERPOLATION_METHOD,
732                    compat_mode: GradientCompatMode::Modern,
733                }
734            },
735            _ => {
736                let e = SelectorParseErrorKind::UnexpectedIdent(ident.clone());
737                return Err(input.new_custom_error(e));
738            },
739        })
740    }

742    fn parse_webkit_gradient_stops<'i, 't>(
743        context: &ParserContext,
744        input: &mut Parser<'i, 't>,
745        reverse_stops: bool,
746    ) -> Result<LengthPercentageItemList, ParseError<'i>> {
747        let mut items = input
748            .try_parse(|i| {
749                i.expect_comma()?;
750                i.parse_comma_separated(|i| {
751                    let function = i.expect_function()?.clone();
752                    let (color, mut p) = i.parse_nested_block(|i| {
753                        let p = match_ignore_ascii_case! { &function,
754                            "color-stop" => {
755                                // TODO(Bug 2037751) - Enable calc()-expressions that can only be resolved at
756                                // computed value time (due to relative lengths, sibling-index(), etc.).
757                                let Some(p) = NumberOrPercentage::parse(context, i)?.to_percentage() else {
758                                    return Err(i.new_custom_error(StyleParseErrorKind::UnspecifiedError));
759                                };
760                                i.expect_comma()?;
761                                p
762                            },
763                            "from" => Percentage::zero(),
764                            "to" => Percentage::hundred(),
765                            _ => {
766                                return Err(i.new_custom_error(
767                                    StyleParseErrorKind::UnexpectedFunction(function.clone())
768                                ))
769                            },
770                        };
771                        let color = Color::parse(context, i)?;
772                        if color == Color::CurrentColor {
773                            return Err(i.new_custom_error(StyleParseErrorKind::UnspecifiedError));
774                        }
775                        Ok((color.into(), p))
776                    })?;
777                    if reverse_stops {
778                        p.reverse();
779                    }
780                    Ok(generic::GradientItem::ComplexColorStop {
781                        color,
782                        position: p.to_length_percentage(),
783                    })
784                })
785            })
786            .unwrap_or(vec![]);
787
788        if items.is_empty() {
789            items = vec![
790                generic::GradientItem::ComplexColorStop {
791                    color: Color::transparent(),
792                    position: LengthPercentage::zero_percent(),
793                },
794                generic::GradientItem::ComplexColorStop {
795                    color: Color::transparent(),
796                    position: LengthPercentage::hundred_percent(),
797                },
798            ];
799        } else if items.len() == 1 {
800            let first = items[0].clone();
801            items.push(first);
802        } else {
803            items.sort_by(|a, b| {
804                match (a, b) {
805                    (
806                        &generic::GradientItem::ComplexColorStop {
807                            position: ref a_position,
808                            ..
809                        },
810                        &generic::GradientItem::ComplexColorStop {
811                            position: ref b_position,
812                            ..
813                        },
814                    ) => match (a_position, b_position) {
815                        (&LengthPercentage::Percentage(a), &LengthPercentage::Percentage(b)) => {
816                            return a.get().partial_cmp(&b.get()).unwrap_or(Ordering::Equal);
817                        },
818                        _ => {},
819                    },
820                    _ => {},
821                }
822                if reverse_stops {
823                    Ordering::Greater
824                } else {
825                    Ordering::Less
826                }
827            })
828        }
829        Ok(items.into())
830    }

859    /// Parses a linear gradient.
860    /// GradientCompatMode can change during `-moz-` prefixed gradient parsing if it come across a `to` keyword.
861    fn parse_linear<'i, 't>(
862        context: &ParserContext,
863        input: &mut Parser<'i, 't>,
864        repeating: bool,
865        mut compat_mode: GradientCompatMode,
866    ) -> Result<Self, ParseError<'i>> {
867        let mut flags = GradientFlags::empty();
868        flags.set(GradientFlags::REPEATING, repeating);
869
870        let mut color_interpolation_method =
871            Self::try_parse_color_interpolation_method(context, input);
872
873        let direction = input
874            .try_parse(|p| LineDirection::parse(context, p, &mut compat_mode))
875            .ok();
876
877        if direction.is_some() && color_interpolation_method.is_none() {
878            color_interpolation_method = Self::try_parse_color_interpolation_method(context, input);
879        }
880
881        // If either of the 2 options were specified, we require a comma.
882        if color_interpolation_method.is_some() || direction.is_some() {
883            input.expect_comma()?;
884        }
885
886        let items = Gradient::parse_stops(context, input)?;
887
888        let default = default_color_interpolation_method(&items);
889        let color_interpolation_method = color_interpolation_method.unwrap_or(default);
890        flags.set(
891            GradientFlags::HAS_DEFAULT_COLOR_INTERPOLATION_METHOD,
892            default == color_interpolation_method,
893        );
894
895        let direction = direction.unwrap_or(match compat_mode {
896            GradientCompatMode::Modern => LineDirection::Vertical(VerticalPositionKeyword::Bottom),
897            _ => LineDirection::Vertical(VerticalPositionKeyword::Top),
898        });
899
900        Ok(Gradient::Linear {
901            direction,
902            color_interpolation_method,
903            items,
904            flags,
905            compat_mode,
906        })
907    }

909    /// Parses a radial gradient.
910    fn parse_radial<'i, 't>(
911        context: &ParserContext,
912        input: &mut Parser<'i, 't>,
913        repeating: bool,
914        compat_mode: GradientCompatMode,
915    ) -> Result<Self, ParseError<'i>> {
916        let mut flags = GradientFlags::empty();
917        flags.set(GradientFlags::REPEATING, repeating);
918
919        let mut color_interpolation_method =
920            Self::try_parse_color_interpolation_method(context, input);
921
922        let (shape, position) = match compat_mode {
923            GradientCompatMode::Modern => {
924                let shape = input.try_parse(|i| EndingShape::parse(context, i, compat_mode));
925                let position = input.try_parse(|i| {
926                    i.expect_ident_matching("at")?;
927                    Position::parse(context, i)
928                });
929                (shape, position.ok())
930            },
931            _ => {
932                let position = input.try_parse(|i| Position::parse(context, i));
933                let shape = input.try_parse(|i| {
934                    if position.is_ok() {
935                        i.expect_comma()?;
936                    }
937                    EndingShape::parse(context, i, compat_mode)
938                });
939                (shape, position.ok())
940            },
941        };
942
943        let has_shape_or_position = shape.is_ok() || position.is_some();
944        if has_shape_or_position && color_interpolation_method.is_none() {
945            color_interpolation_method = Self::try_parse_color_interpolation_method(context, input);
946        }
947
948        if has_shape_or_position || color_interpolation_method.is_some() {
949            input.expect_comma()?;
950        }
951
952        let shape = shape.unwrap_or({
953            generic::EndingShape::Ellipse(Ellipse::Extent(ShapeExtent::FarthestCorner))
954        });
955
956        let position = position.unwrap_or(Position::center());
957
958        let items = Gradient::parse_stops(context, input)?;
959
960        let default = default_color_interpolation_method(&items);
961        let color_interpolation_method = color_interpolation_method.unwrap_or(default);
962        flags.set(
963            GradientFlags::HAS_DEFAULT_COLOR_INTERPOLATION_METHOD,
964            default == color_interpolation_method,
965        );
966
967        Ok(Gradient::Radial {
968            shape,
969            position,
970            color_interpolation_method,
971            items,
972            flags,
973            compat_mode,
974        })
975    }

977    /// Parse a conic gradient.
978    fn parse_conic<'i, 't>(
979        context: &ParserContext,
980        input: &mut Parser<'i, 't>,
981        repeating: bool,
982    ) -> Result<Self, ParseError<'i>> {
983        let mut flags = GradientFlags::empty();
984        flags.set(GradientFlags::REPEATING, repeating);
985
986        let mut color_interpolation_method =
987            Self::try_parse_color_interpolation_method(context, input);
988
989        let angle = input.try_parse(|i| {
990            i.expect_ident_matching("from")?;
991            // Spec allows unitless zero start angles
992            // https://drafts.csswg.org/css-images-4/#valdef-conic-gradient-angle
993            Angle::parse_with_unitless(context, i)
994        });
995        let position = input.try_parse(|i| {
996            i.expect_ident_matching("at")?;
997            Position::parse(context, i)
998        });
999
1000        let has_angle_or_position = angle.is_ok() || position.is_ok();
1001        if has_angle_or_position && color_interpolation_method.is_none() {
1002            color_interpolation_method = Self::try_parse_color_interpolation_method(context, input);
1003        }
1004
1005        if has_angle_or_position || color_interpolation_method.is_some() {
1006            input.expect_comma()?;
1007        }
1008
1009        let angle = angle.unwrap_or(Angle::zero());
1010
1011        let position = position.unwrap_or(Position::center());
1012
1013        let items = generic::GradientItem::parse_comma_separated(
1014            context,
1015            input,
1016            AngleOrPercentage::parse_with_unitless,
1017        )?;
1018
1019        if items.is_empty() {
1020            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
1021        }
1022
1023        let default = default_color_interpolation_method(&items);
1024        let color_interpolation_method = color_interpolation_method.unwrap_or(default);
1025        flags.set(
1026            GradientFlags::HAS_DEFAULT_COLOR_INTERPOLATION_METHOD,
1027            default == color_interpolation_method,
1028        );
1029
1030        Ok(Gradient::Conic {
1031            angle,
1032            position,
1033            color_interpolation_method,
1034            items,
1035            flags,
1036        })
1037    }

1040impl generic::LineDirection for LineDirection {
1041    fn points_downwards(&self, compat_mode: GradientCompatMode) -> bool {
1042        match *self {
1043            LineDirection::Angle(ref angle) => {
1044                angle.as_no_calc().is_some_and(|a| a.degrees() == 180.0)
1045            },
1046            LineDirection::Vertical(VerticalPositionKeyword::Bottom) => {
1047                compat_mode == GradientCompatMode::Modern
1048            },
1049            LineDirection::Vertical(VerticalPositionKeyword::Top) => {
1050                compat_mode != GradientCompatMode::Modern
1051            },
1052            _ => false,
1053        }
1054    }
1055
1056    fn to_css<W>(&self, dest: &mut CssWriter<W>, compat_mode: GradientCompatMode) -> fmt::Result
1057    where
1058        W: Write,
1059    {
1060        match *self {
1061            LineDirection::Angle(ref angle) => angle.to_css(dest),
1062            LineDirection::Horizontal(x) => {
1063                if compat_mode == GradientCompatMode::Modern {
1064                    dest.write_str("to ")?;
1065                }
1066                x.to_css(dest)
1067            },
1068            LineDirection::Vertical(y) => {
1069                if compat_mode == GradientCompatMode::Modern {
1070                    dest.write_str("to ")?;
1071                }
1072                y.to_css(dest)
1073            },
1074            LineDirection::Corner(x, y) => {
1075                if compat_mode == GradientCompatMode::Modern {
1076                    dest.write_str("to ")?;
1077                }
1078                x.to_css(dest)?;
1079                dest.write_char(' ')?;
1080                y.to_css(dest)
1081            },
1082        }
1083    }
1084}

1086impl LineDirection {
1087    fn parse<'i, 't>(
1088        context: &ParserContext,
1089        input: &mut Parser<'i, 't>,
1090        compat_mode: &mut GradientCompatMode,
1091    ) -> Result<Self, ParseError<'i>> {
1092        // Gradients allow unitless zero angles as an exception, see:
1093        // https://github.com/w3c/csswg-drafts/issues/1162
1094        if let Ok(angle) = input.try_parse(|i| Angle::parse_with_unitless(context, i)) {
1095            return Ok(LineDirection::Angle(angle));
1096        }
1097
1098        input.try_parse(|i| {
1099            let to_ident = i.try_parse(|i| i.expect_ident_matching("to"));
1100            match *compat_mode {
1101                // `to` keyword is mandatory in modern syntax.
1102                GradientCompatMode::Modern => to_ident?,
1103                // Fall back to Modern compatibility mode in case there is a `to` keyword.
1104                // According to Gecko, `-moz-linear-gradient(to ...)` should serialize like
1105                // `linear-gradient(to ...)`.
1106                GradientCompatMode::Moz if to_ident.is_ok() => {
1107                    *compat_mode = GradientCompatMode::Modern
1108                },
1109                // There is no `to` keyword in webkit prefixed syntax. If it's consumed,
1110                // parsing should throw an error.
1111                GradientCompatMode::WebKit if to_ident.is_ok() => {
1112                    return Err(
1113                        i.new_custom_error(SelectorParseErrorKind::UnexpectedIdent("to".into()))
1114                    );
1115                },
1116                _ => {},
1117            }
1118
1119            if let Ok(x) = i.try_parse(HorizontalPositionKeyword::parse) {
1120                if let Ok(y) = i.try_parse(VerticalPositionKeyword::parse) {
1121                    return Ok(LineDirection::Corner(x, y));
1122                }
1123                return Ok(LineDirection::Horizontal(x));
1124            }
1125            let y = VerticalPositionKeyword::parse(i)?;
1126            if let Ok(x) = i.try_parse(HorizontalPositionKeyword::parse) {
1127                return Ok(LineDirection::Corner(x, y));
1128            }
1129            Ok(LineDirection::Vertical(y))
1130        })
1131    }
1132}

1134impl EndingShape {
1135    fn parse<'i, 't>(
1136        context: &ParserContext,
1137        input: &mut Parser<'i, 't>,
1138        compat_mode: GradientCompatMode,
1139    ) -> Result<Self, ParseError<'i>> {
1140        if let Ok(extent) = input.try_parse(|i| ShapeExtent::parse_with_compat_mode(i, compat_mode))
1141        {
1142            if input
1143                .try_parse(|i| i.expect_ident_matching("circle"))
1144                .is_ok()
1145            {
1146                return Ok(generic::EndingShape::Circle(Circle::Extent(extent)));
1147            }
1148            let _ = input.try_parse(|i| i.expect_ident_matching("ellipse"));
1149            return Ok(generic::EndingShape::Ellipse(Ellipse::Extent(extent)));
1150        }
1151        if input
1152            .try_parse(|i| i.expect_ident_matching("circle"))
1153            .is_ok()
1154        {
1155            if let Ok(extent) =
1156                input.try_parse(|i| ShapeExtent::parse_with_compat_mode(i, compat_mode))
1157            {
1158                return Ok(generic::EndingShape::Circle(Circle::Extent(extent)));
1159            }
1160            if compat_mode == GradientCompatMode::Modern {
1161                if let Ok(length) = input.try_parse(|i| NonNegativeLength::parse(context, i)) {
1162                    return Ok(generic::EndingShape::Circle(Circle::Radius(length)));
1163                }
1164            }
1165            return Ok(generic::EndingShape::Circle(Circle::Extent(
1166                ShapeExtent::FarthestCorner,
1167            )));
1168        }
1169        if input
1170            .try_parse(|i| i.expect_ident_matching("ellipse"))
1171            .is_ok()
1172        {
1173            if let Ok(extent) =
1174                input.try_parse(|i| ShapeExtent::parse_with_compat_mode(i, compat_mode))
1175            {
1176                return Ok(generic::EndingShape::Ellipse(Ellipse::Extent(extent)));
1177            }
1178            if compat_mode == GradientCompatMode::Modern {
1179                let pair: Result<_, ParseError> = input.try_parse(|i| {
1180                    let x = NonNegativeLengthPercentage::parse(context, i)?;
1181                    let y = NonNegativeLengthPercentage::parse(context, i)?;
1182                    Ok((x, y))
1183                });
1184                if let Ok((x, y)) = pair {
1185                    return Ok(generic::EndingShape::Ellipse(Ellipse::Radii(x, y)));
1186                }
1187            }
1188            return Ok(generic::EndingShape::Ellipse(Ellipse::Extent(
1189                ShapeExtent::FarthestCorner,
1190            )));
1191        }
1192        if let Ok(length) = input.try_parse(|i| NonNegativeLength::parse(context, i)) {
1193            if let Ok(y) = input.try_parse(|i| NonNegativeLengthPercentage::parse(context, i)) {
1194                if compat_mode == GradientCompatMode::Modern {
1195                    let _ = input.try_parse(|i| i.expect_ident_matching("ellipse"));
1196                }
1197                return Ok(generic::EndingShape::Ellipse(Ellipse::Radii(
1198                    NonNegative(LengthPercentage::from(length.0)),
1199                    y,
1200                )));
1201            }
1202            if compat_mode == GradientCompatMode::Modern {
1203                let y = input.try_parse(|i| {
1204                    i.expect_ident_matching("ellipse")?;
1205                    NonNegativeLengthPercentage::parse(context, i)
1206                });
1207                if let Ok(y) = y {
1208                    return Ok(generic::EndingShape::Ellipse(Ellipse::Radii(
1209                        NonNegative(LengthPercentage::from(length.0)),
1210                        y,
1211                    )));
1212                }
1213                let _ = input.try_parse(|i| i.expect_ident_matching("circle"));
1214            }
1215
1216            return Ok(generic::EndingShape::Circle(Circle::Radius(length)));
1217        }
1218        input.try_parse(|i| {
1219            let x = Percentage::parse_non_negative(context, i)?;
1220            let y = if let Ok(y) = i.try_parse(|i| NonNegativeLengthPercentage::parse(context, i)) {
1221                if compat_mode == GradientCompatMode::Modern {
1222                    let _ = i.try_parse(|i| i.expect_ident_matching("ellipse"));
1223                }
1224                y
1225            } else {
1226                if compat_mode == GradientCompatMode::Modern {
1227                    i.expect_ident_matching("ellipse")?;
1228                }
1229                NonNegativeLengthPercentage::parse(context, i)?
1230            };
1231            Ok(generic::EndingShape::Ellipse(Ellipse::Radii(
1232                NonNegative(x.to_length_percentage()),
1233                y,
1234            )))
1235        })
1236    }
1237}

1257impl<T> generic::GradientItem<Color, T> {
1258    fn parse_comma_separated<'i, 't>(
1259        context: &ParserContext,
1260        input: &mut Parser<'i, 't>,
1261        parse_position: impl for<'i1, 't1> Fn(&ParserContext, &mut Parser<'i1, 't1>) -> Result<T, ParseError<'i1>>
1262            + Copy,
1263    ) -> Result<crate::OwnedSlice<Self>, ParseError<'i>> {
1264        let mut items = Vec::new();
1265        let mut seen_stop = false;
1266
1267        loop {
1268            input.parse_until_before(Delimiter::Comma, |input| {
1269                if seen_stop {
1270                    if let Ok(hint) = input.try_parse(|i| parse_position(context, i)) {
1271                        seen_stop = false;
1272                        items.push(generic::GradientItem::InterpolationHint(hint));
1273                        return Ok(());
1274                    }
1275                }
1276
1277                let stop = generic::ColorStop::parse(context, input, parse_position)?;
1278
1279                if let Ok(multi_position) = input.try_parse(|i| parse_position(context, i)) {
1280                    let stop_color = stop.color.clone();
1281                    items.push(stop.into_item());
1282                    items.push(
1283                        generic::ColorStop {
1284                            color: stop_color,
1285                            position: Some(multi_position),
1286                        }
1287                        .into_item(),
1288                    );
1289                } else {
1290                    items.push(stop.into_item());
1291                }
1292
1293                seen_stop = true;
1294                Ok(())
1295            })?;
1296
1297            match input.next() {
1298                Err(_) => break,
1299                Ok(&Token::Comma) => continue,
1300                Ok(_) => unreachable!(),
1301            }
1302        }
1303
1304        if !seen_stop || items.is_empty() {
1305            return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
1306        }
1307        Ok(items.into())
1308    }
1309}

1311impl<T> generic::ColorStop<Color, T> {
1312    fn parse<'i, 't>(
1313        context: &ParserContext,
1314        input: &mut Parser<'i, 't>,
1315        parse_position: impl for<'i1, 't1> Fn(
1316            &ParserContext,
1317            &mut Parser<'i1, 't1>,
1318        ) -> Result<T, ParseError<'i1>>,
1319    ) -> Result<Self, ParseError<'i>> {
1320        Ok(generic::ColorStop {
1321            color: Color::parse(context, input)?,
1322            position: input.try_parse(|i| parse_position(context, i)).ok(),
1323        })
1324    }

1327impl PaintWorklet {

1329    fn parse_args<'i>(
1330        context: &ParserContext,
1331        input: &mut Parser<'i, '_>,
1332    ) -> Result<Self, ParseError<'i>> {
1333        use crate::custom_properties::SpecifiedValue;
1334        use servo_arc::Arc;
1335        let name = Atom::from(&**input.expect_ident()?);
1336        let arguments = input
1337            .try_parse(|input| {
1338                input.expect_comma()?;
1339                input.parse_comma_separated(|input| {
1340                    SpecifiedValue::parse(
1341                        input,
1342                        Some(&context.namespaces.prefixes),
1343                        &context.url_data,
1344                    )
1345                    .map(Arc::new)
1346                })
1347            })
1348            .unwrap_or_default();
1349        Ok(Self { name, arguments })
1350    }

34/// A specified color-mix().
35pub type ColorMix = GenericColorMix<Color, Percentage>;
36
37impl ColorMix {
38    fn parse(
39        context: &ParserContext,
40        input: &mut Parser,
41        preserve_authored: PreserveAuthored,
42    ) -> Result<Self, ParseError> {
43        input.expect_function_matching("color-mix")?;
44
45        input.parse_nested_block(|input| {
46            // If the color interpolation method is omitted, default to "in oklab".
47            // See: https://github.com/web-platform-tests/interop/issues/1166
48            let interpolation = input
49                .try_parse(|input| -> Result<_, ParseError> {
50                    let interpolation = ColorInterpolationMethod::parse(context, input)?;
51                    input.expect_comma()?;
52                    Ok(interpolation)
53                })
54                .unwrap_or_default();
55
56            let try_parse_percentage = |input: &mut Parser| -> Option<Percentage> {
57                input
58                    .try_parse(|input| Percentage::parse_zero_to_a_hundred(context, input))
59                    .ok()
60            };
61
62            let allow_multiple_items = crate::pref!("layout.css.color-mix-multi-color.enabled");
63
64            let mut items = ColorMixItemList::default();
65
66            loop {
67                let mut percentage = try_parse_percentage(input);
68
69                let color = Color::parse_internal(context, input, preserve_authored)?;
70
71                if percentage.is_none() {
72                    percentage = try_parse_percentage(input);
73                }
74
75                // TODO(Bug 2037742) - Enable calc()-expressions that can only be resolved at
76                // computed value time (due to relative lengths, sibling-index(), etc.).
77                if matches!(percentage, Some(ref p) if p.to_percentage().is_none()) {
78                    return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
79                }
80
81                items.push((color, percentage));
82
83                if input.try_parse(|i| i.expect_comma()).is_err() {
84                    break;
85                }
86
87                // Early exit to avoid parsing more than 2 colors if the pref is not enabled.
88                if !allow_multiple_items && items.len() == 2 {
89                    break;
90                }
91            }
92
93            // ...the color-mix() function takes a list of one or more <color> specifications...
94            // <https://drafts.csswg.org/css-color-5/#color-mix>
95            let min_item_count = if allow_multiple_items { 1 } else { 2 };
96            if items.len() < min_item_count {
97                return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
98            }
99
100            // Normalize percentages per:
101            // https://drafts.csswg.org/css-values-5/#normalize-mix-percentages
102            let (mut sum_specified, mut missing) = (0.0, 0);
103            for (_, percentage) in items.iter() {
104                if let Some(p) = percentage {
105                    // Percentage was enforced to be resolvable at parse time.
106                    sum_specified += p.to_percentage().unwrap();
107                } else {
108                    missing += 1;
109                }
110            }
111
112            // When any specified percentage is a calc(), omitted percentages are left
113            // unresolved so they serialize to nothing per
114            // https://drafts.csswg.org/css-color-5/#serial-color-mix, and are filled in
115            // at mix time instead.
116            let any_calc = items
117                .iter()
118                .any(|(_, p)| matches!(p, Some(p) if p.is_calc()));
119
120            let default_for_missing_items = if any_calc {
121                None
122            } else {
123                match missing {
124                    0 => None,
125                    m if m == items.len() => Some(Percentage::new(1.0 / items.len() as f32)),
126                    m => Some(Percentage::new((1.0 - sum_specified) / m as f32)),
127                }
128            };
129
130            if let Some(default) = default_for_missing_items {
131                for (_, percentage) in items.iter_mut() {
132                    if percentage.is_none() {
133                        *percentage = Some(default.clone());
134                    }
135                }
136            }
137
138            let finalized = items
139                .into_iter()
140                .map(|(color, percentage)| GenericColorMixItem {
141                    color,
142                    percentage: percentage.into(),
143                })
144                .collect::<ColorMixItemList<_>>();
145
146            // Pass RESULT_IN_MODERN_SYNTAX here, because the result of the color-mix() function
147            // should always be in the modern color syntax to allow for out of gamut results and
148            // to preserve floating point precision.
149            Ok(ColorMix {
150                interpolation,
151                items: OwnedSlice::from_slice(&finalized),
152                flags: ColorMixFlags::NORMALIZE_WEIGHTS | ColorMixFlags::RESULT_IN_MODERN_SYNTAX,
153            })
154        })
155    }
156}

533/// Whether to preserve authored colors during parsing. That's useful only if we
534/// plan to serialize the color back.

537    No,
538    Yes,
539}

541impl Parse for Color {
542    fn parse(context: &ParserContext, input: &mut Parser) -> Result<Self, ParseError> {
543        Self::parse_internal(context, input, PreserveAuthored::Yes)
544    }

546
547impl Color {
548    fn parse_internal(
549        context: &ParserContext,
550        input: &mut Parser,
551        preserve_authored: PreserveAuthored,
552    ) -> Result<Self, ParseError> {
553        let authored = match preserve_authored {
554            PreserveAuthored::No => None,
555            PreserveAuthored::Yes => {
556                // Currently we only store authored value for color keywords,
557                // because all browsers serialize those values as keywords for
558                // specified value.
559                let start = input.state();
560                let authored = input.expect_ident_cloned().ok();
561                input.reset(&start);
562                authored
563            },
564        };
565
566        match input.try_parse(|i| parsing::parse_color_with(context, i)) {
567            Ok(mut color) => {
568                if let Color::Absolute(ref mut absolute) = color {
569                    // Because we can't set the `authored` value at construction time, we have to set it
570                    // here.
571                    absolute.authored = authored.map(|s| s.to_ascii_lowercase().into_boxed_str());
572                }
573                Ok(color)
574            },
575            Err(e) => {
576                {
577                    #[cfg(feature = "gecko")]
578                    if let Ok(system) = input.try_parse(|i| SystemColor::parse(context, i)) {
579                        return Ok(Color::System(system));
580                    }
581                    #[cfg(feature = "servo")]
582                    if let Ok(system) = input.try_parse(SystemColor::parse) {
583                        return Ok(Color::System(system));
584                    }
585                }
586                if let Ok(mix) = input.try_parse(|i| ColorMix::parse(context, i, preserve_authored))
587                {
588                    return Ok(Color::ColorMix(Box::new(mix)));
589                }
590
591                if let Ok(ld) = input.try_parse(|i| {
592                    GenericLightDark::parse_with(i, |i| {
593                        Self::parse_internal(context, i, preserve_authored)
594                    })
595                }) {
596                    return Ok(Color::LightDark(Box::new(ld)));
597                }
598
599                if let Ok(c) = input.try_parse(|i| {
600                    i.expect_function_matching("contrast-color")?;
601                    i.parse_nested_block(|i| Self::parse_internal(context, i, preserve_authored))
602                }) {
603                    return Ok(Color::ContrastColor(Box::new(c)));
604                }
605
606                Err(e)
607            },
608        }
609    }
610
611    /// Returns whether a given color is valid for authors.
612    pub fn is_valid(context: &ParserContext, input: &mut Parser) -> bool {
613        input
614            .parse_entirely(|input| Self::parse_internal(context, input, PreserveAuthored::No))
615            .is_ok()
616    }
617
618    /// Tries to parse a color and compute it with a given device.
619    pub fn parse_and_compute(
620        context: &ParserContext,
621        input: &mut Parser,
622        device: Option<&Device>,
623    ) -> Result<ComputedColor, ()> {
624        let result = input
625            .parse_entirely(|input| Self::parse_internal(context, input, PreserveAuthored::No));
626
627        let specified = match result {
628            Ok(s) => s,
629            Err(..) => return Err(()),
630        };
631
632        match device {
633            Some(device) => {
634                Context::for_media_query_evaluation(device, device.quirks_mode(), |context| {
635                    specified.to_computed_color(Some(context))
636                })
637            },
638            None => specified.to_computed_color(None),
639        }
640    }
641}

659            Color::System(system) => system.to_css(dest),
660            Color::InheritFromBodyQuirk => dest.write_str("-moz-inherit-from-body-quirk"),
661        }

679impl Color {
680    /// Returns whether this color is allowed in forced-colors mode.
681    pub fn honored_in_forced_colors_mode(
682        &self,
683        context: &Context,
684        allow_transparent: bool,
685    ) -> bool {
686        match *self {
687            Self::InheritFromBodyQuirk => false,
688            Self::CurrentColor => true,
689            Self::System(..) => true,
690            Self::Absolute(ref absolute) => allow_transparent && absolute.color.is_transparent(),
691            Self::ColorFunction(ref color_function) => {
692                // For now we allow transparent colors if we can resolve the color function.
693                // <https://bugzilla.mozilla.org/show_bug.cgi?id=1923053>
694                color_function
695                    .to_computed_color(Some(context))
696                    .ok()
697                    .and_then(|c| c.as_absolute().copied())
698                    .map(|resolved| allow_transparent && resolved.is_transparent())
699                    .unwrap_or(false)
700            },
701            Self::LightDark(ref ld) => {
702                ld.light
703                    .honored_in_forced_colors_mode(context, allow_transparent)
704                    && ld
705                        .dark
706                        .honored_in_forced_colors_mode(context, allow_transparent)
707            },
708            Self::ColorMix(ref mix) => mix.items.iter().all(|item| {
709                item.color
710                    .honored_in_forced_colors_mode(context, allow_transparent)
711            }),
712            Self::ContrastColor(ref c) => {
713                c.honored_in_forced_colors_mode(context, allow_transparent)
714            },
715        }
716    }

739    /// Parse a color, with quirks.
740    ///
741    /// <https://quirks.spec.whatwg.org/#the-hashless-hex-color-quirk>
742    pub fn parse_quirky(
743        context: &ParserContext,
744        input: &mut Parser,
745        allow_quirks: AllowQuirks,
746    ) -> Result<Self, ParseError> {
747        input.try_parse(|i| Self::parse(context, i)).or_else(|e| {
748            if !allow_quirks.allowed(context.quirks_mode) {
749                return Err(e);
750            }
751            Color::parse_quirky_color(input).map_err(|_| e)
752        })
753    }

755    fn parse_hash(bytes: &[u8]) -> Result<Self, ParseError> {
756        match cssparser::color::parse_hash_color(bytes) {
757            Ok((r, g, b, a)) => Ok(Self::from_absolute_color(AbsoluteColor::srgb_legacy(
758                r, g, b, a,
759            ))),
760            Err(()) => Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError)),
761        }
762    }
763
764    /// Parse a <quirky-color> value.
765    ///
766    /// <https://quirks.spec.whatwg.org/#the-hashless-hex-color-quirk>
767    fn parse_quirky_color(input: &mut Parser) -> Result<Self, ParseError> {
768        let (value, unit) = match *input.next()? {
769            Token::Number {
770                int_value: Some(integer),
771                ..
772            } => (integer, None),
773            Token::Dimension {
774                int_value: Some(integer),
775                ref unit,
776                ..
777            } => (integer, Some(unit)),
778            Token::Ident(ref ident) => {
779                if ident.len() != 3 && ident.len() != 6 {
780                    return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
781                }
782                return Self::parse_hash(ident.as_bytes());
783            },
784            _ => {
785                return Err(ParseError::unexpected_token());
786            },
787        };
788        if value < 0 {
789            return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
790        }
791        let length = if value <= 9 {
792            1
793        } else if value <= 99 {
794            2
795        } else if value <= 999 {
796            3
797        } else if value <= 9999 {
798            4
799        } else if value <= 99999 {
800            5
801        } else if value <= 999999 {
802            6
803        } else {
804            return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
805        };
806        let total = length + unit.as_ref().map_or(0, |d| d.len());
807        if total > 6 {
808            return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
809        }
810        let mut serialization = [b'0'; 6];
811        let space_padding = 6 - total;
812        let mut written = space_padding;
813        let mut buf = itoa::Buffer::new();
814        let s = buf.format(value);
815        (&mut serialization[written..])
816            .write_all(s.as_bytes())
817            .unwrap();
818        written += s.len();
819        if let Some(unit) = unit {
820            written += (&mut serialization[written..])
821                .write(unit.as_bytes())
822                .unwrap();
823        }
824        debug_assert_eq!(written, 6);
825        Self::parse_hash(&serialization)
826    }
827}

829impl Color {
830    /// Converts this Color into a ComputedColor.
831    ///
832    /// If `context` is `None`, and the specified color requires data from
833    /// the context to resolve, then `None` is returned.
834    pub fn to_computed_color(&self, context: Option<&Context>) -> Result<ComputedColor, ()> {
835        macro_rules! adjust_absolute_color {
836            ($color:expr) => {{
837                // Computed lightness values can not be NaN.
838                if matches!(
839                    $color.color_space,
840                    ColorSpace::Lab | ColorSpace::Oklab | ColorSpace::Lch | ColorSpace::Oklch
841                ) {
842                    $color.components.0 = normalize($color.components.0);
843                }
844
845                // Computed RGB and XYZ components can not be NaN.
846                if !$color.is_legacy_syntax() && $color.color_space.is_rgb_or_xyz_like() {
847                    $color.components = $color.components.map(normalize);
848                }
849
850                $color.alpha = normalize($color.alpha);
851            }};
852        }
853
854        Ok(match *self {
855            Color::CurrentColor => ComputedColor::CurrentColor,
856            Color::Absolute(ref absolute) => {
857                let mut color = absolute.color;
858                adjust_absolute_color!(color);
859                ComputedColor::Absolute(color)
860            },
861            Color::ColorFunction(ref color_function) => {
862                color_function.to_computed_color(context)?
863            },
864            Color::LightDark(ref ld) => ld.compute(context.ok_or(())?),
865            Color::ColorMix(ref mix) => {
866                let mut items = ColorMixItemList::with_capacity(mix.items.len());
867                for item in mix.items.iter() {
868                    items.push(GenericColorMixItem {
869                        color: item.color.to_computed_color(context)?,
870                        percentage: match item.percentage.as_ref() {
871                            None => Optional::None,
872                            Some(percentage) => Optional::Some(match context {
873                                None => ComputedPercentage(percentage.to_percentage().ok_or(())?),
874                                Some(ctx) => percentage.to_computed_value(ctx),
875                            }),
876                        },
877                    });
878                }
879
880                ComputedColor::from_color_mix(GenericColorMix {
881                    interpolation: mix.interpolation,
882                    items: OwnedSlice::from_slice(items.as_slice()),
883                    flags: mix.flags,
884                })
885            },
886            Color::ContrastColor(ref c) => {
887                let computed = c.to_computed_color(context)?;
888                if let Some(abs) = computed.as_absolute() {
889                    ComputedColor::Absolute(ComputedColor::resolve_contrast_color(abs))
890                } else {
891                    ComputedColor::ContrastColor(Box::new(computed))
892                }
893            },
894            Color::System(system) => system.compute(context.ok_or(())?),
895            Color::InheritFromBodyQuirk => {
896                ComputedColor::Absolute(context.ok_or(())?.device().body_text_color())
897            },
898        })
899    }
900}

927            },
928            ComputedColor::ContrastColor(ref c) => {

1071impl Parse for ColorScheme {
1072    fn parse(_: &ParserContext, input: &mut Parser) -> Result<Self, ParseError> {
1073        let mut idents = vec![];
1074        let mut bits = ColorSchemeFlags::empty();
1075
1076        while let Ok(ident) = input.try_parse(|i| i.expect_ident_cloned()) {
1077            let mut is_only = false;
1078            match_ignore_ascii_case! { &ident,
1079                "normal" => {
1080                    if idents.is_empty() && bits.is_empty() {
1081                        return Ok(Self::normal());
1082                    }
1083                    return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
1084                },
1085                "light" => bits.insert(ColorSchemeFlags::LIGHT),
1086                "dark" => bits.insert(ColorSchemeFlags::DARK),
1087                "only" => {
1088                    if bits.intersects(ColorSchemeFlags::ONLY) {
1089                        return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
1090                    }
1091                    bits.insert(ColorSchemeFlags::ONLY);
1092                    is_only = true;
1093                },
1094                _ => {},
1095            };
1096
1097            if is_only {
1098                if !idents.is_empty() {
1099                    // Only is allowed either at the beginning or at the end,
1100                    // but not in the middle.
1101                    break;
1102                }
1103            } else {
1104                idents.push(CustomIdent::from_ident(&ident, &[])?);
1105            }
1106        }
1107
1108        if idents.is_empty() {
1109            return Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError));
1110        }
1111
1112        Ok(Self {
1113            idents: crate::ArcSlice::from_iter(idents.into_iter()),
1114            bits,
1115        })
1116    }
1117}

1119impl ToCss for ColorScheme {
1120    fn to_css<W>(&self, dest: &mut CssWriter<W>) -> fmt::Result
1121    where
1122        W: Write,
1123    {
1124        if self.idents.is_empty() {
1125            debug_assert!(self.bits.is_empty());
1126            return dest.write_str("normal");
1127        }
1128        let mut first = true;
1129        for ident in self.idents.iter() {
1130            if !first {
1131                dest.write_char(' ')?;
1132            }
1133            first = false;
1134            ident.to_css(dest)?;
1135        }
1136        if self.bits.intersects(ColorSchemeFlags::ONLY) {
1137            dest.write_str(" only")?;
1138        }
1139        Ok(())
1140    }

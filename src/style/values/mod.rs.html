37/// A CSS float value.
38pub type CSSFloat = f32;
39
40/// Normalizes a float value to zero after a set of operations that might turn
41/// it into NaN.
42#[inline]
43pub fn normalize(v: CSSFloat) -> CSSFloat {
44    if v.is_nan() { 0.0 } else { v }
45}
46
47/// Computes the minimum value of the two floats. The CSS Values and Units definition
48/// for min() considers -0 to be less than +0 (whereas Rust considers them equal).
49/// https://drafts.csswg.org/css-values-4/#css-signed-zero
50#[inline]
51pub fn calc_min(a: CSSFloat, b: CSSFloat) -> CSSFloat {
52    match (a.is_sign_negative(), b.is_sign_negative()) {
53        (true, false) => a,
54        (false, true) => b,
55        _ => a.min(b),
56    }
57}
58
59/// Computes the maximum value of the two floats. The CSS Values and Units definition
60/// for max() considers +0 to be greater than -0 (whereas Rust considers them equal).
61/// https://drafts.csswg.org/css-values-4/#css-signed-zero
62#[inline]
63pub fn calc_max(a: CSSFloat, b: CSSFloat) -> CSSFloat {
64    match (a.is_sign_negative(), b.is_sign_negative()) {
65        (true, false) => b,
66        (false, true) => a,
67        _ => a.max(b),
68    }
69}
70
71/// Computes the sign of the given value. The CSS Values and Units definition for
72/// sign() returns +0 or -0 for an input of +0 or -0, respectively (whereas the
73/// Rust f32::signum() function returns +1 or -1, respectively).
74/// https://drafts.csswg.org/css-values-4/#funcdef-sign
75#[inline]
76pub fn calc_sign(value: CSSFloat) -> CSSFloat {
77    if value.is_nan() {
78        f32::NAN
79    } else if value.is_zero() {
80        value
81    } else if value.is_sign_negative() {
82        -1.0
83    } else {
84        1.0
85    }
86}

88/// Generates a random integer given a base value and limit.
89/// https://drafts.csswg.org/css-values-5/#generate-a-random-integer
90pub fn calc_random_integer(base: f32, limit: f32) -> f32 {
91    (base * limit).floor()
92}
93
94/// Evaluates a random() function given a base value and its arguments.
95/// https://drafts.csswg.org/css-values-5/#random-evaluation
96pub fn calc_random(base: f32, min: f32, max: f32, step: Option<f32>) -> f32 {
97    if base.is_nan() || min.is_nan() || max.is_nan() || step.is_some_and(f32::is_nan) {
98        return f32::NAN;
99    }
100
101    // Clamps a <number> into the range of a random base value, using the highest
102    // f32 strictly less than one. The random base value has a half-open range of
103    // [0, 1), but the `fixed <number>` format of <random-key> accepts the closed
104    // range of [0, 1]. So if an author specifies `fixed 1`, this value is used
105    // instead as the random base value.
106    let base = normalize(base.max(0.).min((1.0f32).next_down()));
107
108    // "In random(A, B), if A is infinite, the result is infinite."
109    if min.is_infinite() {
110        return min;
111    }
112
113    // "If A is finite, but the difference between A and B is either infinite or
114    // large enough to be treated as infinite in the user agent, the result is
115    // NaN."
116    if !(max - min).is_finite() {
117        return f32::NAN;
118    }
119
120    // "If the maximum value is less than the minimum value, it behaves as if
121    // it's equal to the minimum value."
122    let max = if max < min { min } else { max };
123    let range = max - min;
124
125    let Some(step) = step else {
126        return min + base * range;
127    };
128
129    // "If C is infinite, the result is A."
130    if step.is_infinite() {
131        return min;
132    }
133
134    // "If C is negative, zero, or positive but close enough to zero that the
135    // range for the step multiplier would be infinite in the user agent, the
136    // step must be ignored."
137    if step <= 0.0 {
138        return min + base * range;
139    }
140
141    // "Let epsilon be step / 1000, or the smallest representable value greater
142    // than zero in the numeric type being used if epsilon would round to zero."
143    let epsilon = match step / 1000.0 {
144        e if e > 0.0 => e,
145        _ => (0.0f32).next_up(),
146    };
147
148    // "Let N be the largest integer such that min + N * step is less than or
149    // equal to max."
150    let mut n = (range / step).floor();
151    if n.is_infinite() {
152        return min + base * range;
153    }
154
155    // "If N produces a value that is not within epsilon of max, but N+1 would
156    // produce a value within epsilon of max, set N to N+1."
157    if (min + n * step - max).abs() > epsilon && (min + (n + 1.0) * step - max).abs() <= epsilon {
158        n += 1.0;
159    }
160
161    // "Let step index be a random integer less than N+1, given R. Let value
162    // be min + step index * step."
163    let step_index = calc_random_integer(base, n + 1.0);
164    let value = min + step_index * step;
165
166    // "If step index is N and value is within epsilon of max, return max."
167    if step_index == n && (value - max).abs() <= epsilon {
168        return max;
169    }
170
171    value
172}

229/// Serialize a specified dimension with unit, calc, and NaN/infinity handling (if enabled)
230pub fn serialize_specified_dimension<W>(
231    v: f32,
232    unit: &str,
233    was_calc: bool,
234    dest: &mut CssWriter<W>,
235) -> fmt::Result
236where
237    W: Write,
238{
239    if was_calc {
240        dest.write_str("calc(")?;
241    }
242
243    if !v.is_finite() {
244        // https://drafts.csswg.org/css-values/#calc-error-constants:
245        // "While not technically numbers, these keywords act as numeric values,
246        // similar to e and pi. Thus to get an infinite length, for example,
247        // requires an expression like calc(infinity * 1px)."
248
249        if v.is_nan() {
250            dest.write_str("NaN")?;
251        } else if v == f32::INFINITY {
252            dest.write_str("infinity")?;
253        } else if v == f32::NEG_INFINITY {
254            dest.write_str("-infinity")?;
255        }
256
257        if !unit.is_empty() {
258            dest.write_str(" * 1")?;
259        }
260    } else {
261        v.to_css(dest)?;
262    }
263
264    dest.write_str(unit)?;
265
266    if was_calc {
267        dest.write_char(')')?;
268    }
269    Ok(())
270}

679impl CustomIdent {
680    /// Parse a <custom-ident>
681    ///
682    /// TODO(zrhoffman, bug 1844501): Use CustomIdent::parse in more places instead of
683    /// CustomIdent::from_ident.
684    pub fn parse(input: &mut Parser, invalid: &[&str]) -> Result<Self, ParseError> {
685        let ident = input.expect_ident()?;
686        CustomIdent::from_ident(ident, invalid)
687    }
688
689    /// Parse an already-tokenizer identifier
690    pub fn from_ident<'i>(ident: &CowRcStr<'i>, excluding: &[&str]) -> Result<Self, ParseError> {
691        if !Self::is_valid(ident, excluding) {
692            return Err(ParseError::custom(SelectorParseErrorKind::UnexpectedIdent));
693        }
694        if excluding.iter().any(|s| ident.eq_ignore_ascii_case(s)) {
695            Err(ParseError::custom(StyleParseErrorKind::UnspecifiedError))
696        } else {
697            Ok(CustomIdent(Atom::from(ident.as_ref())))
698        }
699    }
700
701    fn is_valid(ident: &str, excluding: &[&str]) -> bool {
702        use crate::properties::CSSWideKeyword;
703        // https://drafts.csswg.org/css-values-4/#custom-idents:
704        //
705        //     The CSS-wide keywords are not valid <custom-ident>s. The default
706        //     keyword is reserved and is also not a valid <custom-ident>.
707        if CSSWideKeyword::from_ident(ident).is_ok() || ident.eq_ignore_ascii_case("default") {
708            return false;
709        }
710
711        // https://drafts.csswg.org/css-values-4/#custom-idents:
712        //
713        //     Excluded keywords are excluded in all ASCII case permutations.
714        !excluding.iter().any(|s| ident.eq_ignore_ascii_case(s))
715    }
716}

719    fn to_css<W>(&self, dest: &mut CssWriter<W>) -> fmt::Result
720    where
721        W: Write,
722    {
723        serialize_atom_identifier(&self.0, dest)
724    }
725}

736/// <https://www.w3.org/TR/css-values-4/#dashed-idents>
737/// This is simply an Atom, but will only parse if the identifier starts with "--".
738#[repr(transparent)]
739#[derive(
740    Clone,
741    Debug,
742    Eq,
743    Hash,
744    MallocSizeOf,
745    PartialEq,
746    SpecifiedValueInfo,
747    ToAnimatedValue,
748    ToComputedValue,
749    ToResolvedValue,
750    ToShmem,
751    Serialize,
752    Deserialize,
753)]
754pub struct DashedIdent(pub Atom);
755
756impl DashedIdent {
757    /// Parse an already-tokenizer identifier
758    pub fn from_ident<'i>(ident: &CowRcStr<'i>) -> Result<Self, ParseError> {
759        if !ident.starts_with("--") {
760            return Err(ParseError::custom(SelectorParseErrorKind::UnexpectedIdent));
761        }
762        Ok(Self(Atom::from(ident.as_ref())))
763    }
764
765    /// Special value for internal use. Useful where we can't use Option<>.
766    pub fn empty() -> Self {
767        Self(atom!(""))
768    }
769
770    /// Check for special internal value.
771    pub fn is_empty(&self) -> bool {
772        self.0 == atom!("")
773    }
774
775    /// Returns an atom with the same value, but without the starting "--".
776    ///
777    /// # Panics
778    ///
779    /// Panics when used on the special `DashedIdent::empty()`.
780    pub(crate) fn undashed(&self) -> Atom {
781        assert!(!self.is_empty(), "Can't undash the empty DashedIdent");
782        #[cfg(feature = "gecko")]
783        let name = &self.0.as_slice()[2..];
784        #[cfg(feature = "servo")]
785        let name = &self.0[2..];
786        Atom::from(name)
787    }
788}
789
790impl IsTreeScoped for DashedIdent {
791    fn is_tree_scoped(&self) -> bool {
792        !self.is_empty()
793    }
794}
795
796impl Parse for DashedIdent {
797    fn parse(_: &ParserContext, input: &mut Parser) -> Result<Self, ParseError> {
798        let ident = input.expect_ident()?;
799        Self::from_ident(ident)
800    }
801}

809    }

812/// The <keyframes-name>.
813///
814/// <https://drafts.csswg.org/css-animations/#typedef-keyframes-name>
815///
816/// We use a single atom for this. Empty atom represents `none` animation.
817#[repr(transparent)]
818#[derive(
819    Clone,
820    Debug,
821    Eq,
822    Hash,
823    PartialEq,
824    MallocSizeOf,
825    SpecifiedValueInfo,
826    ToComputedValue,
827    ToResolvedValue,
828    ToShmem,
829)]
830pub struct KeyframesName(Atom);
831
832impl KeyframesName {
833    /// <https://drafts.csswg.org/css-animations/#dom-csskeyframesrule-name>
834    pub fn from_ident(value: &str) -> Self {
835        Self(Atom::from(value))
836    }
837
838    /// Returns the `none` value.
839    pub fn none() -> Self {
840        Self(atom!(""))
841    }
842
843    /// Returns whether this is the special `none` value.
844    pub fn is_none(&self) -> bool {
845        self.0 == atom!("")
846    }
847
848    /// Create a new KeyframesName from Atom.
849    #[cfg(feature = "gecko")]
850    pub fn from_atom(atom: Atom) -> Self {
851        Self(atom)
852    }
853
854    /// The name as an Atom
855    pub fn as_atom(&self) -> &Atom {
856        &self.0
857    }
858}
859
860impl Parse for KeyframesName {
861    fn parse(_: &ParserContext, input: &mut Parser) -> Result<Self, ParseError> {
862        Ok(match *input.next()? {
863            Token::Ident(ref s) => Self(CustomIdent::from_ident(s, &["none"])?.0),
864            // Note that empty <string> should be rejected.
865            Token::QuotedString(ref s) if !s.as_ref().is_empty() => Self(Atom::from(s.as_ref())),
866            _ => return Err(ParseError::unexpected_token()),
867        })
868    }
869}
870
871impl ToCss for KeyframesName {
872    fn to_css<W>(&self, dest: &mut CssWriter<W>) -> fmt::Result
873    where
874        W: Write,
875    {
876        if self.is_none() {
877            return dest.write_str("none");
878        }
879
880        fn serialize<W: Write>(string: &str, dest: &mut CssWriter<W>) -> fmt::Result {
881            if CustomIdent::is_valid(string, &["none"]) {
882                serialize_identifier(string, dest)
883            } else {
884                string.to_css(dest)
885            }
886        }
887
888        #[cfg(feature = "gecko")]
889        return self.0.with_str(|s| serialize(s, dest));
890
891        #[cfg(feature = "servo")]
892        return serialize(self.0.as_ref(), dest);
893    }
894}

896impl ToTyped for KeyframesName {
897    fn to_typed(&self, dest: &mut ThinVec<TypedValue>) -> Result<(), ()> {
898        let s = ToCss::to_css_cssstring(self);
899        dest.push(TypedValue::Keyword(KeywordValue(s)));
900        Ok(())
901    }

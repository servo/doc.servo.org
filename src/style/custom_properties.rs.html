199impl CssEnvironment {
200    /// Get an env() variable, either custom or not.
201    #[inline]
202    pub fn get(
203        &self,
204        name: &Atom,
205        device: &Device,
206        url_data: &UrlExtraData,
207    ) -> Option<VariableValue> {
208        #[cfg(feature = "gecko")]
209        let is_link_parameter = name.as_slice().starts_with(&[b'-' as u16, b'-' as u16]);
210        #[cfg(feature = "servo")]
211        let is_link_parameter = name.starts_with("--");
212        if is_link_parameter {
213            let param = device
214                .link_parameters()?
215                .0
216                .iter()
217                .find(|p| p.name.0 == *name)?;
218            if let LinkParamValueOrNone::Specified(val) = &param.value {
219                let mut input = cssparser::ParserInput::new(val.as_ref());
220                let mut parser = cssparser::Parser::new(&mut input);
221
222                // need to carry around full variable value https://bugzilla.mozilla.org/show_bug.cgi?id=2028998
223                return VariableValue::parse(&mut parser, None, url_data).ok();
224            }
225            return None;
226        }
227
228        if let Some(var) = ENVIRONMENT_VARIABLES.iter().find(|var| var.name == *name) {
229            return Some((var.evaluator)(device, url_data));
230        }
231        if !url_data.chrome_rules_enabled() {
232            return None;
233        }
234        let var = CHROME_ENVIRONMENT_VARIABLES
235            .iter()
236            .find(|var| var.name == *name)?;
237        Some((var.evaluator)(device, url_data))
238    }
239}

358/// Reify a slice of the CSS string into UnparsedSegment entries.
359///
360/// References are stored in source order, with outer substitution functions
361/// inserted before references in their fallback. The shared `reference_index`
362/// relies on this ordering to recurse into fallbacks without reprocessing
363/// nested referecences.
364fn reify_variable_value_range(
365    css: &str,
366    references: &[SubstitutionFunctionReference],
367    reference_index: &mut usize,
368    start: usize,
369    end: usize,
370) -> Result<UnparsedValue, ()> {
371    debug_assert!(start <= end);
372    debug_assert!(end <= css.len());
373
374    let mut values = ThinVec::new();
375    let mut cur_pos = start;
376
377    while *reference_index < references.len() {
378        let reference = &references[*reference_index];
379
380        if reference.start >= end {
381            break;
382        }
383
384        debug_assert!(reference.start >= cur_pos);
385        debug_assert!(reference.start <= reference.end);
386        debug_assert!(reference.end <= css.len());
387
388        if cur_pos < reference.start {
389            values.push(UnparsedSegment::String(CssString::from(
390                &css[cur_pos..reference.start],
391            )));
392        }
393
394        *reference_index += 1;
395
396        if reference.substitution_kind != SubstitutionFunctionKind::Var {
397            return Err(());
398        }
399
400        let (fallback, has_fallback) = if let Some(fallback) = &reference.fallback {
401            debug_assert!(fallback.start.get() <= reference.end - 1);
402
403            (
404                reify_variable_value_range(
405                    css,
406                    references,
407                    reference_index,
408                    fallback.start.get(),
409                    reference.end - 1, // Skip the closing ')'.
410                )?,
411                true,
412            )
413        } else {
414            (ThinVec::new(), false)
415        };
416
417        values.push(UnparsedSegment::VariableReference(VariableReferenceValue {
418            variable: CssString::from(format!("--{}", reference.name)),
419            fallback,
420            has_fallback,
421        }));
422
423        cur_pos = reference.end;
424    }
425
426    if cur_pos < end {
427        values.push(UnparsedSegment::String(CssString::from(&css[cur_pos..end])));
428    }
429
430    Ok(values)
431}

433/// A pair of separate CustomPropertiesMaps, split between custom properties
434/// that have the inherit flag set and those with the flag unset.
435#[repr(C)]
436#[derive(Clone, Debug, Default, PartialEq)]
437pub struct ComputedCustomProperties {
438    /// Map for custom properties with inherit flag set, including non-registered
439    /// ones.
440    pub inherited: CustomPropertiesMap,
441    /// Map for custom properties with inherit flag unset.
442    pub non_inherited: CustomPropertiesMap,
443}
444
445impl ComputedCustomProperties {
446    /// Return whether the inherited and non_inherited maps are none.
447    pub fn is_empty(&self) -> bool {
448        self.inherited.is_empty() && self.non_inherited.is_empty()
449    }
450
451    /// Return the name and value of the property at specified index, if any.
452    pub fn property_at(&self, index: usize) -> Option<(&Name, &Option<ComputedRegisteredValue>)> {
453        // Just expose the custom property items from custom_properties.inherited, followed
454        // by custom property items from custom_properties.non_inherited.
455        self.inherited
456            .get_index(index)
457            .or_else(|| self.non_inherited.get_index(index - self.inherited.len()))
458    }
459
460    /// Insert a custom property in the corresponding inherited/non_inherited
461    /// map, depending on whether the inherit flag is set or unset.
462    pub fn insert(
463        &mut self,
464        registration: &PropertyDescriptors,
465        name: &Name,
466        value: ComputedRegisteredValue,
467    ) {
468        self.map_mut(registration).insert(name, value)
469    }
470
471    /// Remove a custom property from the corresponding inherited/non_inherited
472    /// map, depending on whether the inherit flag is set or unset.
473    pub fn remove(&mut self, registration: &PropertyDescriptors, name: &Name) {
474        self.map_mut(registration).remove(name);
475    }
476
477    /// Shrink the capacity of the inherited maps as much as possible.
478    pub fn shrink_to_fit(&mut self) {
479        self.inherited.shrink_to_fit();
480        self.non_inherited.shrink_to_fit();
481    }
482
483    fn map_mut(&mut self, registration: &PropertyDescriptors) -> &mut CustomPropertiesMap {
484        if registration.inherits() {
485            &mut self.inherited
486        } else {
487            &mut self.non_inherited
488        }
489    }
490
491    /// Returns the relevant custom property value given a registration.
492    pub fn get(
493        &self,
494        registration: &PropertyDescriptors,
495        name: &Name,
496    ) -> Option<&ComputedRegisteredValue> {
497        if registration.inherits() {
498            self.inherited.get(name)
499        } else {
500            self.non_inherited.get(name)
501        }
502    }
503
504    /// Returns a property just by the name. Slightly less efficient than get(), if you already have
505    /// a custom registration handy, thus the longer name.
506    pub fn get_for_cssom(&self, name: &Name) -> Option<&ComputedRegisteredValue> {
507        self.inherited
508            .get(name)
509            .or_else(|| self.non_inherited.get(name))
510    }
511}

520/// Set of flags to references this custom property makes.
521#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, MallocSizeOf, ToShmem)]
522pub struct ReferenceFlags(u8);
523
524bitflags! {
525    impl ReferenceFlags : u8 {
526        /// At least one custom property depends on font-relative units.
527        const FONT_UNITS = 1 << 0;
528        /// At least one custom property depends on root element's font-relative units.
529        const ROOT_FONT_UNITS = 1 << 1;
530        /// At least one custom property depends on line height units.
531        const LH_UNITS = 1 << 2;
532        /// At least one custom property depends on root element's line height units.
533        const ROOT_LH_UNITS = 1 << 3;
534        /// The value depends on the used color-scheme (e.g. a registered `<color>` property).
535        const COLOR_SCHEME = 1 << 4;
536        /// All dependencies not depending on the root element.
537        const NON_ROOT_DEPENDENCIES = Self::FONT_UNITS.0 | Self::LH_UNITS.0;
538        /// All dependencies depending on the root element.
539        const ROOT_DEPENDENCIES = Self::ROOT_FONT_UNITS.0 | Self::ROOT_LH_UNITS.0;
540        /// All non-custom dependencies
541        const NON_CUSTOM = Self::NON_ROOT_DEPENDENCIES.0 | Self::ROOT_DEPENDENCIES.0;
542        /// At least one attr() reference.
543        const ATTR = 1 << 5;
544        /// At least one env() reference.
545        const ENV = 1 << 6;
546        /// At least one var() reference.
547        const VAR = 1 << 7;
548    }
549}
550
551impl ReferenceFlags {
552    /// Iterate for each non custom
553    pub fn for_each_non_custom<F>(mut self, is_root_element: bool, mut f: F)
554    where
555        F: FnMut(SingleNonCustomReference),
556    {
557        // On the root element, rem etc refer to the root's own dependencies.
558        if is_root_element {
559            if self.intersects(Self::ROOT_FONT_UNITS) {
560                self.remove(Self::ROOT_FONT_UNITS);
561                self |= Self::FONT_UNITS;
562            }
563            if self.intersects(Self::ROOT_LH_UNITS) {
564                self.remove(Self::ROOT_FONT_UNITS);
565                self |= Self::LH_UNITS;
566            }
567        }
568
569        for (_, r) in self.iter_names() {
570            let single = match r {
571                Self::FONT_UNITS => SingleNonCustomReference::FontUnits,
572                Self::LH_UNITS => SingleNonCustomReference::LhUnits,
573                Self::COLOR_SCHEME => SingleNonCustomReference::ColorScheme,
574                Self::ROOT_FONT_UNITS
575                | Self::ROOT_LH_UNITS
576                | Self::VAR
577                | Self::ENV
578                | Self::ATTR => continue,
579                _ => unreachable!("Unexpected single bit value"),
580            };
581            f(single);
582        }
583    }
584
585    fn from_unit(value: &CowRcStr) -> Self {
586        // For registered properties, any reference to font-relative dimensions
587        // make it dependent on font-related properties.
588        // TODO(dshin): When we unit algebra gets implemented and handled -
589        // Is it valid to say that `calc(1em / 2em * 3px)` triggers this?
590        if value.eq_ignore_ascii_case(NoCalcLength::LH) {
591            return Self::FONT_UNITS | Self::LH_UNITS;
592        }
593        if value.eq_ignore_ascii_case(NoCalcLength::EM)
594            || value.eq_ignore_ascii_case(NoCalcLength::EX)
595            || value.eq_ignore_ascii_case(NoCalcLength::CAP)
596            || value.eq_ignore_ascii_case(NoCalcLength::CH)
597            || value.eq_ignore_ascii_case(NoCalcLength::IC)
598        {
599            return Self::FONT_UNITS;
600        }
601        if value.eq_ignore_ascii_case(NoCalcLength::RLH) {
602            return Self::ROOT_FONT_UNITS | Self::ROOT_LH_UNITS;
603        }
604        if value.eq_ignore_ascii_case(NoCalcLength::REM)
605            || value.eq_ignore_ascii_case(NoCalcLength::REX)
606            || value.eq_ignore_ascii_case(NoCalcLength::RCH)
607            || value.eq_ignore_ascii_case(NoCalcLength::RCAP)
608            || value.eq_ignore_ascii_case(NoCalcLength::RIC)
609        {
610            return Self::ROOT_FONT_UNITS;
611        }
612        Self::empty()
613    }
614}

770/// In CSS Values and Units, values produced by `attr()` are considered attr()-tainted, as are
771/// functions that contain an attr()-tainted value. Using an attr()-tainted value as or in a <url>
772/// makes a declaration invalid at computed-value time.
773/// https://drafts.csswg.org/css-values-5/#attr-security
774#[derive(Clone, Debug, Default, MallocSizeOf, PartialEq, ToShmem)]
775pub struct AttrTaint(SmallVec<[AttrTaintedRange; 1]>);
776
777impl AttrTaint {
778    /// For a CSS string, determine whether any `<url>` overlapping this `range`
779    /// is disallowed due to attr()-tainting.
780    #[inline(always)]
781    pub fn should_disallow_urls_in_range(&self, range: &AttrTaintedRange) -> bool {
782        self.0
783            .iter()
784            .any(|r| r.start <= range.end && r.end >= range.start)
785    }
786
787    /// Returns true if the attr()-tainted range contains no elements.
788    #[inline(always)]
789    pub fn is_empty(&self) -> bool {
790        self.0.is_empty()
791    }
792
793    #[inline(always)]
794    fn new_fully_tainted(end: usize) -> Self {
795        let mut taint = Self::default();
796        taint.push(0, end);
797        taint
798    }
799
800    #[inline(always)]
801    fn push(&mut self, start: usize, end: usize) {
802        self.0.push(AttrTaintedRange::new(start, end));
803    }
804}

899    fn push<'i>(
900        &mut self,
901        css: &str,
902        css_first_token_type: TokenSerializationType,
903        css_last_token_type: TokenSerializationType,
904        attr_taint: Option<&mut AttrTaint>,
905    ) -> Result<(), ()> {
906        /// Prevent values from getting terribly big since you can use custom
907        /// properties exponentially.
908        ///
909        /// This number (2MB) is somewhat arbitrary, but silly enough that no
910        /// reasonable page should hit it. We could limit by number of total
911        /// substitutions, but that was very easy to work around in practice
912        /// (just choose a larger initial value and boom).
913        const MAX_VALUE_LENGTH_IN_BYTES: usize = 2 * 1024 * 1024;
914
915        if self.css.len() + css.len() > MAX_VALUE_LENGTH_IN_BYTES {
916            return Err(());
917        }
918
919        // This happens e.g. between two subsequent var() functions:
920        // `var(--a)var(--b)`.
921        //
922        // In that case, css_*_token_type is nonsensical.
923        if css.is_empty() {
924            return Ok(());
925        }
926
927        self.first_token_type.set_if_nothing(css_first_token_type);
928        // If self.first_token_type was nothing,
929        // self.last_token_type is also nothing and this will be false:
930        if self
931            .last_token_type
932            .needs_separator_when_before(css_first_token_type)
933        {
934            self.css.push_str("/**/")
935        }
936        let start = self.css.len();
937        self.css.push_str(css);
938        let end = self.css.len();
939        if let Some(taint) = attr_taint {
940            taint.push(start, end);
941        }
942        self.last_token_type = css_last_token_type;
943        Ok(())
944    }

946    /// Parse a custom property value.
947    pub fn parse<'i, 't>(
948        input: &mut Parser<'i, 't>,
949        namespaces: Option<&FxHashMap<Prefix, Namespace>>,
950        url_data: &UrlExtraData,
951    ) -> Result<Self, ParseError<'i>> {
952        let mut references = References::default();
953        let mut missing_closing_characters = String::new();
954        let start_position = input.position();
955        let (first_token_type, last_token_type) = parse_declaration_value(
956            input,
957            start_position,
958            namespaces,
959            &mut references,
960            &mut missing_closing_characters,
961        )?;
962        let mut css = input
963            .slice_from(start_position)
964            .trim_ascii_start()
965            .to_owned();
966        if !missing_closing_characters.is_empty() {
967            // Unescaped backslash at EOF in a quoted string is ignored.
968            if css.ends_with("\\")
969                && matches!(missing_closing_characters.as_bytes()[0], b'"' | b'\'')
970            {
971                css.pop();
972            }
973            css.push_str(&missing_closing_characters);
974        }
975
976        css.truncate(css.trim_ascii_end().len());
977        css.shrink_to_fit();
978        references.refs.shrink_to_fit();
979
980        Ok(Self {
981            css,
982            url_data: url_data.clone(),
983            first_token_type,
984            last_token_type,
985            references,
986        })
987    }

1116/// Like parse_declaration_value, but accept `!` and `;` since they are only invalid at the top level.
1117fn parse_declaration_value_block<'i, 't>(
1118    input: &mut Parser<'i, 't>,
1119    input_start: SourcePosition,
1120    namespaces: Option<&FxHashMap<Prefix, Namespace>>,
1121    references: &mut References,
1122    missing_closing_characters: &mut String,
1123) -> Result<(TokenSerializationType, TokenSerializationType), ParseError<'i>> {
1124    let mut is_first = true;
1125    let mut first_token_type = TokenSerializationType::Nothing;
1126    let mut last_token_type = TokenSerializationType::Nothing;
1127    let mut prev_reference_index: Option<usize> = None;
1128    loop {
1129        let token_start = input.position();
1130        let Ok(token) = input.next_including_whitespace_and_comments() else {
1131            break;
1132        };
1133
1134        let prev_token_type = last_token_type;
1135        let serialization_type = token.serialization_type();
1136        last_token_type = serialization_type;
1137        if is_first {
1138            first_token_type = last_token_type;
1139            is_first = false;
1140        }
1141
1142        macro_rules! nested {
1143            ($closing:expr) => {{
1144                let mut inner_end_position = None;
1145                let result = input.parse_nested_block(|input| {
1146                    let result = parse_declaration_value_block(
1147                        input,
1148                        input_start,
1149                        namespaces,
1150                        references,
1151                        missing_closing_characters,
1152                    )?;
1153                    inner_end_position = Some(input.position());
1154                    Ok(result)
1155                })?;
1156                if inner_end_position.unwrap() == input.position() {
1157                    missing_closing_characters.push_str($closing);
1158                }
1159                result
1160            }};
1161        }
1162        if let Some(index) = prev_reference_index.take() {
1163            references.refs[index].next_token_type = serialization_type;
1164        }
1165        match *token {
1166            Token::Comment(_) => {
1167                let token_slice = input.slice_from(token_start);
1168                if !token_slice.ends_with("*/") {
1169                    missing_closing_characters.push_str(if token_slice.ends_with('*') {
1170                        "/"
1171                    } else {
1172                        "*/"
1173                    })
1174                }
1175            },
1176            Token::BadUrl(ref u) => {
1177                let e = StyleParseErrorKind::BadUrlInDeclarationValueBlock(u.clone());
1178                return Err(input.new_custom_error(e));
1179            },
1180            Token::BadString(ref s) => {
1181                let e = StyleParseErrorKind::BadStringInDeclarationValueBlock(s.clone());
1182                return Err(input.new_custom_error(e));
1183            },
1184            Token::CloseParenthesis => {
1185                let e = StyleParseErrorKind::UnbalancedCloseParenthesisInDeclarationValueBlock;
1186                return Err(input.new_custom_error(e));
1187            },
1188            Token::CloseSquareBracket => {
1189                let e = StyleParseErrorKind::UnbalancedCloseSquareBracketInDeclarationValueBlock;
1190                return Err(input.new_custom_error(e));
1191            },
1192            Token::CloseCurlyBracket => {
1193                let e = StyleParseErrorKind::UnbalancedCloseCurlyBracketInDeclarationValueBlock;
1194                return Err(input.new_custom_error(e));
1195            },
1196            Token::Function(ref name) => {
1197                let substitution_kind = match SubstitutionFunctionKind::from_ident(name).ok() {
1198                    Some(SubstitutionFunctionKind::Attr) => {
1199                        if static_prefs::pref!("layout.css.attr.enabled") {
1200                            Some(SubstitutionFunctionKind::Attr)
1201                        } else {
1202                            None
1203                        }
1204                    },
1205                    kind => kind,
1206                };
1207                if let Some(substitution_kind) = substitution_kind {
1208                    let our_ref_index = references.refs.len();
1209                    let mut input_end_position = None;
1210                    let fallback = input.parse_nested_block(|input| {
1211                        let mut namespace = ParsedNamespace::Known(Namespace::default());
1212                        if substitution_kind == SubstitutionFunctionKind::Attr {
1213                            if let Some(namespaces) = namespaces {
1214                                if let Ok(ns) = input
1215                                    .try_parse(|input| ParsedNamespace::parse(namespaces, input))
1216                                {
1217                                    namespace = ns;
1218                                    let prev = input.state();
1219                                    let next = match *input.next_including_whitespace()? {
1220                                        Token::Ident(_) => Ok(()),
1221                                        ref t => Err(prev
1222                                            .source_location()
1223                                            .new_unexpected_token_error(t.clone())),
1224                                    };
1225                                    input.reset(&prev);
1226                                    next?;
1227                                }
1228                            }
1229                        }
1230                        // TODO(emilio): For env() this should be <custom-ident> per spec, but no other browser does
1231                        // that, see https://github.com/w3c/csswg-drafts/issues/3262.
1232                        let name = input.expect_ident()?;
1233                        let name =
1234                            Atom::from(if substitution_kind == SubstitutionFunctionKind::Var {
1235                                match parse_name(name.as_ref()) {
1236                                    Ok(name) => name,
1237                                    Err(()) => {
1238                                        let name = name.clone();
1239                                        return Err(input.new_custom_error(
1240                                            SelectorParseErrorKind::UnexpectedIdent(name),
1241                                        ));
1242                                    },
1243                                }
1244                            } else {
1245                                name.as_ref()
1246                            });
1247
1248                        let attribute_kind = if substitution_kind == SubstitutionFunctionKind::Attr
1249                        {
1250                            parse_attr_type(input)
1251                        } else {
1252                            AttributeType::None
1253                        };
1254
1255                        // We want the order of the references to match source order. So we need to reserve our slot
1256                        // now, _before_ parsing our fallback. Note that we don't care if parsing fails after all, since
1257                        // if this fails we discard the whole result anyways.
1258                        let start = token_start.byte_index() - input_start.byte_index();
1259                        references.refs.push(SubstitutionFunctionReference {
1260                            name,
1261                            start,
1262                            // To be fixed up after parsing fallback and auto-closing via our_ref_index.
1263                            end: start,
1264                            prev_token_type,
1265                            // To be fixed up (if needed) on the next loop iteration via prev_reference_index.
1266                            next_token_type: TokenSerializationType::Nothing,
1267                            // To be fixed up after parsing fallback.
1268                            fallback: None,
1269                            attribute_data: AttributeData {
1270                                kind: attribute_kind,
1271                                namespace,
1272                            },
1273                            substitution_kind: substitution_kind.clone(),
1274                        });
1275
1276                        let mut fallback = None;
1277                        if input.try_parse(|input| input.expect_comma()).is_ok() {
1278                            input.skip_whitespace();
1279                            let fallback_start = num::NonZeroUsize::new(
1280                                input.position().byte_index() - input_start.byte_index(),
1281                            )
1282                            .unwrap();
1283                            let mut references = References::default();
1284                            // NOTE(emilio): Intentionally using parse_declaration_value rather than
1285                            // parse_declaration_value_block, since that's what parse_fallback used to do.
1286                            let (first, last) = parse_declaration_value(
1287                                input,
1288                                input_start,
1289                                namespaces,
1290                                &mut references,
1291                                missing_closing_characters,
1292                            )?;
1293                            fallback = Some(VariableFallback {
1294                                start: fallback_start,
1295                                first_token_type: first,
1296                                last_token_type: last,
1297                                references,
1298                            });
1299                            input_end_position = Some(input.position());
1300                        } else {
1301                            let state = input.state();
1302                            // We still need to consume the rest of the potentially-unclosed
1303                            // tokens, but make sure to not consume tokens that would otherwise be
1304                            // invalid, by calling reset().
1305                            parse_declaration_value_block(
1306                                input,
1307                                input_start,
1308                                namespaces,
1309                                references,
1310                                missing_closing_characters,
1311                            )?;
1312                            input_end_position = Some(input.position());
1313                            input.reset(&state);
1314                        }
1315                        Ok(fallback)
1316                    })?;
1317                    if input_end_position.unwrap() == input.position() {
1318                        missing_closing_characters.push_str(")");
1319                    }
1320                    prev_reference_index = Some(our_ref_index);
1321                    let reference = &mut references.refs[our_ref_index];
1322                    reference.end = input.position().byte_index() - input_start.byte_index()
1323                        + missing_closing_characters.len();
1324                    reference.fallback = fallback;
1325                    references.flags |= match substitution_kind {
1326                        SubstitutionFunctionKind::Var => ReferenceFlags::VAR,
1327                        SubstitutionFunctionKind::Env => ReferenceFlags::ENV,
1328                        SubstitutionFunctionKind::Attr => ReferenceFlags::ATTR,
1329                    };
1330                    // Bubble up flags from our fallback, so we know what we might reference from the outer scope.
1331                    if let Some(ref fb) = reference.fallback {
1332                        references.flags |= fb.references.flags;
1333                    }
1334                } else {
1335                    nested!(")");
1336                }
1337            },
1338            Token::ParenthesisBlock => {
1339                nested!(")");
1340            },
1341            Token::CurlyBracketBlock => {
1342                nested!("}");
1343            },
1344            Token::SquareBracketBlock => {
1345                nested!("]");
1346            },
1347            Token::QuotedString(_) => {
1348                let token_slice = input.slice_from(token_start);
1349                let quote = &token_slice[..1];
1350                debug_assert!(matches!(quote, "\"" | "'"));
1351                if !(token_slice.ends_with(quote) && token_slice.len() > 1) {
1352                    missing_closing_characters.push_str(quote)
1353                }
1354            },
1355            Token::Ident(ref value)
1356            | Token::AtKeyword(ref value)
1357            | Token::Hash(ref value)
1358            | Token::IDHash(ref value)
1359            | Token::UnquotedUrl(ref value)
1360            | Token::Dimension {
1361                unit: ref value, ..
1362            } => {
1363                references.flags.insert(ReferenceFlags::from_unit(value));
1364                let is_unquoted_url = matches!(token, Token::UnquotedUrl(_));
1365                if value.ends_with("�") && input.slice_from(token_start).ends_with("\\") {
1366                    // Unescaped backslash at EOF in these contexts is interpreted as U+FFFD
1367                    // Check the value in case the final backslash was itself escaped.
1368                    // Serialize as escaped U+FFFD, which is also interpreted as U+FFFD.
1369                    // (Unescaped U+FFFD would also work, but removing the backslash is annoying.)
1370                    missing_closing_characters.push_str("�")
1371                }
1372                if is_unquoted_url && !input.slice_from(token_start).ends_with(")") {
1373                    missing_closing_characters.push_str(")");
1374                }
1375            },
1376            _ => {},
1377        };
1378    }
1379    Ok((first_token_type, last_token_type))
1380}

1382/// Parse <attr-type> = type( <syntax> ) | raw-string | number | <attr-unit>.
1383/// https://drafts.csswg.org/css-values-5/#attr-notation
1384fn parse_attr_type<'i, 't>(input: &mut Parser<'i, 't>) -> AttributeType {
1385    input
1386        .try_parse(|input| {
1387            Ok(match input.next()? {
1388                Token::Function(ref name) if name.eq_ignore_ascii_case("type") => {
1389                    AttributeType::Type(
1390                        input.parse_nested_block(SyntaxDescriptor::from_css_parser)?,
1391                    )
1392                },
1393                Token::Ident(ref ident) => {
1394                    if ident.eq_ignore_ascii_case("raw-string") {
1395                        AttributeType::RawString
1396                    } else if let Ok(unit) = AttrUnit::from_ident(ident) {
1397                        AttributeType::Unit(unit)
1398                    } else {
1399                        AttributeType::Invalid
1400                    }
1401                },
1402                Token::Delim('%') => AttributeType::Unit(AttrUnit::Percentage),
1403                _ => return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError)),
1404            })
1405        })
1406        .unwrap_or(AttributeType::None)
1407}

1409/// Attribute values may reference other substitution functions we may need to process.
1410/// See step 6: https://drafts.csswg.org/css-values-5/#attr-substitution
1411pub fn get_attr_value_for_cycle_resolution(
1412    name: &Name,
1413    attribute_data: &AttributeData,
1414    url_data: &UrlExtraData,
1415    attribute_tracker: &mut AttributeTracker,
1416) -> Result<ComputedRegisteredValue, ()> {
1417    let namespace = match attribute_data.namespace {
1418        ParsedNamespace::Known(ref ns) => ns,
1419        ParsedNamespace::Unknown => return Err(()),
1420    };
1421    let attr = LocalName::with_name(name, |local_name| {
1422        attribute_tracker.query(local_name, namespace).ok_or(())
1423    })?;
1424    let mut input = ParserInput::new(&attr);
1425    let mut parser = Parser::new(&mut input);
1426    // TODO(Bug 2021110): Support namespaced attributes in chained references.
1427    let value = VariableValue::parse(&mut parser, None, &url_data).map_err(|_| ())?;
1428    Ok(ComputedRegisteredValue::universal(Arc::new(value)))
1429}

1431/// See https://drafts.csswg.org/css-variables-2/#invalid-at-computed-value-time
1432pub fn handle_invalid_at_computed_value_time(
1433    name: &Name,
1434    registration: &PropertyDescriptors,
1435    context: &mut computed::Context,
1436) {
1437    if !registration.is_universal() {
1438        // For the root element, inherited maps are empty. We should just
1439        // use the initial value if any, rather than removing the name.
1440        if registration.inherits() && !context.builder.is_root_element {
1441            let inherited = context.builder.inherited_style.custom_properties();
1442            if let Some(value) = inherited.get(registration, name) {
1443                context.builder.substitution_functions.insert_var(
1444                    registration,
1445                    name,
1446                    value.clone(),
1447                );
1448                return;
1449            }
1450        } else if let Some(ref initial_value) = registration.initial_value {
1451            if let Ok(initial_value) = compute_value(
1452                &initial_value.css,
1453                &initial_value.url_data,
1454                registration,
1455                context,
1456                AttrTaint::default(),
1457            ) {
1458                context.builder.substitution_functions.insert_var(
1459                    registration,
1460                    name,
1461                    initial_value,
1462                );
1463                return;
1464            }
1465        }
1466    }
1467    context
1468        .builder
1469        .substitution_functions
1470        .remove_var(registration, name);
1471}

1473/// Replace `var()`, `env()`, and `attr()` functions in a pre-existing variable value.
1474pub fn substitute_references_if_needed_and_apply(
1475    name: &Name,
1476    kind: SubstitutionFunctionKind,
1477    value: &Arc<VariableValue>,
1478    stylist: &Stylist,
1479    context: &mut computed::Context,
1480    attribute_tracker: &mut AttributeTracker,
1481) {
1482    debug_assert_ne!(kind, SubstitutionFunctionKind::Env);
1483    let is_var = matches!(kind, SubstitutionFunctionKind::Var);
1484    let registration = stylist.get_custom_property_registration(&name);
1485    if is_var && !value.has_references() && registration.is_universal() {
1486        // Trivial path: no references and no need to compute the value, just apply it directly.
1487        let computed_value = ComputedRegisteredValue::universal(Arc::clone(value));
1488        context
1489            .builder
1490            .substitution_functions
1491            .insert_var(registration, name, computed_value);
1492        return;
1493    }
1494
1495    let url_data = &value.url_data;
1496    let substitution = substitute_internal(
1497        value,
1498        &context.builder.substitution_functions,
1499        stylist,
1500        context,
1501        attribute_tracker,
1502        &mut SmallVec::new(),
1503        None,
1504    );
1505
1506    let Ok(substitution) = substitution else {
1507        if is_var {
1508            handle_invalid_at_computed_value_time(name, registration, context);
1509        } else {
1510            context.builder.substitution_functions.remove_attr(name);
1511        }
1512        return;
1513    };
1514
1515    // If variable fallback results in a wide keyword, deal with it now.
1516    let inherited = context.builder.inherited_style.custom_properties();
1517    if is_var {
1518        let css = &substitution.css;
1519        let css_wide_kw = {
1520            let mut input = ParserInput::new(&css);
1521            let mut input = Parser::new(&mut input);
1522            input.try_parse(CSSWideKeyword::parse)
1523        };
1524
1525        if let Ok(kw) = css_wide_kw {
1526            // TODO: It's unclear what this should do for revert / revert-layer, see
1527            // https://github.com/w3c/csswg-drafts/issues/9131. For now treating as unset
1528            // seems fine?
1529            match (kw, registration.inherits(), context.is_root_element()) {
1530                (CSSWideKeyword::Initial, _, _)
1531                | (CSSWideKeyword::Revert, false, _)
1532                | (CSSWideKeyword::RevertLayer, false, _)
1533                | (CSSWideKeyword::RevertRule, false, _)
1534                | (CSSWideKeyword::Unset, false, _)
1535                | (CSSWideKeyword::Revert, true, true)
1536                | (CSSWideKeyword::RevertLayer, true, true)
1537                | (CSSWideKeyword::RevertRule, true, true)
1538                | (CSSWideKeyword::Unset, true, true)
1539                | (CSSWideKeyword::Inherit, _, true) => {
1540                    remove_and_insert_initial_value(
1541                        name,
1542                        registration,
1543                        &mut context.builder.substitution_functions,
1544                    );
1545                },
1546                (CSSWideKeyword::Revert, true, false)
1547                | (CSSWideKeyword::RevertLayer, true, false)
1548                | (CSSWideKeyword::RevertRule, true, false)
1549                | (CSSWideKeyword::Inherit, _, false)
1550                | (CSSWideKeyword::Unset, true, false) => {
1551                    match inherited.get(registration, name) {
1552                        Some(value) => {
1553                            context.builder.substitution_functions.insert_var(
1554                                registration,
1555                                name,
1556                                value.clone(),
1557                            );
1558                        },
1559                        None => {
1560                            context
1561                                .builder
1562                                .substitution_functions
1563                                .remove_var(registration, name);
1564                        },
1565                    };
1566                },
1567            }
1568            return;
1569        }
1570    }
1571
1572    match kind {
1573        SubstitutionFunctionKind::Var => {
1574            let value = match substitution.into_value(url_data, registration, context) {
1575                Ok(v) => v,
1576                Err(()) => {
1577                    handle_invalid_at_computed_value_time(name, registration, context);
1578                    return;
1579                },
1580            };
1581            context
1582                .builder
1583                .substitution_functions
1584                .insert_var(registration, name, value);
1585        },
1586        SubstitutionFunctionKind::Attr => {
1587            let mut value = ComputedRegisteredValue::universal(Arc::new(VariableValue::new(
1588                substitution.css.into_owned(),
1589                url_data,
1590                substitution.first_token_type,
1591                substitution.last_token_type,
1592            )));
1593            value.attr_tainted |= substitution.attr_tainted;
1594            context
1595                .builder
1596                .substitution_functions
1597                .insert_attr(name, value);
1598        },
1599        SubstitutionFunctionKind::Env => unreachable!("Kind cannot be env."),
1600    }
1601}

1603#[derive(Default, Debug)]
1604struct Substitution<'a> {
1605    css: Cow<'a, str>,
1606    first_token_type: TokenSerializationType,
1607    last_token_type: TokenSerializationType,
1608    attr_tainted: bool,
1609}
1610
1611impl<'a> Substitution<'a> {
1612    fn from_value(v: VariableValue, attr_tainted: bool) -> Self {
1613        Substitution {
1614            css: v.css.into(),
1615            first_token_type: v.first_token_type,
1616            last_token_type: v.last_token_type,
1617            attr_tainted,
1618        }
1619    }
1620
1621    fn into_value(
1622        self,
1623        url_data: &UrlExtraData,
1624        registration: &PropertyDescriptors,
1625        computed_context: &computed::Context,
1626    ) -> Result<ComputedRegisteredValue, ()> {
1627        if registration.is_universal() {
1628            let mut value = ComputedRegisteredValue::universal(Arc::new(VariableValue::new(
1629                self.css.into_owned(),
1630                url_data,
1631                self.first_token_type,
1632                self.last_token_type,
1633            )));
1634            value.attr_tainted |= self.attr_tainted;
1635            return Ok(value);
1636        }
1637        let taint = if self.attr_tainted {
1638            // Per spec: substitution value of an arbitrary substitution function is
1639            // attr()-tainted as a whole if any attr()-tainted values were involved
1640            // in creating that substitution value.
1641            // https://drafts.csswg.org/css-values-5/#attr-security
1642            AttrTaint::new_fully_tainted(self.css.len())
1643        } else {
1644            AttrTaint::default()
1645        };
1646        let mut v = compute_value(&self.css, url_data, registration, computed_context, taint)?;
1647        v.attr_tainted |= self.attr_tainted;
1648        Ok(v)
1649    }
1650
1651    fn new(
1652        css: Cow<'a, str>,
1653        first_token_type: TokenSerializationType,
1654        last_token_type: TokenSerializationType,
1655        attr_tainted: bool,
1656    ) -> Self {
1657        Self {
1658            css,
1659            first_token_type,
1660            last_token_type,
1661            attr_tainted,
1662        }
1663    }
1664}

1669    /// The resolved CSS string after substitution.
1670    pub css: Cow<'a, str>,
1671    /// Regions in the `css` string that are attr()-tainted, if any.
1672    pub attr_taint: AttrTaint,

1675fn compute_value(
1676    css: &str,
1677    url_data: &UrlExtraData,
1678    registration: &PropertyDescriptors,
1679    computed_context: &computed::Context,
1680    attr_taint: AttrTaint,
1681) -> Result<ComputedRegisteredValue, ()> {
1682    debug_assert!(!registration.is_universal());
1683
1684    let mut input = ParserInput::new(&css);
1685    let mut input = Parser::new(&mut input);

1688        &mut input,
1689        registration,
1690        None,
1691        url_data,
1692        computed_context,

1694        attr_taint,
1695    )

1711fn do_substitute_chunk<'a>(
1712    css: &'a str,
1713    start: usize,
1714    end: usize,
1715    first_token_type: TokenSerializationType,
1716    last_token_type: TokenSerializationType,
1717    url_data: &UrlExtraData,
1718    substitution_functions: &'a ComputedSubstitutionFunctions,
1719    stylist: &Stylist,
1720    computed_context: &computed::Context,
1721    references: &'a [SubstitutionFunctionReference],
1722    attribute_tracker: &mut AttributeTracker,
1723    seen: &mut SmallVec<[&'a Name; 8]>,
1724    mut attr_taint: Option<&mut AttrTaint>,
1725) -> Result<Substitution<'a>, ()> {
1726    if start == end {
1727        // Empty string. Easy.
1728        return Ok(Substitution::default());
1729    }
1730    // Easy case: no references involved.
1731    if references.is_empty() {
1732        let result = &css[start..end];
1733        return Ok(Substitution::new(
1734            Cow::Borrowed(result),
1735            first_token_type,
1736            last_token_type,
1737            Default::default(),
1738        ));
1739    }
1740
1741    let mut substituted = ComputedValue::empty(url_data);
1742    let mut next_token_type = first_token_type;
1743    let mut cur_pos = start;
1744    let mut attr_tainted = false;
1745    let mut references = references.iter();
1746    while let Some(reference) = references.next() {
1747        if reference.start != cur_pos {
1748            substituted.push(
1749                &css[cur_pos..reference.start],
1750                next_token_type,
1751                reference.prev_token_type,
1752                /* attr_taint */ None,
1753            )?;
1754        }
1755
1756        let substitution = substitute_one_reference(
1757            css,
1758            url_data,
1759            substitution_functions,
1760            reference,
1761            stylist,
1762            computed_context,
1763            attribute_tracker,
1764            seen,
1765        )?;
1766
1767        // Optimize the property: var(--...) case to avoid allocating at all.
1768        if reference.start == start && reference.end == end {
1769            if let Some(taint) = attr_taint.filter(|_| substitution.attr_tainted) {
1770                taint.push(start, end);
1771            }
1772            return Ok(substitution);
1773        }
1774
1775        substituted.push(
1776            &substitution.css,
1777            substitution.first_token_type,
1778            substitution.last_token_type,
1779            attr_taint
1780                .as_deref_mut()
1781                .filter(|_| substitution.attr_tainted),
1782        )?;
1783        attr_tainted |= substitution.attr_tainted;
1784        next_token_type = reference.next_token_type;
1785        cur_pos = reference.end;
1786    }
1787    // Push the rest of the value if needed.
1788    if cur_pos != end {
1789        substituted.push(
1790            &css[cur_pos..end],
1791            next_token_type,
1792            last_token_type,
1793            /* attr_taint */ None,
1794        )?;
1795    }
1796    Ok(Substitution::from_value(substituted, attr_tainted))
1797}

1805fn substitute_one_reference<'a>(
1806    css: &'a str,
1807    url_data: &UrlExtraData,
1808    substitution_functions: &'a ComputedSubstitutionFunctions,
1809    reference: &'a SubstitutionFunctionReference,
1810    stylist: &Stylist,
1811    computed_context: &computed::Context,
1812    attribute_tracker: &mut AttributeTracker,
1813    seen: &mut SmallVec<[&'a Name; 8]>,
1814) -> Result<Substitution<'a>, ()> {
1815    let simple_attr_subst = |s: &str| {
1816        Some(Substitution::new(
1817            Cow::Owned(quoted_css_string(s)),
1818            TokenSerializationType::Nothing,
1819            TokenSerializationType::Nothing,
1820            /* attr_tainted */ true,
1821        ))
1822    };
1823    let substitution: Option<_> = match reference.substitution_kind {
1824        SubstitutionFunctionKind::Var => {
1825            let registration = stylist.get_custom_property_registration(&reference.name);
1826            match substitution_functions.get_var(registration, &reference.name) {
1827                None => None,
1828                // If the referenced value is itself still unresolved (i.e. it has references), we
1829                // are substituting against a partially-resolved map -- this happens while applying
1830                // a prioritary property in the middle of custom-property resolution. Resolve it
1831                // recursively, guarding against cycles (which the resolver will also remove, but it
1832                // may not have gotten to this variable yet).
1833                Some(v) => match v.as_universal() {
1834                    Some(u) if u.has_references() => {
1835                        if seen.contains(&&reference.name) {
1836                            // Cycle: the primary is guaranteed-invalid, so fall through to the
1837                            // fallback (if any), like any other invalid primary.
1838                            None
1839                        } else {
1840                            seen.push(&reference.name);
1841                            let result = substitute_internal(
1842                                u,
1843                                substitution_functions,
1844                                stylist,
1845                                computed_context,
1846                                attribute_tracker,
1847                                seen,
1848                                /* attr_taint */ None,
1849                            );
1850                            seen.pop();
1851                            match result {
1852                                Ok(mut substitution) => {
1853                                    substitution.attr_tainted |= v.attr_tainted;
1854                                    Some(substitution)
1855                                },
1856                                // The primary couldn't be resolved (invalid); use the fallback.
1857                                Err(()) => None,
1858                            }
1859                        }
1860                    },
1861                    _ => Some(Substitution::from_value(
1862                        v.to_variable_value(),
1863                        v.attr_tainted,
1864                    )),
1865                },
1866            }
1867        },
1868        SubstitutionFunctionKind::Env => {
1869            let device = stylist.device();
1870            device
1871                .environment()
1872                .get(&reference.name, device, url_data)
1873                .map(|v| Substitution::from_value(v, /* attr_tainted */ false))
1874        },
1875        // https://drafts.csswg.org/css-values-5/#attr-substitution
1876        SubstitutionFunctionKind::Attr => {
1877            let namespace = match reference.attribute_data.namespace {
1878                ParsedNamespace::Known(ref ns) => Some(ns),
1879                ParsedNamespace::Unknown => None,
1880            };
1881            namespace
1882                .and_then(|namespace| {
1883                    LocalName::with_name(&reference.name, |local_name| {
1884                        attribute_tracker.query(local_name, namespace)
1885                    })
1886                })
1887                .map_or_else(
1888                    || {
1889                        // Special case when fallback and <attr-type> are omitted.
1890                        // See FAILURE: https://drafts.csswg.org/css-values-5/#attr-substitution
1891                        if reference.fallback.is_none()
1892                            && reference.attribute_data.kind == AttributeType::None
1893                        {
1894                            simple_attr_subst("")
1895                        } else {
1896                            None
1897                        }
1898                    },
1899                    |attr| {
1900                        let attr = if let AttributeType::Type(_) = &reference.attribute_data.kind {
1901                            // If we're evaluating a container query, we haven't run the cascade
1902                            // and populated substitution_functions.attributes, so we can't do the
1903                            // get_attr() lookup here.
1904                            // TODO: This means chained attr() references will not work reliably in
1905                            // container style queries:
1906                            // https://bugzilla.mozilla.org/show_bug.cgi?id=2028861
1907                            if computed_context.in_container_query {
1908                                attr
1909                            } else {
1910                                substitution_functions
1911                                    .get_attr(&reference.name)
1912                                    .map(|v| v.to_variable_value())?
1913                                    .css
1914                            }
1915                        } else {
1916                            attr
1917                        };
1918                        let mut input = ParserInput::new(&attr);
1919                        let mut parser = Parser::new(&mut input);
1920                        match &reference.attribute_data.kind {
1921                            AttributeType::Unit(unit) => {
1922                                let css = {
1923                                    // Verify that attribute data is a <number-token>.
1924                                    parser.expect_number().ok()?;
1925                                    let mut s = attr.clone();
1926                                    s.push_str(unit.as_ref());
1927                                    s
1928                                };
1929                                let serialization = match unit {
1930                                    AttrUnit::Number => TokenSerializationType::Number,
1931                                    AttrUnit::Percentage => TokenSerializationType::Percentage,
1932                                    _ => TokenSerializationType::Dimension,
1933                                };
1934                                let value =
1935                                    ComputedValue::new(css, url_data, serialization, serialization);
1936                                Some(Substitution::from_value(
1937                                    value, /* attr_tainted */ true,
1938                                ))
1939                            },
1940                            AttributeType::Type(syntax) => {
1941                                let value = SpecifiedRegisteredValue::parse(
1942                                    &mut parser,
1943                                    &syntax,
1944                                    url_data,
1945                                    None,
1946                                    AllowComputationallyDependent::Yes,
1947                                    AttrTaint::default(),
1948                                )
1949                                .ok()?;
1950                                let value = value.to_variable_value();
1951                                Some(Substitution::from_value(
1952                                    value, /* attr_tainted */ true,
1953                                ))
1954                            },
1955                            AttributeType::RawString | AttributeType::None => {
1956                                simple_attr_subst(&attr)
1957                            },
1958                            AttributeType::Invalid => None,
1959                        }
1960                    },
1961                )
1962        },
1963    };
1964
1965    if let Some(s) = substitution {
1966        return Ok(s);
1967    }
1968
1969    let Some(ref fallback) = reference.fallback else {
1970        return Err(());
1971    };
1972
1973    do_substitute_chunk(
1974        css,
1975        fallback.start.get(),
1976        reference.end - 1, // Skip the closing parenthesis of the reference value.
1977        fallback.first_token_type,
1978        fallback.last_token_type,
1979        url_data,
1980        substitution_functions,
1981        stylist,
1982        computed_context,
1983        &fallback.references.refs,
1984        attribute_tracker,
1985        seen,
1986        /* attr_taint */ None,
1987    )
1988}

1990/// Replace `var()`, `env()`, and `attr()` functions. Return `Err(..)` for invalid at computed time.
1991fn substitute_internal<'a>(
1992    variable_value: &'a VariableValue,
1993    substitution_functions: &'a ComputedSubstitutionFunctions,
1994    stylist: &Stylist,
1995    computed_context: &computed::Context,
1996    attribute_tracker: &mut AttributeTracker,
1997    seen: &mut SmallVec<[&'a Name; 8]>,
1998    mut attr_taint: Option<&mut AttrTaint>,
1999) -> Result<Substitution<'a>, ()> {
2000    do_substitute_chunk(
2001        &variable_value.css,
2002        /* start = */ 0,
2003        /* end = */ variable_value.css.len(),
2004        variable_value.first_token_type,
2005        variable_value.last_token_type,
2006        &variable_value.url_data,
2007        substitution_functions,
2008        stylist,
2009        computed_context,
2010        &variable_value.references.refs,
2011        attribute_tracker,
2012        seen,
2013        attr_taint.as_deref_mut(),
2014    )
2015}

2018pub fn substitute<'a>(

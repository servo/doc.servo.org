27/// A struct that implements all the adjustment methods.
28///
29/// NOTE(emilio): If new adjustments are introduced that depend on reset
30/// properties of the parent, you may need tweaking the
31/// `ChildCascadeRequirement` code in `matching.rs`.
32///
33/// NOTE(emilio): Also, if new adjustments are introduced that break the
34/// following invariant:
35///
36///   Given same tag name, namespace, rules and parent style, two elements would
37///   end up with exactly the same style.
38///
39/// Then you need to adjust the lookup_by_rules conditions in the sharing cache.
40pub struct StyleAdjuster<'a, 'b: 'a> {
41    style: &'a mut StyleBuilder<'b>,
42}

44#[cfg(feature = "gecko")]
45fn is_topmost_svg_svg_element<E>(e: E) -> bool
46where
47    E: TElement,
48{
49    debug_assert!(e.is_svg_element());
50    if e.local_name() != &*atom!("svg") {
51        return false;
52    }
53
54    let parent = match e.traversal_parent() {
55        Some(n) => n,
56        None => return true,
57    };
58
59    if !parent.is_svg_element() {
60        return true;
61    }
62
63    parent.local_name() == &*atom!("foreignObject")
64}

66// https://drafts.csswg.org/css-display/#unbox
67#[cfg(feature = "gecko")]
68fn is_effective_display_none_for_display_contents<E>(element: E) -> bool
69where
70    E: TElement,
71{
72    use crate::Atom;
73
74    const SPECIAL_HTML_ELEMENTS: [Atom; 16] = [
75        atom!("br"),
76        atom!("wbr"),
77        atom!("meter"),
78        atom!("progress"),
79        atom!("canvas"),
80        atom!("embed"),
81        atom!("object"),
82        atom!("audio"),
83        atom!("iframe"),
84        atom!("img"),
85        atom!("video"),
86        atom!("frame"),
87        atom!("frameset"),
88        atom!("input"),
89        atom!("textarea"),
90        atom!("select"),
91    ];
92
93    // https://drafts.csswg.org/css-display/#unbox-svg
94    //
95    // There's a note about "Unknown elements", but there's not a good way to
96    // know what that means, or to get that information from here, and no other
97    // UA implements this either.
98    const SPECIAL_SVG_ELEMENTS: [Atom; 6] = [
99        atom!("svg"),
100        atom!("a"),
101        atom!("g"),
102        atom!("use"),
103        atom!("tspan"),
104        atom!("textPath"),
105    ];
106
107    // https://drafts.csswg.org/css-display/#unbox-html
108    if element.is_html_element() {
109        let local_name = element.local_name();
110        return SPECIAL_HTML_ELEMENTS
111            .iter()
112            .any(|name| &**name == local_name);
113    }
114
115    // https://drafts.csswg.org/css-display/#unbox-svg
116    if element.is_svg_element() {
117        if is_topmost_svg_svg_element(element) {
118            return true;
119        }
120        let local_name = element.local_name();
121        return !SPECIAL_SVG_ELEMENTS
122            .iter()
123            .any(|name| &**name == local_name);
124    }
125
126    // https://drafts.csswg.org/css-display/#unbox-mathml
127    if element.is_mathml_element() {
128        return true;
129    }
130
131    false
132}

141    /// <https://fullscreen.spec.whatwg.org/#new-stacking-layer>
142    ///
143    ///    Any position value other than 'absolute' and 'fixed' are
144    ///    computed to 'absolute' if the element is in a top layer.
145    ///
146    fn adjust_for_top_layer(&mut self) {
147        if !self.style.in_top_layer() {
148            return;
149        }
150        if !self.style.is_absolutely_positioned() {
151            self.style.mutate_box().set_position(Position::Absolute);
152        }
153        if self.style.get_box().clone_display().is_contents() {
154            self.style.mutate_box().set_display(Display::Block);
155        }
156    }

158    /// -webkit-box with line-clamp and vertical orientation gets turned into
159    /// flow-root at computed-value time.
160    ///
161    /// This makes the element not be a flex container, with all that it
162    /// implies, but it should be safe. It matches blink, see
163    /// https://bugzilla.mozilla.org/show_bug.cgi?id=1786147#c10
164    #[cfg(feature = "gecko")]
165    fn adjust_for_webkit_line_clamp(&mut self) {
166        use crate::properties::longhands::_moz_box_orient::computed_value::T as BoxOrient;
167        use crate::values::specified::box_::{DisplayInside, DisplayOutside};
168        let box_style = self.style.get_box();
169        if box_style.clone__webkit_line_clamp().is_none() {
170            return;
171        }
172        let line_clamp = box_style.clone_line_clamp();
173        let disp = box_style.clone_display();
174        if disp.inside() != DisplayInside::WebkitBox && line_clamp.webkit_legacy
175            || disp.inside() == DisplayInside::WebkitBox
176                && self.style.get_xul().clone__moz_box_orient() != BoxOrient::Vertical
177        {
178            return;
179        }
180        // Inline elements should not have line-clamp applied.
181        if disp.inside() == DisplayInside::Flow && disp.outside() == DisplayOutside::Inline {
182            return;
183        }
184        let new_display = if disp.outside() == DisplayOutside::Block {
185            Display::FlowRoot
186        } else {
187            debug_assert_eq!(disp.outside(), DisplayOutside::Inline);
188            Display::InlineBlock
189        };
190        self.style
191            .mutate_box()
192            .set_adjusted_display(new_display, false);
193    }

206    /// Whether we should skip any item-based display property blockification on
207    /// this element.
208    fn skip_item_display_fixup<E>(&self, element: Option<E>) -> bool
209    where
210        E: TElement,
211    {
212        if let Some(pseudo) = self.style.pseudo {
213            return pseudo.skip_item_display_fixup();
214        }
215
216        element.is_some_and(|e| e.skip_item_display_fixup())

219    /// Apply the blockification rules based on the table in CSS 2.2 section 9.7.
220    /// <https://drafts.csswg.org/css2/visuren.html#dis-pos-flo>
221    /// A ::marker pseudo-element with 'list-style-position:outside' needs to
222    /// have its 'display' blockified, unless the ::marker is for an inline
223    /// list-item (for which 'list-style-position:outside' behaves as 'inside').
224    /// https://drafts.csswg.org/css-lists-3/#list-style-position-property
225    fn blockify_if_necessary<E>(&mut self, layout_parent_style: &ComputedValues, element: Option<E>)
226    where
227        E: TElement,
228    {
229        let mut blockify = false;
230        macro_rules! blockify_if {
231            ($if_what:expr) => {
232                if !blockify {
233                    blockify = $if_what;
234                }
235            };
236        }
237
238        blockify_if!(self.style.is_root_element);
239        if !self.skip_item_display_fixup(element) {
240            let parent_display = layout_parent_style.get_box().clone_display();
241            blockify_if!(parent_display.is_item_container());
242        }
243
244        let is_item_or_root = blockify;
245
246        blockify_if!(self.style.is_floating());
247        blockify_if!(self.style.is_absolutely_positioned());
248
249        if !blockify {
250            return;
251        }
252
253        let display = self.style.get_box().clone_display();
254        let blockified_display = display.equivalent_block_display(self.style.is_root_element);
255        if display != blockified_display {
256            self.style
257                .mutate_box()
258                .set_adjusted_display(blockified_display, is_item_or_root);
259        }
260    }

262    /// Compute a few common flags for both text and element's style.
263    fn set_bits(&mut self) {
264        let box_style = self.style.get_box();
265        let display = box_style.clone_display();
266
267        if !display.is_contents() {
268            if !self
269                .style
270                .get_text()
271                .clone_text_decoration_line()
272                .is_empty()
273            {
274                self.style
275                    .add_flags(ComputedValueFlags::HAS_TEXT_DECORATION_LINES);
276            }
277
278            if self.style.get_effects().clone_opacity() == 0. {
279                self.style
280                    .add_flags(ComputedValueFlags::IS_IN_OPACITY_ZERO_SUBTREE);
281            }
282        } else if self
283            .style
284            .get_parent_box()
285            .clone_display()
286            .is_item_container()
287            || self
288                .style
289                .get_parent_flags()
290                .contains(ComputedValueFlags::DISPLAY_CONTENTS_IN_ITEM_CONTAINER)
291        {
292            self.style
293                .add_flags(ComputedValueFlags::DISPLAY_CONTENTS_IN_ITEM_CONTAINER);
294        }
295
296        if self.style.pseudo.is_some_and(|p| p.is_first_line()) {
297            self.style
298                .add_flags(ComputedValueFlags::IS_IN_FIRST_LINE_SUBTREE);
299        }
300
301        if self.style.is_root_element {
302            self.style
303                .add_flags(ComputedValueFlags::IS_ROOT_ELEMENT_STYLE);
304        }
305
306        #[cfg(feature = "gecko")]
307        if box_style
308            .clone_effective_containment()
309            .contains(Contain::STYLE)
310        {
311            self.style
312                .add_flags(ComputedValueFlags::SELF_OR_ANCESTOR_HAS_CONTAIN_STYLE);
313        }
314
315        if box_style.clone_container_type().is_size_container_type() {
316            self.style
317                .add_flags(ComputedValueFlags::SELF_OR_ANCESTOR_HAS_SIZE_CONTAINER_TYPE);
318        }
319    }

335    /// Change writing mode of the text frame for text-combine-upright.
336    ///
337    /// It is safe to look at our own style because we are looking at inherited
338    /// properties, and text is just plain inheritance.
339    ///
340    /// TODO(emilio): we should (Gecko too) revise these adjustments in presence
341    /// of display: contents.
342    ///
343    /// FIXME(emilio): How does this play with logical properties? Doesn't
344    /// mutating writing-mode change the potential physical sides chosen?
345    #[cfg(feature = "gecko")]
346    fn adjust_for_text_combine_upright(&mut self) {
347        use crate::computed_values::text_combine_upright::T as TextCombineUpright;
348        use crate::computed_values::writing_mode::T as WritingMode;
349        use crate::logical_geometry;
350
351        let writing_mode = self.style.get_inherited_box().clone_writing_mode();
352        let text_combine_upright = self.style.get_inherited_text().clone_text_combine_upright();
353
354        if matches!(
355            writing_mode,
356            WritingMode::VerticalRl | WritingMode::VerticalLr

362                .set_writing_mode(WritingMode::HorizontalTb);
363            self.style.writing_mode =
364                logical_geometry::WritingMode::new(self.style.get_inherited_box());

388    /// <https://drafts.csswg.org/css-writing-modes-3/#block-flow:>
389    ///
390    ///    If a box has a different writing-mode value than its containing
391    ///    block:
392    ///
393    ///        - If the box has a specified display of inline, its display
394    ///          computes to inline-block. [CSS21]
395    ///
396    /// This matches the adjustment that Gecko does, not exactly following
397    /// the spec. See also:
398    ///
399    /// <https://lists.w3.org/Archives/Public/www-style/2017Mar/0045.html>
400    /// <https://github.com/servo/servo/issues/15754>
401    fn adjust_for_writing_mode(&mut self, layout_parent_style: &ComputedValues) {
402        let our_writing_mode = self.style.get_inherited_box().clone_writing_mode();
403        let parent_writing_mode = layout_parent_style.get_inherited_box().clone_writing_mode();
404
405        if our_writing_mode != parent_writing_mode
406            && self.style.get_box().clone_display() == Display::Inline
407        {
408            // TODO(emilio): Figure out if we can just set the adjusted display
409            // on Gecko too and unify this code path.
410            if cfg!(feature = "servo") {
411                self.style
412                    .mutate_box()
413                    .set_adjusted_display(Display::InlineBlock, false);
414            } else {
415                self.style.mutate_box().set_display(Display::InlineBlock);
416            }
417        }
418    }

420    /// CSS overflow-x and overflow-y require some fixup as well in some cases.
421    /// https://drafts.csswg.org/css-overflow-3/#overflow-properties
422    /// "Computed value: as specified, except with `visible`/`clip` computing to
423    /// `auto`/`hidden` (respectively) if one of `overflow-x` or `overflow-y` is
424    /// neither `visible` nor `clip`."
425    fn adjust_for_overflow(&mut self) {
426        let overflow_x = self.style.get_box().clone_overflow_x();
427        let overflow_y = self.style.get_box().clone_overflow_y();
428        if overflow_x == overflow_y {
429            return; // optimization for the common case
430        }
431
432        if overflow_x.is_scrollable() != overflow_y.is_scrollable() {
433            let box_style = self.style.mutate_box();
434            box_style.set_overflow_x(overflow_x.to_scrollable());
435            box_style.set_overflow_y(overflow_y.to_scrollable());
436        }
437    }

439    #[cfg(feature = "gecko")]
440    fn adjust_for_contain(&mut self) {
441        let box_style = self.style.get_box();
442        let container_type = box_style.clone_container_type();
443        let content_visibility = box_style.clone_content_visibility();
444        if !container_type.is_size_container_type()
445            && content_visibility == ContentVisibility::Visible
446        {
447            debug_assert_eq!(
448                box_style.clone_contain(),
449                box_style.clone_effective_containment()
450            );
451            return;
452        }
453        let old_contain = box_style.clone_contain();
454        let mut new_contain = old_contain;
455        match content_visibility {
456            ContentVisibility::Visible => {},
457            // `content-visibility:auto` also applies size containment when content
458            // is not relevant (and therefore skipped). This is checked in
459            // nsIFrame::GetContainSizeAxes.
460            ContentVisibility::Auto => {
461                new_contain.insert(Contain::LAYOUT | Contain::PAINT | Contain::STYLE)
462            },
463            ContentVisibility::Hidden => new_contain
464                .insert(Contain::LAYOUT | Contain::PAINT | Contain::SIZE | Contain::STYLE),
465        }
466        if container_type.intersects(ContainerType::INLINE_SIZE) {
467            // https://drafts.csswg.org/css-contain-3/#valdef-container-type-inline-size:
468            //     Applies layout containment, style containment, and inline-size
469            //     containment to the principal box.
470            new_contain.insert(Contain::STYLE | Contain::INLINE_SIZE);
471        } else if container_type.intersects(ContainerType::SIZE) {
472            // https://drafts.csswg.org/css-contain-3/#valdef-container-type-size:
473            //     Applies layout containment, style containment, and size
474            //     containment to the principal box.
475            new_contain.insert(Contain::STYLE | Contain::SIZE);
476        }
477        if new_contain == old_contain {
478            debug_assert_eq!(
479                box_style.clone_contain(),
480                box_style.clone_effective_containment()
481            );
482            return;
483        }
484        self.style
485            .mutate_box()
486            .set_effective_containment(new_contain);
487    }

489    /// content-visibility: auto should force contain-intrinsic-size to gain
490    /// an auto value
491    ///
492    /// <https://github.com/w3c/csswg-drafts/issues/8407>
493    #[cfg(feature = "gecko")]
494    fn adjust_for_contain_intrinsic_size(&mut self) {
495        let content_visibility = self.style.get_box().clone_content_visibility();
496        if content_visibility != ContentVisibility::Auto {
497            return;
498        }
499
500        let pos = self.style.get_position();
501        let new_width = pos.clone_contain_intrinsic_width().add_auto_if_needed();
502        let new_height = pos.clone_contain_intrinsic_height().add_auto_if_needed();
503        if new_width.is_none() && new_height.is_none() {
504            return;
505        }
506
507        let pos = self.style.mutate_position();
508        if let Some(width) = new_width {
509            pos.set_contain_intrinsic_width(width);
510        }
511        if let Some(height) = new_height {
512            pos.set_contain_intrinsic_height(height);
513        }
514    }

516    /// Handles the relevant sections in:
517    ///
518    /// https://drafts.csswg.org/css-display/#unbox-html
519    ///
520    /// And forbidding display: contents in pseudo-elements, at least for now.
521    #[cfg(feature = "gecko")]
522    fn adjust_for_prohibited_display_contents<E>(&mut self, element: Option<E>)
523    where
524        E: TElement,
525    {
526        if self.style.get_box().clone_display() != Display::Contents {
527            return;
528        }
529
530        // FIXME(emilio): ::before and ::after should support display: contents, see bug 1418138.
531        if self.style.pseudo.is_some_and(|p| !p.is_element_backed()) {
532            self.style.mutate_box().set_display(Display::Inline);
533            return;
534        }
535
536        let element = match element {
537            Some(e) => e,
538            None => return,
539        };
540
541        if is_effective_display_none_for_display_contents(element) {
542            self.style.mutate_box().set_display(Display::None);
543        }
544    }

572    /// this type into its ::-moz-fieldset-content anonymous box.

574    fn adjust_for_fieldset_content(&mut self) {

586            Display::Grid | Display::InlineGrid => Some(Display::Grid),

616    #[cfg(feature = "gecko")]
617    fn should_suppress_linebreak<E>(&self, element: Option<E>) -> bool
618    where
619        E: TElement,
620    {
621        // Line break suppression should only be propagated to in-flow children.
622        if self.style.is_floating() || self.style.is_absolutely_positioned() {
623            return false;
624        }
625        let parent_display = self.style.get_parent_box().clone_display();
626        if self
627            .style
628            .get_parent_flags()
629            .contains(ComputedValueFlags::SHOULD_SUPPRESS_LINEBREAK)
630        {
631            // Line break suppression is propagated to any children of
632            // line participants, and across display: contents boundaries.
633            if parent_display.is_line_participant() || parent_display.is_contents() {
634                return true;
635            }
636        }
637        match self.style.get_box().clone_display() {
638            // Ruby base and text are always non-breakable.
639            Display::RubyBase | Display::RubyText => true,
640            // Ruby base container and text container are breakable.
641            // Non-HTML elements may not form ruby base / text container because
642            // they may not respect ruby-internal display values, so we can't
643            // make them escaped from line break suppression.
644            // Note that, when certain HTML tags, e.g. form controls, have ruby
645            // level container display type, they could also escape from the
646            // line break suppression flag while they shouldn't. However, it is
647            // generally fine as far as they can't break the line inside them.
648            Display::RubyBaseContainer | Display::RubyTextContainer
649                if element.map_or(true, |e| e.is_html_element()) =>
650            {
651                false
652            },
653            // Anything else is non-breakable if and only if its layout parent
654            // has a ruby display type, because any of the ruby boxes can be
655            // anonymous.
656            _ => parent_display.is_ruby_type(),
657        }
658    }

660    /// Do ruby-related style adjustments, which include:
661    /// * propagate the line break suppression flag,
662    /// * inlinify block descendants,
663    /// * suppress border and padding for ruby level containers,
664    /// * correct unicode-bidi.
665    #[cfg(feature = "gecko")]
666    fn adjust_for_ruby<E>(&mut self, element: Option<E>)
667    where
668        E: TElement,
669    {
670        use crate::properties::longhands::unicode_bidi::computed_value::T as UnicodeBidi;
671
672        let self_display = self.style.get_box().clone_display();
673        // Check whether line break should be suppressed for this element.
674        if self.should_suppress_linebreak(element) {
675            self.style
676                .add_flags(ComputedValueFlags::SHOULD_SUPPRESS_LINEBREAK);
677            // Inlinify the display type if allowed.
678            if !self.skip_item_display_fixup(element) {
679                let inline_display = self_display.inlinify();
680                if self_display != inline_display {
681                    self.style
682                        .mutate_box()
683                        .set_adjusted_display(inline_display, false);
684                }
685            }
686        }
687        // Suppress border and padding for ruby level containers.
688        // This is actually not part of the spec. It is currently unspecified
689        // how border and padding should be handled for ruby level container,
690        // and suppressing them here make it easier for layout to handle.
691        if self_display.is_ruby_level_container() {
692            self.style.reset_border_struct();
693            self.style.reset_padding_struct();
694        }
695
696        // Force bidi isolation on all internal ruby boxes and ruby container
697        // per spec https://drafts.csswg.org/css-ruby-1/#bidi
698        if self_display.is_ruby_type() {
699            let new_value = match self.style.get_text().clone_unicode_bidi() {
700                UnicodeBidi::Normal | UnicodeBidi::Embed => Some(UnicodeBidi::Isolate),
701                UnicodeBidi::BidiOverride => Some(UnicodeBidi::IsolateOverride),
702                _ => None,
703            };
704            if let Some(new_value) = new_value {
705                self.style.mutate_text().set_unicode_bidi(new_value);
706            }
707        }
708    }

710    /// Computes the RELEVANT_LINK_VISITED flag based on the parent style and on
711    /// whether we're a relevant link.
712    ///
713    /// NOTE(emilio): We don't do this for text styles, which is... dubious, but
714    /// Gecko doesn't seem to do it either. It's extremely easy to do if needed
715    /// though.
716    ///
717    /// FIXME(emilio): This isn't technically a style adjustment thingie, could
718    /// it move somewhere else?
719    fn adjust_for_visited<E>(&mut self, element: Option<E>)
720    where
721        E: TElement,
722    {
723        if !self.style.has_visited_style() {
724            return;
725        }
726
727        let is_link_element = self.style.pseudo.is_none() && element.map_or(false, |e| e.is_link());
728
729        if !is_link_element {
730            return;
731        }
732
733        if element.unwrap().is_visited_link() {
734            self.style
735                .add_flags(ComputedValueFlags::IS_RELEVANT_LINK_VISITED);
736        } else {
737            // Need to remove to handle unvisited link inside visited.
738            self.style
739                .remove_flags(ComputedValueFlags::IS_RELEVANT_LINK_VISITED);
740        }
741    }

744    /// to comply with:

746    /// <https://drafts.csswg.org/css-align/#valdef-justify-items-legacy>
747    #[cfg(feature = "gecko")]
748    fn adjust_for_justify_items(&mut self) {
749        use crate::values::specified::align;

770    /// If '-webkit-appearance' is 'menulist' on a <select> element then
771    /// the computed value of 'line-height' is 'normal'.
772    ///
773    /// https://github.com/w3c/csswg-drafts/issues/3257
774    fn adjust_for_appearance<E>(&mut self, element: Option<E>)
775    where
776        E: TElement,
777    {
778        use crate::properties::longhands::appearance::computed_value::T as Appearance;
779        use crate::properties::longhands::line_height::computed_value::T as LineHeight;
780
781        let box_ = self.style.get_box();
782        let appearance = match box_.clone_appearance() {
783            Appearance::Auto => box_.clone__moz_default_appearance(),
784            a => a,
785        };
786
787        if appearance == Appearance::Menulist {
788            if self.style.get_font().clone_line_height() == LineHeight::normal() {
789                return;
790            }
791            if self.style.pseudo.is_some() {
792                return;
793            }
794            let is_html_select_element = element.map_or(false, |e| {
795                e.is_html_element() && e.local_name() == &*atom!("select")
796            });
797            if !is_html_select_element {
798                return;
799            }
800            self.style
801                .mutate_font()
802                .set_line_height(LineHeight::normal());
803        }
804    }

806    /// A legacy ::marker (i.e. no 'content') without an author-specified 'font-family'
807    /// and 'list-style-type:disc|circle|square|disclosure-closed|disclosure-open'
808    /// is assigned 'font-family:-moz-bullet-font'. (This is for <ul><li> etc.)
809    /// We don't want synthesized italic/bold for this font, so turn that off too.
810    /// Likewise for 'letter/word-spacing' -- unless the author specified it then reset
811    /// them to their initial value because traditionally we never added such spacing
812    /// between a legacy bullet and the list item's content, so we keep that behavior
813    /// for web-compat reasons.
814    /// We intentionally don't check 'list-style-image' below since we want it to use
815    /// the same font as its fallback ('list-style-type') in case it fails to load.
816    #[cfg(feature = "gecko")]
817    fn adjust_for_marker_pseudo(&mut self, author_specified_properties: &LonghandIdSet) {
818        use crate::values::computed::counters::Content;
819        use crate::values::computed::font::{FontFamily, FontSynthesis, FontSynthesisStyle};
820        use crate::values::computed::text::{LetterSpacing, WordSpacing};
821
822        let is_legacy_marker = self.style.pseudo.map_or(false, |p| p.is_marker())
823            && self.style.get_list().clone_list_style_type().is_bullet()
824            && self.style.get_counters().clone_content() == Content::Normal;
825        if !is_legacy_marker {
826            return;
827        }
828        if !author_specified_properties.contains(LonghandId::FontFamily) {
829            self.style
830                .mutate_font()
831                .set_font_family(FontFamily::moz_bullet().clone());
832
833            // FIXME(mats): We can remove this if support for font-synthesis is added to @font-face rules.
834            // Then we can add it to the @font-face rule in html.css instead.
835            // https://github.com/w3c/csswg-drafts/issues/6081
836            if !author_specified_properties.contains(LonghandId::FontSynthesisWeight) {
837                self.style
838                    .mutate_font()
839                    .set_font_synthesis_weight(FontSynthesis::None);
840            }
841            if !author_specified_properties.contains(LonghandId::FontSynthesisStyle) {
842                self.style
843                    .mutate_font()
844                    .set_font_synthesis_style(FontSynthesisStyle::None);
845            }
846        }
847        if !author_specified_properties.contains(LonghandId::LetterSpacing) {
848            self.style
849                .mutate_inherited_text()
850                .set_letter_spacing(LetterSpacing::normal());
851        }
852        if !author_specified_properties.contains(LonghandId::WordSpacing) {
853            self.style
854                .mutate_inherited_text()
855                .set_word_spacing(WordSpacing::normal());
856        }
857    }

859    /// Performs adjustments for position-try-fallbacks. The properties that need adjustments here
860    /// are luckily not affected by previous adjustments nor by other computed-value-time effects,
861    /// so we can just perform them here.
862    ///
863    /// NOTE(emilio): If we ever perform the interleaving dance, this could / should probably move
864    /// around to the specific properties' to_computed_value implementations, but that seems
865    /// overkill for now.
866    fn adjust_for_try_tactic(&mut self, tactic: &PositionTryFallbacksTryTactic) {
867        debug_assert!(!tactic.is_empty());
868        // TODO: This is supposed to use the containing block's WM (bug 1995256).
869        let wm = self.style.writing_mode;
870        // TODO: Flip inset / margin / sizes percentages and anchor lookup sides as necessary.
871        for tactic in tactic.iter() {
872            use PositionTryFallbacksTryTacticKeyword::*;
873            match tactic {
874                FlipBlock => {
875                    self.flip_self_alignment(/* block = */ true);
876                    self.flip_insets_and_margins(/* horizontal = */ wm.is_vertical());
877                },
878                FlipInline => {
879                    self.flip_self_alignment(/* block = */ false);
880                    self.flip_insets_and_margins(/* horizontal = */ wm.is_horizontal());
881                },
882                FlipX => {
883                    self.flip_self_alignment(/* block = */ wm.is_vertical());
884                    self.flip_insets_and_margins(/* horizontal = */ true);
885                },
886                FlipY => {
887                    self.flip_self_alignment(/* block = */ wm.is_horizontal());
888                    self.flip_insets_and_margins(/* horizontal = */ false);
889                },
890                FlipStart => {
891                    self.flip_start();
892                },
893            }
894            self.apply_position_area_tactic(*tactic);
895        }
896    }
897
898    fn apply_position_area_tactic(&mut self, tactic: PositionTryFallbacksTryTacticKeyword) {
899        let pos = self.style.get_position();
900        let old = pos.clone_position_area();
901        let wm = self.style.writing_mode;
902        let new = old.with_tactic(wm, tactic);
903        if new == old {
904            return;
905        }
906        let pos = self.style.mutate_position();
907        pos.set_position_area(new);
908    }

911    fn swap_insets(&mut self, a_side: PhysicalSide, b_side: PhysicalSide) {
912        debug_assert_ne!(a_side, b_side);
913        let pos = self.style.mutate_position();
914        let mut a = pos.get_inset(a_side).clone();
915        a.try_tactic_adjustment(a_side, b_side);
916        let mut b = pos.get_inset(b_side).clone();
917        b.try_tactic_adjustment(b_side, a_side);
918        pos.set_inset(a_side, b);
919        pos.set_inset(b_side, a);
920    }

922    fn swap_margins(&mut self, a_side: PhysicalSide, b_side: PhysicalSide) {
923        debug_assert_ne!(a_side, b_side);
924        let margin = self.style.get_margin();
925        let mut a = margin.get_margin(a_side).clone();
926        a.try_tactic_adjustment(a_side, b_side);
927        let mut b = margin.get_margin(b_side).clone();
928        b.try_tactic_adjustment(b_side, a_side);
929        let margin = self.style.mutate_margin();
930        margin.set_margin(a_side, b);
931        margin.set_margin(b_side, a);
932    }

934    fn swap_sizes(&mut self, block_start: PhysicalSide, inline_start: PhysicalSide) {
935        let pos = self.style.mutate_position();
936        let mut min_width = pos.clone_min_width();
937        min_width.try_tactic_adjustment(inline_start, block_start);
938        let mut max_width = pos.clone_max_width();
939        max_width.try_tactic_adjustment(inline_start, block_start);
940        let mut width = pos.clone_width();
941        width.try_tactic_adjustment(inline_start, block_start);
942
943        let mut min_height = pos.clone_min_height();
944        min_height.try_tactic_adjustment(block_start, inline_start);
945        let mut max_height = pos.clone_max_height();
946        max_height.try_tactic_adjustment(block_start, inline_start);
947        let mut height = pos.clone_height();
948        height.try_tactic_adjustment(block_start, inline_start);
949

951        pos.set_width(height);
952        pos.set_height(width);
953        pos.set_max_width(max_height);
954        pos.set_max_height(max_width);
955        pos.set_min_width(min_height);
956        pos.set_min_height(min_width);

959    fn flip_start(&mut self) {
960        let wm = self.style.writing_mode;
961        let bs = wm.block_start_physical_side();
962        let is = wm.inline_start_physical_side();
963        let be = wm.block_end_physical_side();
964        let ie = wm.inline_end_physical_side();
965        self.swap_sizes(bs, is);
966        self.swap_insets(bs, is);
967        self.swap_insets(ie, be);
968        self.swap_margins(bs, is);
969        self.swap_margins(ie, be);
970        self.flip_alignment_start();
971    }

973    fn flip_insets_and_margins(&mut self, horizontal: bool) {
974        if horizontal {
975            self.swap_insets(PhysicalSide::Left, PhysicalSide::Right);
976            self.swap_margins(PhysicalSide::Left, PhysicalSide::Right);
977        } else {
978            self.swap_insets(PhysicalSide::Top, PhysicalSide::Bottom);
979            self.swap_margins(PhysicalSide::Top, PhysicalSide::Bottom);
980        }
981    }

983    fn flip_alignment_start(&mut self) {
984        let pos = self.style.get_position();
985        let align = pos.clone_align_self();
986        let mut justify = pos.clone_justify_self();
987        if align == justify {
988            return;
989        }
990
991        // Fix-up potential justify-self: {left, right} values which might end up as alignment
992        // values.
993        if matches!(justify.value(), AlignFlags::LEFT | AlignFlags::RIGHT) {
994            let left = justify.value() == AlignFlags::LEFT;
995            let ltr = self.style.writing_mode.is_bidi_ltr();
996            justify = justify.with_value(if left == ltr {
997                AlignFlags::SELF_START
998            } else {
999                AlignFlags::SELF_END
1000            });
1001        }
1002
1003        let pos = self.style.mutate_position();
1004        pos.set_align_self(justify);
1005        pos.set_justify_self(align);
1006    }

1008    fn flip_self_alignment(&mut self, block: bool) {
1009        let pos = self.style.get_position();
1010        let cur = if block {
1011            pos.clone_align_self()
1012        } else {
1013            pos.clone_justify_self()
1014        };
1015        let flipped = cur.flip_position();
1016        if flipped == cur {
1017            return;
1018        }
1019        let pos = self.style.mutate_position();
1020        if block {
1021            pos.set_align_self(flipped);
1022        } else {
1023            pos.set_justify_self(flipped);
1024        }
1025    }

1027    /// Adjusts the style to account for various fixups that don't fit naturally into the cascade.
1028    #[allow(unused_variables)]
1029    pub fn adjust<E>(
1030        &mut self,
1031        layout_parent_style: &ComputedValues,
1032        element: Option<E>,
1033        try_tactic: &PositionTryFallbacksTryTactic,
1034        author_specified_properties: &LonghandIdSet,
1035    ) where
1036        E: TElement,
1037    {
1038        if cfg!(debug_assertions) {
1039            if let Some(e) = element {
1040                if let Some(p) = e.implemented_pseudo_element() {
1041                    // It'd be nice to assert `self.style.pseudo == Some(&pseudo)`,
1042                    // but we do resolve ::-moz-list pseudos on ::before / ::after
1043                    // content, sigh.
1044                    debug_assert!(
1045                        self.style.pseudo.is_some(),
1046                        "Someone really messed up (no pseudo style for {e:?}, {p:?})"
1047                    );
1048                }
1049            }
1050        }
1051        // FIXME(emilio): The apply_declarations callsite in Servo's
1052        // animation, and the font stuff for Gecko
1053        // (Stylist::compute_for_declarations) should pass an element to
1054        // cascade(), then we can make this assertion hold everywhere.
1055        // debug_assert!(
1056        //     element.is_some() || self.style.pseudo.is_some(),
1057        //     "Should always have an element around for non-pseudo styles"
1058        // );
1059
1060        self.adjust_for_visited(element);
1061        #[cfg(feature = "gecko")]
1062        {
1063            self.adjust_for_prohibited_display_contents(element);
1064            self.adjust_for_fieldset_content();
1065            self.adjust_for_text_control_editing_root();
1066        }
1067        self.adjust_for_top_layer();
1068        self.blockify_if_necessary(layout_parent_style, element);
1069        #[cfg(feature = "gecko")]
1070        self.adjust_for_webkit_line_clamp();
1071        self.adjust_for_position();
1072        self.adjust_for_overflow();
1073        #[cfg(feature = "gecko")]
1074        {
1075            self.adjust_for_contain();
1076            self.adjust_for_contain_intrinsic_size();
1077            self.adjust_for_justify_items();
1078        }
1079        self.adjust_for_table_text_align();
1080        self.adjust_for_writing_mode(layout_parent_style);
1081        #[cfg(feature = "gecko")]
1082        self.adjust_for_ruby(element);
1083        self.adjust_for_appearance(element);
1084        #[cfg(feature = "gecko")]
1085        self.adjust_for_marker_pseudo(author_specified_properties);
1086        if !try_tactic.is_empty() {
1087            self.adjust_for_try_tactic(try_tactic);
1088        }
1089        self.set_bits();
1090    }

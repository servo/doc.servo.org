68impl<L, N, P, LP, C, Image, U, Integer, A, T, R, Transform>
69    GenericValueComponent<L, N, P, LP, C, Image, U, Integer, A, T, R, Transform>
70{
71    fn serialization_types(&self) -> (TokenSerializationType, TokenSerializationType) {
72        let first_token_type = match self {
73            Self::Length(_) | Self::Angle(_) | Self::Time(_) | Self::Resolution(_) => {
74                TokenSerializationType::Dimension
75            },
76            Self::Number(_) | Self::Integer(_) => TokenSerializationType::Number,
77            Self::Percentage(_) | Self::LengthPercentage(_) => TokenSerializationType::Percentage,
78            Self::Color(_)
79            | Self::Image(_)
80            | Self::Url(_)
81            | Self::TransformFunction(_)
82            | Self::TransformList(_) => TokenSerializationType::Function,
83            Self::CustomIdent(_) => TokenSerializationType::Ident,
84            Self::String(_) => TokenSerializationType::Other,
85        };
86        let last_token_type = if first_token_type == TokenSerializationType::Function {
87            TokenSerializationType::Other
88        } else {
89            first_token_type
90        };
91        (first_token_type, last_token_type)
92    }
93}

152/// A list of component values, including the list's multiplier.
153#[derive(Clone, ToComputedValue, ToResolvedValue, Debug, MallocSizeOf, PartialEq, ToShmem)]
154pub struct ComponentList<Component> {
155    /// Multiplier
156    pub multiplier: Multiplier,
157    /// The list of components contained.
158    pub components: crate::OwnedSlice<Component>,
159}
160
161impl<Component: Animate> Animate for ComponentList<Component> {
162    fn animate(&self, other: &Self, procedure: Procedure) -> Result<Self, ()> {
163        if self.multiplier != other.multiplier {
164            return Err(());
165        }
166        let components = animated::lists::by_computed_value::animate(
167            &self.components,
168            &other.components,
169            procedure,
170        )?;
171        Ok(Self {
172            multiplier: self.multiplier,
173            components,
174        })
175    }
176}

178impl<Component: ToCss> ToCss for ComponentList<Component> {
179    fn to_css<W>(&self, dest: &mut CssWriter<W>) -> fmt::Result
180    where
181        W: Write,
182    {
183        let mut iter = self.components.iter();
184        let Some(first) = iter.next() else {
185            return Ok(());
186        };
187        first.to_css(dest)?;
188
189        // The separator implied by the multiplier for this list.
190        let separator = match self.multiplier {
191            // <https://drafts.csswg.org/cssom-1/#serialize-a-whitespace-separated-list>
192            Multiplier::Space => " ",
193            // <https://drafts.csswg.org/cssom-1/#serialize-a-comma-separated-list>
194            Multiplier::Comma => ", ",
195        };
196        for component in iter {
197            dest.write_str(separator)?;
198            component.to_css(dest)?;
199        }
200        Ok(())
201    }
202}

204/// A struct for a single specified registered custom property value that includes its original URL
205/// data so the value can be uncomputed later.
206#[derive(Clone, Debug, MallocSizeOf, ToCss, ToComputedValue, ToResolvedValue, ToShmem, ToTyped)]
207#[typed(todo_derive_fields)]
208pub struct Value<Component> {
209    /// The registered custom property value.
210    pub(crate) v: ValueInner<Component>,
211    /// The URL data of the registered custom property from before it was computed. This is
212    /// necessary to uncompute registered custom properties.
213    #[css(skip)]
214    url_data: UrlExtraData,
215    /// Flag indicating whether this value is tainted by an attr().
216    #[css(skip)]
217    pub attr_tainted: bool,
218}

220impl<Component: PartialEq> PartialEq for Value<Component> {
221    // Ignore the url_data field when comparing values for equality.
222    // attr_tainted is compared so the cascade doesn't treat a tainted
223    // value as equal to an untainted one, which could lose the taint.
224    fn eq(&self, other: &Self) -> bool {
225        self.v == other.v && self.attr_tainted == other.attr_tainted
226    }
227}

232        Ok(Value {
233            v,
234            url_data: self.url_data.clone(),
235            attr_tainted: self.attr_tainted,
236        })
237    }

263impl<L, N, P, LP, C, Image, U, Integer, A, T, R, Transform>
264    Value<GenericValueComponent<L, N, P, LP, C, Image, U, Integer, A, T, R, Transform>>
265where
266    Self: ToCss,
267{
268    fn serialization_types(&self) -> (TokenSerializationType, TokenSerializationType) {
269        match &self.v {
270            ValueInner::Component(component) => component.serialization_types(),
271            ValueInner::Universal(_) => unreachable!(),
272            ValueInner::List(list) => list
273                .components
274                .first()
275                .map_or(Default::default(), |f| f.serialization_types()),
276        }
277    }
278
279    /// Convert to an untyped variable value.
280    pub fn to_variable_value(&self) -> ComputedPropertyValue {
281        if let ValueInner::Universal(ref value) = self.v {
282            return (**value).clone();
283        }
284        let serialization_types = self.serialization_types();
285        ComputedPropertyValue::new(
286            self.to_css_string(),
287            &self.url_data,
288            serialization_types.0,
289            serialization_types.1,
290        )
291    }

315impl SpecifiedValue {
316    /// Convert a registered custom property to a Computed custom property value, given input and a
317    /// property registration.
318    pub fn compute<'i, 't>(
319        input: &mut CSSParser<'i, 't>,
320        registration: &PropertyDescriptors,
321        namespaces: Option<&FxHashMap<Prefix, Namespace>>,
322        url_data: &UrlExtraData,
323        context: &computed::Context,
324        allow_computationally_dependent: AllowComputationallyDependent,
325        attr_taint: AttrTaint,
326    ) -> Result<ComputedValue, ()> {
327        debug_assert!(!registration.is_universal(), "Shouldn't be needed");
328        let Some(ref syntax) = registration.syntax else {
329            return Err(());
330        };
331        let Ok(value) = Self::parse(
332            input,
333            syntax,
334            url_data,
335            namespaces,
336            allow_computationally_dependent,
337            attr_taint,
338        ) else {
339            return Err(());
340        };
341
342        Ok(value.to_computed_value(context))
343    }
344
345    /// Parse and validate a registered custom property value according to its syntax descriptor,
346    /// and check for computational independence.
347    pub fn parse<'i, 't>(
348        mut input: &mut CSSParser<'i, 't>,
349        syntax: &Descriptor,
350        url_data: &UrlExtraData,
351        namespaces: Option<&FxHashMap<Prefix, Namespace>>,
352        allow_computationally_dependent: AllowComputationallyDependent,
353        attr_taint: AttrTaint,
354    ) -> Result<Self, StyleParseError<'i>> {
355        if syntax.is_universal() {
356            let parsed = ComputedPropertyValue::parse(&mut input, namespaces, url_data)?;
357            return Ok(Self::new(
358                ValueInner::Universal(Arc::new(parsed)),
359                url_data.clone(),
360            ));
361        }
362
363        let mut values = SmallComponentVec::new();
364        let mut multiplier = None;
365        {
366            let mut parser = Parser::new(syntax, &mut values, &mut multiplier);
367            parser.parse(
368                &mut input,
369                url_data,
370                allow_computationally_dependent,
371                attr_taint,
372            )?;
373        }
374        let v = if let Some(multiplier) = multiplier {
375            ValueInner::List(ComponentList {
376                multiplier,
377                components: values.to_vec().into(),
378            })
379        } else {
380            ValueInner::Component(values[0].clone())
381        };
382        Ok(Self::new(v, url_data.clone()))
383    }
384}

418type SmallComponentVec = SmallVec<[SpecifiedValueComponent; 1]>;
419
420struct Parser<'a> {
421    syntax: &'a Descriptor,
422    output: &'a mut SmallComponentVec,
423    output_multiplier: &'a mut Option<Multiplier>,
424}
425
426impl<'a> Parser<'a> {
427    fn new(
428        syntax: &'a Descriptor,
429        output: &'a mut SmallComponentVec,
430        output_multiplier: &'a mut Option<Multiplier>,
431    ) -> Self {
432        Self {
433            syntax,
434            output,
435            output_multiplier,
436        }
437    }
438
439    fn parse<'i, 't>(
440        &mut self,
441        input: &mut CSSParser<'i, 't>,
442        url_data: &UrlExtraData,
443        allow_computationally_dependent: AllowComputationallyDependent,
444        attr_taint: AttrTaint,
445    ) -> Result<(), StyleParseError<'i>> {
446        use self::AllowComputationallyDependent::*;
447        let parsing_mode = match allow_computationally_dependent {
448            No => ParsingMode::DISALLOW_COMPUTATIONALLY_DEPENDENT,
449            Yes => ParsingMode::DEFAULT,
450        };
451        let ref context = ParserContext::new(
452            Origin::Author,
453            url_data,
454            Some(CssRuleType::Style),
455            parsing_mode,
456            QuirksMode::NoQuirks,
457            /* namespaces = */ Default::default(),
458            None,
459            None,
460            attr_taint,
461        );
462        for component in self.syntax.components.iter() {
463            let result = input.try_parse(|input| {
464                input.parse_entirely(|input| {
465                    Self::parse_value(context, input, &component.unpremultiplied())
466                })
467            });
468            let Ok(values) = result else { continue };
469            self.output.extend(values);
470            *self.output_multiplier = component.multiplier();
471            break;
472        }
473        if self.output.is_empty() {
474            return Err(input.new_error(BasicParseErrorKind::EndOfInput));
475        }
476        Ok(())
477    }
478
479    fn parse_value<'i, 't>(
480        context: &ParserContext,
481        input: &mut CSSParser<'i, 't>,
482        component: &SyntaxComponent,
483    ) -> Result<SmallComponentVec, StyleParseError<'i>> {
484        let mut values = SmallComponentVec::new();
485        values.push(Self::parse_component_without_multiplier(
486            context, input, component,
487        )?);
488
489        if let Some(multiplier) = component.multiplier() {
490            loop {
491                let result = Self::expect_multiplier(input, &multiplier);
492                if Self::expect_multiplier_yielded_eof_error(&result) {
493                    break;
494                }
495                result?;
496                values.push(Self::parse_component_without_multiplier(
497                    context, input, component,
498                )?);
499            }
500        }
501        Ok(values)
502    }
503
504    fn parse_component_without_multiplier<'i, 't>(
505        context: &ParserContext,
506        input: &mut CSSParser<'i, 't>,
507        component: &SyntaxComponent,
508    ) -> Result<SpecifiedValueComponent, StyleParseError<'i>> {
509        let data_type = match component.name() {
510            ComponentName::DataType(ty) => ty,
511            ComponentName::Ident(ref name) => {
512                let ident = CustomIdent::parse(input, &[])?;
513                if ident != *name {
514                    return Err(input.new_custom_error(StyleParseErrorKind::UnspecifiedError));
515                }
516                return Ok(SpecifiedValueComponent::CustomIdent(ident));
517            },
518        };
519
520        let value = match data_type {
521            DataType::Length => {
522                SpecifiedValueComponent::Length(specified::Length::parse(context, input)?)
523            },
524            DataType::Number => {
525                SpecifiedValueComponent::Number(specified::Number::parse(context, input)?)
526            },
527            DataType::Percentage => {
528                SpecifiedValueComponent::Percentage(specified::Percentage::parse(context, input)?)
529            },
530            DataType::LengthPercentage => SpecifiedValueComponent::LengthPercentage(
531                specified::LengthPercentage::parse(context, input)?,
532            ),
533            DataType::Color => {
534                SpecifiedValueComponent::Color(specified::Color::parse(context, input)?)
535            },
536            DataType::Image => {
537                SpecifiedValueComponent::Image(specified::Image::parse_forbid_none(context, input)?)
538            },
539            DataType::Url => {
540                SpecifiedValueComponent::Url(specified::url::SpecifiedUrl::parse(context, input)?)
541            },
542            DataType::Integer => {
543                SpecifiedValueComponent::Integer(specified::Integer::parse(context, input)?)
544            },
545            DataType::Angle => {
546                SpecifiedValueComponent::Angle(specified::Angle::parse(context, input)?)
547            },
548            DataType::Time => {
549                SpecifiedValueComponent::Time(specified::Time::parse(context, input)?)
550            },
551            DataType::Resolution => {
552                SpecifiedValueComponent::Resolution(specified::Resolution::parse(context, input)?)
553            },
554            DataType::TransformFunction => SpecifiedValueComponent::TransformFunction(
555                specified::Transform::parse(context, input)?,
556            ),
557            DataType::CustomIdent => {
558                let name = CustomIdent::parse(input, &[])?;
559                SpecifiedValueComponent::CustomIdent(name)
560            },
561            DataType::TransformList => {
562                let mut values = vec![];
563                let Some(multiplier) = component.unpremultiplied().multiplier() else {
564                    debug_assert!(false, "Unpremultiplied <transform-list> had no multiplier?");
565                    return Err(
566                        input.new_custom_error(StyleParseErrorKind::PropertySyntaxField(
567                            PropertySyntaxParseError::UnexpectedEOF,
568                        )),
569                    );
570                };
571                debug_assert_eq!(multiplier, Multiplier::Space);
572                loop {
573                    values.push(SpecifiedValueComponent::TransformFunction(
574                        specified::Transform::parse(context, input)?,
575                    ));
576                    let result = Self::expect_multiplier(input, &multiplier);
577                    if Self::expect_multiplier_yielded_eof_error(&result) {
578                        break;
579                    }
580                    result?;
581                }
582                let list = ComponentList {
583                    multiplier,
584                    components: values.into(),
585                };
586                SpecifiedValueComponent::TransformList(list)
587            },
588            DataType::String => {
589                let string = input.expect_string()?;
590                SpecifiedValueComponent::String(string.as_ref().to_owned().into())
591            },
592        };
593        Ok(value)
594    }
595
596    fn expect_multiplier_yielded_eof_error<'i>(result: &Result<(), StyleParseError<'i>>) -> bool {
597        matches!(
598            result,
599            Err(StyleParseError {
600                kind: ParseErrorKind::Basic(BasicParseErrorKind::EndOfInput),
601                ..
602            })
603        )
604    }
605
606    fn expect_multiplier<'i, 't>(
607        input: &mut CSSParser<'i, 't>,
608        multiplier: &Multiplier,
609    ) -> Result<(), StyleParseError<'i>> {
610        match multiplier {
611            Multiplier::Space => {
612                input.expect_whitespace()?;
613                if input.is_exhausted() {
614                    // If there was trailing whitespace, do not interpret it as a multiplier
615                    return Err(input.new_error(BasicParseErrorKind::EndOfInput));
616                }
617                Ok(())
618            },
619            Multiplier::Comma => Ok(input.expect_comma()?),
620        }
621    }
622}

624/// An animated value for custom property.
625#[derive(Clone, Debug, MallocSizeOf, PartialEq)]
626pub struct CustomAnimatedValue {
627    /// The name of the custom property.
628    pub(crate) name: crate::custom_properties::Name,

634impl Animate for CustomAnimatedValue {
635    fn animate(&self, other: &Self, procedure: Procedure) -> Result<Self, ()> {
636        if self.name != other.name {
637            return Err(());
638        }
639        let value = self.value.animate(&other.value, procedure)?;
640        Ok(Self {
641            name: self.name.clone(),
642            value,
643        })
644    }

647impl CustomAnimatedValue {
648    pub(crate) fn from_computed(
649        name: &crate::custom_properties::Name,
650        value: Option<&ComputedValue>,
651    ) -> Self {
652        Self {
653            name: name.clone(),
654            value: value.cloned(),
655        }
656    }
657
658    pub(crate) fn from_declaration(
659        declaration: &properties::CustomDeclaration,
660        context: &mut computed::Context,
661    ) -> Option<Self> {
662        let computed_value = match declaration.value {
663            properties::CustomDeclarationValue::Unparsed(ref value) => Some({
664                debug_assert!(
665                    context.builder.stylist.is_some(),
666                    "Need a Stylist to get property registration!"
667                );
668                let registration = context
669                    .builder
670                    .stylist
671                    .unwrap()
672                    .get_custom_property_registration(&declaration.name);
673                if registration.is_universal() {
674                    // FIXME: Do we need to perform substitution here somehow?
675                    ComputedValue::new(
676                        ValueInner::Universal(Arc::clone(value)),
677                        value.url_data.clone(),
678                    )
679                } else {
680                    let mut input = cssparser::ParserInput::new(&value.css);
681                    let mut input = CSSParser::new(&mut input);
682                    SpecifiedValue::compute(
683                        &mut input,
684                        registration,
685                        None,
686                        &value.url_data,
687                        context,
688                        AllowComputationallyDependent::Yes,
689                        /* attr_taint */ Default::default(),
690                    )
691                    .unwrap_or_else(|_| {
692                        ComputedValue::new(
693                            ValueInner::Universal(Arc::clone(value)),
694                            value.url_data.clone(),
695                        )
696                    })
697                }
698            }),
699            properties::CustomDeclarationValue::Parsed(ref v) => Some(v.to_computed_value(context)),
700            properties::CustomDeclarationValue::CSSWideKeyword(keyword) => {
701                let stylist = context.builder.stylist.unwrap();
702                let registration = stylist.get_custom_property_registration(&declaration.name);
703                match keyword {
704                    CSSWideKeyword::Initial => stylist
705                        .get_custom_property_initial_values()
706                        .get(registration, &declaration.name),
707                    CSSWideKeyword::Inherit => context
708                        .builder
709                        .inherited_custom_properties()
710                        .get(registration, &declaration.name),
711                    CSSWideKeyword::Unset => {
712                        if registration.inherits() {
713                            context
714                                .builder
715                                .inherited_custom_properties()
716                                .get(registration, &declaration.name)
717                        } else {
718                            stylist
719                                .get_custom_property_initial_values()
720                                .get(registration, &declaration.name)
721                        }
722                    },
723                    // FIXME(emilio, bug 1533327): I think revert (and
724                    // revert-layer) handling is not fine here, but what to
725                    // do instead?
726                    //
727                    // Seems we'd need the computed value as if it was
728                    // revert, somehow. Returning `None` seems fine for now...
729                    //
730                    // Note that once this is fixed, this method should be
731                    // able to return `Self` instead of Option<Self>`.
732                    CSSWideKeyword::Revert
733                    | CSSWideKeyword::RevertRule
734                    | CSSWideKeyword::RevertLayer => return None,
735                }
736                .cloned()
737            },
738        };
739        Some(Self {
740            name: declaration.name.clone(),
741            value: computed_value,
742        })
743    }
744
745    pub(crate) fn to_declaration(&self) -> properties::PropertyDeclaration {
746        properties::PropertyDeclaration::Custom(properties::CustomDeclaration {
747            name: self.name.clone(),
748            value: match &self.value {
749                Some(value) => value.to_declared_value(),
750                None => CustomDeclarationValue::CSSWideKeyword(CSSWideKeyword::Initial),
751            },
752        })
753    }

13/// Derive implementation of the `ToTyped` trait.
14///
15/// This derive supports both enums and structs:
16///
17/// * Enums
18///   * Pure keyword enums: Enums made up entirely of unit variants
19///     (e.g. `enum Visibility { Visible, Hidden, Collapse }`). In this case,
20///     the derive optimizes by delegating to `ToCss` once for the whole value,
21///     then wrapping the result in `TypedValue::Keyword`.
22///
23///   * Mixed enums with unit variants: Enums that contain both unit
24///     variants (keywords) and data-carrying variants. The derive generates a
25///     `match` implementation where unit variants are reified as
26///     `TypedValue::Keyword`, and fielded variants reify by calling
27///     `.to_typed()` on their inner values by default. Variants or types
28///     marked with `skip_derive_fields` return `Err(())` instead.
29///
30/// * Structs
31///   * Bitflags structs marked with `#[css(bitflags(...))]` are handled by
32///     deriving keyword reification from their bitflag metadata.
33///
34///   * Other structs are handled similarly to data-carrying variants in mixed
35///     enums. Unless `skip_derive_fields` is set, `.to_typed()` is generated
36///     for their inner values; otherwise, they return `Err(())`.
37///
38/// Unit variants are mapped to keywords using their Rust identifier converted
39/// via `to_css_identifier`. Attributes like `#[css(keyword = "...")]` will
40/// override the behavior and use the provided keyword instead.
41///
42/// For other kinds of types (e.g. unions), no `to_typed` method is generated;
43/// the default implementation applies, which always returns `Err(())`.
44///
45/// This allows keywords to be reified automatically into `CSSKeywordValue`
46/// objects, while leaving more complex value types to be implemented
47/// incrementally as Typed OM support expands.
48///
49/// Summary of derive attributes recognized by this derive:
50///
51/// * `#[css(bitflags(single = "...", mixed = "...", overlapping_bits))]` on a
52///   struct generates keyword reification for CSS bitflags types. Values that
53///   can be represented as a single CSS keyword are reified as
54///   `TypedValue::Keyword`; values that would serialize to multiple CSS
55///   keywords are treated as unsupported and return `Err(())`.
56///
57///   `overlapping_bits` is supported for bitflags where one keyword subsumes
58///   other internal bits, such as `contain: size`.
59///
60/// * `#[typed(skip_derive_fields)]` on the type disables field recursion for
61///   structs and data-carrying enum variants.
62///
63/// * `#[css(skip)]`, `#[typed(skip)]`, or `#[typed(todo)]` on a variant mark
64///   it as unsupported and cause the generated arm to return
65///   `Err(())`.
66///
67/// * `#[css(skip)]` on a field disables reification for that field.
68///
69/// * `#[css(skip_if = "...")]` / `#[typed(skip_if = "...")]` on a field
70///   conditionally disables reification for that field. If the provided
71///   function returns `true` for the field value, the field is ignored.
72///
73/// * `#[css(contextual_skip_if = "...")]` /
74///   `#[typed(contextual_skip_if = "...")]` on a field conditionally disables
75///   reification for that field. The provided function is called with all
76///   fields in the current struct or variant. If it returns `true`, the field
77///   is ignored.
78///
79///   Typed skip annotations override CSS skip annotations when both are
80///   present.
81///
82/// * `#[css(keyword = "...")]` on a unit variant overrides the keyword
83///   string.
84///
85/// * `#[css(comma)]` on the variant indicates that fields may reify to
86///   multiple separate values. When present, multiple `TypedValue`s may be
87///   produced across the supported fields, unless
88///   `#[typed(no_multiple_values)]` is also present. If multiple values are
89///   not allowed and the derived implementation would produce more than one
90///   item, it treats the value as unsupported and returns `Err(())`.
91///
92/// * `#[typed(no_multiple_values)]` on a variant prevents it from reifying to
93///   multiple `TypedValue`s, even if `#[css(comma)]` is present.
94///
95/// * `#[css(iterable)]` on a field indicates that the field is an iterable
96///   collection whose elements should be reified individually.
97///
98/// * `#[css(if_empty = "...")]` on an iterable field causes the provided
99///   keyword to be emitted when the iterable contains no elements.
100///
101/// * `#[css(represents_keyword)]` on a bool field causes the field name to be
102///   reified as a keyword when the field is true.
103pub fn derive(mut input: DeriveInput) -> TokenStream {
104    // The mutable `where_clause` is passed down to helper functions so they
105    // can append trait bounds only when necessary. In particular, a bound of
106    // the form `T: ToTyped` is added only if the generated code actually calls
107    // `.to_typed()` on that inner value. This avoids forcing unrelated types
108    // to implement `ToTyped` when field recursion is disabled, keeping
109    // compilation requirements minimal and isolated to cases where reification
110    // recursion is actually performed.
111    let mut where_clause = input.generics.where_clause.take();
112
113    let css_input_attrs = cg::parse_input_attrs::<CssInputAttrs>(&input);
114
115    let input_attrs = cg::parse_input_attrs::<TypedInputAttrs>(&input);
116
117    if matches!(input.data, Data::Enum(..)) || css_input_attrs.bitflags.is_some() {
118        assert!(
119            !css_input_attrs.comma,
120            "#[css(comma)] is not allowed on enums or bitflags"
121        );
122    }
123
124    if let Some(ref bitflags) = css_input_attrs.bitflags {
125        assert!(where_clause.is_none(), "Generic bitflags?");
126        return derive_bitflags(&input, bitflags);
127    }
128
129    let body = match &input.data {
130        // Handle enums.
131        Data::Enum(DataEnum { variants, .. }) => {
132            // Check if this enum consists entirely of unit variants (no fields).
133            let all_unit = variants.iter().all(|v| matches!(v.fields, Fields::Unit));
134
135            if all_unit {
136                // Optimization: for all-unit enums, reuse `ToCss` once and
137                // wrap the result in a `TypedValue::Keyword`, instead of
138                // generating a full match. This avoids code bloat while
139                // producing the same runtime behavior.
140                quote! {
141                    fn to_typed(&self, dest: &mut thin_vec::ThinVec<crate::typed_om::TypedValue>) -> Result<(), ()> {
142                      let s = style_traits::ToCss::to_css_cssstring(self);
143                      dest.push(crate::typed_om::TypedValue::Keyword(crate::typed_om::KeywordValue(s)));
144                      Ok(())
145                    }
146                }
147            } else {
148                // Mixed enums: generate a `match` where unit variants map to
149                // `TypedValue::Keyword` and all other variants return
150                // `Err(())`. This is more verbose in code size, but allows
151                // selective handling of individual variants.
152                let s = Structure::new(&input);
153                let match_body = s.each_variant(|variant| {
154                    derive_variant_arm(
155                        variant,
156                        input_attrs.skip_derive_fields || input_attrs.todo_derive_fields,
157                        &mut where_clause,
158                    )
159                });
160
161                quote! {
162                    fn to_typed(&self, dest: &mut thin_vec::ThinVec<crate::typed_om::TypedValue>) -> Result<(), ()> {
163                        match *self {
164                            #match_body
165                        }
166                    }
167                }
168            }
169        },
170
171        // Handle structs that are not bitflags.
172        Data::Struct(_) => {
173            if css_input_attrs.bitflags.is_none() {
174                let s = Structure::new(&input);
175                let match_body = s.each_variant(|variant| {
176                    derive_variant_arm(
177                        variant,
178                        input_attrs.skip_derive_fields || input_attrs.todo_derive_fields,
179                        &mut where_clause,
180                    )
181                });
182
183                quote! {
184                    fn to_typed(&self, dest: &mut thin_vec::ThinVec<crate::typed_om::TypedValue>) -> Result<(), ()> {
185                        match *self {
186                            #match_body
187                        }
188                    }
189                }
190            } else {
191                quote! {}
192            }
193        },
194
195        // Otherwise, don’t emit any `to_typed` method body. The default
196        // implementation (returning `Err(())`) will apply.
197        _ => quote! {},
198    };
199
200    input.generics.where_clause = where_clause;
201
202    let name = &input.ident;
203
204    // Split the input type’s generics into pieces we can use for impl.
205    let (impl_generics, ty_generics, where_clause) = input.generics.split_for_impl();
206
207    // Put it all together into the impl block.
208    quote! {
209        impl #impl_generics crate::typed_om::ToTyped for #name #ty_generics #where_clause {
210            #body
211        }
212    }
213}

215/// Derive a `ToTyped` implementation for CSS bitflags types.
216///
217/// This mirrors the `ToCss` bitflags metadata, but with stricter output:
218/// Typed OM reification can only produce one `TypedValue::Keyword`. Mixed
219/// values that would serialize to multiple keywords return `Err(())`.
220///
221/// `overlapping_bits` needs special handling because some keywords are stored
222/// as combinations of lower-level bits. For example, `contain: size` also
223/// contains the `inline-size` bit internally. The generated code tracks
224/// consumed flags so such values still reify to `"size"` rather than being
225/// rejected as multi-keyword values.
226fn derive_bitflags(input: &syn::DeriveInput, bitflags: &CssBitflagAttrs) -> TokenStream {
227    let name = &input.ident;
228    let mut body = TokenStream::new();
229
230    for (rust_name, css_name) in bitflags.single_flags() {
231        let rust_ident = Ident::new(&rust_name, Span::call_site());
232        body.append_all(quote! {
233            if *self == Self::#rust_ident {
234                dest.push(crate::typed_om::TypedValue::Keyword(
235                    crate::typed_om::KeywordValue(style_traits::CssString::from(#css_name)),
236                ));
237                return Ok(());
238            }
239        });
240    }
241
242    body.append_all(quote! {
243        let mut flag: Option<&'static str> = None;
244    });
245
246    if bitflags.overlapping_bits {
247        // Track already-consumed flags so overlapping serialized keywords
248        // such as `contain: size` are not mistaken for multi-keyword values.
249        body.append_all(quote! {
250            let mut serialized = Self::empty();
251        });
252    }
253
254    for (rust_name, css_name) in bitflags.mixed_flags() {
255        let rust_ident = Ident::new(&rust_name, Span::call_site());
256        let init_flag = quote! {
257            if flag.is_some() {
258                return Err(());
259            }
260            flag = Some(#css_name);
261        };
262        if bitflags.overlapping_bits {
263            body.append_all(quote! {
264                if self.contains(Self::#rust_ident) && !serialized.intersects(Self::#rust_ident) {
265                    #init_flag
266                    serialized.insert(Self::#rust_ident);
267                }
268            });
269        } else {
270            body.append_all(quote! {
271                if self.intersects(Self::#rust_ident) {
272                    #init_flag
273                }
274            });
275        }
276    }
277
278    body.append_all(quote! {
279        let Some(flag) = flag else {
280            return Err(());
281        };
282
283        dest.push(crate::typed_om::TypedValue::Keyword(
284            crate::typed_om::KeywordValue(style_traits::CssString::from(flag)),
285        ));
286        Ok(())
287    });
288
289    quote! {
290        impl crate::typed_om::ToTyped for #name {
291            #[allow(unused_variables)]
292            #[inline]
293            fn to_typed(
294                &self,
295                dest: &mut thin_vec::ThinVec<crate::typed_om::TypedValue>,
296            ) -> Result<(), ()> {
297                #body
298            }
299        }
300    }
301}

303/// Generate the match arm expression for a struct or enum variant in a derived
304/// `ToTyped` implementation.
305///
306/// * Unit variants are reified into `TypedValue::Keyword`, using the variant’s
307///   identifier converted with `cg::to_css_identifier` or a custom keyword if
308///   provided through `#[css(keyword = "...")]`.
309/// * Variants marked with `#[css(skip)]` or `#[typed(skip)]` or
310///   `#[typed(todo)]` return `Err(())`.
311/// * Variants with fields delegate to `derive_variant_fields_expr()` by
312///   default; if `skip_derive_fields` is set, they return `Err(())`.
313///
314/// Note: `#[css(keyword = "...")]` overrides are now recognized in this
315/// `derive_variant_arm` path, but the support is not yet exercised because we
316/// currently have no mixed enums that use keyword overrides together with
317/// `ToTyped`. This keeps the behavior the same as before, all existing enums
318/// with keyword overrides (e.g. `#[css(keyword = "preserve-3d")]`) are still
319/// pure keyword enums and are handled through the all-unit `ToCss` path.
320fn derive_variant_arm(
321    variant: &synstructure::VariantInfo,
322    skip_derive_fields: bool,
323    where_clause: &mut Option<WhereClause>,
324) -> TokenStream {
325    let bindings = variant.bindings();
326    // Get the underlying syn AST node for this variant.
327    let ast = variant.ast();
328    let identifier = &ast.ident;
329
330    // Parse any #[css(...)] attributes attached to this variant.
331    let css_variant_attrs = cg::parse_variant_attrs_from_ast::<CssVariantAttrs>(&ast);
332
333    // Parse any #[typed(...)] attributes attached to this variant.
334    let variant_attrs = cg::parse_variant_attrs_from_ast::<TypedVariantAttrs>(&ast);
335
336    // If the variant is explicitly marked #[css(skip)], don’t generate
337    // anything for it, always return Err(()).
338    if css_variant_attrs.skip {
339        return quote! {Err(())};
340    }
341
342    // If the variant is explicitly marked #[typed(skip)] or #[typed(todo)],
343    // don’t generate anything for it, always return Err(()).
344    if variant_attrs.skip || variant_attrs.todo {
345        return quote! {Err(())};
346    }
347
348    // If the variant has no bindings (i.e. no data fields), treat it as a unit
349    // variant and reify it as a keyword.
350    if bindings.is_empty() {
351        // If #[css(keyword = "...")] is present, use it.
352        // Else convert the Rust variant name into its CSS identifier form
353        // (e.g. AvoidColumn -> "avoid-column").
354        let keyword = css_variant_attrs
355            .keyword
356            .unwrap_or_else(|| cg::to_css_identifier(&identifier.to_string()));
357
358        // Emit code to wrap this keyword into a TypedValue.
359        quote! {
360            dest.push(crate::typed_om::TypedValue::Keyword(
361                crate::typed_om::KeywordValue(style_traits::CssString::from(#keyword))
362            ));
363            Ok(())
364        }
365    } else if !skip_derive_fields {
366        let allow_multiple_values = css_variant_attrs.comma && !variant_attrs.no_multiple_values;
367
368        derive_variant_fields_expr(bindings, where_clause, allow_multiple_values)
369    } else {
370        // This variant has one or more fields, but field reification is
371        // disabled. With `skip_derive_fields`, this variant simply returns
372        // `Err(())`.
373        quote! {
374            Err(())
375        }
376    }
377}

379/// Generate the match arm expression for fields of a struct or enum variant
380/// in a derived `ToTyped` implementation.
381///
382/// This helper examines the variant’s fields and generates reification code
383/// for the usable fields.
384///
385/// * If the variant has exactly one non-iterable field, it emits a direct
386///   call to that field’s `ToTyped` implementation and adds the corresponding
387///   trait bound (e.g. `T: ToTyped`) to the `where` clause.
388///
389/// * Otherwise, it appends the reified output of the supported fields to the
390///   destination and then validates whether the enclosing variant is allowed
391///   to produce multiple `TypedValue`s.
392///
393/// Fields marked with `#[css(skip)]`, or skipped by css/typed `skip_if` or
394/// `contextual_skip_if` annotations, are ignored.
395fn derive_variant_fields_expr(
396    bindings: &[BindingInfo],
397    where_clause: &mut Option<WhereClause>,
398    allow_multiple_values: bool,
399) -> TokenStream {
400    // Filter out fields marked with #[css(skip)] so they are ignored during
401    // reification.
402    let mut iter = bindings
403        .iter()
404        .filter_map(|binding| {
405            let css_field_attrs = cg::parse_field_attrs::<CssFieldAttrs>(&binding.ast());
406            let field_attrs = cg::parse_field_attrs::<TypedFieldAttrs>(&binding.ast());
407            if css_field_attrs.skip {
408                return None;
409            }
410            Some((binding, css_field_attrs, field_attrs))
411        })
412        .peekable();
413
414    // If no usable fields remain, generate code that just returns Err(()).
415    let (first, css_field_attrs, field_attrs) = match iter.next() {
416        Some(triple) => triple,
417        None => return quote! { Err(()) },
418    };
419
420    // At this point we have at least one usable field in `first`.
421
422    // Handle the simple case of exactly one non-iterable field.
423    if !css_field_attrs.iterable && iter.peek().is_none() {
424        let expr =
425            derive_single_field_expr(first, css_field_attrs, field_attrs, where_clause, bindings);
426
427        return quote! {{
428            #expr
429            Ok(())
430        }};
431    }
432
433    // Handle the general case by appending reified output from the supported
434    // fields directly to the destination.
435    let mut expr =
436        derive_single_field_expr(first, css_field_attrs, field_attrs, where_clause, bindings);
437    for (binding, css_field_attrs, field_attrs) in iter {
438        derive_single_field_expr(
439            binding,
440            css_field_attrs,
441            field_attrs,
442            where_clause,
443            bindings,
444        )
445        .to_tokens(&mut expr)
446    }
447
448    quote! {{
449        let old_len = dest.len();
450        #expr
451        if !#allow_multiple_values && dest.len() - old_len > 1 {
452            dest.truncate(old_len);
453            return Err(());
454        }
455        Ok(())
456    }}
457}

459/// Generate the expression used to reify a single field in a derived
460/// `ToTyped` implementation.
461///
462/// For fields marked with `#[css(iterable)]`, this helper generates code that
463/// iterates over the field and calls `ToTyped::to_typed` for each element. If
464/// `#[css(if_empty = "...")]` is present, the generated code emits the
465/// specified keyword when the iterable is empty.
466///
467/// For non-iterable fields, it generates a direct `ToTyped::to_typed` call
468/// for the field value.
469///
470/// If a css/typed `skip_if` or `contextual_skip_if` annotation is present and
471/// the provided function returns `true`, the field contributes no reified
472/// output. Typed skip annotations override CSS skip annotations.
473///
474/// The appropriate `T: ToTyped` bounds for the field type or iterable element
475/// type(s) are added to the `where` clause.
476fn derive_single_field_expr(
477    field: &BindingInfo,
478    css_field_attrs: CssFieldAttrs,
479    field_attrs: TypedFieldAttrs,
480    where_clause: &mut Option<WhereClause>,
481    bindings: &[BindingInfo],
482) -> TokenStream {
483    let mut expr = if css_field_attrs.iterable {
484        // We add `ToTyped` bounds for the iterable's element type(s), rather
485        // than for the container type itself. This avoids ToTyped forcing
486        // unrelated container types to implement ToTyped.
487        //
488        // This is a bit more involved than other derives, but it matches how
489        // Typed OM reification is structured today. If this approach works
490        // well, the helper that extracts the field types
491        // (field_generic_arguments) can be moved into cg.rs alongside other
492        // derive helpers.
493        //
494        // See also the comment in the beginning of the main `derive` fn.
495        for item_ty in field_generic_arguments(field) {
496            cg::add_predicate(
497                where_clause,
498                parse_quote!(#item_ty: crate::typed_om::ToTyped),
499            );
500        }
501
502        if let Some(if_empty) = css_field_attrs.if_empty {
503            quote! {
504                let mut iter = #field.iter().peekable();
505                if iter.peek().is_none() {
506                    dest.push(crate::typed_om::TypedValue::Keyword(
507                        crate::typed_om::KeywordValue(style_traits::CssString::from(#if_empty)),
508                    ));
509                } else {
510                    for item in iter {
511                        crate::typed_om::ToTyped::to_typed(&item, dest)?;
512                    }
513                }
514            }
515        } else {
516            quote! {
517                for item in #field.iter() {
518                    crate::typed_om::ToTyped::to_typed(&item, dest)?;
519                }
520            }
521        }
522    } else if css_field_attrs.represents_keyword {
523        let ident = field
524            .ast()
525            .ident
526            .as_ref()
527            .expect("Unnamed field with represents_keyword?");
528        let ident = cg::to_css_identifier(&ident.to_string()).replace("_", "-");
529        quote! {
530            if *#field {
531                dest.push(crate::typed_om::TypedValue::Keyword(
532                    crate::typed_om::KeywordValue(style_traits::CssString::from(#ident)),
533                ));
534            }
535        }
536    } else {
537        // Add a trait bound `T: ToTyped` to ensure the field type implements
538        // the required conversion, and emit a call to its `.to_typed()`
539        // method.
540        let ty = &field.ast().ty;
541        cg::add_predicate(where_clause, parse_quote!(#ty: crate::typed_om::ToTyped));
542
543        quote! {
544           crate::typed_om::ToTyped::to_typed(#field, dest)?;
545        }
546    };
547
548    let typed_has_skip = field_attrs.skip_if.is_some() || field_attrs.contextual_skip_if.is_some();
549
550    let skip_if = field_attrs.skip_if.or_else(|| {
551        if typed_has_skip {
552            None
553        } else {
554            css_field_attrs.skip_if
555        }
556    });
557
558    let contextual_skip_if = field_attrs.contextual_skip_if.or_else(|| {
559        if typed_has_skip {
560            None
561        } else {
562            css_field_attrs.contextual_skip_if
563        }
564    });
565
566    assert!(
567        skip_if.is_none() || contextual_skip_if.is_none(),
568        "Field should not have both skip_if and contextual_skip_if"
569    );
570
571    if let Some(condition) = skip_if {
572        expr = quote! {
573            if !#condition(#field) {
574                #expr
575            }
576        }
577    }
578
579    if let Some(condition) = contextual_skip_if {
580        expr = quote! {
581            if !#condition(#(#bindings), *) {
582                #expr
583            }
584        }
585    }
586
587    expr
588}

590/// Extract generic type arguments from a field type.
591///
592/// This helper is used by the `derive_variant_fields_expr` when handling
593/// iterable fields. The function needs to add `T: ToTyped` bounds for the
594/// item type produced by iteration (since the generated code calls
595/// `.to_typed()` on each item).
596///
597/// For example:
598///   * `Vec<T>` / `OwnedSlice<T>` -> `T`
599///   * `SmallVec<[T; N]>` -> `T`
600///
601/// The function inspects the last path segment of the field’s type and
602/// returns any generic type arguments it finds, unwrapping array forms such
603/// as `[T; N]` used by containers like `SmallVec`.
604///
605/// This is intentionally minimal and currently supports the container shapes
606/// used in style structs.
607pub(crate) fn field_generic_arguments(field: &BindingInfo) -> Vec<syn::Type> {
608    use syn::{GenericArgument, PathArguments, Type};
609
610    let ty = &field.ast().ty;
611
612    let Type::Path(type_path) = ty else {
613        return vec![];
614    };
615    let Some(seg) = type_path.path.segments.last() else {
616        return vec![];
617    };
618    let PathArguments::AngleBracketed(args) = &seg.arguments else {
619        return vec![];
620    };
621
622    let mut result = Vec::new();
623    for arg in &args.args {
624        let GenericArgument::Type(arg_ty) = arg else {
625            continue;
626        };
627
628        // If it's something like SmallVec<[T; N]>, take T.
629        match arg_ty {
630            Type::Array(arr) => result.push((*arr.elem).clone()),
631            _ => result.push(arg_ty.clone()),
632        }
633    }
634    result
635}

637#[derive(Default, FromDeriveInput)]
638#[darling(attributes(typed), default)]
639pub struct TypedInputAttrs {
640    /// Disables field-level recursion when deriving `ToTyped`.
641    ///
642    /// When set, the derive will not call `.to_typed()` on inner values (for
643    /// example, struct fields or data-carrying enum variants), and instead
644    /// the generated code will return `Err(())` for those cases.
645    ///
646    /// This is useful to avoid requiring inner types to implement `ToTyped`
647    /// when reification of those fields is not yet supported.
648    pub skip_derive_fields: bool,
649
650    /// Temporarily disables field-level recursion while marking it as TODO.
651    ///
652    /// When set, the derive will not call `.to_typed()` on inner values and
653    /// instead return `Err(())` for those cases.
654    ///
655    /// Unlike `skip_derive_fields`, this indicates that reification is
656    /// expected to be revisited later, either to implement it or to switch to
657    /// `skip_derive_fields` if it turns out to be unsupported.
658    pub todo_derive_fields: bool,
659
660    /// Prevents this type from reifying to multiple Typed OM values.
661    ///
662    /// This is useful for types that use `#[css(comma)]` for CSS
663    /// serialization, but should still be treated as unsupported when they
664    /// would produce more than one `TypedValue`.
665    pub no_multiple_values: bool,
666}

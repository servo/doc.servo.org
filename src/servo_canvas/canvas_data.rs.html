56    pub(crate) fn collect_memory_report(
57        &self,
58        canvas_id: CanvasId,
59        ops: &mut MallocSizeOfOps,
60    ) -> Vec<Report> {
61        let dimensions = self.draw_target.get_size();
62        let canvas_info_string = format!(
63            "canvas(id={}, {}x{})",
64            canvas_id.0, dimensions.width, dimensions.height
65        );
66        self.draw_target
67            .canvas_store_sizes(ops)
68            .unwrap_or_default()
69            .into_iter()
70            .map(|canvas_store_type| Report {
71                path: profile_traits::path!["canvas", canvas_info_string, canvas_store_type.name],
72                kind: canvas_store_type.kind,
73                size: canvas_store_type.size,
74            })
75            .collect()
76    }

78    #[expect(clippy::too_many_arguments)]
79    pub(crate) fn draw_image(
80        &mut self,
81        snapshot: Snapshot,
82        dest_rect: Rect<f64>,
83        source_rect: Rect<f64>,
84        smoothing_enabled: bool,
85        shadow_options: ShadowOptions,
86        composition_options: CompositionOptions,
87        transform: Transform2D<f64>,
88    ) {
89        // We round up the floating pixel values to draw the pixels
90        let source_rect = source_rect.ceil();
91        // It discards the extra pixels (if any) that won't be painted
92        let snapshot = if Rect::from_size(snapshot.size().to_f64()).contains_rect(&source_rect) {
93            snapshot.get_rect(source_rect.to_u32())
94        } else {
95            snapshot
96        };
97
98        let writer = |draw_target: &mut DrawTarget, transform| {
99            write_image::<DrawTarget>(
100                draw_target,
101                snapshot,
102                dest_rect,
103                smoothing_enabled,
104                composition_options,
105                transform,
106            );
107        };
108
109        if shadow_options.need_to_draw_shadow() {
110            let rect = Rect::new(
111                Point2D::new(dest_rect.origin.x as f32, dest_rect.origin.y as f32),
112                Size2D::new(dest_rect.size.width as f32, dest_rect.size.height as f32),
113            );
114
115            self.draw_with_shadow(
116                &rect,
117                shadow_options,
118                composition_options,
119                transform,
120                writer,
121            );
122        } else {
123            writer(&mut self.draw_target, transform);
124        }
125    }

176    pub(crate) fn fill_rect(
177        &mut self,
178        rect: &Rect<f32>,
179        style: FillOrStrokeStyle,
180        shadow_options: ShadowOptions,
181        composition_options: CompositionOptions,
182        transform: Transform2D<f64>,

197            );
198        } else {
199            self.maybe_bound_shape_with_pattern(
200                style,
201                composition_options,
202                &rect.cast(),
203                transform,
204                |self_, style| {
205                    self_
206                        .draw_target
207                        .fill_rect(rect, style, composition_options, transform);
208                },
209            );
210        }

326    /// <https://html.spec.whatwg.org/multipage/#reset-the-rendering-context-to-its-default-state>
327    pub(crate) fn recreate(&mut self, size: Option<Size2D<u64>>) {
328        let size = size
329            .unwrap_or_else(|| self.draw_target.get_size().to_u64())
330            .max(MIN_WR_IMAGE_SIZE);
331
332        // Step 1. Clear canvas's bitmap to transparent black.
333        self.draw_target = self
334            .draw_target
335            .create_similar_draw_target(&Size2D::new(size.width, size.height).cast());
336
337        self.update_image_rendering(None);
338    }

341    pub(crate) fn update_image_rendering(&mut self, canvas_epoch: Option<Epoch>) {
342        let Some(image_key) = self.image_key else {
343            return;
344        };
345
346        let (descriptor, data) = {
347            let _span =
348                profile_traits::trace_span!("image_descriptor_and_serializable_data").entered();
349            self.draw_target.image_descriptor_and_serializable_data()
350        };
351
352        self.paint_api
353            .update_image(image_key, descriptor, data, canvas_epoch);
354    }

377    fn draw_with_shadow<F>(
378        &self,
379        rect: &Rect<f32>,
380        shadow_options: ShadowOptions,
381        composition_options: CompositionOptions,
382        transform: Transform2D<f64>,
383        draw_shadow_source: F,
384    ) where
385        F: FnOnce(&mut DrawTarget, Transform2D<f64>),
386    {
387        let shadow_src_rect = transform.outer_transformed_rect(&rect.cast());
388        // Because this comes from the rect on f32 precision, casting it down should be ok.
389        let mut new_draw_target = self.create_draw_target_for_shadow(&shadow_src_rect.cast());
390        let shadow_transform = transform
391            .then(&Transform2D::identity().pre_translate(-shadow_src_rect.origin.to_vector()));
392        draw_shadow_source(&mut new_draw_target, shadow_transform);
393        self.draw_target.draw_surface_with_shadow(
394            new_draw_target.surface(),
395            &Point2D::new(
396                shadow_src_rect.origin.x as f32,
397                shadow_src_rect.origin.y as f32,
398            ),
399            shadow_options,
400            composition_options,
401        );
402    }

404    /// Push a clip to the draw target to respect the non-repeating bound (either x, y, or both)
405    /// of the given pattern.
406    fn maybe_bound_shape_with_pattern<F>(
407        &mut self,
408        style: FillOrStrokeStyle,
409        composition_options: CompositionOptions,
410        path_bound_box: &Rect<f64>,
411        transform: Transform2D<f64>,
412        draw_shape: F,
413    ) where
414        F: FnOnce(&mut Self, FillOrStrokeStyle),
415    {
416        let x_bound = style.x_bound();
417        let y_bound = style.y_bound();
418        // Clear operations are also unbounded.
419        if matches!(
420            composition_options.composition_operation,
421            CompositionOrBlending::Composition(CompositionStyle::Clear)
422        ) || (x_bound.is_none() && y_bound.is_none())
423        {
424            draw_shape(self, style);
425            return;
426        }
427        let rect = Rect::from_size(Size2D::new(
428            x_bound.unwrap_or(path_bound_box.size.width.ceil() as u32),
429            y_bound.unwrap_or(path_bound_box.size.height.ceil() as u32),
430        ))
431        .cast();
432        let rect = transform.outer_transformed_rect(&rect);
433        self.draw_target.push_clip_rect(&rect.cast());
434        draw_shape(self, style);
435        self.draw_target.pop_clip();
436    }

438    /// It reads image data from the canvas
439    /// canvas_size: The size of the canvas we're reading from
440    /// read_rect: The area of the canvas we want to read from
441    #[servo_tracing::instrument(skip_all)]
442    pub(crate) fn read_pixels(&mut self, read_rect: Option<Rect<u32>>) -> Snapshot {
443        let canvas_size = self.draw_target.get_size().cast();
444
445        if let Some(read_rect) = read_rect {
446            let canvas_rect = Rect::from_size(canvas_size);
447            if canvas_rect
448                .intersection(&read_rect)
449                .is_none_or(|rect| rect.is_empty())
450            {
451                Snapshot::empty()
452            } else {
453                self.draw_target.snapshot().get_rect(read_rect)
454            }
455        } else {
456            self.draw_target.snapshot()
457        }
458    }

475/// It writes an image to the destination target
476/// draw_target: the destination target where the image_data will be copied
477/// image_data: Pixel information of the image to be written. It takes RGBA8
478/// image_size: The size of the image to be written
479/// dest_rect: Area of the destination target where the pixels will be copied
480/// smoothing_enabled: It determines if smoothing is applied to the image result
481/// premultiply: Determines whenever the image data should be premultiplied or not
482fn write_image<DrawTarget: GenericDrawTarget>(
483    draw_target: &mut DrawTarget,
484    snapshot: Snapshot,
485    dest_rect: Rect<f64>,
486    smoothing_enabled: bool,
487    composition_options: CompositionOptions,
488    transform: Transform2D<f64>,
489) {
490    if snapshot.size().is_empty() {
491        return;
492    }
493
494    let image_rect = Rect::new(Point2D::zero(), snapshot.size().cast());
495
496    // From spec https://html.spec.whatwg.org/multipage/#dom-context-2d-drawimage
497    // When scaling up, if the imageSmoothingEnabled attribute is set to true, the user agent should attempt
498    // to apply a smoothing algorithm to the image data when it is scaled.
499    // Otherwise, the image must be rendered using nearest-neighbor interpolation.
500    let filter = if smoothing_enabled {
501        Filter::Bilinear
502    } else {
503        Filter::Nearest
504    };
505
506    let source_surface = draw_target
507        .create_source_surface_from_data(snapshot)
508        .unwrap();
509
510    draw_target.draw_surface(
511        source_surface,
512        dest_rect,
513        image_rect,
514        filter,
515        composition_options,
516        transform,
517    );
518}

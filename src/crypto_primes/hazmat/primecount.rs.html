7/// Estimate the number of primes smaller than x using the asymptotic expansion of `Li(x)` with 4 terms, i.e.:
8///
9///   $$\pi(x) \approx \frac{x}{\ln x} \left(1 + \frac{1!}{\ln x} + \frac{2!}{\ln^2 x} + \frac{3!}{\ln^3 x}\right)$$
10///
11/// For values of x up to ~$10^{29}$, consider using precalculated values for $\pi(x)$ from
12/// e.g. <https://sweet.ua.pt/tos/primes.html>.
13///
14/// # Error considerations
15///
16/// In addition to the usual floating point math limitations, there are two components to the error:
17///
18/// 1. The truncation error from using a limited number of terms (4) of the asymptotic expansion of Li(x): $|Li(x) -
19///    Li_{approx}(x)|$
20/// 2. The theoretical error, given by the absolute difference between $\pi(x)$ and Li(x): $|\pi(x) - Li(x)|$
21///
22/// ## Truncation Error
23///
24/// The size of the truncation error for the 4-term approximation $Li_{approx}(x)$ is roughly the size of the first
25/// omitted term, which is $\frac{24x}{\ln^5 x}$.
26///
27/// | Bits | Truncation error (abs)     | Truncation error (rel) |
28/// | :--- | :------------------------- | :----------------------|
29/// | 1024 | $\approx 2^{981}$          | $\approx 2^{-33}$      |
30/// | 2048 | $\approx 2^{2000}$         | $\approx 2^{-37}$      |
31/// | 4096 | $\approx 2^{4043}$         | $\approx 2^{-41}$      |
32///
33/// *(Relative error is calculated as the first omitted term divided by $Li_{approx}(x)$)*
34///
35/// ## Theoretical Error
36///
37/// Assuming RH, we can use Schoenfeld’s bound $|\pi(x) - \text{Li}(x)| < \frac{\sqrt{x} \ln x}{8\pi}$.
38/// This gives a very small theoretical error bound compared to the best known unconditional bounds[^Trudgian2014].
39///
40/// | Bits | Error Bound (abs) | Error Bound (rel) |
41/// | :--- | :---------------- | :-----------------|
42/// | 1024 | < $2^{517}$       | < $2^{-497}$      |
43/// | 2048 | < $2^{1030}$      | < $2^{-1007}$     |
44/// | 4096 | < $2^{2055}$      | < $2^{-2029}$     |
45///
46/// *(Relative error bound is calculated as Schoenfeld’s Bound (abs) divided by $Li_{approx}(x)$)*
47///
48/// ## Discussion
49///
50/// Assuming RH, the dominant error term in estimating $\pi(x)$ with $Li_{approx}(x)$
51/// comes from truncating the asymptotic expansion of Li(x).
52/// While this truncation error is extremely small in relative terms, it is large in absolute terms.
53/// Improving the Li(x) approximation to be the same order of magnitude as the Schoenfeld bound would require using
54/// hundreds of terms from the asymptotic series (or employing more complex methods like continued fractions for the
55/// remainder, which is beyond the scope of this simple approximation). In relative terms the error bound is
56/// approximately $2^{-33}$ (for 1024 bits) down to $\sim 2^{-41}$ (for 4096 bits).
57/// This corresponds to an extremely small percentage error (significantly less than $10^{-8}$ %).
58///
59/// It should be noted that while Li(x) is generally smaller than $\pi(x)$ for 'small' x, it is known that the sign of
60/// $\pi(x) - Li(x)$ changes infinitely often. It has been proven that there must be a crossing below $\sim 10^{316}$
61/// ($\sim 2^{1051}$), which is well within the ranges used in this library. Thus, users should be aware that the
62/// estimate provided here can be both greater than and smaller than the actual value of $\pi(x)$.
63///
64/// [^Dusart2010]: P. Dusart, "Estimates of Some Functions Over Primes without R.H.",
65///   [arXiv:1002.0442](https://arxiv.org/pdf/1002.0442) (2010)
66///
67/// [^Dusart2018]: P. Dusart, "Explicit estimates of some functions over primes",
68///   The Ramanujan Journal 45(1) 227-251 (2018),
69///   DOI: [10.1007/s11139-016-9839-4](https://link.springer.com/article/10.1007/s11139-016-9839-4)
70///
71/// [^Wikipedia-pcf]: <https://en.wikipedia.org/wiki/Prime-counting_function>
72///
73/// [^Schoenfeld1976]: L. Schoenfeld, "Sharper Bounds for the Chebyshev Functions θ(x) and ψ(x). II",
74///   Math. Comp. 30(134) 337-360 (1976),
75///   DOI: [10.2307/2005976](https://www.jstor.org/stable/2005976)
76///
77/// [^Trudgian2014]: T. Trudgian, "Updating the error term in the prime number theorem",
78///   [arXiv:1401.2689](https://arxiv.org/abs/1401.2689) (2014)
79#[must_use]
80pub fn estimate_primecount<const LIMBS: usize, const RHS_LIMBS: usize>(x: &Uint<LIMBS>) -> Uint<LIMBS>
81where
82    Uint<LIMBS>: Concat<LIMBS, Output = Uint<RHS_LIMBS>>,
83{
84    // Number of bits to scale by (fractional bits during division)
85    // On 64 bit CPUs and x is a U64, we use 32 bits (1 limb * 32). Else 64.
86    // On 32 bit CPUs and x is a U64, we use 32 bits (2 limb * 16). Else 64.
87    let limbs = u32::try_from(LIMBS).expect("the number of limbs fits into `u32`");
88    let scale_bits = {
89        cpubits! {
90            32 => { limbs.min(4).checked_mul(16) }
91            64 => { limbs.min(2).checked_mul(32) }
92        }
93    }
94    .expect("scale bits fit into `u32`");
95
96    let total_scale_bits = scale_bits.checked_mul(2).expect("scale bits times 2 fit into `u32`");
97
98    // Scaling factor for the denominators of the expansion terms, 2^64.
99    //
100    // The number of limbs needed for precision loss is very large,
101    // and in any case some precision loss is acceptable here.
102    #[expect(clippy::as_conversions, clippy::cast_precision_loss)]
103    let denom_scale_factor = (1u128 << scale_bits) as f64;
104
105    let ln_x = ln(x);
106
107    if !ln_x.is_finite() || ln_x <= 1.0 {
108        return Uint::ZERO;
109    }
110
111    // Calculate f64 denominators L, L^2, L^3, L^4
112    let ln_x_2 = ln_x * ln_x;
113    let ln_x_3 = ln_x_2 * ln_x;
114    let ln_x_4 = ln_x_3 * ln_x;
115
116    // Scale up a float by 2^64, round, cast to u128 and then to a wide `Uint`.
117    let f64_to_scaled_uint = |value: f64| -> NonZero<Uint<RHS_LIMBS>> {
118        let scaled = value * denom_scale_factor;
119
120        #[expect(
121            clippy::as_conversions,
122            // We checked above that ln_x > 1, and `value` will be `ln_x` to a small power.
123            clippy::cast_sign_loss,
124            // `value` won't be big enough for truncation
125            clippy::cast_possible_truncation,
126        )]
127        let scaled = round(scaled).max(1.0) as u128;
128
129        let denom = Uint::<RHS_LIMBS>::from_u128(scaled);
130        NonZero::new(denom).expect("max(1.0) ensures value is at least 1")
131    };
132
133    let d1 = f64_to_scaled_uint(ln_x);
134    let d2 = f64_to_scaled_uint(ln_x_2);
135    let d3 = f64_to_scaled_uint(ln_x_3);
136    let d4 = f64_to_scaled_uint(ln_x_4);
137
138    let x_wide: Uint<RHS_LIMBS> = x.concat(&Uint::ZERO);
139    let x_wide_scaled = x_wide.wrapping_shl_vartime(total_scale_bits);
140
141    let term1_scaled = x_wide_scaled.wrapping_div(&d1);
142    let term2_scaled = x_wide_scaled.wrapping_div(&d2);
143
144    // Term 3: (2x << total_scale_bits) / d3  (Factorial 2!)
145    let term3_scaled = x_wide_scaled.wrapping_shl_vartime(1).wrapping_div(&d3);
146
147    // Term 4: (6x << total_scale_bits) / d4  (Factorial 3!)
148    let six = Uint::<LIMBS>::from(6u64);
149    let x_times_6 = x_wide_scaled.saturating_mul(&six);
150    let term4_scaled = x_times_6.wrapping_div(&d4);
151
152    let sum_scaled = term1_scaled
153        .wrapping_add(&term2_scaled)
154        .wrapping_add(&term3_scaled)
155        .wrapping_add(&term4_scaled);
156
157    // Descale by right-shifting
158    let li_x = sum_scaled
159        .overflowing_shr_vartime(scale_bits)
160        .expect("the scaling factor is small enough to not overflow");
161    li_x.resize_checked()
162        .expect("de-scaling should leave the high half zero")
163}

232    fn assert_bit_difference<const LIMBS: usize>(candidate: Uint<LIMBS>, reference: Uint<LIMBS>, min_bit_diff: u32) {
233        let delta = if reference > candidate {
234            reference - candidate
235        } else {
236            candidate - reference
237        };
238        assert!(
239            reference.bits_vartime().abs_diff(delta.bits_vartime()) >= min_bit_diff,
240            concat!(
241                "Estimate not close enough: delta has {} bits, reference has {} bits. ",
242                "Difference should be >= {}\nEstimate: {},\nReference: {}"
243            ),
244            delta.bits_vartime(),
245            reference.bits_vartime(),
246            min_bit_diff,
247            candidate,
248            reference,
249        );
250    }

311    fn uint_to_u128<const LIMBS: usize>(x: &Uint<LIMBS>) -> u128 {
312        let limbs = x.as_limbs();
313
314        cpubits! {
315            32 => {
316                u128::from(limbs[3].0) << 96
317                | u128::from(limbs[2].0) << 64
318                | u128::from(limbs[1].0) << 32
319                | u128::from(limbs[0].0)
320            }
321            64 => {
322                (u128::from(limbs[1].0) << 64) | u128::from(limbs[0].0)
323            }
324        }
325    }

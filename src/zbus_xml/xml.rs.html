1//! A parser for D-Bus introspection XML, and a matching attribute-value escaping helper.
2//!
3//! Introspection XML only uses elements and attributes, so rather than reaching for a
4//! general-purpose XML library this is a small [`winnow`] parser that recognises exactly the
5//! D-Bus introspection grammar — `<node>`, `<interface>`, `<method>`, `<signal>`, `<property>`,
6//! `<arg>` and `<annotation>` — and builds the [`Node`] tree directly, one combinator per
7//! element deciding which attributes are required, which are optional, and which child elements
8//! are expected. Everything else a real document may carry (the XML declaration, a doctype,
9//! comments, processing instructions, CDATA and foreign elements such as Telepathy's
10//! `<tp:docstring>`) is recognised only well enough to skip over it.
11

30/// The maximum element nesting depth accepted when parsing.
31///
32/// Far deeper than any real introspection tree (four levels, plus one per nested `<node>`). The
33/// parser iterates rather than recurses over the two axes where a document can nest arbitrarily
34/// deep — nested `<node>`s and skipped-over foreign elements — so this cap only bounds the memory
35/// spent tracking the open elements, not the call stack.
36const MAX_DEPTH: usize = 1024;
37
38/// Parse a D-Bus introspection document into its [`Node`] tree.
39pub(crate) fn parse<'a>(document: &str) -> Result<Node<'a>> {
40    parse_with_warnings(document).map(|(node, _)| node)
41}
42
43/// Parse a D-Bus introspection document, also returning the warnings collected on the way.
44pub(crate) fn parse_with_warnings<'a>(document: &str) -> Result<(Node<'a>, Vec<Warning>)> {
45    let mut input = Input {
46        input: LocatingSlice::new(document),
47        state: State {
48            document,
49            warnings: Vec::new(),
50        },
51    };
52    match document_node(&mut input) {
53        // The tree owns its data, so it outlives `document` and satisfies any caller lifetime.
54        Ok(node) => Ok((node, input.state.warnings)),
55        Err(ErrMode::Backtrack(error) | ErrMode::Cut(error)) => Err(error.into_error()),
56        Err(ErrMode::Incomplete(_)) => Err(Error::Xml(XmlError::new(
57            "unexpected end of document",
58            document.len(),
59        ))),
60    }
61}

63/// The document's root element, preceded by an optional prolog (declaration, doctype, comments).
64///
65/// The root element's name is not checked, for compatibility with servers that don't name it
66/// `node` and with how previous (quick-xml-based) versions of this crate behaved.
67fn document_node<'i>(input: &mut Input<'i>) -> PResult<Node<'static>> {
68    ignorable(input)?;
69    if opt(eof).parse_next(input)?.is_some() {
70        return Err(error("missing root element", input));
71    }
72    let (tag, attrs, self_closing) = start_element(input)?;
73
74    node(input, tag, attrs, self_closing)
75}

77/// The input stream: a [`LocatingSlice`] (byte offsets for error reporting) wrapped in the
78/// parser [`State`], so the whole document and the collected warnings are reachable anywhere.
79type Input<'i> = Stateful<LocatingSlice<&'i str>, State<'i>>;
80
81/// The ambient state threaded through the parser.
82#[derive(Debug)]
83struct State<'i> {
84    /// The whole document, for slicing out the raw content of captured elements.
85    document: &'i str,
86    /// Warnings about ignored content, handed back by [`parse_with_warnings`].
87    warnings: Vec<Warning>,
88}
89
90/// Record a warning about ignored document content.
91fn warn(input: &mut Input<'_>, warning: Warning) {
92    input.state.warnings.push(warning);
93}

95/// A parser result, using [`ParseError`] as winnow's error type.
96type PResult<O> = ModalResult<O, ParseError>;

98/// A parse failure: either a positioned structural/XML error, or a rejected value (an invalid
99/// name or signature) that carries the matching [`Error`] variant.
100#[derive(Debug)]
101enum ParseError {
102    Xml {
103        message: Cow<'static, str>,
104        offset: usize,
105    },
106    Domain(Error),
107}
108
109impl ParseError {
110    /// A fatal XML error at `offset`.
111    fn xml(message: impl Into<Cow<'static, str>>, offset: usize) -> ErrMode<Self> {
112        ErrMode::Cut(ParseError::Xml {
113            message: message.into(),
114            offset,
115        })
116    }
117
118    /// A fatal error from rejecting a value (an invalid name or signature).
119    fn domain(error: Error) -> ErrMode<Self> {
120        ErrMode::Cut(ParseError::Domain(error))
121    }
122
123    fn into_error(self) -> Error {
124        match self {
125            ParseError::Xml { message, offset } => Error::Xml(XmlError::new(message, offset)),
126            ParseError::Domain(error) => error,
127        }
128    }
129}
130
131impl<'i> ParserError<Input<'i>> for ParseError {
132    type Inner = Self;
133
134    fn from_input(input: &Input<'i>) -> Self {
135        ParseError::Xml {
136            message: Cow::Borrowed("malformed markup"),
137            offset: input.current_token_start(),
138        }
139    }
140
141    fn into_inner(self) -> std::result::Result<Self::Inner, Self> {
142        Ok(self)
143    }
144}

146/// A fatal XML error at the input's current position.
147fn error(message: impl Into<Cow<'static, str>>, input: &Input<'_>) -> ErrMode<ParseError> {
148    ParseError::xml(message, input.current_token_start())
149}

151/// A `<node>` and its subtree.
152///
153/// Nested `<node>`s — the one axis on which an introspection tree itself can nest arbitrarily
154/// deep — are walked iteratively on an explicit stack, so that a deeply nested document grows
155/// this `Vec` rather than the call stack. Interfaces (and their fixed, shallow subtrees) recurse
156/// normally.
157fn node<'i>(
158    input: &mut Input<'i>,
159    tag: &'i str,
160    attrs: Attrs<'i>,
161    self_closing: bool,
162) -> PResult<Node<'static>> {
163    let root = empty_node(&attrs);
164    if self_closing {
165        return Ok(root);
166    }
167
168    let mut open: Vec<(&'i str, Node<'static>)> = vec![(tag, root)];
169    loop {
170        ignorable(input)?;
171        let tag = open.last().expect("the root is popped only by returning").0;
172        if opt(eof).parse_next(input)?.is_some() {
173            return Err(error(format!("missing `</{tag}>`"), input));
174        }
175        if let Some(close) = opt(closing_tag).parse_next(input)? {
176            if close != tag {
177                return Err(error(
178                    format!("unexpected `</{close}>` while parsing `<{tag}>`"),
179                    input,
180                ));
181            }
182            let (_, finished) = open.pop().expect("a close tag was just matched");
183            match open.last_mut() {
184                Some((_, parent)) => parent.nodes.push(finished),
185                None => return Ok(finished),
186            }
187            continue;
188        }
189
190        let (child, child_attrs, child_self_closing) = start_element(input)?;
191        match child {
192            "interface" => {
193                let interface = interface(input, child, child_attrs, child_self_closing)?;
194                open.last_mut()
195                    .expect("non-empty")
196                    .1
197                    .interfaces
198                    .push(interface);
199            }
200            "node" if child_self_closing => {
201                open.last_mut()
202                    .expect("non-empty")
203                    .1
204                    .nodes
205                    .push(empty_node(&child_attrs));
206            }
207            "node" => {
208                if open.len() >= MAX_DEPTH {
209                    return Err(error("maximum element nesting depth exceeded", input));
210                }
211                open.push((child, empty_node(&child_attrs)));
212            }
213            other if is_docstring(other) => {
214                let docstring = capture_docstring(input, other, child_self_closing)?;
215                let node = &mut open.last_mut().expect("non-empty").1;
216                node.docstring = docstring.or(node.docstring.take());
217            }
218            other => {
219                if let Some(def) =
220                    telepathy_type_def(input, other, &child_attrs, child_self_closing)?
221                {
222                    open.last_mut()
223                        .expect("non-empty")
224                        .1
225                        .telepathy_types
226                        .push(def);
227                }
228            }
229        }
230    }
231}

244/// An `<interface>` and its members.
245fn interface<'i>(
246    input: &mut Input<'i>,
247    tag: &'i str,
248    attrs: Attrs<'i>,
249    self_closing: bool,
250) -> PResult<Interface<'static>> {
251    let name = attrs.name(|n| InterfaceName::try_from(n).map_err(Error::Name))?;
252    let mut methods = Vec::new();
253    let mut properties = Vec::new();
254    let mut signals = Vec::new();
255    let mut annotations = Vec::new();
256    let mut docstring = None;
257    let mut telepathy_types = Vec::new();
258    children(
259        input,
260        tag,
261        self_closing,
262        |input, child, attrs, sc| match child {
263            "method" => {
264                methods.push(method(input, child, attrs, sc)?);
265                Ok(true)
266            }
267            "property" => {
268                properties.push(property(input, child, attrs, sc)?);
269                Ok(true)
270            }
271            "signal" => {
272                signals.push(signal(input, child, attrs, sc)?);
273                Ok(true)
274            }
275            "annotation" => {
276                annotations.push(annotation(input, child, attrs, sc)?);
277                Ok(true)
278            }
279            other if is_docstring(other) => {
280                docstring = capture_docstring(input, other, sc)?.or(docstring.take());
281                Ok(true)
282            }
283            other => {
284                if let Some(def) = telepathy_type_def(input, other, &attrs, sc)? {
285                    telepathy_types.push(def);
286                }
287                Ok(true)
288            }
289        },
290    )?;
291
292    Ok(Interface {
293        name,
294        methods,
295        properties,
296        signals,
297        annotations,
298        docstring,
299        telepathy_types,
300    })
301}

462/// An `<annotation>`: a `name`/`value` pair. Its content, if any, is ignored.
463fn annotation<'i>(
464    input: &mut Input<'i>,
465    tag: &'i str,
466    attrs: Attrs<'i>,
467    self_closing: bool,
468) -> PResult<Annotation> {
469    let name = attrs.required("name")?.to_owned();
470    let value = attrs.required("value")?.to_owned();
471    children(input, tag, self_closing, |input, child, attrs, sc| {
472        skip_unsupported(input, child, &attrs, sc)?;
473        Ok(true)
474    })?;
475
476    Ok(Annotation { name, value })
477}

479/// Dispatch each child element of the just-opened `<tag>` to `handle`, skipping over content it
480/// does not claim, until the matching `</tag>`.
481///
482/// `handle` returns whether it consumed the element's body; elements it declines (returns
483/// `Ok(false)` for) are skipped along with their subtree.
484fn children<'i>(
485    input: &mut Input<'i>,
486    tag: &'i str,
487    self_closing: bool,
488    mut handle: impl FnMut(&mut Input<'i>, &'i str, Attrs<'i>, bool) -> PResult<bool>,
489) -> PResult<()> {
490    if self_closing {
491        return Ok(());
492    }
493    loop {
494        ignorable(input)?;
495        if opt(eof).parse_next(input)?.is_some() {
496            return Err(error(format!("missing `</{tag}>`"), input));
497        }
498        if let Some(close) = opt(closing_tag).parse_next(input)? {
499            if close != tag {
500                return Err(error(
501                    format!("unexpected `</{close}>` while parsing `<{tag}>`"),
502                    input,
503                ));
504            }
505            return Ok(());
506        }
507        let (child, attrs, self_closing) = start_element(input)?;
508        if !handle(input, child, attrs, self_closing)? {
509            skip_element(input, child, self_closing)?;
510        }
511    }
512}

514/// Skip an element whose start tag has been read, along with its whole subtree.
515///
516/// Iterative (tracking the open elements on a `Vec`) so that a deeply nested foreign element
517/// cannot exhaust the call stack.
518fn skip_element<'i>(input: &mut Input<'i>, tag: &'i str, self_closing: bool) -> PResult<()> {
519    if self_closing {
520        return Ok(());
521    }
522    let mut open = vec![tag];
523    while let Some(&expected) = open.last() {
524        ignorable(input)?;
525        if opt(eof).parse_next(input)?.is_some() {
526            return Err(error(format!("missing `</{expected}>`"), input));
527        }
528        if let Some(close) = opt(closing_tag).parse_next(input)? {
529            if close != expected {
530                return Err(error(
531                    format!("unexpected `</{close}>` while parsing `<{expected}>`"),
532                    input,
533                ));
534            }
535            open.pop();
536            continue;
537        }
538        let (child, _, self_closing) = start_element(input)?;
539        if !self_closing {
540            if open.len() >= MAX_DEPTH {
541                return Err(error("maximum element nesting depth exceeded", input));
542            }
543            open.push(child);
544        }
545    }
546
547    Ok(())
548}

550/// Skip an element that has no place in the introspection format, recording a warning.
551fn skip_unsupported<'i>(
552    input: &mut Input<'i>,
553    tag: &'i str,
554    attrs: &Attrs<'i>,
555    self_closing: bool,
556) -> PResult<()> {
557    warn(input, Warning::unsupported(tag, attrs.offset - 1));
558    skip_element(input, tag, self_closing)
559}

561/// The fallback child handler for an element with no type definitions of its own: capture a
562/// Telepathy docstring into `slot` (last non-empty wins), or skip an unknown element with a
563/// warning. Always reports the child consumed.
564fn docstring_or_skip<'i>(
565    input: &mut Input<'i>,
566    child: &'i str,
567    attrs: &Attrs<'i>,
568    self_closing: bool,
569    slot: &mut Option<String>,
570) -> PResult<bool> {
571    if is_docstring(child) {
572        *slot = capture_docstring(input, child, self_closing)?.or(slot.take());
573    } else {
574        skip_unsupported(input, child, attrs, self_closing)?;
575    }
576    Ok(true)
577}

579/// Consume a Telepathy docstring element, returning its content.
580///
581/// The content is returned verbatim — typically it is HTML — with the surrounding whitespace
582/// trimmed; `None` if it is empty.
583fn capture_docstring<'i>(
584    input: &mut Input<'i>,
585    tag: &'i str,
586    self_closing: bool,
587) -> PResult<Option<String>> {
588    if self_closing {
589        return Ok(None);
590    }
591    let start = input.current_token_start();
592    let end = skip_to_close(input, tag)?;
593    let content = input.state.document[start..end].trim();
594
595    Ok((!content.is_empty()).then(|| content.to_owned()))
596}

598/// Consume an element's subtree, returning the byte offset of the `<` of its matching closing
599/// tag. The `tag`'s (non-self-closing) start tag has already been read.
600///
601/// Iterative, like [`skip_element`], so deeply nested content cannot exhaust the call stack.
602fn skip_to_close<'i>(input: &mut Input<'i>, tag: &'i str) -> PResult<usize> {
603    let mut open = vec![tag];
604    loop {
605        ignorable(input)?;
606        let here = input.current_token_start();
607        let expected = *open.last().expect("non-empty until returning");
608        if opt(eof).parse_next(input)?.is_some() {
609            return Err(error(format!("missing `</{expected}>`"), input));
610        }
611        if let Some(close) = opt(closing_tag).parse_next(input)? {
612            if close != expected {
613                return Err(error(
614                    format!("unexpected `</{close}>` while parsing `<{expected}>`"),
615                    input,
616                ));
617            }
618            open.pop();
619            if open.is_empty() {
620                return Ok(here);
621            }
622            continue;
623        }
624        let (child, _, self_closing) = start_element(input)?;
625        if !self_closing {
626            if open.len() >= MAX_DEPTH {
627                return Err(error("maximum element nesting depth exceeded", input));
628            }
629            open.push(child);
630        }
631    }
632}

634/// The part of an element or attribute name after the namespace prefix, if any.
635fn local_name(name: &str) -> &str {
636    name.rsplit(':').next().unwrap_or(name)
637}

639/// Whether `name` is a Telepathy `tp:docstring` element, under any namespace prefix.
640fn is_docstring(name: &str) -> bool {
641    local_name(name) == "docstring"
642}

644/// Whether `name` is a Telepathy `tp:type` attribute.
645///
646/// The prefix is required: without one, `type` is the signature attribute.
647fn is_tp_type(name: &str) -> bool {
648    name.split_once(':')
649        .is_some_and(|(prefix, local)| !prefix.is_empty() && local == "type")
650}

652/// A `type` attribute holding a D-Bus signature, in a Telepathy type definition.
653enum SignatureAttr {
654    Missing,
655    Invalid,
656    Value(Signature),
657}
658
659impl SignatureAttr {
660    fn parse(value: Option<&str>) -> Self {
661        match value {
662            None => SignatureAttr::Missing,
663            Some(value) => match zvariant::Signature::try_from(value.as_bytes()) {
664                // The empty signature parses as `Unit`, which is only valid as a top-level
665                // signature — inside a composed signature (`Struct`/`Mapping::signature`) it
666                // produces invalid signatures.
667                Ok(zvariant::Signature::Unit) | Err(_) => SignatureAttr::Invalid,
668                Ok(signature) => SignatureAttr::Value(Signature(signature)),
669            },
670        }
671    }
672}

674/// Parse a Telepathy type-definition element, if `tag` is one.
675///
676/// The element is consumed in all cases; `Ok(None)` — with a warning recorded — is returned
677/// when `tag` is not a type definition, or is one that cannot be parsed.
678fn telepathy_type_def<'i>(
679    input: &mut Input<'i>,
680    tag: &'i str,
681    attrs: &Attrs<'i>,
682    self_closing: bool,
683) -> PResult<Option<TypeDef>> {
684    match local_name(tag) {
685        "simple-type" => simple_type(input, tag, attrs, self_closing),
686        "enum" => enum_def(input, tag, attrs, self_closing),
687        "struct" => struct_def(input, tag, attrs, self_closing),
688        "mapping" => mapping_def(input, tag, attrs, self_closing),
689        _ => {
690            skip_unsupported(input, tag, attrs, self_closing)?;
691            Ok(None)
692        }
693    }
694}

707/// Parse a `<tp:simple-type>`: a `name` given to a plain D-Bus `type`.
708fn simple_type<'i>(
709    input: &mut Input<'i>,
710    tag: &'i str,
711    attrs: &Attrs<'i>,
712    self_closing: bool,
713) -> PResult<Option<TypeDef>> {
714    let position = attrs.offset - 1;
715    let name = attrs.optional("name").map(str::to_owned);
716    let ty = SignatureAttr::parse(attrs.optional("type"));
717    let mut docstring = None;
718    children(input, tag, self_closing, |input, child, attrs, sc| {
719        docstring_or_skip(input, child, &attrs, sc, &mut docstring)
720    })?;
721
722    let Some(name) = name else {
723        return malformed(input, tag, position, "missing attribute `name`");
724    };
725    let ty = match ty {
726        SignatureAttr::Value(ty) => ty,
727        SignatureAttr::Missing => {

729        }
730        SignatureAttr::Invalid => {
731            return malformed(input, tag, position, "invalid signature in `type`");
732        }
733    };
734
735    Ok(Some(TypeDef::SimpleType(telepathy::SimpleType {
736        name,
737        ty,
738        docstring,
739    })))
740}

742/// Parse a `<tp:enum>`: a `name`, an underlying `type` and its `<tp:enumvalue>`s.
743fn enum_def<'i>(
744    input: &mut Input<'i>,
745    tag: &'i str,
746    attrs: &Attrs<'i>,
747    self_closing: bool,
748) -> PResult<Option<TypeDef>> {
749    let position = attrs.offset - 1;
750    let name = attrs.optional("name").map(str::to_owned);
751    let ty = SignatureAttr::parse(attrs.optional("type"));
752    let mut values = Vec::new();
753    let mut incomplete_value = false;
754    let mut docstring = None;
755    children(input, tag, self_closing, |input, child, attrs, sc| {
756        if local_name(child) == "enumvalue" {
757            match enum_value(input, child, &attrs, sc)? {
758                Some(value) => values.push(value),
759                None => incomplete_value = true,
760            }
761            Ok(true)
762        } else {
763            docstring_or_skip(input, child, &attrs, sc, &mut docstring)
764        }
765    })?;
766
767    let Some(name) = name else {
768        return malformed(input, tag, position, "missing attribute `name`");
769    };
770    let ty = match ty {
771        SignatureAttr::Value(ty) => ty,
772        SignatureAttr::Missing => {
773            return malformed(input, tag, position, "missing attribute `type`");
774        }
775        SignatureAttr::Invalid => {
776            return malformed(input, tag, position, "invalid signature in `type`");
777        }
778    };
779    if incomplete_value {
780        let reason = "an enumvalue is missing its `suffix` or `value` attribute";
781        return malformed(input, tag, position, reason);
782    }
783
784    Ok(Some(TypeDef::Enum(telepathy::Enum {
785        name,
786        ty,
787        values,
788        docstring,
789    })))
790}

793fn enum_value<'i>(
794    input: &mut Input<'i>,
795    tag: &'i str,
796    attrs: &Attrs<'i>,
797    self_closing: bool,
798) -> PResult<Option<telepathy::EnumValue>> {
799    let suffix = attrs.optional("suffix").map(str::to_owned);
800    let value = attrs.optional("value").map(str::to_owned);

802    children(input, tag, self_closing, |input, child, attrs, sc| {
803        docstring_or_skip(input, child, &attrs, sc, &mut docstring)
804    })?;

806    Ok(suffix

810            value,
811            docstring,

815/// Parse a `<tp:struct>`: a `name` and its `<tp:member>`s.
816fn struct_def<'i>(
817    input: &mut Input<'i>,
818    tag: &'i str,
819    attrs: &Attrs<'i>,
820    self_closing: bool,
821) -> PResult<Option<TypeDef>> {
822    let position = attrs.offset - 1;
823    let name = attrs.optional("name").map(str::to_owned);
824    let (members, incomplete_member, docstring) = members(input, tag, self_closing)?;
825
826    let Some(name) = name else {
827        return malformed(input, tag, position, "missing attribute `name`");
828    };
829    if incomplete_member {
830        return malformed(input, tag, position, INCOMPLETE_MEMBER);
831    }
832    if members.is_empty() {
833        return malformed(input, tag, position, "no members");
834    }
835
836    Ok(Some(TypeDef::Struct(telepathy::Struct {
837        name,
838        members,
839        docstring,
840    })))
841}

843/// Parse a `<tp:mapping>`: a `name` and exactly two `<tp:member>`s (a basic key and a value).
844fn mapping_def<'i>(
845    input: &mut Input<'i>,
846    tag: &'i str,
847    attrs: &Attrs<'i>,
848    self_closing: bool,
849) -> PResult<Option<TypeDef>> {
850    let position = attrs.offset - 1;
851    let name = attrs.optional("name").map(str::to_owned);
852    let (members, incomplete_member, docstring) = members(input, tag, self_closing)?;
853
854    let Some(name) = name else {
855        return malformed(input, tag, position, "missing attribute `name`");
856    };
857    if incomplete_member {
858        return malformed(input, tag, position, INCOMPLETE_MEMBER);
859    }
860    let mut members = members.into_iter();
861    let (key, value) = match (members.next(), members.next(), members.next()) {
862        (Some(key), Some(value), None) => (key, value),
863        _ => return malformed(input, tag, position, "expected exactly 2 members"),
864    };
865    if !telepathy::is_basic(key.ty()) {
866        return malformed(input, tag, position, "the key is not a basic type");
867    }
868
869    Ok(Some(TypeDef::Mapping(telepathy::Mapping {
870        name,
871        key,
872        value,
873        docstring,
874    })))
875}

971/// A markup declaration such as `<!DOCTYPE …>`.
972///
973/// Tried after [`comment`] and [`cdata`], which also open with `<!`.
974fn markup_declaration<'i>(input: &mut Input<'i>) -> PResult<()> {
975    preceded("<!", cut_err(markup_body)).parse_next(input)
976}

978/// The body of a markup declaration, up to and including the terminating `>`.
979///
980/// A `>` inside a quoted literal or an internal subset (`[ … ]`, which may itself contain `>`s)
981/// does not terminate the declaration.
982fn markup_body<'i>(input: &mut Input<'i>) -> PResult<()> {
983    let mut bracket_depth = 0usize;
984    loop {
985        match any.parse_next(input)? {
986            quote @ ('"' | '\'') => {
987                (take_until(0.., quote), any).void().parse_next(input)?;
988            }
989            '[' => bracket_depth += 1,
990            ']' => bracket_depth = bracket_depth.saturating_sub(1),
991            '>' if bracket_depth == 0 => return Ok(()),
992            _ => (),
993        }
994    }
995}

1007/// A start tag such as `<arg name="foo" type="s">`, or the self-closing `<arg … />`, yielding
1008/// the element name, its [`Attrs`] and whether it is self-closing.
1009fn start_element<'i>(input: &mut Input<'i>) -> PResult<(&'i str, Attrs<'i>, bool)> {
1010    let (name, span) = preceded('<', xml_name.with_span()).parse_next(input)?;
1011    let pairs = attributes(input)?;
1012    let self_closing =
1013        preceded(whitespace, alt(("/>".value(true), ">".value(false)))).parse_next(input)?;
1014
1015    Ok((
1016        name,
1017        Attrs {
1018            element: name,
1019            offset: span.start,
1020            pairs,
1021        },
1022        self_closing,
1023    ))

1026/// The attributes of an element: names paired with their unescaped values, in document order.
1027struct Attrs<'i> {
1028    /// The element name, for error messages.
1029    element: &'i str,
1030    /// The byte offset of the element name, for missing-attribute errors.
1031    offset: usize,
1032    pairs: Vec<(&'i str, Cow<'i, str>)>,
1033}
1034
1035impl<'i> Attrs<'i> {
1036    /// The value of `key`, if present.
1037    fn optional(&self, key: &str) -> Option<&str> {
1038        self.pairs
1039            .iter()
1040            .find(|(name, _)| *name == key)
1041            .map(|(_, value)| value.as_ref())
1042    }
1043
1044    /// The value of `key`, or a "missing attribute" error anchored at the element.
1045    fn required(&self, key: &str) -> PResult<&str> {
1046        self.optional(key).ok_or_else(|| {
1047            ParseError::xml(
1048                format!("missing attribute `{key}` on `<{}>`", self.element),
1049                self.offset,
1050            )
1051        })
1052    }
1053
1054    /// The required `name` attribute, validated by `parse` (e. g. into an [`InterfaceName`]).
1055    fn name<T>(&self, parse: impl FnOnce(String) -> std::result::Result<T, Error>) -> PResult<T> {
1056        parse(self.required("name")?.to_owned()).map_err(ParseError::domain)
1057    }
1058
1059    /// The required `type` attribute, parsed as a signature.
1060    fn signature(&self) -> PResult<Signature> {
1061        zvariant::Signature::try_from(self.required("type")?.as_bytes())
1062            .map(Signature)
1063            .map_err(|e| ParseError::domain(zvariant::Error::from(e).into()))
1064    }
1065
1066    /// The value of the Telepathy `tp:type` attribute, if present.
1067    fn tp_type(&self) -> Option<&str> {
1068        self.pairs
1069            .iter()
1070            .find(|(name, _)| is_tp_type(name))
1071            .map(|(_, value)| value.as_ref())
1072    }
1073}
1074
1075/// The attributes of an element, unescaped and with duplicates rejected.
1076fn attributes<'i>(input: &mut Input<'i>) -> PResult<Vec<(&'i str, Cow<'i, str>)>> {
1077    let raw: Vec<Attribute<'i>> = repeat(0.., attribute).parse_next(input)?;
1078    let mut pairs: Vec<(&'i str, Cow<'i, str>)> = Vec::with_capacity(raw.len());
1079    // Real elements carry only a few attributes, for which a linear duplicate scan is cheapest;
1080    // but that scan is O(n²), so once a (only ever hostile) tag grows past a threshold, track the
1081    // names in a set to keep the work linear.
1082    let mut seen: Option<HashSet<&'i str>> = None;
1083    for attr in raw {
1084        let duplicate = match seen {
1085            Some(ref mut seen) => !seen.insert(attr.name),
1086            None => {
1087                let duplicate = pairs.iter().any(|(name, _)| *name == attr.name);
1088                if !duplicate && pairs.len() >= 32 {
1089                    let mut set: HashSet<&'i str> = pairs.iter().map(|(name, _)| *name).collect();
1090                    set.insert(attr.name);
1091                    seen = Some(set);
1092                }
1093                duplicate
1094            }
1095        };
1096        if duplicate {
1097            return Err(ParseError::xml(
1098                format!("duplicate attribute `{}`", attr.name),
1099                attr.name_offset,
1100            ));
1101        }
1102        let value = unescape(attr.value)
1103            .map_err(|(message, at)| ParseError::xml(message, attr.value_offset + at))?;
1104        pairs.push((attr.name, value));
1105    }
1106
1107    Ok(pairs)
1108}

1110/// A single attribute with the byte offsets needed to anchor errors precisely.
1111struct Attribute<'i> {
1112    name: &'i str,
1113    name_offset: usize,
1114    /// The raw (still escaped) value.
1115    value: &'i str,
1116    value_offset: usize,
1117}

1119/// A single attribute, e. g. `name="foo"`, along with the whitespace XML requires before it.
1120///
1121/// The mandatory leading whitespace is what rejects run-together attributes like
1122/// `name="a"type="s"`.
1123fn attribute<'i>(input: &mut Input<'i>) -> PResult<Attribute<'i>> {
1124    let (name, name_span) = preceded(whitespace1, xml_name.with_span()).parse_next(input)?;
1125    (whitespace, '=', whitespace).parse_next(input)?;
1126    let (value, value_offset) = quoted_value(input)?;
1127
1128    Ok(Attribute {
1129        name,
1130        name_offset: name_span.start,
1131        value,
1132        value_offset,
1133    })
1134}

1136/// An element or attribute name: everything up to a delimiter.
1137///
1138/// Names are only tokenized here, not validated against the XML grammar: the parser only ever
1139/// compares them against the fixed set of introspection names.
1140fn xml_name<'i>(input: &mut Input<'i>) -> PResult<&'i str> {
1141    take_while(1.., |c: char| {
1142        !c.is_ascii_whitespace() && !matches!(c, '=' | '/' | '>' | '<')
1143    })
1144    .parse_next(input)
1145}

1147/// A quoted attribute value, returning its raw (still escaped) contents and their byte offset.
1148fn quoted_value<'i>(input: &mut Input<'i>) -> PResult<(&'i str, usize)> {
1149    alt((
1150        delimited('"', take_until(0.., '"').with_span(), '"'),
1151        delimited('\'', take_until(0.., '\'').with_span(), '\''),
1152    ))
1153    .map(|(value, span)| (value, span.start))
1154    .parse_next(input)
1155}

1157/// Optional run of XML whitespace.
1158fn whitespace<'i>(input: &mut Input<'i>) -> PResult<&'i str> {
1159    take_while(0.., |c: char| c.is_ascii_whitespace()).parse_next(input)
1160}

1162/// At least one XML whitespace character.
1163fn whitespace1<'i>(input: &mut Input<'i>) -> PResult<&'i str> {

1167/// Resolve entity and character references in an attribute value and normalize whitespace.
1168///
1169/// Per the XML attribute-value normalization rules, literal whitespace characters are replaced
1170/// with spaces, while whitespace escaped through character references (e.g. `&#10;`) is kept. A
1171/// literal `\r\n` collapses to a single space, as line-ending normalization (which folds it to a
1172/// lone `\n`) precedes attribute-value normalization. On failure, the error carries the byte
1173/// offset of the offending reference within `value`.
1174fn unescape(value: &str) -> std::result::Result<Cow<'_, str>, (String, usize)> {
1175    if !value.contains(['&', '\t', '\n', '\r']) {
1176        return Ok(Cow::Borrowed(value));
1177    }
1178
1179    let mut unescaped = String::with_capacity(value.len());
1180    let mut rest = value;
1181    while let Some(i) = rest.find(['&', '\t', '\n', '\r']) {
1182        unescaped.push_str(&rest[..i]);
1183        let reference = value.len() - rest.len() + i;
1184        if rest.as_bytes()[i] != b'&' {
1185            unescaped.push(' ');
1186            // A `\r\n` pair is a single line ending, so it normalizes to one space, not two.
1187            let width = if rest.as_bytes()[i] == b'\r' && rest.as_bytes().get(i + 1) == Some(&b'\n')
1188            {
1189                2
1190            } else {
1191                1
1192            };
1193            rest = &rest[i + width..];
1194            continue;
1195        }
1196        rest = &rest[i + 1..];
1197        let end = rest
1198            .find(';')
1199            .ok_or(("unterminated entity reference".to_string(), reference))?;
1200        let entity = &rest[..end];
1201        match entity {
1202            "amp" => unescaped.push('&'),
1203            "lt" => unescaped.push('<'),
1204            "gt" => unescaped.push('>'),
1205            "quot" => unescaped.push('"'),
1206            "apos" => unescaped.push('\''),
1207            _ => unescaped.push(char_reference(entity).map_err(|e| (e, reference))?),
1208        }
1209        rest = &rest[end + 1..];
1210    }
1211    unescaped.push_str(rest);
1212
1213    Ok(Cow::Owned(unescaped))
1214}

1216/// Resolve a numeric character reference (the part of `&#...;` between `&` and `;`).
1217fn char_reference(entity: &str) -> std::result::Result<char, String> {
1218    let invalid = || format!("invalid character reference `&{entity};`");
1219
1220    let code = if let Some(hex) = entity
1221        .strip_prefix("#x")
1222        .or_else(|| entity.strip_prefix("#X"))
1223    {
1224        if hex.is_empty() || !hex.bytes().all(|b| b.is_ascii_hexdigit()) {
1225            return Err(invalid());
1226        }
1227        u32::from_str_radix(hex, 16).ok()
1228    } else if let Some(dec) = entity.strip_prefix('#') {
1229        if dec.is_empty() || !dec.bytes().all(|b| b.is_ascii_digit()) {
1230            return Err(invalid());
1231        }
1232        dec.parse().ok()
1233    } else {
1234        return Err(format!("unknown entity `&{entity};`"));
1235    };
1236
1237    code.and_then(char::from_u32)
1238        .filter(|c| is_xml_char(*c))
1239        .ok_or_else(invalid)
1240}
1241
1242/// Whether `c` is a valid XML 1.0 character (the `Char` production).
1243fn is_xml_char(c: char) -> bool {
1244    matches!(
1245        c,
1246        '\u{9}' | '\u{A}' | '\u{D}' | '\u{20}'..='\u{D7FF}' | '\u{E000}'..='\u{FFFD}' | '\u{10000}'..='\u{10FFFF}'
1247    )
1248}

1250/// Escape a string for use as an attribute value.
1251///
1252/// Whitespace other than the space character is escaped as character references so that it
1253/// survives the attribute-value normalization done by parsers.
1254pub(crate) fn escape(value: &str) -> Cow<'_, str> {
1255    if !value.contains(['&', '<', '>', '"', '\'', '\t', '\n', '\r']) {
1256        return Cow::Borrowed(value);
1257    }
1258
1259    let mut escaped = String::with_capacity(value.len() + 8);
1260    for c in value.chars() {
1261        match c {
1262            '&' => escaped.push_str("&amp;"),
1263            '<' => escaped.push_str("&lt;"),
1264            '>' => escaped.push_str("&gt;"),
1265            '"' => escaped.push_str("&quot;"),
1266            '\'' => escaped.push_str("&apos;"),
1267            '\t' => escaped.push_str("&#9;"),
1268            '\n' => escaped.push_str("&#10;"),
1269            '\r' => escaped.push_str("&#13;"),
1270            c => escaped.push(c),
1271        }
1272    }
1273
1274    Cow::Owned(escaped)

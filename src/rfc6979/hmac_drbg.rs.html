7/// Implementation of [NIST SP800-90A]'s `HMAC_DRBG` HMAC-based deterministic random bit generator.
8///
9/// [NIST SP800-90A]: https://csrc.nist.gov/publications/detail/sp/800-90a/rev-1/final
10// TODO(tarcieri): extract this into the `hmacdrbg` crate: RustCrypto/CSRNGs#7
11pub(crate) struct HmacDrbg<D>
12where
13    D: OutputSizeUser,
14{
15    /// HMAC key `K` (see RFC 6979 Section 3.2.c)
16    k: Array<u8, D::OutputSize>,
17
18    /// Chaining value `V` (see RFC 6979 Section 3.2.c)
19    v: Array<u8, D::OutputSize>,
20}
21
22impl<D> HmacDrbg<D>
23where
24    D: BlockSizeUser + Digest,
25{
26    /// Initialize `HMAC_DRBG`.
27    #[must_use]
28    pub(crate) fn new(entropy_input: &[u8], nonce: &[u8], personalization_string: &[u8]) -> Self {
29        let mut drbg = Self {
30            k: Array::default(),
31            v: Array::default(),
32        };
33        drbg.v.fill(0x01);
34        drbg.update(&[entropy_input, nonce, personalization_string]);
35        drbg
36    }
37
38    /// Write the next `HMAC_DRBG` output to the given byte slice.
39    pub(crate) fn fill_bytes(&mut self, out: &mut [u8]) {
40        for out_chunk in out.chunks_mut(self.v.len()) {
41            self.update_v();
42            out_chunk.copy_from_slice(&self.v[..out_chunk.len()]);
43        }
44
45        self.update(&[]);
46    }
47
48    /// Update `K` and `V` using the provided seed material.
49    fn update(&mut self, inputs: &[&[u8]]) {
50        let mut hmac = self.hmac_k();
51        hmac.update(&self.v);
52        hmac.update(&[0x00]);
53        inputs.iter().for_each(|&input| hmac.update(input));
54
55        self.k = hmac.finalize().into_bytes();
56        self.update_v();
57
58        if !inputs.is_empty() {
59            let mut hmac = self.hmac_k();
60            hmac.update(&self.v);
61            hmac.update(&[0x01]);
62            inputs.iter().for_each(|&input| hmac.update(input));
63
64            self.k = hmac.finalize().into_bytes();
65            self.update_v();
66        }
67    }
68
69    /// Update `V = HMAC(K, V)`.
70    fn update_v(&mut self) {
71        let mut hmac = self.hmac_k();
72        hmac.update(&self.v);
73        self.v = hmac.finalize().into_bytes();
74    }
75
76    /// Initialize HMAC using the current value of `K`.
77    fn hmac_k(&self) -> SimpleHmac<D> {
78        SimpleHmac::<D>::new_from_slice(&self.k).expect("HMAC key should be valid")
79    }
80}

92mod tests {
93    use super::HmacDrbg;
94    use hex_literal::hex;
95    use sha2::Sha256;

97    // ECDSA with K-163+SHA-256.
98    #[test]
99    fn rfc6979_appendix_a1_k163() {
100        let x = hex!("009A4D6792295A7F730FC3F2B49CBC0F62E862272F");
101
102        // NOTE: this input is `SHA256("sample")` with `bits2octets` applied in advance
103        let h = hex!("01795EDF0D54DB760F156D0DAC04C0322B3A204224");
104
105        let mut drbg = HmacDrbg::<Sha256>::new(&x, &h, &[]);
106        let mut out = [0u8; 21];
107        drbg.fill_bytes(&mut out);
108        assert_eq!(out, hex!("9305A46DE7FF8EB107194DEBD3FD48AA20D5E7656C"));
109    }

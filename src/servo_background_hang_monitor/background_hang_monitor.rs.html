24    monitoring_enabled: bool,
25}

27impl HangMonitorRegister {
28    /// Start a new hang monitor worker, and return a handle to register components for monitoring,
29    /// as well as a join handle on the worker thread.
30    pub fn init(
31        constellation_chan: GenericSender<HangAlert>,
32        control_port: GenericReceiver<BackgroundHangMonitorControlMsg>,
33        monitoring_enabled: bool,
34    ) -> (Box<dyn BackgroundHangMonitorRegister>, JoinHandle<()>) {
35        let (sender, port) = unbounded();
36
37        let join_handle = Builder::new()
38            .name("BackgroundHangMonitor".to_owned())
39            .spawn(move || {
40                let mut monitor = BackgroundHangMonitorWorker::new(
41                    constellation_chan,
42                    control_port,
43                    port,
44                    monitoring_enabled,
45                );
46                while monitor.run() {
47                    // Monitoring until all senders have been dropped...
48                }
49            })
50            .expect("Couldn't start BHM worker.");
51        (
52            Box::new(HangMonitorRegister {
53                sender,
54                monitoring_enabled,
55            }),
56            join_handle,
57        )
58    }
59}

61impl BackgroundHangMonitorRegister for HangMonitorRegister {
62    /// Register a component for monitoring.
63    /// Returns a dedicated wrapper around a sender
64    /// to be used for communication with the hang monitor worker.
65    fn register_component(
66        &self,
67        component_id: MonitoredComponentId,
68        transient_hang_timeout: Duration,
69        permanent_hang_timeout: Duration,
70        exit_signal: Box<dyn BackgroundHangMonitorExitSignal>,
71    ) -> Box<dyn BackgroundHangMonitor> {
72        let bhm_chan = BackgroundHangMonitorChan::new(
73            self.sender.clone(),
74            component_id,
75            self.monitoring_enabled,
76        );
77
78        bhm_chan.send(MonitoredComponentMsg::Register(
79            SamplerImpl::default(),
80            thread::current().name().map(str::to_owned),
81            transient_hang_timeout,
82            permanent_hang_timeout,
83            exit_signal,
84        ));
85        Box::new(bhm_chan)
86    }
87}

95/// Messages sent from monitored components to the monitor.
96enum MonitoredComponentMsg {
97    /// Register component for monitoring,
98    Register(
99        SamplerImpl,
100        Option<String>,
101        Duration,
102        Duration,
103        Box<dyn BackgroundHangMonitorExitSignal>,
104    ),
105    /// Unregister component for monitoring.
106    Unregister,
107    /// Notify start of new activity for a given component,
108    NotifyActivity(HangAnnotation),
109    /// Notify start of waiting for a new task to come-in for processing.
110    NotifyWait,
111}

113/// A wrapper around a sender to the monitor,
114/// which will send the Id of the monitored component along with each message,
115/// and keep track of whether the monitor is still listening on the other end.
116struct BackgroundHangMonitorChan {
117    sender: MonitoredComponentSender,
118    component_id: MonitoredComponentId,
119    monitoring_enabled: bool,
120}
121
122impl BackgroundHangMonitorChan {
123    fn new(
124        sender: MonitoredComponentSender,
125        component_id: MonitoredComponentId,
126        monitoring_enabled: bool,
127    ) -> Self {
128        BackgroundHangMonitorChan {
129            sender,
130            component_id,
131            monitoring_enabled,
132        }
133    }
134
135    fn send(&self, msg: MonitoredComponentMsg) {
136        self.sender
137            .send((self.component_id.clone(), msg))
138            .expect("BHM is gone");
139    }
140}

142impl BackgroundHangMonitor for BackgroundHangMonitorChan {
143    fn notify_activity(&self, annotation: HangAnnotation) {
144        if self.monitoring_enabled {
145            let msg = MonitoredComponentMsg::NotifyActivity(annotation);
146            self.send(msg);
147        }
148    }
149    fn notify_wait(&self) {
150        if self.monitoring_enabled {
151            let msg = MonitoredComponentMsg::NotifyWait;
152            self.send(msg);
153        }
154    }
155    fn unregister(&self) {
156        let msg = MonitoredComponentMsg::Unregister;
157        self.send(msg);
158    }
159}

161struct MonitoredComponent {
162    sampler: SamplerImpl,
163    last_activity: Instant,
164    last_annotation: Option<HangAnnotation>,
165    transient_hang_timeout: Duration,
166    permanent_hang_timeout: Duration,
167    sent_transient_alert: bool,
168    sent_permanent_alert: bool,
169    is_waiting: bool,
170    exit_signal: Box<dyn BackgroundHangMonitorExitSignal>,

173struct BackgroundHangMonitorWorker {
174    component_names: FxHashMap<MonitoredComponentId, String>,
175    monitored_components: FxHashMap<MonitoredComponentId, MonitoredComponent>,
176    constellation_chan: GenericSender<HangAlert>,
177    port: Receiver<(MonitoredComponentId, MonitoredComponentMsg)>,
178    control_port: Option<RoutedReceiver<BackgroundHangMonitorControlMsg>>,
179    monitoring_enabled: bool,
180    shutting_down: bool,
181}
182
183type MonitoredComponentSender = Sender<(MonitoredComponentId, MonitoredComponentMsg)>;
184type MonitoredComponentReceiver = Receiver<(MonitoredComponentId, MonitoredComponentMsg)>;
185
186impl BackgroundHangMonitorWorker {
187    fn new(
188        constellation_chan: GenericSender<HangAlert>,
189        control_port: GenericReceiver<BackgroundHangMonitorControlMsg>,
190        port: MonitoredComponentReceiver,
191        monitoring_enabled: bool,
192    ) -> Self {
193        let control_port = control_port.route_preserving_errors();
194        Self {
195            component_names: Default::default(),
196            monitored_components: Default::default(),
197            constellation_chan,
198            port,
199            control_port: Some(control_port),
200            monitoring_enabled,
201            shutting_down: Default::default(),
202        }
203    }
204
205    fn run(&mut self) -> bool {
206        let tick = if self.monitoring_enabled {
207            after(Duration::from_millis(100))
208        } else {
209            never()
210        };
211
212        // Helper enum to collect messages from different ports depending
213        // on the BHM state.
214        enum BhmMessage {
215            ComponentMessage((MonitoredComponentId, MonitoredComponentMsg)),
216            Exit,
217            ControlError(String),
218            ControlDisconnected,
219            Tick,
220            PortDisconnected,
221        }
222
223        let result = if let Some(ref control_port) = self.control_port {
224            select! {
225                recv(self.port) -> event => {
226                    match event {
227                        Ok(msg) => BhmMessage::ComponentMessage(msg),
228                        Err(_) => BhmMessage::PortDisconnected,
229                    }
230                },
231                recv(control_port) -> event => {
232                    match event {
233                        Ok(Ok(BackgroundHangMonitorControlMsg::Exit)) => BhmMessage::Exit,
234                        Ok(Err(e)) => BhmMessage::ControlError(format!("{e:?}")),
235                        Err(_) => BhmMessage::ControlDisconnected,
236                    }
237                }
238                recv(tick) -> _ => BhmMessage::Tick,
239            }
240        } else {
241            // control_port is already disconnected, just wait on port and tick
242            select! {
243                recv(self.port) -> event => {
244                    match event {
245                        Ok(msg) => BhmMessage::ComponentMessage(msg),
246                        Err(_) => BhmMessage::PortDisconnected,
247                    }
248                },
249                recv(tick) -> _ => BhmMessage::Tick,
250            }
251        };
252
253        match result {
254            BhmMessage::PortDisconnected => {
255                // All senders have dropped,
256                // which means all monitored components have shut down,
257                // and so we can as well.
258                return false;
259            },
260            BhmMessage::Exit => {
261                for component in self.monitored_components.values_mut() {
262                    component.exit_signal.signal_to_exit();
263                }
264
265                // Note the start of shutdown to ensure exit propagates,
266                // even to components that have yet to register themselves,
267                // from this point on.
268                self.shutting_down = true;
269            },
270            BhmMessage::ControlError(e) => {
271                warn!("BackgroundHangMonitorWorker control message deserialization error: {e}");
272            },
273            BhmMessage::ControlDisconnected => {
274                // The control port has disconnected. This can happen during
275                // shutdown when the EventLoop drops before all script threads
276                // have finished. Instead of exiting immediately, we signal
277                // all components to exit and continue running until they all
278                // unregister (indicated by `self.port` disconnecting).
279                if !self.shutting_down {
280                    error!("BHM control disconnected before shutting down!");
281                    for component in self.monitored_components.values_mut() {
282                        component.exit_signal.signal_to_exit();
283                    }
284                    self.shutting_down = true;
285                }
286                self.control_port = None;
287            },
288            BhmMessage::ComponentMessage(msg) => {
289                self.handle_msg(msg);
290                while let Ok(another_msg) = self.port.try_recv() {
291                    // Handle any other incoming messages before performing a hang checkpoint.
292                    self.handle_msg(another_msg);
293                }
294            },
295            BhmMessage::Tick => {
296                // Just proceed to checkpoint
297            },
298        }
299
300        self.perform_a_hang_monitor_checkpoint();
301
302        true
303    }
304
305    fn handle_msg(&mut self, msg: (MonitoredComponentId, MonitoredComponentMsg)) {
306        match msg {
307            (
308                component_id,
309                MonitoredComponentMsg::Register(
310                    sampler,
311                    name,
312                    transient_hang_timeout,
313                    permanent_hang_timeout,
314                    exit_signal,
315                ),
316            ) => {
317                // If we are shutting down,
318                // propagate it to the component,
319                // and register it(the component will unregister itself
320                // as part of handling the exit).
321                if self.shutting_down {
322                    exit_signal.signal_to_exit();
323                }
324
325                let component = MonitoredComponent {
326                    sampler,
327                    last_activity: Instant::now(),
328                    last_annotation: None,
329                    transient_hang_timeout,
330                    permanent_hang_timeout,
331                    sent_transient_alert: false,
332                    sent_permanent_alert: false,
333                    is_waiting: true,
334                    exit_signal,
335                };
336                if let Some(name) = name {
337                    self.component_names.insert(component_id.clone(), name);
338                }
339                assert!(
340                    self.monitored_components
341                        .insert(component_id, component)
342                        .is_none(),
343                    "This component was already registered for monitoring."
344                );
345            },
346            (component_id, MonitoredComponentMsg::Unregister) => {
347                self.monitored_components
348                    .remove_entry(&component_id)
349                    .expect("Received Unregister for an unknown component");
350            },
351            (component_id, MonitoredComponentMsg::NotifyActivity(annotation)) => {
352                let component = self
353                    .monitored_components
354                    .get_mut(&component_id)
355                    .expect("Received NotifyActivity for an unknown component");
356                component.last_activity = Instant::now();
357                component.last_annotation = Some(annotation);
358                component.sent_transient_alert = false;
359                component.sent_permanent_alert = false;
360                component.is_waiting = false;
361            },
362            (component_id, MonitoredComponentMsg::NotifyWait) => {
363                let component = self
364                    .monitored_components
365                    .get_mut(&component_id)
366                    .expect("Received NotifyWait for an unknown component");
367                component.last_activity = Instant::now();
368                component.sent_transient_alert = false;
369                component.sent_permanent_alert = false;
370                component.is_waiting = true;
371            },
372        }
373    }
374
375    fn perform_a_hang_monitor_checkpoint(&mut self) {
376        for (component_id, monitored) in self.monitored_components.iter_mut() {
377            if monitored.is_waiting {
378                continue;
379            }
380            let last_annotation = monitored.last_annotation.unwrap();
381            if monitored.last_activity.elapsed() > monitored.permanent_hang_timeout {
382                if monitored.sent_permanent_alert {
383                    continue;
384                }
385                let profile = match monitored.sampler.suspend_and_sample_thread() {
386                    Ok(native_stack) => Some(native_stack.to_hangprofile()),
387                    Err(()) => None,
388                };
389                let _ = self.constellation_chan.send(HangAlert::Permanent(
390                    component_id.clone(),
391                    last_annotation,
392                    profile,
393                ));
394                monitored.sent_permanent_alert = true;
395                continue;
396            }
397            if monitored.last_activity.elapsed() > monitored.transient_hang_timeout {
398                if monitored.sent_transient_alert {
399                    continue;
400                }
401                let _ = self
402                    .constellation_chan
403                    .send(HangAlert::Transient(component_id.clone(), last_annotation));
404                monitored.sent_transient_alert = true;
405            }
406        }
407    }

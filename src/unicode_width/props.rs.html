19/// Whether this character is a zero-width character with
20/// `Joining_Type=Transparent`. Used by the Alef-Lamed ligatures.
21/// See also [`is_ligature_transparent`], a near-subset of this (only ZWJ is excepted)
22/// which is transparent for non-Arabic ligatures.
23pub(crate) fn is_transparent_zero_width(c: char) -> bool {
24    if crate::lookup::lookup_width(c).0 != 0 {
25        // Not zero-width
26        false
27    } else {
28        let cp: u32 = c.into();
29        NON_TRANSPARENT_ZERO_WIDTHS
30            .binary_search_by(|&(lo, hi)| {
31                let lo = u32::from_le_bytes([lo[0], lo[1], lo[2], 0]);
32                let hi = u32::from_le_bytes([hi[0], hi[1], hi[2], 0]);
33                if cp < lo {
34                    Ordering::Greater
35                } else if cp > hi {
36                    Ordering::Less
37                } else {
38                    Ordering::Equal
39                }
40            })
41            .is_err()
42    }
43}

45/// Whether this character is transparent wrt the effect of
46/// U+0338 COMBINING LONG SOLIDUS OVERLAY
47/// on its base character.
48#[cfg(feature = "cjk")]
49pub fn is_solidus_transparent(c: char) -> bool {
50    let cp: u32 = c.into();
51    is_ligature_transparent(c)
52        || SOLIDUS_TRANSPARENT
53            .binary_search_by(|&(lo, hi)| {
54                let lo = u32::from_le_bytes([lo[0], lo[1], lo[2], 0]);
55                let hi = u32::from_le_bytes([hi[0], hi[1], hi[2], 0]);
56                if cp < lo {
57                    Ordering::Greater
58                } else if cp > hi {
59                    Ordering::Less
60                } else {
61                    Ordering::Equal
62                }
63            })
64            .is_ok()
65}

72/// OCB3: generic over a block cipher implementation, nonce size, and tag size.
73///
74/// - `NonceSize`: max of 15-bytes, default and recommended size of 12-bytes (96-bits).
75///   We further restrict the minimum nonce size to 6-bytes to prevent an attack described in
76///   the following paper: <https://eprint.iacr.org/2023/326.pdf>.
77/// - `TagSize`: non-zero, max of 16-bytes, default and recommended size of 16-bytes.
78///
79/// Compilation will fail if the size conditions are not satisfied:
80///
81/// ```rust,compile_fail
82/// # use aes::Aes128;
83/// # use ocb3::{aead::{consts::U5, KeyInit}, Ocb3};
84/// # let key = [42; 16].into();
85/// // Invalid nonce size equal to 5 bytes
86/// let cipher = Ocb3::<Aes128, U5>::new(&key);
87/// ```
88///
89/// ```rust,compile_fail
90/// # use aes::Aes128;
91/// # use ocb3::{aead::{consts::U16, KeyInit}, Ocb3};
92/// # let key = [42; 16].into();
93/// // Invalid nonce size equal to 16 bytes
94/// let cipher = Ocb3::<Aes128, U16>::new(&key);
95/// ```
96///
97/// ```rust,compile_fail
98/// # use aes::Aes128;
99/// # use ocb3::{aead::{consts::{U12, U0}, KeyInit}, Ocb3};
100/// # let key = [42; 16].into();
101/// // Invalid tag size equal to 0 bytes
102/// let cipher = Ocb3::<Aes128, U12, U0>::new(&key);
103/// ```
104///
105/// ```rust,compile_fail
106/// # use aes::Aes128;
107/// # use ocb3::{aead::{consts::{U12, U20}, KeyInit}, Ocb3};
108/// # let key = [42; 16].into();
109/// // Invalid tag size equal to 20 bytes
110/// let cipher = Ocb3::<Aes128, U12, U20>::new(&key);
111/// ```
112///
113/// # Maximum plaintext and associated data lengths
114///
115/// The maximum lengths depend on the `L_TABLE_SIZE` constant which sets
116/// number of L values to be precomputed. Precomputing `m` values, allows
117/// processing inputs of length up to 2^m blocks (2^m * 16 bytes) without
118/// needing to calculate L values at runtime.
119///
120/// By default `L_TABLE_SIZE` is equal to 26, which allows to handle up to 1 GiB of data.
121#[derive(Clone)]
122pub struct Ocb3<Cipher, NonceSize = U12, TagSize = U16, const L_TABLE_SIZE: usize = 26>
123where
124    NonceSize: sealed::NonceSizes,
125    TagSize: sealed::TagSizes,
126{
127    cipher: Cipher,
128    nonce_size: PhantomData<NonceSize>,
129    tag_size: PhantomData<TagSize>,
130    // precomputed key-dependent variables
131    ll_star: Block,
132    ll_dollar: Block,
133    // list of pre-computed L values
134    ll: [Block; L_TABLE_SIZE],
135}

191impl<Cipher, NonceSize, TagSize, const L_TABLE_SIZE: usize> AeadInOut
192    for Ocb3<Cipher, NonceSize, TagSize, L_TABLE_SIZE>
193where
194    Cipher: BlockSizeUser<BlockSize = U16> + BlockCipherEncrypt + BlockCipherDecrypt,
195    NonceSize: sealed::NonceSizes,
196    TagSize: sealed::TagSizes,
197{
198    fn encrypt_inout_detached(
199        &self,
200        nonce: &Nonce<NonceSize>,
201        associated_data: &[u8],
202        buffer: InOutBuf<'_, '_, u8>,
203    ) -> aead::Result<aead::Tag<Self>> {
204        let max_len = 1 << (L_TABLE_SIZE + 4);
205        if (buffer.len() >= max_len) || (associated_data.len() >= max_len) {
206            return Err(Error);
207        }
208
209        // First, try to process many blocks at once.
210        let (tail, index, mut offset_i, mut checksum_i) = self.wide_encrypt(nonce, buffer);
211
212        let mut i = index;
213
214        // Then, process the remaining blocks.
215        let (blocks, mut tail): (InOutBuf<'_, '_, Block>, _) = tail.into_chunks();
216
217        for p_i in blocks {
218            // offset_i = offset_{i-1} xor L_{ntz(i)}
219            inplace_xor(&mut offset_i, &self.ll[ntz(i)]);
220            // checksum_i = checksum_{i-1} xor p_i
221            inplace_xor(&mut checksum_i, p_i.get_in());
222            // c_i = offset_i xor ENCIPHER(K, p_i xor offset_i)
223            let mut c_i = p_i;
224            c_i.xor_in2out(&offset_i);
225            self.cipher.encrypt_block(c_i.get_out());
226            inplace_xor(c_i.get_out(), &offset_i);
227
228            i += 1;
229        }
230
231        // Process any partial blocks.
232        if !tail.is_empty() {
233            let remaining_bytes = tail.len();
234
235            // offset_* = offset_m xor L_*
236            inplace_xor(&mut offset_i, &self.ll_star);
237            // Pad = ENCIPHER(K, offset_*)
238            let mut pad = Block::default();
239            inplace_xor(&mut pad, &offset_i);
240            self.cipher.encrypt_block(&mut pad);
241            // checksum_* = checksum_m xor (P_* || 1 || zeros(127-bitlen(P_*)))
242            let checksum_rhs = &mut [0u8; 16];
243            checksum_rhs[..remaining_bytes].copy_from_slice(tail.get_in());
244            checksum_rhs[remaining_bytes] = 0b1000_0000;
245            inplace_xor(&mut checksum_i, checksum_rhs.as_array_ref());
246            // C_* = P_* xor Pad[1..bitlen(P_*)]
247            let p_star = tail.get_out();
248            let pad = &mut pad[..p_star.len()];
249            tail.xor_in2out(pad);
250        }
251
252        let tag = self.compute_tag(associated_data, &mut checksum_i, &offset_i);
253
254        Ok(tag)
255    }
256
257    fn decrypt_inout_detached(
258        &self,
259        nonce: &Nonce<NonceSize>,
260        associated_data: &[u8],
261        buffer: InOutBuf<'_, '_, u8>,
262        tag: &aead::Tag<Self>,
263    ) -> aead::Result<()> {
264        let expected_tag = self.decrypt_inout_return_tag(nonce, associated_data, buffer)?;
265        if expected_tag.ct_eq(tag).into() {
266            Ok(())
267        } else {
268            Err(Error)
269        }
270    }
271}

280    /// Decrypts in place and returns expected tag.
281    pub(crate) fn decrypt_inout_return_tag(
282        &self,
283        nonce: &Nonce<NonceSize>,
284        associated_data: &[u8],
285        buffer: InOutBuf<'_, '_, u8>,
286    ) -> aead::Result<aead::Tag<Self>> {
287        let max_len = 1 << (L_TABLE_SIZE + 4);
288        if (buffer.len() > max_len) || (associated_data.len() > max_len) {
289            return Err(Error);
290        }
291
292        // First, try to process many blocks at once.
293        let (tail, index, mut offset_i, mut checksum_i) = self.wide_decrypt(nonce, buffer);
294
295        let mut i = index;
296
297        // Then, process the remaining blocks.
298        let (blocks, mut tail): (InOutBuf<'_, '_, Block>, _) = tail.into_chunks();
299        for c_i in blocks {
300            // offset_i = offset_{i-1} xor L_{ntz(i)}
301            inplace_xor(&mut offset_i, &self.ll[ntz(i)]);
302            // p_i = offset_i xor DECIPHER(K, c_i xor offset_i)
303            let mut p_i = c_i;
304            p_i.xor_in2out(&offset_i);
305            self.cipher.decrypt_block(p_i.get_out());
306            inplace_xor(p_i.get_out(), &offset_i);
307            // checksum_i = checksum_{i-1} xor p_i
308            inplace_xor(&mut checksum_i, p_i.get_out());
309
310            i += 1;
311        }
312
313        // Process any partial blocks.
314        if !tail.is_empty() {
315            let remaining_bytes = tail.len();
316
317            // offset_* = offset_m xor L_*
318            inplace_xor(&mut offset_i, &self.ll_star);
319            // Pad = ENCIPHER(K, offset_*)
320            let mut pad = Block::default();
321            inplace_xor(&mut pad, &offset_i);
322            self.cipher.encrypt_block(&mut pad);
323            // P_* = C_* xor Pad[1..bitlen(C_*)]
324            let c_star = tail.get_in();
325            let pad = &mut pad[..c_star.len()];
326            tail.xor_in2out(pad);
327            // checksum_* = checksum_m xor (P_* || 1 || zeros(127-bitlen(P_*)))
328            let checksum_rhs = &mut [0u8; 16];
329            checksum_rhs[..remaining_bytes].copy_from_slice(tail.get_out());
330            checksum_rhs[remaining_bytes] = 0b1000_0000;
331            inplace_xor(&mut checksum_i, checksum_rhs.as_array_ref());
332        }
333
334        Ok(self.compute_tag(associated_data, &mut checksum_i, &offset_i))
335    }

337    /// Encrypts plaintext in groups of two.
338    ///
339    /// Adapted from https://www.cs.ucdavis.edu/~rogaway/ocb/news/code/ocb.c
340    fn wide_encrypt<'i, 'o>(
341        &self,
342        nonce: &Nonce<NonceSize>,
343        buffer: InOutBuf<'i, 'o, u8>,
344    ) -> (InOutBuf<'i, 'o, u8>, usize, Block, Block) {
345        const WIDTH: usize = 2;
346
347        let mut i = 1;
348
349        let mut offset_i = [Block::default(); WIDTH];
350        offset_i[1] = initial_offset(&self.cipher, nonce, TagSize::to_u32());
351        let mut checksum_i = Block::default();
352
353        let (wide_blocks, tail): (InOutBuf<'_, '_, DoubleBlock>, _) = buffer.into_chunks();
354        for wide_block in wide_blocks.into_iter() {
355            let mut p_i = split_into_two_blocks(wide_block);
356            // checksum_i = checksum_{i-1} xor p_i
357            for p_ij in &p_i {
358                inplace_xor(&mut checksum_i, p_ij.get_in());
359            }
360
361            // offset_i = offset_{i-1} xor L_{ntz(i)}
362            offset_i[0] = offset_i[1];
363            inplace_xor(&mut offset_i[0], &self.ll[ntz(i)]);
364            offset_i[1] = offset_i[0];
365            inplace_xor(&mut offset_i[1], &self.ll[ntz(i + 1)]);
366
367            // c_i = offset_i xor ENCIPHER(K, p_i xor offset_i)
368            for j in 0..p_i.len() {
369                p_i[j].xor_in2out(&offset_i[j]);
370                self.cipher.encrypt_block(p_i[j].get_out());
371                inplace_xor(p_i[j].get_out(), &offset_i[j]);
372            }
373
374            i += WIDTH;
375        }
376
377        (tail, i, offset_i[offset_i.len() - 1], checksum_i)
378    }

380    /// Decrypts plaintext in groups of two.
381    ///
382    /// Adapted from https://www.cs.ucdavis.edu/~rogaway/ocb/news/code/ocb.c
383    fn wide_decrypt<'i, 'o>(
384        &self,
385        nonce: &Nonce<NonceSize>,
386        buffer: InOutBuf<'i, 'o, u8>,
387    ) -> (InOutBuf<'i, 'o, u8>, usize, Block, Block) {
388        const WIDTH: usize = 2;
389
390        let mut i = 1;
391
392        let mut offset_i = [Block::default(); WIDTH];
393        offset_i[1] = initial_offset(&self.cipher, nonce, TagSize::to_u32());
394        let mut checksum_i = Block::default();
395
396        let (wide_blocks, tail): (InOutBuf<'_, '_, DoubleBlock>, _) = buffer.into_chunks();
397        for wide_block in wide_blocks.into_iter() {
398            let mut c_i = split_into_two_blocks(wide_block);
399
400            // offset_i = offset_{i-1} xor L_{ntz(i)}
401            offset_i[0] = offset_i[1];
402            inplace_xor(&mut offset_i[0], &self.ll[ntz(i)]);
403            offset_i[1] = offset_i[0];
404            inplace_xor(&mut offset_i[1], &self.ll[ntz(i + 1)]);
405
406            // p_i = offset_i xor DECIPHER(K, c_i xor offset_i)
407            // checksum_i = checksum_{i-1} xor p_i
408            for j in 0..c_i.len() {
409                c_i[j].xor_in2out(&offset_i[j]);
410                self.cipher.decrypt_block(c_i[j].get_out());
411                inplace_xor(c_i[j].get_out(), &offset_i[j]);
412                inplace_xor(&mut checksum_i, c_i[j].get_out());
413            }
414
415            i += WIDTH;
416        }
417
418        (tail, i, offset_i[offset_i.len() - 1], checksum_i)
419    }

421    /// Computes HASH function defined in https://www.rfc-editor.org/rfc/rfc7253.html#section-4.1
422    fn hash(&self, associated_data: &[u8]) -> Block {
423        let mut offset_i = Block::default();
424        let mut sum_i = Block::default();
425
426        let mut i = 1;
427        let (blocks, remaining) = Block::slice_as_chunks(associated_data);
428        for a_i in blocks {
429            // offset_i = offset_{i-1} xor L_{ntz(i)}
430            inplace_xor(&mut offset_i, &self.ll[ntz(i)]);
431            // Sum_i = Sum_{i-1} xor ENCIPHER(K, A_i xor offset_i)
432            let mut a_i = *a_i;
433            inplace_xor(&mut a_i, &offset_i);
434            self.cipher.encrypt_block(&mut a_i);
435            inplace_xor(&mut sum_i, &a_i);
436
437            i += 1;
438        }
439
440        // Process any partial blocks.
441        if !remaining.is_empty() {
442            let processed_bytes = (i - 1) * 16;
443            let remaining_bytes = associated_data.len() - processed_bytes;
444
445            // offset_* = offset_m xor L_*
446            inplace_xor(&mut offset_i, &self.ll_star);
447            // CipherInput = (A_* || 1 || zeros(127-bitlen(A_*))) xor offset_*
448            let mut cipher_input = Block::default();
449            cipher_input[..remaining_bytes].copy_from_slice(&associated_data[processed_bytes..]);
450            cipher_input[remaining_bytes] = 0b1000_0000;
451            //let cipher_input = Block::from_mut_slice(cipher_input);
452            inplace_xor(&mut cipher_input, &offset_i);
453            // Sum = Sum_m xor ENCIPHER(K, CipherInput)
454            self.cipher.encrypt_block(&mut cipher_input);
455            inplace_xor(&mut sum_i, &cipher_input);
456        }
457
458        sum_i
459    }
460
461    fn compute_tag(
462        &self,
463        associated_data: &[u8],
464        checksum_m: &mut Block,
465        offset_m: &Block,
466    ) -> Tag<TagSize> {
467        // Tag = ENCIPHER(K, checksum_m xor offset_m xor L_$) xor HASH(K,A)
468        let full_tag = checksum_m;
469        inplace_xor(full_tag, offset_m);
470        inplace_xor(full_tag, &self.ll_dollar);
471        self.cipher.encrypt_block(full_tag);
472        inplace_xor(full_tag, &self.hash(associated_data));
473
474        // truncate the tag to the required length
475        Tag::try_from(&full_tag[..TagSize::to_usize()]).expect("tag size mismatch")
476    }
477}

491/// Computes key-dependent variables defined in
492/// https://www.rfc-editor.org/rfc/rfc7253.html#section-4.1
493fn key_dependent_variables<Cipher, const L_TABLE_SIZE: usize>(
494    cipher: &Cipher,
495) -> (Block, Block, [Block; L_TABLE_SIZE])
496where
497    Cipher: BlockSizeUser<BlockSize = U16> + BlockCipherEncrypt,
498{
499    let mut ll_star = Block::default();
500    cipher.encrypt_block(&mut ll_star);
501    let ll_dollar = ll_star.dbl();
502
503    let mut ll = [Block::default(); L_TABLE_SIZE];
504    let mut ll_i = ll_dollar;
505    #[allow(clippy::needless_range_loop)]
506    for i in 0..L_TABLE_SIZE {
507        ll_i = ll_i.dbl();
508        ll[i] = ll_i;
509    }
510    (ll_star, ll_dollar, ll)
511}

513/// Computes nonce-dependent variables as defined
514/// in https://www.rfc-editor.org/rfc/rfc7253.html#section-4.2
515#[allow(clippy::cast_possible_truncation, reason = "TODO")]
516fn nonce_dependent_variables<
517    Cipher: BlockSizeUser<BlockSize = U16> + BlockCipherEncrypt,
518    NonceSize: sealed::NonceSizes,
519>(
520    cipher: &Cipher,
521    nn: &Nonce<NonceSize>,
522    tag_len: u32,
523) -> (usize, [u8; 24]) {
524    // Nonce = num2str(TAGLEN mod 128,7) || zeros(120-bitlen(N)) || 1 || N
525    let mut nonce = [0u8; 16];
526    nonce[0] = (((tag_len * 8) % 128) << 1) as u8;
527
528    let start = 16 - NonceSize::to_usize();
529    nonce[start..16].copy_from_slice(nn.as_slice());
530    nonce[16 - NonceSize::to_usize() - 1] |= 1;
531
532    // Separate the last 6 bits into `bottom`, and the rest into `top`.
533    let bottom = nonce[15] & 0b111111;
534
535    let nonce = u128::from_be_bytes(nonce);
536    let top = nonce & !0b111111;
537
538    let mut ktop = Block::from(top.to_be_bytes());
539    cipher.encrypt_block(&mut ktop);
540    let ktop = ktop.as_mut_slice();
541
542    // stretch = Ktop || (Ktop[1..64] xor Ktop[9..72])
543    let mut stretch = [0u8; 24];
544    stretch[..16].copy_from_slice(ktop);
545    for i in 0..8 {
546        ktop[i] ^= ktop[i + 1];
547    }
548    stretch[16..].copy_from_slice(&ktop[..8]);
549
550    (bottom as usize, stretch)
551}

553/// Computes the initial offset as defined
554/// in https://www.rfc-editor.org/rfc/rfc7253.html#section-4.2
555#[allow(clippy::unwrap_used)]
556fn initial_offset<
557    Cipher: BlockSizeUser<BlockSize = U16> + BlockCipherEncrypt,
558    NonceSize: sealed::NonceSizes,
559>(
560    cipher: &Cipher,
561    nn: &Nonce<NonceSize>,
562    tag_size: u32,
563) -> Block {
564    let (bottom, stretch) = nonce_dependent_variables(cipher, nn, tag_size);
565    let stretch_low = u128::from_be_bytes((&stretch[..16]).try_into().unwrap());
566    let stretch_hi = u64::from_be_bytes((&stretch[16..24]).try_into().unwrap());
567    let stretch_hi = u128::from(stretch_hi);
568
569    // offset_0 = stretch[1+bottom..128+bottom]
570    let offset = (stretch_low << bottom) | (stretch_hi >> (64 - bottom));
571    offset.to_be_bytes().into()
572}

574#[inline]
575pub(crate) fn inplace_xor<T, U>(a: &mut Array<T, U>, b: &Array<T, U>)
576where
577    U: ArraySize,
578    T: core::ops::BitXor<Output = T> + Copy,
579{
580    for (aa, bb) in a.as_mut_slice().iter_mut().zip(b.as_slice()) {
581        *aa = *aa ^ *bb;
582    }
583}

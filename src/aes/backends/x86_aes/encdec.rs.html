13#[inline]
14#[target_feature(enable = "aes")]
15pub(crate) fn encrypt<const RK: usize>(keys: &RoundKeys<RK>, mut block: InOut<'_, '_, Block>) {
16    const { assert!(matches!(RK, 11 | 13 | 15)) }
17
18    let mut b = utils::load_block(block.get_in());
19    b = _mm_xor_si128(b, keys[0]);
20    for &key in &keys[1..RK - 1] {
21        b = _mm_aesenc_si128(b, key);
22    }
23    b = _mm_aesenclast_si128(b, keys[RK - 1]);
24    utils::store_block(block.get_out(), b);
25}

41#[inline]
42#[target_feature(enable = "aes")]
43pub(super) fn batch_encrypt<const RK: usize, ParBlocks: ArraySize>(
44    keys: &RoundKeys<RK>,
45    mut blocks: InOut<'_, '_, Array<Block, ParBlocks>>,
46) {
47    const { assert!(matches!(RK, 11 | 13 | 15)) }
48
49    let mut b = utils::load_batch_blocks(blocks.get_in());
50
51    // Loop over keys is intentionally not used here to force inlining
52    batch_xor(&mut b, keys[0]);
53    batch_aesenc(&mut b, keys[1]);
54    batch_aesenc(&mut b, keys[2]);
55    batch_aesenc(&mut b, keys[3]);
56    batch_aesenc(&mut b, keys[4]);
57    batch_aesenc(&mut b, keys[5]);
58    batch_aesenc(&mut b, keys[6]);
59    batch_aesenc(&mut b, keys[7]);
60    batch_aesenc(&mut b, keys[8]);
61    batch_aesenc(&mut b, keys[9]);
62    if RK >= 13 {
63        batch_aesenc(&mut b, keys[10]);
64        batch_aesenc(&mut b, keys[11]);
65    }
66    if RK == 15 {
67        batch_aesenc(&mut b, keys[12]);
68        batch_aesenc(&mut b, keys[13]);
69    }
70    batch_aesenclast(&mut b, keys[RK - 1]);
71    utils::store_batch_blocks(blocks.get_out(), b);
72}

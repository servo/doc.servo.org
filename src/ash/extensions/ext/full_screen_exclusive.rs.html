29    pub unsafe fn get_device_group_surface_present_modes2(
30        &self,
31        surface_info: &vk::PhysicalDeviceSurfaceInfo2KHR<'_>,
32    ) -> VkResult<vk::DeviceGroupPresentModeFlagsKHR> {
33        let mut present_modes = mem::MaybeUninit::uninit();
34        (self.fp.get_device_group_surface_present_modes2_ext)(

39        .assume_init_on_success(present_modes)
40    }

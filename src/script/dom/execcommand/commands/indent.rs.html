24/// <https://w3c.github.io/editing/docs/execCommand/#indent>
25pub(crate) fn indent(cx: &mut JSContext, context_object: &Document, node_list: Vec<DomRoot<Node>>) {
26    // Step 1. If node list is empty, do nothing and abort these steps.
27    if node_list.is_empty() {
28        return;
29    }
30
31    // Step 2. Let first node be the first member of node list.
32    let first_node = node_list
33        .first()
34        .expect("Must have a first node by now.")
35        .clone();
36
37    // Step 3. If first node's parent is an ol or ul:
38    if let Some(parent) = first_node.GetParentElement() &&
39        (parent.is::<HTMLOListElement>() || parent.is::<HTMLUListElement>())
40    {
41        // Step 3.1. Let tag be the local name of the parent of first node.
42        let tag = parent.local_name();
43
44        // Step 3.2. Wrap node list,
45        //           with sibling criteria returning true for an HTML element with local name tag and false otherwise,
46        //           and new parent instructions returning the result of calling createElement(tag) on the ownerDocument of first node.
47        wrap_node_list(
48            cx,
49            node_list,
50            |_cx, sibling| {
51                sibling
52                    .downcast::<Element>()
53                    .is_some_and(|sibling| sibling.local_name() == tag)
54            },
55            |cx| {
56                Some(DomRoot::upcast(
57                    first_node.owner_doc().create_element(cx, tag),
58                ))
59            },
60        );
61
62        // Step 3.3. Abort these steps.
63        return;
64    }
65
66    // Step 4. Wrap node list,
67    //         with sibling criteria returning true for a simple indentation element and false otherwise,
68    //         and new parent instructions returning the result of calling createElement("blockquote") on the ownerDocument of first node.
69    //         Let new parent be the result.
70    let new_parent = wrap_node_list(
71        cx,
72        node_list,
73        |_cx, sibling| {
74            sibling
75                .downcast::<Element>()
76                .is_some_and(|sibling| sibling.is_simple_indentation_element())
77        },
78        |cx| {
79            Some(DomRoot::upcast(
80                first_node.owner_doc().create_element(cx, "blockquote"),
81            ))
82        },
83    );
84
85    // Step 5. Fix disallowed ancestors of new parent.
86    if let Some(new_parent) = new_parent {
87        new_parent.fix_disallowed_ancestors(cx, context_object);
88    }
89}

91/// <https://w3c.github.io/editing/docs/execCommand/#the-indent-command>
92pub(crate) fn execute_indent_command(
93    cx: &mut JSContext,
94    document: &Document,
95    selection: &Selection,
96) -> bool {
97    let mut active_range = selection
98        .active_range(cx)
99        .expect("Must always have an active range.");
100    // Step 1. Let items be a list of all lis that are inclusive ancestors of the active range's start and/or end node.
101    let items: HashSet<DomRoot<Node>> = active_range
102        .start_container()
103        .ancestors()
104        .chain(active_range.end_container().ancestors())
105        .filter(|ancestor| ancestor.is::<HTMLLIElement>())
106        .collect();
107
108    // Step 2. For each item in items, normalize sublists of item.
109    for item in items {
110        item.normalize_sublists(cx);
111    }
112
113    // Normalizing sublists probably messes up the range
114    active_range = selection
115        .active_range(cx)
116        .expect("Must always have an active range.");
117
118    // Step 3. Block-extend the active range, and let new range be the result.
119    let new_range = active_range.block_extend(cx, document);
120
121    // Step 4. Let node list be a list of nodes, initially empty.
122    let mut node_list: Vec<DomRoot<Node>> = vec![];
123
124    // Step 5. For each node node contained in new range,
125    //         if node is editable and is an allowed child of "div" or "ol"
126    //         and if the last member of node list (if any) is not an ancestor of node,
127    //         append node to node list.
128    let mut contained_nodes = new_range.contained_nodes_rooted();
129    while let Some(node) = contained_nodes.next(cx.no_gc()) {
130        if node.is_editable(cx) &&
131            (is_allowed_child(
132                NodeOrString::Node(UnrootedDom::from_ref(&*node, cx.no_gc())),
133                NodeOrString::String("div".to_owned()),
134            ) || is_allowed_child(
135                NodeOrString::Node(UnrootedDom::from_ref(&*node, cx.no_gc())),
136                NodeOrString::String("ol".to_owned()),
137            )) &&
138            node_list
139                .last()
140                .is_none_or(|last| !last.is_ancestor_of(&node))
141        {
142            node_list.push(node);
143        }
144    }
145
146    // Step 6. If the first visible member of node list is an li whose parent is an ol or ul:
147    if let Some(first_visible_member) = node_list.iter().find(|node| node.is_visible(cx)) &&
148        first_visible_member.is::<HTMLLIElement>() &&
149        let Some(parent) = first_visible_member.GetParentNode() &&
150        (parent.is::<HTMLOListElement>() || parent.is::<HTMLUListElement>())
151    {
152        // Step 6.1. Let sibling be node list's first visible member's previousSibling.
153        let mut sibling = first_visible_member.GetPreviousSibling();
154
155        // Step 6.2. While sibling is invisible, set sibling to its previousSibling.
156        while let Some(ref some_sibling) = sibling &&
157            some_sibling.is_invisible(cx)
158        {
159            sibling = some_sibling.GetPreviousSibling();
160        }
161
162        // Step 6.3. If sibling is an li, normalize sublists of sibling.
163        if let Some(sibling) = sibling &&
164            sibling.is::<HTMLLIElement>()
165        {
166            sibling.normalize_sublists(cx);
167        }
168    }
169
170    // Step 7. While node list is not empty:
171    let mut node_list_iter = node_list.iter().peekable();
172    while node_list_iter.peek().is_some() {
173        // Step 7.1. Let sublist be a list of nodes, initially empty.
174        let mut sublist: Vec<DomRoot<Node>> = vec![];
175
176        // Step 7.2. Remove the first member of node list and append it to sublist.
177        sublist.push(
178            node_list_iter
179                .next()
180                .expect("Must always have a next item")
181                .clone(),
182        );
183
184        // Step 7.3. While the first member of node list is the nextSibling of the last member of
185        //           sublist, remove the first member of node list and append it to sublist.
186        while node_list_iter.peek().is_some_and(|node| {
187            sublist
188                .last()
189                .expect("Must always have last element here.")
190                .GetNextSibling()
191                .is_some_and(|next_sibling| &&next_sibling == node)
192        }) {
193            sublist.push(
194                node_list_iter
195                    .next()
196                    .expect("Must always have a next item")
197                    .clone(),
198            );
199        }
200
201        // Step 7.4. Indent sublist.
202        indent(cx, document, sublist);
203    }
204
205    // Step 8. Return true.
206    // Note: This isn't in the spec (yet), see https://github.com/w3c/editing/pull/547
207    true
208}

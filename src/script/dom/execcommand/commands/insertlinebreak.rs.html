23/// <https://w3c.github.io/editing/docs/execCommand/#the-insertlinebreak-command>
24pub(crate) fn execute_insert_line_break_command(
25    cx: &mut JSContext,
26    document: &Document,
27    selection: &Selection,
28) -> bool {
29    // Step 1. Delete the selection, with strip wrappers false.
30    selection.delete_the_selection(
31        cx,
32        document,
33        Default::default(),
34        SelectionDeletionStripWrappers::NoStrip,
35        Default::default(),
36    );
37
38    // Step 2. If the active range's start node is neither editable nor an editing host, return
39    //         true.
40    let mut active_range = selection
41        .active_range(cx)
42        .expect("Must always have an active range.");
43    if !active_range.start_container().is_editable_or_editing_host() {
44        return true;
45    }
46
47    // Step 3. If the active range's start node is an Element, and "br" is not an allowed child of
48    //         it, return true.
49    if active_range.start_container().is::<Element>() &&
50        !is_allowed_child(
51            NodeOrString::String("br".to_owned()),
52            NodeOrString::from_node(&active_range.start_container(), cx.no_gc()),
53        )
54    {
55        return false;
56    }
57
58    // Step 4. If the active range's start node is not an Element, and "br" is not an allowed child
59    //         of the active range's start node's parent, return true.
60    if !active_range.start_container().is::<Element>() &&
61        !is_allowed_child(
62            NodeOrString::String("br".to_owned()),
63            NodeOrString::from_node(
64                &active_range
65                    .start_container()
66                    .GetParentNode()
67                    .expect("Must always have a parent."),
68                cx.no_gc(),
69            ),
70        )
71    {
72        return false;
73    }
74
75    // Step 5. If the active range's start node is a Text node and its start offset is zero, call
76    //         collapse() on the context object's selection, with first argument equal to the
77    //         active range's start node's parent and second argument equal to the active range's
78    //         start node's index.
79    if active_range.start_container().is::<Text>() && active_range.start_offset() == 0 {
80        if selection
81            .Collapse(
82                cx,
83                active_range.start_container().GetParentNode().as_deref(),
84                active_range.start_container().index(),
85            )
86            .is_err()
87        {
88            unreachable!("Should always be able to collapse the selection.");
89        }
90        active_range = selection
91            .active_range(cx)
92            .expect("Must always have an active range");
93    }
94
95    // Step 6. If the active range's start node is a Text node and its start offset is the length
96    //         of its start node, call collapse() on the context object's selection, with first
97    //         argument equal to the active range's start node's parent and second argument equal
98    //         to one plus the active range's start node's index.
99    if active_range.start_container().is::<Text>() &&
100        active_range.start_offset() == active_range.start_container().len()
101    {
102        if selection
103            .Collapse(
104                cx,
105                active_range.start_container().GetParentNode().as_deref(),
106                1 + active_range.start_container().index(),
107            )
108            .is_err()
109        {
110            unreachable!("Should always be able to collapse the selection.");
111        }
112        active_range = selection
113            .active_range(cx)
114            .expect("Must always have an active range");
115    }
116
117    // Step 7. Let br be the result of calling createElement("br") on the context object.
118    //
119    // Browsers use a newline character for insertion when the CSS style for "white-space"
120    // preserves whitespace (that's the case for "pre", "pre-line" and "pre-wrap"). However,
121    // confusingly they don't do that when the node is exactly a `<pre>`.
122    let has_preformatted_styling = active_range
123        .start_container()
124        .downcast::<Element>()
125        .filter(|element| !element.is::<HTMLPreElement>())
126        .and_then(|element| element.style(cx))
127        .is_some_and(|style| {
128            let white_space_collapse = style.get_inherited_text().white_space_collapse;
129            white_space_collapse == WhiteSpaceCollapse::Preserve ||
130                white_space_collapse == WhiteSpaceCollapse::PreserveBreaks
131        });
132
133    let newline_node = if has_preformatted_styling {
134        DomRoot::upcast(document.CreateTextNode(cx, DOMString::from_static("\n")))
135    } else {
136        DomRoot::upcast(document.create_element(cx, "br"))
137    };
138
139    // Step 8. Call insertNode(br) on the active range.
140    if active_range.InsertNode(cx, &newline_node).is_err() {
141        unreachable!("The node should always be insertable.");
142    }
143
144    // Step 9. Call collapse() on the context object's selection, with br's parent as the first
145    //         argument and one plus br's index as the second argument.
146    if selection
147        .Collapse(
148            cx,
149            newline_node.GetParentNode().as_deref(),
150            1 + newline_node.index(),
151        )
152        .is_err()
153    {
154        unreachable!("Should always be able to collapse the selection.");
155    }
156
157    // Step 10. If br is a collapsed line break, call createElement("br") on the context object and
158    //          let extra br be the result, then call insertNode(extra br) on the active range.
159    if newline_node.is_collapsed_line_break(cx) {
160        // Same as above at step 7, we insert a newline character when appropriate
161        let extra_newline_node = if has_preformatted_styling {
162            DomRoot::upcast(document.CreateTextNode(cx, DOMString::from_static("\n")))
163        } else {
164            DomRoot::upcast(document.create_element(cx, "br"))
165        };
166
167        if active_range.InsertNode(cx, &extra_newline_node).is_err() {
168            unreachable!("The node should always be insertable.");
169        }
170    }
171
172    // Step 11. Return true.
173    true
174}

149impl VTTCueMethods<crate::DomTypeHolder> for VTTCue {
150    /// <https://w3c.github.io/webvtt/#dom-vttcue-vttcue>
151    fn Constructor(
152        cx: &mut JSContext,
153        window: &Window,
154        proto: Option<HandleObject>,
155        start_time: Finite<f64>,
156        end_time: Finite<f64>,
157        text: DOMString,
158    ) -> Fallible<DomRoot<Self>> {
159        // Step 3. If the value of the endTime argument is negative Infinity or a Not-a-Number (NaN) value,
160        // then throw a TypeError exception.
161        // Otherwise, let cue’s text track cue end time be the value of the endTime argument.
162        if (end_time.is_infinite() && end_time.is_sign_negative()) || end_time.is_nan() {
163            return Err(Error::Type(
164                c"End time is negative Infinity or Not-a-Number".to_owned(),
165            ));
166        }
167        // Step 1. Create a new WebVTT cue. Let cue be that WebVTT cue.
168        // Step 16. Return the VTTCue object representing cue.
169        Ok(VTTCue::new(
170            cx,
171            window,
172            proto,
173            // Step 2. Let cue’s text track cue start time be the value of the startTime argument.
174            *start_time,
175            *end_time,
176            // Step 4. Let cue’s cue text be the value of the text argument,
177            // and let the rules for extracting the chapter title be the
178            // WebVTT rules for extracting the chapter title.
179            // TODO: Extract title
180            text,
181            // Step 5. Let cue’s text track cue identifier be the empty string.
182            Default::default(),
183            // Step 8. Let cue’s WebVTT cue writing direction be horizontal.
184            DirectionSetting::_empty,
185            // Step 9. Let cue’s WebVTT cue snap-to-lines flag be true.
186            true,
187            // Step 10. Let cue’s WebVTT cue line be auto.
188            LineAndPositionSetting::Auto,
189            // Step 11. Let cue’s WebVTT cue line alignment be start alignment.
190            LineAlignSetting::Start,
191            // Step 12. Let cue’s WebVTT cue position be auto.
192            LineAndPositionSetting::Auto,
193            // Step 13. Let cue’s WebVTT cue position alignment be auto.
194            PositionAlignSetting::Auto,
195            // Step 14. Let cue’s WebVTT cue size be 100.
196            100.,
197            // Step 15. Let cue’s WebVTT cue text alignment be center alignment.
198            AlignSetting::Center,
199            None,
200        ))
201    }

328#[derive(Clone, JSTraceable, MallocSizeOf)]
329enum LineAndPositionSetting {
330    Double(f64),
331    Auto,
332}
333
334impl From<VTTCueBinding::LineAndPositionSetting> for LineAndPositionSetting {
335    fn from(value: VTTCueBinding::LineAndPositionSetting) -> Self {
336        match value {
337            VTTCueBinding::LineAndPositionSetting::Double(x) => LineAndPositionSetting::Double(*x),
338            VTTCueBinding::LineAndPositionSetting::AutoKeyword(_) => LineAndPositionSetting::Auto,
339        }
340    }
341}
342
343impl From<LineAndPositionSetting> for VTTCueBinding::LineAndPositionSetting {
344    fn from(value: LineAndPositionSetting) -> Self {
345        match value {
346            LineAndPositionSetting::Double(x) => {
347                VTTCueBinding::LineAndPositionSetting::Double(Finite::wrap(x))
348            },
349            LineAndPositionSetting::Auto => {
350                VTTCueBinding::LineAndPositionSetting::AutoKeyword(AutoKeyword::Auto)
351            },
352        }
353    }
354}

367        match self {

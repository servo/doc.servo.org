73    // https://dom.spec.whatwg.org/#dom-text-splittext
74    /// <https://dom.spec.whatwg.org/#concept-text-split>
75    fn SplitText(&self, cx: &mut JSContext, offset: u32) -> Fallible<DomRoot<Text>> {
76        let cdata = self.upcast::<CharacterData>();
77        // Step 1: Let length be node’s length.
78        let length = cdata.Length();
79        // Step 2: If offset is greater than length, then throw an "IndexSizeError" DOMException.
80        if offset > length {
81            return Err(Error::IndexSize(None));
82        }
83        // Step 3: Let count be length − offset.
84        let count = length - offset;
85        // Step 4: Let newData be the result of substringing data of node with offset and count.
86        let new_data = cdata.SubstringData(offset, count).unwrap();
87        // Step 5: Let newNode be the result of creating a text node given node’s node document and newData.
88        let node = self.upcast::<Node>();
89        let owner_doc = node.owner_doc();
90        let new_text_node = owner_doc.CreateTextNode(cx, new_data);
91        // Step 6: Let parent be node’s parent.
92        let parent = node.GetParentNode();
93        // Step 7: If parent is non-null:
94        if let Some(ref parent) = parent {
95            // Step 7.1: Insert newNode into parent before node’s next sibling.
96            let new_node = new_text_node.upcast();
97            parent
98                .InsertBefore(cx, new_node, node.GetNextSibling().as_deref())
99                .unwrap();
100
101            // Steps 7.2-7.5: The live range update steps.
102            if let Some(selection) = owner_doc.selection() {
103                selection.text_split_steps(node, offset, parent, new_node);
104            }
105            live_range_text_split_steps(parent, node, offset, new_node);
106        }
107        // Step 8.
108        cdata.DeleteData(cx, offset, count).unwrap();
109        // Step 9.
110        Ok(new_text_node)
111    }

113    /// <https://dom.spec.whatwg.org/#dom-text-wholetext>
114    fn WholeText(&self, cx: &JSContext) -> DOMString {
115        let first = self
116            .upcast::<Node>()
117            .inclusively_preceding_siblings_unrooted(cx.no_gc())
118            .take_while(|node| node.is::<Text>())
119            .last()
120            .unwrap();
121        let nodes = first
122            .inclusively_following_siblings_unrooted(cx.no_gc())
123            .take_while(|node| node.is::<Text>());
124        let mut text = String::new();
125        for ref node in nodes {
126            let cdata = node.downcast::<CharacterData>().unwrap();
127            text.push_str(&cdata.data());
128        }
129        DOMString::from(text)
130    }

132    /// <https://dom.spec.whatwg.org/#dom-slotable-assignedslot>
133    fn GetAssignedSlot(&self, cx: &JSContext) -> Option<DomRoot<HTMLSlotElement>> {
134        // > The assignedSlot getter steps are to return the result of
135        // > find a slot given this and with the open flag set.
136        rooted!(&in(cx) let slottable = Slottable(Dom::from_ref(self.upcast::<Node>())));
137        slottable.find_a_slot(cx.no_gc(), true)
138    }

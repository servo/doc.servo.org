58impl AudioContext {
59    #[cfg_attr(crown, expect(crown::unrooted_must_root))]
60    /// <https://webaudio.github.io/web-audio-api/#AudioContext-constructors>
61    fn new_inherited(
62        options: &AudioContextOptions,
63        pipeline_id: PipelineId,
64    ) -> Fallible<AudioContext> {
65        // Steps 1-3.
66        let context = BaseAudioContext::new_inherited(
67            BaseAudioContextOptions::AudioContext(options.convert()),
68            pipeline_id,
69        )?;
70
71        // Step 4.1.
72        let latency_hint = match options.latencyHint {
73            AudioContextLatencyCategoryOrDouble::AudioContextLatencyCategory(category) => category,
74            AudioContextLatencyCategoryOrDouble::Double(_) => {
75                AudioContextLatencyCategory::Interactive
76            }, // TODO
77        };
78
79        // Step 4.2. The sample rate is set during the creation of the BaseAudioContext.
80        // servo-media takes care of setting the default sample rate of the output device
81        // and of resampling the audio output if needed.
82
83        // Steps 5 and 6 of the construction algorithm will happen in `resume`,
84        // after reflecting dom object.
85
86        Ok(AudioContext {
87            context,
88            latency_hint,
89            base_latency: 0.,   // TODO
90            output_latency: 0., // TODO
91        })
92    }
93
94    #[cfg_attr(crown, expect(crown::unrooted_must_root))]
95    fn new(
96        cx: &mut js::context::JSContext,
97        window: &Window,
98        proto: Option<HandleObject>,
99        options: &AudioContextOptions,
100    ) -> Fallible<DomRoot<AudioContext>> {
101        let pipeline_id = window.pipeline_id();
102        let context = AudioContext::new_inherited(options, pipeline_id)?;
103        let context = reflect_dom_object_with_proto(cx, Box::new(context), window, proto);
104        context.resume();
105        Ok(context)
106    }
107
108    fn resume(&self) {
109        // Step 5.
110        if self.context.is_allowed_to_start() {
111            // Step 6.
112            self.context.resume();
113        }
114    }
115
116    pub(crate) fn base(&self) -> DomRoot<BaseAudioContext> {
117        DomRoot::from_ref(&self.context)
118    }
119}

151    /// <https://webaudio.github.io/web-audio-api/#dom-audiocontext-suspend>
152    fn Suspend(&self, cx: &mut CurrentRealm) -> RootedPromise {
153        // Step 1.
154        let promise = Promise::new_in_realm(cx);
155
156        // Step 2.
157        if self.context.control_thread_state() == ProcessingState::Closed {
158            promise.reject_error(cx, Error::InvalidState(None));
159            return promise;
160        }
161
162        // Step 3.
163        if self.context.State() == AudioContextState::Suspended {
164            promise.resolve_native(cx, &());
165            return promise;
166        }
167
168        // Steps 4 and 5.
169        let trusted_promise = TrustedPromise::from(&promise);
170        match self.context.audio_context_impl().lock().unwrap().suspend() {
171            Some(_) => {
172                let base_context = Trusted::new(&self.context);
173                let context = Trusted::new(self);
174                self.global().task_manager().dom_manipulation_task_source().queue(
175                    task!(suspend_ok: move |cx| {
176                        let base_context = base_context.root();
177                        let context = context.root();
178                        let promise = trusted_promise.root(cx);
179                        promise.resolve_native(cx, &());
180                        if base_context.State() != AudioContextState::Suspended {
181                            base_context.set_state_attribute(AudioContextState::Suspended);
182                            context.global().task_manager().dom_manipulation_task_source().queue_simple_event(
183                                context.upcast(),
184                                atom!("statechange"),
185                            );
186                        }
187                    })
188                );
189            },
190            None => {
191                // The spec does not define the error case and `suspend` should
192                // never fail, but we handle the case here for completion.
193                self.global()
194                    .task_manager()
195                    .dom_manipulation_task_source()
196                    .queue(task!(suspend_error: move |cx| {
197                        let promise = trusted_promise.root(cx);
198                        promise.reject_error(cx, Error::Type(c"Something went wrong".to_owned()));
199                    }));
200            },
201        };
202
203        // Step 6.
204        promise
205    }

322impl Convert<LatencyCategory> for AudioContextLatencyCategory {
323    fn convert(self) -> LatencyCategory {
324        match self {
325            AudioContextLatencyCategory::Balanced => LatencyCategory::Balanced,
326            AudioContextLatencyCategory::Interactive => LatencyCategory::Interactive,
327            AudioContextLatencyCategory::Playback => LatencyCategory::Playback,
328        }
329    }
330}

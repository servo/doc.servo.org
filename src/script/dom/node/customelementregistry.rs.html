155    /// <https://html.spec.whatwg.org/multipage/#look-up-a-custom-element-registry>
156    pub(crate) fn lookup_a_custom_element_registry(
157        node: &Node,
158    ) -> Option<DomRoot<CustomElementRegistry>> {
159        match node.type_id() {
160            // Step 1. If node is an Element object, then return node's custom element registry.
161            NodeTypeId::Element(_) => node
162                .downcast::<Element>()
163                .expect("Nodes with element type must be an element")
164                .custom_element_registry(),
165            // Step 2. If node is a ShadowRoot object, then return node's custom element registry.
166            NodeTypeId::DocumentFragment(DocumentFragmentTypeId::ShadowRoot) => node
167                .downcast::<ShadowRoot>()
168                .expect("Nodes with ShadowRoot type must be a ShadowRoot")
169                .custom_element_registry(),
170            // Step 3. If node is a Document object, then return node's custom element registry.
171            NodeTypeId::Document(_) => node
172                .downcast::<Document>()
173                .expect("Nodes with document type must be a document")
174                .custom_element_registry(),
175            // Step 4. Return null.
176            _ => None,
177        }
178    }

180    /// <https://html.spec.whatwg.org/multipage/#look-up-a-custom-element-definition>
181    pub(crate) fn lookup_custom_element_definition(
182        registry: Option<&CustomElementRegistry>,
183        namespace: &Namespace,
184        local_name: &LocalName,
185        is: Option<&LocalName>,
186    ) -> Option<Rc<CustomElementDefinition>> {
187        // Step 1. If registry is null, then return null.
188        let registry = registry?;
189
190        // Step 2. If namespace is not the HTML namespace, then return null.
191        if *namespace != ns!(html) {
192            return None;
193        }
194
195        // Step 3. If registry's custom element definition set contains an item
196        // with name and local name both equal to localName, then return that
197        // item.
198        // Step 4. If registry's custom element definition set contains an item
199        // with name equal to is and local name equal to localName, then return
200        // that item.
201        // Step 5. Return null.
202        registry
203            .definitions
204            .borrow()
205            .0
206            .values()
207            .find(|definition| {
208                definition.local_name == *local_name &&
209                    (definition.name == *local_name || Some(&definition.name) == is)
210            })
211            .cloned()
212    }

288    /// <https://html.spec.whatwg.org/multipage/#upgrade-particular-elements-within-a-document>
289    fn upgrade_particular_elements_within_a_document(
290        &self,
291        cx: &JSContext,
292        document: &Document,
293        definition: &Rc<CustomElementDefinition>,
294        local_name: &LocalName,
295        name: &LocalName,
296    ) {
297        // Step 1. Let upgradeCandidates be all elements that are shadow-including
298        // descendants of document, whose custom element registry is registry, whose
299        // namespace is the HTML namespace, and whose local name is localName,
300        // in shadow-including tree order. Additionally, if name is not localName,
301        // only include elements whose is value is equal to name.
302        for candidate in document
303            .upcast::<Node>()
304            .traverse_preorder_non_rooting(cx, ShadowIncluding::Yes)
305            .filter_map(UnrootedDom::downcast::<Element>)
306        {
307            // Note: If the registry is scoped, only include elements whose custom
308            // element registry is explicitly set to this registry. Otherwise,
309            // include elements with no explicit registry (they inherit the
310            // document's global registry) as well.
311            let registry_matches = if self.is_scoped.get() {
312                candidate
313                    .custom_element_registry()
314                    .is_some_and(|registry| *registry == *self)
315            } else {
316                candidate
317                    .custom_element_registry()
318                    .is_none_or(|registry| *registry == *self)
319            };
320            if *candidate.local_name() == *local_name &&
321                *candidate.namespace() == ns!(html) &&
322                registry_matches &&
323                (*name == *local_name || candidate.get_is().as_ref() == Some(name))
324            {
325                // Step 2. For each element element of upgradeCandidates: enqueue a
326                // custom element upgrade reaction given element and definition.
327                ScriptThread::enqueue_upgrade_reaction(cx, &candidate, definition.clone());
328            }
329        }
330    }

339/// <https://html.spec.whatwg.org/multipage/#dom-customelementregistry-define>
340/// Step 14.4: Get `callbackValue` for all `callbackName` in `lifecycleCallbacks`.
341#[expect(unsafe_code)]
342fn get_callback(
343    cx: &mut JSContext,
344    prototype: HandleObject,
345    name: &CStr,
346) -> Fallible<Option<Rc<Function>>> {
347    rooted!(&in(cx) let mut callback = UndefinedValue());
348    unsafe {
349        // Step 10.4.1
350        if !JS_GetProperty(cx, prototype, name.as_ptr(), callback.handle_mut()) {
351            return Err(Error::JSFailed);
352        }
353
354        // Step 10.4.2
355        if !callback.is_undefined() {
356            if !callback.is_object() || !IsCallable(callback.to_object()) {
357                return Err(Error::Type(
358                    c"Lifecycle callback is not callable".to_owned(),
359                ));
360            }
361            Ok(Some(Function::new(cx, callback.to_object())))
362        } else {
363            Ok(None)
364        }
365    }

382    #[expect(unsafe_code)]
383    /// <https://html.spec.whatwg.org/multipage/#dom-customelementregistry-define>
384    fn Define(
385        &self,
386        cx: &mut JSContext,
387        name: DOMString,
388        constructor_: Rc<CustomElementConstructor>,
389        options: &ElementDefinitionOptions,
390    ) -> ErrorResult {
391        rooted!(&in(cx) let constructor = constructor_.callback());
392        let name = LocalName::from(name);
393
394        // Step 1. If IsConstructor(constructor) is false, then throw a TypeError.
395        // We must unwrap the constructor as all wrappers are constructable if they are callable.
396        rooted!(&in(cx) let unwrapped_constructor = unsafe { UnwrapObjectStatic(constructor.get()) });
397
398        if unwrapped_constructor.is_null() {
399            // We do not have permission to access the unwrapped constructor.
400            return Err(Error::Security(None));
401        }
402
403        if unsafe { !IsConstructor(unwrapped_constructor.get()) } {
404            return Err(Error::Type(
405                c"Second argument of CustomElementRegistry.define is not a constructor".to_owned(),
406            ));
407        }
408
409        // Step 2. If name is not a valid custom element name, then throw a "SyntaxError" DOMException.
410        if !is_valid_custom_element_name(&name) {
411            return Err(Error::Syntax(Some(format!(
412                "{} name is not a valid custom element name",
413                name
414            ))));
415        }
416
417        // Step 3. If this's custom element definition set contains an item with name name,
418        // then throw a "NotSupportedError" DOMException.
419        if self.definitions.borrow().contains_key(&name) {
420            return Err(Error::NotSupported(Some(format!(
421                "{} has already been defined as a custom element",
422                name
423            ))));
424        }
425
426        // Step 4. If this's custom element definition set contains an
427        // item with constructor constructor, then throw a "NotSupportedError" DOMException.
428        if self
429            .definitions
430            .borrow()
431            .iter()
432            .any(|(_, def)| def.constructor == constructor_)
433        {
434            return Err(Error::NotSupported(None));
435        }
436
437        // Step 6. Let extends be options["extends"] if it exists; otherwise null.
438        let extends = &options.extends;
439
440        // Steps 5, 7
441        let local_name = if let Some(ref extended_name) = *extends {
442            // Step 7.1 If this's is scoped is true, then throw a "NotSupportedError" DOMException.
443            if self.is_scoped.get() {
444                return Err(Error::NotSupported(Some(
445                    "Scoped custom element registries cannot define customized built-in elements"
446                        .to_owned(),
447                )));
448            }
449
450            // Step 7.2 If extends is a valid custom element name, then throw a "NotSupportedError" DOMException.
451            if is_valid_custom_element_name(&extended_name.str()) {
452                return Err(Error::NotSupported(None));
453            }
454
455            // Step 7.3 If the element interface for extends and the HTML namespace is HTMLUnknownElement
456            // (e.g., if extends does not indicate an element definition in this specification)
457            // then throw a "NotSupportedError" DOMException.
458            if !is_extendable_element_interface(&extended_name.str()) {
459                return Err(Error::NotSupported(None));
460            }
461
462            // Step 7.4 Set localName to extends.
463            LocalName::from(extended_name)
464        } else {
465            // Step 5. Let localName be name.
466            name.clone()
467        };
468
469        // Step 8
470        if self.element_definition_is_running.get() {
471            return Err(Error::NotSupported(None));
472        }
473
474        // Step 9
475        self.element_definition_is_running.set(true);
476
477        // Steps 10-13: Initialize `formAssociated`, `disableInternals`, `disableShadow`, and
478        // `observedAttributes` with default values, but this is done later.
479
480        // Steps 14.1 - 14.2: Get the value of the prototype.
481        rooted!(&in(cx) let mut prototype = UndefinedValue());
482        {
483            let mut realm = AutoRealm::new_from_handle(cx, constructor.handle());
484            if let Err(error) =
485                self.check_prototype(&mut realm, constructor.handle(), prototype.handle_mut())
486            {
487                self.element_definition_is_running.set(false);
488                return Err(error);
489            }
490        };
491
492        // Steps 10.3 - 10.4
493        // It would be easier to get all the callbacks in one pass after
494        // we know whether this definition is going to be form-associated,
495        // but the order of operations is specified and it's observable
496        // if one of the callback getters throws an exception.
497        rooted!(&in(cx) let proto_object = prototype.to_object());
498        let mut callbacks = {
499            let mut realm = AutoRealm::new_from_handle(cx, proto_object.handle());
500            match self.get_callbacks(&mut realm, proto_object.handle()) {
501                Ok(callbacks) => callbacks,
502                Err(error) => {
503                    self.element_definition_is_running.set(false);
504                    return Err(error);
505                },
506            }
507        };
508
509        // Step 14.5: Handle the case where with `attributeChangedCallback` on `lifecycleCallbacks`
510        // is not null.
511        let observed_attributes: Vec<DOMString> = if callbacks.attribute_changed_callback.is_some()
512        {
513            let mut realm = AutoRealm::new_from_handle(cx, constructor.handle());
514            match get_property(
515                &mut realm,
516                constructor.handle(),
517                c"observedAttributes",
518                StringificationBehavior::Default,
519            ) {
520                Ok(attributes) => attributes.unwrap_or_default(),
521                Err(error) => {
522                    self.element_definition_is_running.set(false);
523                    return Err(error);
524                },
525            }
526        } else {
527            Vec::new()
528        };
529
530        // Steps 14.6 - 14.10: Handle `disabledFeatures`.
531        let (disable_internals, disable_shadow) = {
532            let mut realm = AutoRealm::new_from_handle(cx, constructor.handle());
533            match get_property::<Vec<DOMString>>(
534                &mut realm,
535                constructor.handle(),
536                c"disabledFeatures",
537                StringificationBehavior::Default,
538            ) {
539                Ok(sequence) => {
540                    let sequence = sequence.unwrap_or_default();
541                    (
542                        sequence.iter().any(|s| *s == "internals"),
543                        sequence.iter().any(|s| *s == "shadow"),
544                    )
545                },
546                Err(error) => {
547                    self.element_definition_is_running.set(false);
548                    return Err(error);
549                },
550            }
551        };
552
553        // Step 14.11 - 14.12: Handle `formAssociated`.
554        let form_associated: bool = {
555            let mut realm = AutoRealm::new_from_handle(cx, constructor.handle());
556            match get_property(&mut realm, constructor.handle(), c"formAssociated", ()) {
557                Ok(flag) => flag.unwrap_or_default(),
558                Err(error) => {
559                    self.element_definition_is_running.set(false);
560                    return Err(error);
561                },
562            }
563        };
564
565        // Steps 14.13: Add the `formAssociated` callbacks.
566        if form_associated {
567            let mut realm = AutoRealm::new_from_handle(cx, proto_object.handle());
568            unsafe {
569                if let Err(error) = self.add_form_associated_callbacks(
570                    &mut realm,
571                    proto_object.handle(),
572                    &mut callbacks,
573                ) {
574                    self.element_definition_is_running.set(false);
575                    return Err(error);
576                }
577            }
578        }
579
580        self.element_definition_is_running.set(false);
581
582        // Step 15: Let definition be a new custom element definition with name name,
583        // local name localName, constructor constructor, observed attributes
584        // observedAttributes, lifecycle callbacks lifecycleCallbacks,
585        // form-associated formAssociated, disable internals disableInternals,
586        // and disable shadow disableShadow.
587        let definition = Rc::new(CustomElementDefinition::new(
588            name.clone(),
589            local_name.clone(),
590            constructor_,
591            observed_attributes,
592            callbacks,
593            form_associated,
594            disable_internals,
595            disable_shadow,
596        ));
597
598        // Step 16: Append definition to this's custom element definition set.
599        self.definitions
600            .borrow_mut()
601            .insert(name.clone(), definition.clone());
602
603        // Step 17: If this's is scoped is true, then for each document of
604        // this's scoped document set: upgrade particular elements within a
605        // document given this, document, definition, and localName.
606        if self.is_scoped.get() {
607            for document in self.scoped_document_set.borrow().iter() {
608                self.upgrade_particular_elements_within_a_document(
609                    cx,
610                    document,
611                    &definition,
612                    &local_name,
613                    &local_name,
614                );
615            }
616        } else {
617            // Step 18: Otherwise, upgrade particular elements within a document given
618            // this, this's relevant global object's associated Document, definition,
619            // localName, and name.
620            self.upgrade_particular_elements_within_a_document(
621                cx,
622                &self.window.Document(),
623                &definition,
624                &local_name,
625                &name,
626            );
627        }
628
629        // Step 19: If this's when-defined promise map[name] exists:
630        // Step 19.2: Remove this's when-defined promise map[name].
631        let promise = self.when_defined.borrow_mut().remove(&name);
632        if let Some(promise) = promise {
633            rooted!(&in(cx) let mut constructor = UndefinedValue());
634            definition

661    /// <https://html.spec.whatwg.org/multipage/#dom-customelementregistry-whendefined>
662    fn WhenDefined(&self, realm: &mut CurrentRealm, name: DOMString) -> Rc<Promise> {
663        let name = LocalName::from(name);
664
665        // Step 1
666        if !is_valid_custom_element_name(&name) {
667            let promise = Promise::new_in_realm(realm);
668            let error = DOMException::new(
669                realm,
670                self.window.as_global_scope(),
671                DOMErrorName::SyntaxError,
672            );
673            promise.reject_native(realm, &error);
674            return promise;
675        }
676
677        // Step 2
678        if let Some(definition) = self.definitions.borrow().get(&LocalName::from(&*name)) {
679            rooted!(&in(*realm) let mut constructor = UndefinedValue());
680            definition
681                .constructor
682                .to_jsval(realm, constructor.handle_mut());
683            let promise = Promise::new_in_realm(realm);
684            promise.resolve_native(realm, &constructor.get());
685            return promise;
686        }
687
688        // Steps 3, 4, 5, 6
689        let existing_promise = self.when_defined.borrow().get(&name).cloned();
690        existing_promise.unwrap_or_else(|| {
691            let promise = Promise::new_in_realm(realm);
692            self.when_defined.borrow_mut().insert(name, promise.clone());
693            promise
694        })
695    }

697    /// <https://html.spec.whatwg.org/multipage/#dom-customelementregistry-upgrade>
698    fn Upgrade(&self, cx: &JSContext, node: &Node) {
699        // Step 1. For each shadow-including inclusive descendant candidate of
700        // root, in shadow-including tree order:
701        for node in node.traverse_preorder_non_rooting(cx, ShadowIncluding::Yes) {
702            // Step 1.1. If candidate is not an Element node, then continue.
703            let Some(element) = node.downcast::<Element>() else {
704                continue;
705            };
706            // Step 1.2. If candidate's custom element registry is not this,
707            // then continue.
708            if element
709                .custom_element_registry()
710                .is_some_and(|registry| *registry != *self)
711            {
712                continue;
713            }
714            // Step 1.3. Try to upgrade candidate.
715            try_upgrade_element(cx, element);
716        }
717    }

719    /// <https://html.spec.whatwg.org/multipage/#dom-customelementregistry-initialize>
720    fn Initialize(&self, cx: &JSContext, root: &Node) -> ErrorResult {
721        // Step 1. If this's is scoped is false and either root is a Document node
722        // or root's node document's custom element registry is not this, then
723        // throw a "NotSupportedError" DOMException.
724        if !self.is_scoped.get() {
725            let is_document = root.is::<Document>();
726            let registry_mismatch = root
727                .owner_doc()
728                .custom_element_registry()
729                .is_some_and(|registry| *registry != *self);
730            if is_document || registry_mismatch {
731                return Err(Error::NotSupported(Some(
732                    "Initialize is not allowed on a non-scoped registry for this root".to_owned(),
733                )));
734            }
735        }
736
737        // Step 2. If root is a Document node whose custom element registry is null,
738        // then set root's custom element registry to this.
739        if let Some(document) = root.downcast::<Document>() {
740            if document.custom_element_registry().is_none() {
741                document.set_custom_element_registry(self);
742            }
743        }
744        // Step 3. Otherwise, if root is a ShadowRoot node whose custom element registry
745        // is null, then set root's custom element registry to this.
746        else if let Some(shadow_root) = root.downcast::<ShadowRoot>() &&
747            shadow_root.custom_element_registry().is_none()
748        {
749            shadow_root.set_custom_element_registry(Some(self));
750        }
751
752        // Step 4. For each inclusive descendant inclusiveDescendant of root, in tree order:
753        for node in root.traverse_preorder(ShadowIncluding::No) {
754            // Step 4.1. If inclusiveDescendant is not an Element node, then continue.
755            let Some(element) = node.downcast::<Element>() else {
756                continue;
757            };
758
759            // Step 4.2. If inclusiveDescendant's custom element registry is null:
760            // Note: A global registry (set by Node::adopt) is treated the same as null.
761            //       Both mean the element has no scoped registry, so it should get this one.
762            if element
763                .custom_element_registry()
764                .is_none_or(|registry| Self::is_a_global_element_registry(Some(&*registry)))
765            {
766                // Step 4.2.1. Set inclusiveDescendant's custom element registry to this.
767                element.set_custom_element_registry(Some(self), cx.no_gc());
768
769                // Step 4.2.2. If this's is scoped is true, then append
770                // inclusiveDescendant's node document to this's scoped document set.
771                if self.is_scoped.get() {
772                    let document = element.upcast::<Node>().owner_doc();
773                    self.scoped_document_set
774                        .borrow_mut()
775                        .push(Dom::from_ref(&document));
776                }
777            // Step 4.3. If inclusiveDescendant's custom element registry is not this, then continue.
778            } else if element
779                .custom_element_registry()
780                .is_none_or(|registry| *registry != *self)
781            {
782                continue;
783            }
784
785            // Step 4.4. Try to upgrade inclusiveDescendant.
786            try_upgrade_element(cx, element);
787        }
788        Ok(())
789    }

892    /// <https://dom.spec.whatwg.org/#concept-create-element> Step 5.1
893    #[expect(unsafe_code)]
894    pub(crate) fn create_element(
895        &self,
896        cx: &mut JSContext,
897        document: &Document,
898        prefix: Option<Prefix>,
899        registry: Option<&CustomElementRegistry>,
900    ) -> Fallible<DomRoot<Element>> {
901        let window = document.window();
902
903        // Step 5.1.1. Let C be definition’s constructor.
904        rooted!(&in(cx) let constructor = ObjectValue(self.constructor.callback()));
905        rooted!(&in(cx) let mut element = ptr::null_mut::<JSObject>());
906        {
907            // Go into the constructor's realm
908            let mut realm = AutoRealm::new(cx, NonNull::new(self.constructor.callback()).unwrap());
909            let cx = &mut realm;
910
911            // Step 5.3.1. Set result to the result of constructing C, with no arguments.
912            // https://webidl.spec.whatwg.org/#construct-a-callback-function
913            run_a_script::<DomTypeHolder, _, _>(cx, window.upcast(), |cx| {
914                run_a_callback::<DomTypeHolder, _>(window.upcast(), || {
915                    let args = HandleValueArray::empty();
916                    if unsafe { !Construct1(cx, constructor.handle(), &args, element.handle_mut()) }
917                    {
918                        Err(Error::JSFailed)
919                    } else {
920                        Ok(())
921                    }
922                })
923            })?;
924        }
925
926        rooted!(&in(cx) let element_val = ObjectValue(element.get()));
927        let element: DomRoot<Element> =
928            match FromJSValConvertible::from_jsval(cx, element_val.handle(), ()) {
929                Ok(ConversionResult::Success(element)) => element,
930                Ok(ConversionResult::Failure(..)) => {
931                    return Err(Error::Type(
932                        c"Constructor did not return a DOM node".to_owned(),
933                    ));
934                },
935                _ => return Err(Error::JSFailed),
936            };
937
938        // Step 5.1.3.2 Assert: result’s custom element state and custom element definition are initialized.
939        // Step 5.1.3.3 Assert: result’s namespace is the HTML namespace.
940        // Note: IDL enforces that result is an HTMLElement object, which all use the HTML namespace.
941        // Note: the custom element definition is initialized by the caller if
942        // this method returns a success value.
943        assert!(element.is::<HTMLElement>());
944
945        // Step 5.1.3.4. If result’s attribute list is not empty, then throw a "NotSupportedError" DOMException.
946        // Step 5.1.3.5. If result has children, then throw a "NotSupportedError" DOMException.
947        // Step 5.1.3.6. If result’s parent is not null, then throw a "NotSupportedError" DOMException.
948        // Step 5.1.3.7. If result’s node document is not document, then throw a "NotSupportedError" DOMException.
949        // Step 5.1.3.8. If result’s local name is not equal to localName then throw a "NotSupportedError" DOMException.
950        if element.HasAttributes() ||
951            element.upcast::<Node>().children_count() > 0 ||
952            element.upcast::<Node>().has_parent() ||
953            &*element.upcast::<Node>().owner_doc() != document ||
954            *element.namespace() != ns!(html) ||
955            *element.local_name() != self.local_name
956        {
957            return Err(Error::NotSupported(None));
958        }
959
960        // Step 5.1.3.9. Set result’s namespace prefix to prefix.
961        element.set_prefix(prefix);
962
963        // Step 5.1.3.10. Set result’s is value to null.
964        // Element's `is` is None by default
965
966        // Step 5.1.3.11. Set result’s custom element registry to registry.
967        element.set_custom_element_registry(registry, cx.no_gc());
968
969        Ok(element)
970    }

977/// <https://html.spec.whatwg.org/multipage/#concept-upgrade-an-element>
978pub(crate) fn upgrade_element(
979    cx: &mut JSContext,
980    definition: Rc<CustomElementDefinition>,
981    element: &Element,
982) {
983    // Step 1. If element's custom element state is not "undefined" or "uncustomized", then return.
984    let state = element.get_custom_element_state();
985    if state != CustomElementState::Undefined && state != CustomElementState::Uncustomized {
986        return;
987    }
988
989    // Step 2. Set element's custom element definition to definition.
990    element.set_custom_element_definition(Rc::clone(&definition), cx.no_gc());
991
992    // Step 3. Set element's custom element state to "failed".
993    element.set_custom_element_state(CustomElementState::Failed, cx.no_gc());
994
995    // Step 4. For each attribute in element's attribute list, in order, enqueue a custom element callback reaction
996    // with element, callback name "attributeChangedCallback", and « attribute's local name, null, attribute's value,
997    // attribute's namespace ».
998    let custom_element_reaction_stack = ScriptThread::custom_element_reaction_stack();
999    for attr in element.attrs().borrow().iter() {
1000        let local_name = attr.local_name().clone();
1001        let namespace = attr.namespace().clone();
1002        custom_element_reaction_stack.enqueue_callback_reaction(
1003            cx,
1004            element,
1005            CallbackReaction::AttributeChanged(local_name, None, Some(&*attr.value()), namespace),
1006            Some(definition.clone()),
1007        );
1008    }
1009
1010    // Step 5. If element is connected, then enqueue a custom element callback reaction with element,
1011    // callback name "connectedCallback", and « ».
1012    if element.is_connected() {
1013        custom_element_reaction_stack.enqueue_callback_reaction(
1014            cx,
1015            element,
1016            CallbackReaction::Connected,
1017            Some(definition.clone()),
1018        );
1019    }
1020
1021    // Step 6. Add element to the end of definition's construction stack.
1022    definition
1023        .construction_stack
1024        .borrow_mut()
1025        .push(ConstructionStackEntry::Element(DomRoot::from_ref(element)));
1026
1027    // Steps 7-8, successful case
1028    let result = run_upgrade_constructor(cx, &definition, element);
1029
1030    // "regardless of whether the above steps threw an exception" step
1031    definition.construction_stack.borrow_mut().pop();
1032
1033    // Step 8 exception handling
1034    if let Err(error) = result {
1035        // Step 8.exception.1
1036        element.clear_custom_element_definition(cx.no_gc());
1037
1038        // Step 8.exception.2
1039        element.clear_reaction_queue();
1040
1041        // Step 8.exception.3
1042        let global = GlobalScope::current().expect("No current global");
1043
1044        let mut realm = enter_auto_realm(cx, &*global);
1045        let cx = &mut realm.current_realm();
1046
1047        throw_dom_exception(cx, &global, error);
1048        report_pending_exception(cx);
1049
1050        return;
1051    }
1052
1053    // Step 9: handle with form-associated custom element
1054    if let Some(html_element) = element.downcast::<HTMLElement>() &&
1055        html_element.is_form_associated_custom_element()
1056    {
1057        // We know this element is is form-associated, so we can use the implementation of
1058        // `FormControl` for HTMLElement, which makes that assumption.
1059        // Step 9.1: Reset the form owner of element
1060        html_element.reset_form_owner(cx);
1061        if let Some(form) = html_element.form_owner() {
1062            // Even though the tree hasn't structurally mutated,
1063            // HTMLCollections need to be invalidated.
1064            form.upcast::<Node>().rev_version(cx.no_gc());
1065            // The spec tells us specifically to enqueue a formAssociated reaction
1066            // here, but it also says to do that for resetting form owner in general,
1067            // and we don't need two reactions.
1068        }
1069
1070        // Either enabled_state or disabled_state needs to be set,
1071        // and the possibility of a disabled fieldset ancestor needs
1072        // to be accounted for. (In the spec, being disabled is
1073        // a fact that's true or false about a node at a given time,
1074        // not a flag that belongs to the node and is updated,
1075        // so it doesn't describe this check as an action.)
1076        element.check_disabled_attribute();
1077        element.check_ancestors_disabled_state_for_form_control();
1078        element.update_read_write_state_from_readonly_attribute();
1079
1080        // Step 9.2: If element is disabled, then enqueue a custom element callback reaction
1081        // with element.
1082        if element.disabled_state() {
1083            custom_element_reaction_stack.enqueue_callback_reaction(
1084                cx,
1085                element,
1086                CallbackReaction::FormDisabled(true),
1087                Some(definition),
1088            )
1089        }
1090    }
1091
1092    // Step 10
1093    element.set_custom_element_state(CustomElementState::Custom, cx.no_gc());
1094}

1096/// <https://html.spec.whatwg.org/multipage/#concept-upgrade-an-element>
1097/// Steps 9.1-9.4
1098#[expect(unsafe_code)]
1099fn run_upgrade_constructor(
1100    cx: &mut JSContext,
1101    definition: &CustomElementDefinition,
1102    element: &Element,
1103) -> ErrorResult {
1104    let constructor = &definition.constructor;
1105    let window = element.owner_window();
1106    rooted!(&in(cx) let constructor_val = ObjectValue(constructor.callback()));
1107    rooted!(&in(cx) let mut element_val = UndefinedValue());
1108    element.to_jsval(cx, element_val.handle_mut());
1109    rooted!(&in(cx) let mut construct_result = ptr::null_mut::<JSObject>());
1110    {
1111        // Step 9.1. If definition's disable shadow is true and element's shadow root is non-null,
1112        // then throw a "NotSupportedError" DOMException.
1113        if definition.disable_shadow && element.is_shadow_host() {
1114            return Err(Error::NotSupported(None));
1115        }
1116
1117        // Go into the constructor's realm
1118        let mut realm = AutoRealm::new(cx, NonNull::new(constructor.callback()).unwrap());
1119        let cx = &mut *realm;
1120
1121        let args = HandleValueArray::empty();
1122        // Step 8.2. Set element's custom element state to "precustomized".
1123        element.set_custom_element_state(CustomElementState::Precustomized, cx.no_gc());
1124
1125        // Step 9.3. Let constructResult be the result of constructing C, with no arguments.
1126        // https://webidl.spec.whatwg.org/#construct-a-callback-function
1127        run_a_script::<DomTypeHolder, _, _>(cx, window.upcast(), |cx| {
1128            run_a_callback::<DomTypeHolder, _>(window.upcast(), || {
1129                if unsafe {
1130                    !Construct1(
1131                        cx,
1132                        constructor_val.handle(),
1133                        &args,
1134                        construct_result.handle_mut(),
1135                    )
1136                } {
1137                    Err(Error::JSFailed)
1138                } else {
1139                    Ok(())
1140                }
1141            })
1142        })?;
1143
1144        let mut same = false;
1145        rooted!(&in(cx) let construct_result_val = ObjectValue(construct_result.get()));
1146
1147        // Step 9.4. If SameValue(constructResult, element) is false, then throw a TypeError.
1148        if unsafe {
1149            !SameValue(
1150                cx,
1151                construct_result_val.handle(),
1152                element_val.handle(),
1153                &mut same,
1154            )
1155        } {
1156            return Err(Error::JSFailed);
1157        }
1158        if !same {
1159            return Err(Error::Type(
1160                c"Returned element is not SameValue as the upgraded element".to_owned(),
1161            ));
1162        }
1163    }
1164    Ok(())
1165}

1167/// <https://html.spec.whatwg.org/multipage/#concept-try-upgrade>
1168pub(crate) fn try_upgrade_element(cx: &JSContext, element: &Element) {
1169    // Step 1. Let definition be the result of looking up a custom element
1170    // definition given element's custom element registry, element's namespace,
1171    // element's local name, and element's is value.
1172    let lookup_registry = {
1173        // TODO: Remove this fallback when Node::adopt is aligned according to specs.
1174        //       Currently elements carry stale global registry from another document.
1175        let registry = element.custom_element_registry();
1176        if registry
1177            .as_ref()
1178            .is_some_and(|registry| registry.is_scoped())
1179        {
1180            registry
1181        } else {
1182            element.owner_document().custom_element_registry()
1183        }
1184    };
1185    if let Some(definition) = CustomElementRegistry::lookup_custom_element_definition(
1186        lookup_registry.as_deref(),
1187        element.namespace(),
1188        element.local_name(),
1189        element.get_is().as_ref(),
1190    ) {
1191        // Step 2. If definition is not null, then enqueue a custom element
1192        // upgrade reaction given element and definition.
1193        ScriptThread::enqueue_upgrade_reaction(cx, element, definition);
1194    }
1195}

1254/// <https://html.spec.whatwg.org/multipage/#custom-element-reactions-stack>
1255/// # Safety
1256/// This can be shared inside an Rc because one of those Rc copies lives
1257/// inside ScriptThread, so the GC can always reach this structure.
1258#[derive(JSTraceable, MallocSizeOf)]
1259#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
1260#[cfg_attr(crown, crown::unrooted_must_root_lint::allow_unrooted_in_rc)]
1261pub(crate) struct CustomElementReactionStack {
1262    stack: DomRefCell<Vec<ElementQueue>>,
1263    backup_queue: ElementQueue,
1264    processing_backup_element_queue: Cell<BackupElementQueueFlag>,
1265}
1266
1267impl CustomElementReactionStack {
1268    pub(crate) fn new() -> CustomElementReactionStack {
1269        CustomElementReactionStack {
1270            stack: DomRefCell::new(Vec::new()),
1271            backup_queue: ElementQueue::new(),
1272            processing_backup_element_queue: Cell::new(BackupElementQueueFlag::NotProcessing),
1273        }
1274    }
1275
1276    pub(crate) fn push_new_element_queue(&self) {
1277        self.stack.borrow_mut().push(ElementQueue::new());
1278    }
1279
1280    pub(crate) fn pop_current_element_queue(&self, cx: &mut JSContext) {
1281        rooted_vec!(let mut stack);
1282        mem::swap(&mut *stack, &mut *self.stack.borrow_mut());
1283
1284        if let Some(current_queue) = stack.last() {
1285            current_queue.invoke_reactions(cx);
1286        }
1287        stack.pop();
1288
1289        mem::swap(&mut *self.stack.borrow_mut(), &mut *stack);
1290        self.stack.borrow_mut().append(&mut *stack);
1291    }
1292
1293    /// <https://html.spec.whatwg.org/multipage/#enqueue-an-element-on-the-appropriate-element-queue>
1294    /// Step 4
1295    pub(crate) fn invoke_backup_element_queue(&self, cx: &mut JSContext) {
1296        // Step 4.1
1297        self.backup_queue.invoke_reactions(cx);
1298
1299        // Step 4.2
1300        self.processing_backup_element_queue
1301            .set(BackupElementQueueFlag::NotProcessing);
1302    }
1303
1304    /// <https://html.spec.whatwg.org/multipage/#enqueue-an-element-on-the-appropriate-element-queue>
1305    pub(crate) fn enqueue_element(&self, cx: &JSContext, element: &Element) {
1306        if let Some(current_queue) = self.stack.borrow().last() {
1307            // Step 2
1308            current_queue.append_element(element);
1309        } else {
1310            // Step 1.1
1311            self.backup_queue.append_element(element);
1312
1313            // Step 1.2
1314            if self.processing_backup_element_queue.get() == BackupElementQueueFlag::Processing {
1315                return;
1316            }
1317
1318            // Step 1.3
1319            self.processing_backup_element_queue
1320                .set(BackupElementQueueFlag::Processing);
1321
1322            // Step 4
1323            ScriptThread::enqueue_microtask(cx, Box::new(CustomElementReactionMicrotask::new()));
1324        }
1325    }

1327    /// <https://html.spec.whatwg.org/multipage/#enqueue-a-custom-element-callback-reaction>
1328    #[cfg_attr(crown, expect(crown::unrooted_must_root))]
1329    pub(crate) fn enqueue_callback_reaction(
1330        &self,
1331        cx: &mut JSContext,
1332        element: &Element,
1333        reaction: CallbackReaction,
1334        definition: Option<Rc<CustomElementDefinition>>,
1335    ) {
1336        // Step 1. Let definition be element's custom element definition.
1337        let definition = match definition.or_else(|| element.get_custom_element_definition()) {
1338            Some(definition) => definition,
1339            None => return,
1340        };
1341
1342        // Step 2. Let callback be the value of the entry in definition's lifecycle callbacks with
1343        // key callbackName.
1344        let (callback, args) = match reaction {
1345            CallbackReaction::Connected => {
1346                (definition.callbacks.connected_callback.clone(), Vec::new())
1347            },
1348            CallbackReaction::Disconnected => (
1349                definition.callbacks.disconnected_callback.clone(),
1350                Vec::new(),
1351            ),
1352            CallbackReaction::Adopted(ref old_doc, ref new_doc) => {
1353                let args = vec![Heap::default(), Heap::default()];
1354                args[0].set(ObjectValue(old_doc.reflector().get_jsobject().get()));
1355                args[1].set(ObjectValue(new_doc.reflector().get_jsobject().get()));
1356                (definition.callbacks.adopted_callback.clone(), args)
1357            },
1358            CallbackReaction::AttributeChanged(local_name, old_val, val, namespace) => {
1359                // Step 5.
1360                if !definition
1361                    .observed_attributes
1362                    .iter()
1363                    .any(|attr| *attr == *local_name)
1364                {
1365                    return;
1366                }
1367
1368                // We might be here during HTML parsing, rather than
1369                // during Javscript execution, and so we typically aren't
1370                // already in a realm here.
1371                let mut realm = enter_auto_realm(cx, &*element.global());
1372                let cx = &mut realm;
1373
1374                let local_name = DOMString::from(&*local_name);
1375                rooted!(&in(cx) let mut name_value = UndefinedValue());
1376                local_name.to_jsval(cx, name_value.handle_mut());
1377
1378                rooted!(&in(cx) let mut old_value = NullValue());
1379                if let Some(old_val) = old_val {
1380                    old_val.to_jsval(cx, old_value.handle_mut());
1381                }
1382
1383                rooted!(&in(cx) let mut value = NullValue());
1384                if let Some(val) = val {
1385                    val.to_jsval(cx, value.handle_mut());
1386                }
1387
1388                rooted!(&in(cx) let mut namespace_value = NullValue());
1389                if namespace != ns!() {
1390                    let namespace = DOMString::from(&*namespace);
1391                    namespace.to_jsval(cx, namespace_value.handle_mut());
1392                }
1393
1394                let args = vec![
1395                    Heap::default(),
1396                    Heap::default(),
1397                    Heap::default(),
1398                    Heap::default(),
1399                ];
1400                args[0].set(name_value.get());
1401                args[1].set(old_value.get());
1402                args[2].set(value.get());
1403                args[3].set(namespace_value.get());
1404
1405                (
1406                    definition.callbacks.attribute_changed_callback.clone(),
1407                    args,
1408                )
1409            },
1410            CallbackReaction::FormAssociated(form) => {
1411                let args = vec![Heap::default()];
1412                if let Some(form) = form {
1413                    args[0].set(ObjectValue(form.reflector().get_jsobject().get()));
1414                } else {
1415                    args[0].set(NullValue());
1416                }
1417                (definition.callbacks.form_associated_callback.clone(), args)
1418            },
1419            CallbackReaction::FormDisabled(disabled) => {
1420                rooted!(&in(cx) let disabled_value = BooleanValue(disabled));
1421                let args = vec![Heap::default()];
1422                args[0].set(disabled_value.get());
1423                (definition.callbacks.form_disabled_callback.clone(), args)
1424            },
1425            CallbackReaction::FormReset => {
1426                (definition.callbacks.form_reset_callback.clone(), Vec::new())
1427            },
1428            CallbackReaction::ConnectedMove => {
1429                let callback = definition.callbacks.connected_move_callback.clone();
1430                // Step 3. If callbackName is "connectedMoveCallback" and callback is null:
1431                if callback.is_none() {
1432                    // Step 3.1. Let disconnectedCallback be the value of the entry in
1433                    // definition's lifecycle callbacks with key "disconnectedCallback".
1434                    let disconnected_callback = definition.callbacks.disconnected_callback.clone();
1435
1436                    // Step 3.2. Let connectedCallback be the value of the entry in
1437                    // definition's lifecycle callbacks with key "connectedCallback".
1438                    let connected_callback = definition.callbacks.connected_callback.clone();
1439
1440                    // Step 3.3. If connectedCallback and disconnectedCallback are null,
1441                    // then return.
1442                    if disconnected_callback.is_none() && connected_callback.is_none() {
1443                        return;
1444                    }
1445
1446                    // Step 3.4. Set callback to the following steps:
1447                    // Step 3.4.1. If disconnectedCallback is not null, then call
1448                    // disconnectedCallback with no arguments.
1449                    if let Some(disconnected_callback) = disconnected_callback {
1450                        element.push_callback_reaction(
1451                            disconnected_callback,
1452                            Box::new([]),
1453                            cx.no_gc(),
1454                        );
1455                    }
1456                    // Step 3.4.2. If connectedCallback is not null, then call
1457                    // connectedCallback with no arguments.
1458                    if let Some(connected_callback) = connected_callback {
1459                        element.push_callback_reaction(
1460                            connected_callback,
1461                            Box::new([]),
1462                            cx.no_gc(),
1463                        );
1464                    }
1465
1466                    self.enqueue_element(cx, element);
1467                    return;
1468                }
1469
1470                (callback, Vec::new())
1471            },
1472        };
1473
1474        // Step 4. If callback is null, then return.
1475        let callback = match callback {
1476            Some(callback) => callback,
1477            None => return,
1478        };
1479
1480        // Step 6. Add a new callback reaction to element's custom element reaction queue, with
1481        // callback function callback and arguments args.
1482        element.push_callback_reaction(callback, args.into_boxed_slice(), cx.no_gc());
1483
1484        // Step 7. Enqueue an element on the appropriate element queue given element.
1485        self.enqueue_element(cx, element);
1486    }

1488    /// <https://html.spec.whatwg.org/multipage/#enqueue-a-custom-element-upgrade-reaction>
1489    pub(crate) fn enqueue_upgrade_reaction(
1490        &self,
1491        cx: &JSContext,
1492        element: &Element,
1493        definition: Rc<CustomElementDefinition>,
1494    ) {
1495        // Step 1. Add a new upgrade reaction to element's custom element reaction queue,
1496        // with custom element definition definition.
1497        element.push_upgrade_reaction(definition, cx.no_gc());
1498
1499        // Step 2. Enqueue an element on the appropriate element queue given element.
1500        self.enqueue_element(cx, element);
1501    }

1504/// <https://html.spec.whatwg.org/multipage/#element-queue>
1505#[derive(JSTraceable, MallocSizeOf)]
1506#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
1507struct ElementQueue {
1508    queue: DomRefCell<VecDeque<Dom<Element>>>,
1509}
1510
1511impl ElementQueue {
1512    fn new() -> ElementQueue {
1513        ElementQueue {
1514            queue: Default::default(),
1515        }
1516    }
1517
1518    /// <https://html.spec.whatwg.org/multipage/#invoke-custom-element-reactions>
1519    fn invoke_reactions(&self, cx: &mut JSContext) {
1520        // Steps 1-2
1521        while let Some(element) = self.next_element() {
1522            element.invoke_reactions(cx)
1523        }
1524        self.queue.borrow_mut().clear();
1525    }
1526
1527    fn next_element(&self) -> Option<DomRoot<Element>> {
1528        self.queue
1529            .borrow_mut()
1530            .pop_front()
1531            .as_deref()
1532            .map(DomRoot::from_ref)
1533    }
1534
1535    fn append_element(&self, element: &Element) {
1536        self.queue.borrow_mut().push_back(Dom::from_ref(element));
1537    }
1538}

1540/// <https://html.spec.whatwg.org/multipage/#valid-custom-element-name>
1541pub(crate) fn is_valid_custom_element_name(name: &str) -> bool {
1542    // Custom elment names must match:
1543    // PotentialCustomElementName ::= [a-z] (PCENChar)* '-' (PCENChar)*
1544    let mut chars = name.chars();
1545    if !chars.next().is_some_and(|c| c.is_ascii_lowercase()) {
1546        return false;
1547    }
1548
1549    let mut has_dash = false;
1550
1551    for c in chars {
1552        if c == '-' {
1553            has_dash = true;
1554            continue;
1555        }
1556
1557        if !is_potential_custom_element_char(c) {
1558            return false;
1559        }
1560    }
1561
1562    if !has_dash {
1563        return false;
1564    }
1565
1566    if name == "annotation-xml" ||
1567        name == "color-profile" ||
1568        name == "font-face" ||
1569        name == "font-face-src" ||
1570        name == "font-face-uri" ||
1571        name == "font-face-format" ||
1572        name == "font-face-name" ||
1573        name == "missing-glyph"
1574    {
1575        return false;
1576    }
1577
1578    true
1579}

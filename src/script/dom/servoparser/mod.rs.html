110/// The parser maintains two input streams: one for input from script through
111/// document.write(), and one for input from network.
112///
113/// There is no concrete representation of the insertion point, instead it
114/// always points to just before the next character from the network input,
115/// with all of the script input before itself.
116///
117/// ```text
118///     ... script input ... | ... network input ...
119///                          ^
120///                 insertion point
121/// ```
122pub(crate) struct ServoParser {
123    reflector: Reflector,
124    /// The document associated with this parser.
125    document: Dom<Document>,
126    /// The decoder used for the network input.
127    network_decoder: DomRefCell<NetworkDecoderState>,
128    /// Input received from network.
129    #[ignore_malloc_size_of = "Defined in html5ever"]
130    #[no_trace]
131    network_input: BufferQueue,
132    /// Input received from script. Used only to support document.write().
133    #[ignore_malloc_size_of = "Defined in html5ever"]
134    #[no_trace]
135    script_input: BufferQueue,
136    /// The tokenizer of this parser.
137    tokenizer: Tokenizer,
138    /// Whether to expect any further input from the associated network request.
139    last_chunk_received: Cell<bool>,
140    /// Whether this parser should avoid passing any further data to the tokenizer.
141    suspended: Cell<bool>,
142    /// <https://html.spec.whatwg.org/multipage/#script-nesting-level>
143    script_nesting_level: Cell<usize>,
144    /// <https://html.spec.whatwg.org/multipage/#abort-a-parser>
145    aborted: Cell<bool>,
146    /// <https://html.spec.whatwg.org/multipage/#stop-parsing>
147    stopped: Cell<bool>,
148    /// <https://html.spec.whatwg.org/multipage/#script-created-parser>
149    script_created_parser: bool,
150    /// A decoder exclusively for input to the prefetch tokenizer.
151    ///
152    /// Unlike the actual decoder, this one takes a best guess at the encoding and starts
153    /// decoding immediately.
154    #[no_trace]
155    prefetch_decoder: RefCell<LossyDecoder<NetworkSink>>,
156    /// We do a quick-and-dirty parse of the input looking for resources to prefetch.
157    // TODO: if we had speculative parsing, we could do this when speculatively
158    // building the DOM. https://github.com/servo/servo/pull/19203
159    prefetch_tokenizer: prefetch::Tokenizer,
160    #[ignore_malloc_size_of = "Defined in html5ever"]
161    #[no_trace]
162    prefetch_input: BufferQueue,
163    // The whole input as a string, if needed for the devtools Sources panel.
164    // TODO: use a faster type for concatenating strings?
165    content_for_devtools: Option<DomRefCell<String>>,
166}

187    pub(crate) fn parse_html_document(
188        cx: &mut JSContext,
189        document: &Document,
190        input: Option<DOMString>,
191        url: ServoUrl,
192        encoding_hint_from_content_type: Option<&'static Encoding>,
193        encoding_of_container_document: Option<&'static Encoding>,
194    ) {
195        // Step 1. Set document's type to "html".
196        //
197        // Set by callers of this function and asserted here
198        assert!(document.is_html_document());
199
200        // Step 2. Create an HTML parser parser, associated with document.
201        let parser = ServoParser::new(
202            cx,
203            document,
204            if pref!(dom_servoparser_async_html_tokenizer_enabled) {
205                Tokenizer::AsyncHtml(self::async_html::Tokenizer::new(document, url, None))
206            } else {
207                Tokenizer::Html(self::html::Tokenizer::new(
208                    document,
209                    url,
210                    None,
211                    ParsingAlgorithm::Normal,
212                ))
213            },
214            ParserKind::Normal,
215            encoding_hint_from_content_type,
216            encoding_of_container_document,
217        );
218
219        // Step 3. Place html into the input stream for parser. The encoding confidence is irrelevant.
220        // Step 4. Start parser and let it run until it has consumed all the
221        // characters just inserted into the input stream.
222        //
223        // Set as the document's current parser and initialize with `input`, if given.
224        if let Some(input) = input {
225            parser.parse_complete_string_chunk(cx, String::from(input));
226        } else {
227            parser.document.set_current_parser(Some(&parser));
228        }
229    }

233        cx: &mut JSContext,
234        context: &'el Element,

270            context_document.image_cache(),
271        );
272
273        // Step 2. If context's node document is in quirks mode, then set document's mode to "quirks".

275        // mode to "limited-quirks".

279        // Step 4. If allowDeclarativeShadowRoots is true, then set document's
280        // allow declarative shadow roots to true.
281        // Step 5. Create a new HTML parser, and associate it with document.
282
283        // Step 11.
284        let form = context_node
285            .inclusive_ancestors(ShadowIncluding::No)
286            .find(|element| element.is::<HTMLFormElement>());
287
288        let fragment_context = FragmentContext {
289            context_elem: context_node,
290            form_elem: form.as_deref(),
291            context_element_allows_scripting: context_document.scripting_enabled(),
292        };
293
294        let parser = ServoParser::new(
295            cx,
296            &document,
297            Tokenizer::Html(self::html::Tokenizer::new(
298                &document,
299                url,
300                Some(fragment_context),
301                ParsingAlgorithm::Fragment,
302            )),
303            ParserKind::Normal,
304            None,
305            None,
306        );
307        parser.parse_complete_string_chunk(cx, String::from(input));
308
309        // Step 14.
310        let root_element = document.GetDocumentElement().expect("no document element");
311        FragmentParsingResult {
312            inner: root_element.upcast::<Node>().children(),
313        }
314    }

369    /// Corresponds to the latter part of the "Otherwise" branch of the 'An end
370    /// tag whose tag name is "script"' of
371    /// <https://html.spec.whatwg.org/multipage/#parsing-main-incdata>
372    ///
373    /// This first moves everything from the script input to the beginning of
374    /// the network input, effectively resetting the insertion point to just
375    /// before the next character to be consumed.
376    ///
377    ///
378    /// ```text
379    ///     | ... script input ... network input ...
380    ///     ^
381    ///     insertion point
382    /// ```
383    pub(crate) fn resume_with_pending_parsing_blocking_script(
384        &self,
385        cx: &mut JSContext,
386        script: &HTMLScriptElement,
387        result: ScriptResult,
388    ) {
389        assert!(self.suspended.get());
390        self.suspended.set(false);
391
392        self.script_input.swap_with(&self.network_input);
393        while let Some(chunk) = self.script_input.pop_front() {
394            self.network_input.push_back(chunk);
395        }
396
397        let script_nesting_level = self.script_nesting_level.get();
398        assert_eq!(script_nesting_level, 0);
399
400        self.script_nesting_level.set(script_nesting_level + 1);
401        script.execute(cx, result);
402        self.script_nesting_level.set(script_nesting_level);
403
404        if !self.suspended.get() && !self.aborted.get() {
405            self.parse_sync(cx);
406        }
407    }

413    /// Steps 6-8 of <https://html.spec.whatwg.org/multipage/#document.write()>
414    pub(crate) fn write(&self, cx: &mut JSContext, text: DOMString) {
415        assert!(self.can_write());
416
417        if self.document.has_pending_parsing_blocking_script() {
418            // There is already a pending parsing blocking script so the
419            // parser is suspended, we just append everything to the
420            // script input and abort these steps.
421            self.script_input.push_back(String::from(text).into());
422            return;
423        }
424
425        // There is no pending parsing blocking script, so all previous calls
426        // to document.write() should have seen their entire input tokenized
427        // and process, with nothing pushed to the parser script input.
428        assert!(self.script_input.is_empty());
429
430        let input = BufferQueue::default();
431        input.push_back(String::from(text).into());
432
433        let profiler_chan = self
434            .document
435            .window()
436            .as_global_scope()
437            .time_profiler_chan()
438            .clone();
439        let profiler_metadata = TimerMetadata {
440            url: self.document.url().as_str().into(),
441            iframe: TimerMetadataFrameType::RootWindow,
442            incremental: TimerMetadataReflowType::FirstReflow,
443        };
444        self.tokenize(cx, |cx, tokenizer| {
445            tokenizer.feed(cx, &input, profiler_chan.clone(), profiler_metadata.clone())
446        });
447
448        if self.suspended.get() {
449            // Parser got suspended, insert remaining input at end of
450            // script input, following anything written by scripts executed
451            // reentrantly during this call.
452            while let Some(chunk) = input.pop_front() {
453                self.script_input.push_back(chunk);
454            }
455            return;
456        }
457
458        assert!(input.is_empty());
459    }

461    /// Steps 4-6 of <https://html.spec.whatwg.org/multipage/#dom-document-close>
462    pub(crate) fn close(&self, cx: &mut JSContext) {
463        assert!(self.script_created_parser);
464
465        // Step 4. Insert an explicit "EOF" character at the end of the parser's input stream.
466        self.last_chunk_received.set(true);
467
468        // Step 5. If this's pending parsing-blocking script is not null, then return.
469        if self.suspended.get() {
470            return;
471        }
472
473        // Step 6. Run the tokenizer, processing resulting tokens as they are emitted,
474        // and stopping when the tokenizer reaches the explicit "EOF" character or spins the event loop.
475        self.parse_sync(cx);
476    }

478    // https://html.spec.whatwg.org/multipage/#abort-a-parser
479    pub(crate) fn abort(&self, cx: &mut JSContext) {
480        assert!(!self.aborted.get());
481        self.aborted.set(true);
482
483        // Step 1.
484        self.script_input.replace_with(BufferQueue::default());
485        self.network_input.replace_with(BufferQueue::default());
486
487        // Step 2.
488        self.document
489            .update_the_current_document_readiness(cx, DocumentReadyState::Interactive);
490
491        // Step 3.
492        self.tokenizer.end(cx);
493        self.document.set_current_parser(None);
494
495        // Step 4.
496        self.document
497            .update_the_current_document_readiness(cx, DocumentReadyState::Complete);
498    }

567    fn push_tendril_input_chunk(&self, chunk: StrTendril) {
568        if let Some(mut content_for_devtools) = self
569            .content_for_devtools
570            .as_ref()
571            .map(|content| content.borrow_mut())
572        {
573            // TODO: append these chunks more efficiently
574            content_for_devtools.push_str(chunk.as_ref());
575        }
576
577        if chunk.is_empty() {
578            return;
579        }
580
581        // Push the chunk into the network input stream,
582        // which is tokenized lazily.
583        self.network_input.push_back(chunk);
584    }

586    fn push_bytes_input_chunk(&self, chunk: &[u8]) {
587        // For byte input, we convert it to text using the network decoder.
588        if let Some(decoded_chunk) = self
589            .network_decoder
590            .borrow_mut()
591            .push(chunk, &self.document)
592        {
593            self.push_tendril_input_chunk(decoded_chunk);
594        }
595
596        if self.should_prefetch() {
597            // Push the chunk into the prefetch input stream,
598            // which is tokenized eagerly, to scan for resources
599            // to prefetch. If the user script uses `document.write()`
600            // to overwrite the network input, this prefetching may
601            // have been wasted, but in most cases it won't.
602            let mut prefetch_decoder = self.prefetch_decoder.borrow_mut();
603            prefetch_decoder.process(ByteTendril::from(chunk));
604
605            self.prefetch_input
606                .push_back(mem::take(&mut prefetch_decoder.inner_sink_mut().output));
607            self.prefetch_tokenizer.feed(&self.prefetch_input);
608        }
609    }

611    fn should_prefetch(&self) -> bool {
612        // Per https://github.com/whatwg/html/issues/1495
613        // stylesheets should not be loaded for documents
614        // without browsing contexts.
615        // https://github.com/whatwg/html/issues/1495#issuecomment-230334047
616        // suggests that no content should be preloaded in such a case.
617        // We're conservative, and only prefetch for documents
618        // with browsing contexts.
619        self.document.browsing_context().is_some()
620    }

629    fn parse_sync(&self, cx: &mut JSContext) {
630        assert!(self.script_input.is_empty());
631
632        // This parser will continue to parse while there is either pending input or
633        // the parser remains unsuspended.
634
635        if self.last_chunk_received.get() {
636            let chunk = self.network_decoder.borrow_mut().finish(&self.document);
637            if !chunk.is_empty() {
638                self.push_tendril_input_chunk(chunk);
639            }
640        }
641
642        if self.aborted.get() {
643            return;
644        }
645
646        let profiler_chan = self
647            .document
648            .window()
649            .as_global_scope()
650            .time_profiler_chan()
651            .clone();
652        let profiler_metadata = TimerMetadata {
653            url: self.document.url().as_str().into(),
654            iframe: TimerMetadataFrameType::RootWindow,
655            incremental: TimerMetadataReflowType::FirstReflow,
656        };
657        self.tokenize(cx, |cx, tokenizer| {
658            tokenizer.feed(
659                cx,
660                &self.network_input,
661                profiler_chan.clone(),
662                profiler_metadata.clone(),
663            )
664        });
665
666        if self.suspended.get() {
667            return;
668        }
669
670        assert!(self.network_input.is_empty());
671
672        if self.last_chunk_received.get() {
673            self.finish(cx);
674        }
675    }
676
677    fn parse_complete_string_chunk(&self, cx: &mut JSContext, input: String) {
678        self.document.set_current_parser(Some(self));
679        self.push_string_input_chunk(input);
680        self.last_chunk_received.set(true);
681        if !self.suspended.get() {
682            self.parse_sync(cx);
683        }
684    }
685
686    fn parse_bytes_chunk(&self, cx: &mut JSContext, input: &[u8]) {
687        let mut realm = enter_auto_realm(cx, &*self.document);
688        let cx = &mut realm.current_realm();
689        self.document.set_current_parser(Some(self));
690        self.push_bytes_input_chunk(input.as_ref());
691        if !self.suspended.get() {
692            self.parse_sync(cx);
693        }
694    }
695
696    fn tokenize<F>(&self, cx: &mut JSContext, feed: F)
697    where
698        F: Fn(&mut JSContext, &Tokenizer) -> TokenizerResult<DomRoot<HTMLScriptElement>>,
699    {
700        loop {
701            assert!(!self.suspended.get());
702            assert!(!self.aborted.get());
703
704            self.document.window().reflow_if_reflow_timer_expired(cx);
705            let script = match feed(cx, &self.tokenizer) {
706                TokenizerResult::Done => return,
707                TokenizerResult::EncodingIndicator(_) => continue,
708                TokenizerResult::Script(script) => script,
709            };
710
711            // https://html.spec.whatwg.org/multipage/#parsing-main-incdata
712            // branch "An end tag whose tag name is "script"
713            // The spec says to perform the microtask checkpoint before
714            // setting the insertion mode back from Text, but this is not
715            // possible with the way servo and html5ever currently
716            // relate to each other, and hopefully it is not observable.
717            if is_execution_stack_empty() {
718                self.document.window().perform_a_microtask_checkpoint(cx);
719            }
720
721            let script_nesting_level = self.script_nesting_level.get();
722
723            self.script_nesting_level.set(script_nesting_level + 1);
724            script.set_initial_script_text();
725            let introduction_type_override =
726                (script_nesting_level > 0).then_some(IntroductionType::INJECTED_SCRIPT);
727            script.prepare(cx, introduction_type_override);
728            self.script_nesting_level.set(script_nesting_level);
729
730            if self.document.has_pending_parsing_blocking_script() {
731                self.suspended.set(true);
732                return;
733            }
734            if self.aborted.get() {
735                return;
736            }
737        }
738    }
739
740    /// <https://html.spec.whatwg.org/multipage/#abort-a-parser>
741    pub(crate) fn has_aborted(&self) -> bool {
742        self.aborted.get()
743    }
744
745    /// <https://html.spec.whatwg.org/multipage/#stop-parsing>
746    pub(crate) fn has_stopped(&self) -> bool {
747        self.stopped.get()
748    }
749
750    /// <https://html.spec.whatwg.org/multipage/#the-end>
751    fn finish(&self, cx: &mut JSContext) {
752        assert!(!self.suspended.get());
753        assert!(self.last_chunk_received.get());
754        assert!(self.script_input.is_empty());
755        assert!(self.network_input.is_empty());
756        assert!(self.network_decoder.borrow().is_finished());
757
758        self.stopped.set(true);
759
760        // Step 1. If the active speculative HTML parser is not null,
761        // then stop the speculative HTML parser and return.
762        // TODO
763        // Step 2. Set the insertion point to undefined.
764        self.tokenizer.end(cx);
765        // Step 3. Update the current document readiness to "interactive".
766        self.document
767            .update_the_current_document_readiness(cx, DocumentReadyState::Interactive);
768        // Step 4. Pop all the nodes off the stack of open elements.
769        self.document.set_current_parser(None);
770        // Step 5. While the list of scripts that will execute when the document has finished parsing is not empty:
771        self.document.start_the_end_loading_phase();
772        let url = self.tokenizer.url().clone();
773        self.document.finish_load(LoadType::PageSource(url), cx);
774
775        // Send the source contents to devtools, if needed.
776        if let Some(content_for_devtools) = self
777            .content_for_devtools
778            .as_ref()
779            .map(|content| content.take())
780        {
781            let global = self.document.global();
782            let chan = global.devtools_chan().expect("Guaranteed by new");
783            let pipeline_id = self.document.global().pipeline_id();
784            let _ = chan.send(ScriptToDevtoolsControlMsg::UpdateSourceContent(
785                pipeline_id,
786                content_for_devtools,
787            ));
788        }
789    }

1067    /// <https://html.spec.whatwg.org/multipage/#loading-a-document>
1068    fn load_document(
1069        &mut self,
1070        cx: &mut JSContext,
1071        parser: Option<&ServoParser>,
1072        document: &Document,
1073    ) {
1074        assert!(!self.has_loaded_document);
1075        self.has_loaded_document = true;
1076        // Step 1. Let type be the computed type of navigationParams's response.
1077        let content_type = &self.navigation_params.content_type;
1078        let mime_type = MimeClassifier::default().classify(
1079            LoadContext::Browsing,
1080            NoSniffFlag::Off,
1081            ApacheBugFlag::from_content_type(content_type.as_ref()),
1082            content_type,
1083            &self.navigation_params.resource_header,
1084        );
1085        // Step 2. If the user agent has been configured to process resources of the given type using
1086        // some mechanism other than rendering the content in a navigable, then skip this step.
1087        // Otherwise, if the type is one of the following types:
1088        let Some(media_type) = MimeClassifier::get_media_type(&mime_type) else {
1089            let page = format!(
1090                "<html><body><p>Unknown content type ({}).</p></body></html>",
1091                mime_type,
1092            );
1093            self.load_inline_unknown_content(
1094                cx,
1095                parser.expect("Must have a parser for unknown content"),
1096                page,
1097            );
1098            return;
1099        };
1100        match media_type {
1101            // Return the result of loading an HTML document, given navigationParams.
1102            MediaType::Html => self.load_html_document(cx, document),
1103            // Return the result of loading an XML document given navigationParams and type.
1104            MediaType::Xml => self.load_xml_document(cx, document),
1105            // Return the result of loading a text document given navigationParams and type.
1106            MediaType::JavaScript | MediaType::Text | MediaType::Css => {
1107                self.load_text_document(cx, parser.expect("Must have a parser for text"))
1108            },
1109            // Return the result of loading a json document given navigationParams and type.
1110            MediaType::Json => {
1111                self.load_json_document(cx, parser.expect("Must have a parser for JSON"))
1112            },
1113            // Return the result of loading a media document given navigationParams and type.
1114            MediaType::Image | MediaType::AudioVideo => {
1115                self.load_media_document(
1116                    cx,
1117                    parser.expect("Must have a parser for media"),
1118                    media_type,
1119                    &mime_type,
1120                );
1121                return;
1122            },
1123            MediaType::Font => {
1124                let page = format!(
1125                    "<html><body><p>Unable to load font with content type ({}).</p></body></html>",
1126                    mime_type,
1127                );
1128                self.load_inline_unknown_content(
1129                    cx,
1130                    parser.expect("Must have a parser for inline unknown"),
1131                    page,
1132                );
1133                return;
1134            },
1135        };
1136
1137        if let Some(parser) = parser {
1138            parser.parse_bytes_chunk(
1139                cx,
1140                std::mem::take(&mut self.navigation_params.resource_header).as_ref(),
1141            );
1142        }
1143    }

1328    pub(crate) fn process_response(

1330        script_thread: &ScriptThread,
1331        cx: &mut JSContext,
1332        meta_result: Result<FetchMetadata, NetworkError>,
1333    ) {
1334        let (metadata, mut error) = match meta_result {
1335            Ok(meta) => (
1336                Some(match meta {
1337                    FetchMetadata::Unfiltered(m) => m,
1338                    FetchMetadata::Filtered { unsafe_, .. } => unsafe_,
1339                }),
1340                None,
1341            ),
1342            Err(error) => (
1343                // Check variant without moving
1344                match &error {
1345                    NetworkError::LoadCancelled => {
1346                        return;
1347                    },
1348                    _ => {
1349                        let mut meta = Metadata::default(self.url.clone());
1350                        let mime: Option<Mime> = "text/html".parse().ok();
1351                        meta.set_content_type(mime.as_ref());
1352                        Some(meta)
1353                    },
1354                },
1355                Some(error),
1356            ),
1357        };
1358        let content_type: Option<Mime> = metadata
1359            .clone()
1360            .and_then(|meta| meta.content_type)
1361            .map(Serde::into_inner)
1362            .map(Into::into);
1363
1364        // <https://html.spec.whatwg.org/multipage/#create-navigation-params-by-fetching>
1365        // Step 21.9. Set responsePolicyContainer to the result of creating a
1366        // policy container from a fetch response given response and request's
1367        // reserved client.
1368        let (policy_container, endpoints_list, link_headers) = match metadata.as_ref() {
1369            None => (PolicyContainer::default(), None, vec![]),
1370            Some(metadata) => (

1372                ReportingEndpoint::parse_reporting_endpoints_header(
1373                    &self.url.clone(),
1374                    &metadata.headers,
1375                ),
1376                extract_links_from_headers(&metadata.headers),
1377            ),
1378        };
1379
1380        // Step 21.10. Set finalSandboxFlags to the union of targetSnapshotParams's
1381        // sandboxing flags and responsePolicyContainer's CSP list's CSP-derived
1382        // sandboxing flags.
1383        let final_sandboxing_flag_set = policy_container
1384            .csp_list
1385            .as_ref()
1386            .and_then(|csp| csp.get_sandboxing_flag_set_for_document())
1387            .unwrap_or(SandboxingFlagSet::empty())
1388            .union(self.target_snapshot_params.sandboxing_flags);
1389
1390        // Step 21.11. Set responseOrigin to the result of determining the origin
1391        // given response's URL, finalSandboxFlags, and entry's document state's
1392        // initiator origin.
1393        let source_origin = match self.load_origin {
1394            LoadOrigin::Script(ref snapshot) => {
1395                Some(MutableOrigin::from_snapshot(snapshot.clone()))
1396            },
1397            _ => None,
1398        };

1400            metadata.as_ref().map(|metadata| &metadata.final_url),
1401            final_sandboxing_flag_set,
1402            source_origin,
1403        );
1404

1406            self.webview_id,
1407            self.pipeline_id,
1408            metadata.as_ref(),
1409            origin.clone(),
1410            cx,
1411        ) else {
1412            return;
1413        };

1416

1418            .get_current_parser()
1419            .is_some_and(|parser| parser.aborted.get())
1420        {
1421            return;
1422        }

1425        let cx = &mut realm;

1430        if
1431        // navigationParams is null;
1432        // TODO
1433        // the result of should navigation response to navigation request of
1434        // type in target be blocked by Content Security Policy? given
1435        // navigationParams's request, navigationParams's response, navigationParams's policy container's CSP list,
1436        // cspNavigationType, and navigable is "Blocked";
1437        policy_container.csp_list.should_navigation_response_to_navigation_request_be_blocked(

1439            window,
1440            self.url.clone().into_url(),
1441            &origin.immutable().clone().into_url_origin(),
1442        )

1444        // checking a navigation response's adherence to its embedder policy given navigationParams's response,
1445        // navigable, and navigationParams's policy container's embedder policy is false; or

1448        // given navigationParams's response, navigable, navigationParams's policy container's CSP list,
1449        // and navigationParams's origin is false,
1450        || !check_a_navigation_response_adherence_to_x_frame_options(

1454            metadata
1455                .as_ref()

1467            // TODO

1556    pub(crate) fn process_response_chunk(&mut self, cx: &mut JSContext, payload: Bytes) {
1557        if self.is_synthesized_document {
1558            return;
1559        }
1560        let Some(parser) = self.parser.as_ref().map(|p| p.root()) else {
1561            return;
1562        };
1563        let Some(document) = self.document.as_ref().map(|document| document.root()) else {
1564            return;
1565        };
1566        if parser.aborted.get() {
1567            return;
1568        }
1569        if !self.has_loaded_document {
1570            // https://mimesniff.spec.whatwg.org/#read-the-resource-header
1571            self.navigation_params
1572                .resource_header
1573                .extend_from_slice(payload.as_ref());
1574            // the number of bytes in buffer is greater than or equal to 1445.
1575            if self.navigation_params.resource_header.len() >= 1445 {
1576                self.load_document(cx, Some(&parser), &document);
1577            }
1578        } else {
1579            parser.parse_bytes_chunk(cx, payload.as_ref());
1580        }
1581    }

1584    // submit_resource_timing in this function
1585    // Resource listeners are called via net_traits::Action::process, which handles submission for them
1586    pub(crate) fn process_response_eof(
1587        mut self,
1588        cx: &mut JSContext,
1589        status: Result<(), NetworkError>,
1590        timing: ResourceFetchTiming,
1591    ) {
1592        let parser = self.parser.as_ref().map(|parser| parser.root());
1593        if parser.as_ref().is_some_and(|parser| parser.aborted.get()) ||
1594            self.is_synthesized_document
1595        {
1596            return;
1597        }
1598
1599        if let Err(error) = &status {
1600            // TODO(Savago): we should send a notification to callers #5463.
1601            debug!("Failed to load page URL {}, error: {error:?}", self.url);
1602        }
1603
1604        let Some(document) = self.document.as_ref().map(|document| document.root()) else {
1605            return;
1606        };
1607
1608        // https://mimesniff.spec.whatwg.org/#read-the-resource-header
1609        //
1610        // the end of the resource is reached.
1611        if !self.has_loaded_document {
1612            self.load_document(cx, parser.as_deref(), &document);
1613        }
1614
1615        let mut realm = enter_auto_realm(cx, &*document);
1616        let cx = &mut realm;
1617
1618        if status.is_ok() {
1619            document.set_resource_fetch_timing(timing);
1620        }
1621
1622        if let Some(parser) = parser {
1623            parser.last_chunk_received.set(true);
1624            if !parser.suspended.get() {
1625                parser.parse_sync(cx);
1626            }
1627        }
1628
1629        // TODO: Only update if this is the current document resource.
1630        if let Some(pushed_index) = self.pushed_entry_index {
1631            let performance_entry =
1632                PerformanceNavigationTiming::new(cx, &document.global(), &document);
1633            document
1634                .global()
1635                .performance(cx)
1636                .update_entry(pushed_index, performance_entry.upcast::<PerformanceEntry>());
1637        }
1638
1639        if document.is_initial_about_blank() {
1640            self.finish_synchronous_load_for_initial_about_blank(cx, &document);
1641        }
1642    }
1643}

1651/// <https://html.spec.whatwg.org/multipage/#insert-an-element-at-the-adjusted-insertion-location>
1652#[cfg_attr(crown, expect(crown::unrooted_must_root))]
1653fn insert_an_element_at_the_adjusted_insertion_location(
1654    cx: &mut JSContext,
1655    node_to_insert: Dom<Node>,
1656    adjusted_insertion_location_parent: &Node,
1657    adjusted_insertion_location_child: Option<&Node>,
1658    parsing_algorithm: ParsingAlgorithm,
1659    custom_element_reaction_stack: &CustomElementReactionStack,
1660) {
1661    // Step 1: Let the adjusted insertion location be the appropriate place for inserting a node.
1662    //
1663    // Note: This is handled as part of the input.
1664
1665    // Step 2: If it is not possible to insert element at the adjusted insertion location,
1666    // abort these steps.
1667    if Node::ensure_pre_insertion_validity(
1668        cx.no_gc(),
1669        &node_to_insert,
1670        adjusted_insertion_location_parent,
1671        adjusted_insertion_location_child,
1672    )
1673    .is_err()
1674    {
1675        return;
1676    }
1677
1678    // Step 3. If the parser was not created as part of the HTML fragment parsing algorithm,
1679    // then push a new element queue onto element's relevant agent's custom element reactions
1680    // stack.
1681    let element_in_non_fragment =
1682        parsing_algorithm != ParsingAlgorithm::Fragment && node_to_insert.is::<Element>();
1683    if element_in_non_fragment {
1684        custom_element_reaction_stack.push_new_element_queue();
1685    }
1686
1687    // Step 4: Insert element at the adjusted insertion location.
1688    Node::insert(
1689        cx,
1690        &node_to_insert,
1691        adjusted_insertion_location_parent,
1692        adjusted_insertion_location_child,
1693        SuppressObserver::Unsuppressed,
1694    );
1695
1696    // Step 5: If the parser was not created as part of the HTML fragment parsing algorithm,
1697    // then pop the element queue from element's relevant agent's custom element reactions
1698    // stack, and invoke custom element reactions in that queue.
1699    //
1700    // Note: Handled as part of `pop_current_element_queue()`.
1701    if element_in_non_fragment {
1702        custom_element_reaction_stack.pop_current_element_queue(cx);
1703    }
1704}

1706#[cfg_attr(crown, expect(crown::unrooted_must_root))]
1707fn insert(
1708    cx: &mut JSContext,
1709    parent: &Node,
1710    reference_child: Option<&Node>,
1711    child: NodeOrText<Dom<Node>>,
1712    parsing_algorithm: ParsingAlgorithm,
1713    custom_element_reaction_stack: &CustomElementReactionStack,
1714) {
1715    match child {
1716        NodeOrText::AppendNode(node) => {
1717            // This encompasses two parts of the specification:
1718            //  - https://html.spec.whatwg.org/multipage/#insert-a-foreign-element
1719            //  - https://html.spec.whatwg.org/multipage/#insert-a-comment
1720            //
1721            // TODO: This part of the code should match the specification more closely.
1722            insert_an_element_at_the_adjusted_insertion_location(
1723                cx,
1724                node,
1725                parent,
1726                reference_child,
1727                parsing_algorithm,
1728                custom_element_reaction_stack,
1729            );
1730        },
1731        NodeOrText::AppendText(t) => {
1732            // https://html.spec.whatwg.org/multipage/#insert-a-character
1733            let text = reference_child
1734                .and_then(Node::GetPreviousSibling)
1735                .or_else(|| parent.GetLastChild())
1736                .and_then(DomRoot::downcast::<Text>);
1737
1738            if let Some(text) = text {
1739                text.upcast::<CharacterData>().append_data(cx, &t);
1740            } else {
1741                let text = Text::new(cx, String::from(t).into(), &parent.owner_doc());
1742                parent
1743                    .InsertBefore(cx, text.upcast(), reference_child)
1744                    .unwrap();
1745            }
1746        },
1747    }
1748}

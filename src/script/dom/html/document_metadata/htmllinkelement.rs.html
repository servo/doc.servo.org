174    fn remove_stylesheet(&self, no_gc: &NoGC) {
175        if let Some(stylesheet) = self.stylesheet.borrow_mut().take() {
176            let owner = self.stylesheet_list_owner();
177            owner.remove_stylesheet(
178                StylesheetSource::Element(Dom::from_ref(self.upcast())),

181            self.clean_stylesheet_ownership();
182            owner.invalidate_stylesheets(no_gc);
183        }

251    fn attribute_mutated(
252        &self,
253        cx: &mut js::context::JSContext,
254        attr: AttrRef<'_>,
255        mutation: AttributeMutation,
256    ) {
257        self.super_type()
258            .unwrap()
259            .attribute_mutated(cx, attr, mutation);
260
261        let local_name = attr.local_name();
262        let is_removal = mutation.is_removal();
263        match *local_name {
264            local_name!("disabled") => {
265                self.handle_disabled_attribute_change(cx.no_gc(), is_removal);
266                return;
267            },
268            local_name!("rel") | local_name!("rev") => {
269                let previous_relations = self.relations.get();
270                self.relations
271                    .set(LinkRelations::for_element(self.upcast()));
272
273                // If relations haven't changed, we shouldn't do anything
274                if previous_relations == self.relations.get() {
275                    return;
276                }
277            },
278            _ => {},
279        }
280
281        let node = self.upcast::<Node>();
282        if !node.is_connected() {
283            return;
284        }
285
286        // For stylesheets, we should only refetch when the actual attribute value
287        // has been changed.
288        if self.relations.get().contains(LinkRelations::STYLESHEET) &&
289            let AttributeMutation::Set(Some(previous_value), _) = mutation &&
290            **previous_value == **attr.value()
291        {
292            return;
293        }
294
295        match *local_name {
296            local_name!("rel") | local_name!("rev") => {
297                // https://html.spec.whatwg.org/multipage/#link-type-stylesheet:fetch-and-process-the-linked-resource
298                // > When the external resource link is created on a link element that is already browsing-context connected.
299                if self.relations.get().contains(LinkRelations::STYLESHEET) {
300                    self.handle_stylesheet_url(cx);
301                } else {
302                    self.remove_stylesheet(cx.no_gc());
303                }
304
305                if self.relations.get().contains(LinkRelations::MODULE_PRELOAD) {
306                    self.fetch_and_process_modulepreload(cx);
307                }
308            },
309            local_name!("href") => {
310                // https://html.spec.whatwg.org/multipage/#attr-link-href
311                // > If both the href and imagesrcset attributes are absent, then the element does not define a link.
312                if is_removal {
313                    if self.relations.get().contains(LinkRelations::STYLESHEET) {
314                        self.remove_stylesheet(cx.no_gc());
315                    }
316                    return;
317                }
318                // https://html.spec.whatwg.org/multipage/#link-type-stylesheet:fetch-and-process-the-linked-resource
319                // > When the href attribute of the link element of an external resource link
320                // > that is already browsing-context connected is changed.
321                if self.relations.get().contains(LinkRelations::STYLESHEET) {
322                    self.handle_stylesheet_url(cx);
323                }
324
325                if self.relations.get().contains(LinkRelations::ICON) {
326                    self.handle_favicon_url(&attr.value());
327                }
328
329                // https://html.spec.whatwg.org/multipage/#link-type-prefetch
330                // When the href attribute of the link element of an external resource link
331                // that is already browsing-context connected is changed.
332                if self.relations.get().contains(LinkRelations::PREFETCH) {
333                    self.fetch_and_process_prefetch_link(&attr.value());
334                }
335
336                // https://html.spec.whatwg.org/multipage/#link-type-preload
337                // When the href attribute of the link element of an external resource link
338                // that is already browsing-context connected is changed.
339                if self.relations.get().contains(LinkRelations::PRELOAD) {
340                    self.handle_preload_url();
341                }
342
343                // https://html.spec.whatwg.org/multipage/#link-type-modulepreload
344                if self.relations.get().contains(LinkRelations::MODULE_PRELOAD) {
345                    self.fetch_and_process_modulepreload(cx);
346                }
347            },
348            local_name!("imagesrcset") => {
349                self.source_set
350                    .borrow_mut()
351                    .update_source_set(self.upcast::<Element>());
352            },
353            local_name!("imagesizes") => {
354                if self
355                    .upcast::<Element>()
356                    .has_attribute(&local_name!("imagesrcset"))
357                {
358                    self.source_set
359                        .borrow_mut()
360                        .update_source_set(self.upcast::<Element>());
361                }
362            },
363            local_name!("sizes") if self.relations.get().contains(LinkRelations::ICON) => {
364                self.handle_favicon_url(&attr.value());
365            },
366            local_name!("crossorigin") => {
367                // https://html.spec.whatwg.org/multipage/#link-type-prefetch
368                // When the crossorigin attribute of the link element of an external resource link
369                // that is already browsing-context connected is set, changed, or removed.
370                if self.relations.get().contains(LinkRelations::PREFETCH) {
371                    self.fetch_and_process_prefetch_link(&attr.value());
372                }
373
374                // https://html.spec.whatwg.org/multipage/#link-type-stylesheet
375                // When the crossorigin attribute of the link element of an external resource link
376                // that is already browsing-context connected is set, changed, or removed.
377                if self.relations.get().contains(LinkRelations::STYLESHEET) {
378                    self.handle_stylesheet_url(cx);
379                }
380            },
381            local_name!("as") => {
382                // https://html.spec.whatwg.org/multipage/#link-type-preload
383                // When the as attribute of the link element of an external resource link
384                // that is already browsing-context connected is changed.
385                if self.relations.get().contains(LinkRelations::PRELOAD) &&
386                    let AttributeMutation::Set(Some(_), _) = mutation
387                {
388                    self.handle_preload_url();
389                }
390            },
391            local_name!("type") => {
392                // https://html.spec.whatwg.org/multipage/#link-type-stylesheet:fetch-and-process-the-linked-resource
393                // When the type attribute of the link element of an external resource link that
394                // is already browsing-context connected is set or changed to a value that does
395                // not or no longer matches the Content-Type metadata of the previous obtained
396                // external resource, if any.
397                //
398                // TODO: Match Content-Type metadata to check if it needs to be updated
399                if self.relations.get().contains(LinkRelations::STYLESHEET) {
400                    self.handle_stylesheet_url(cx);
401                }
402
403                // https://html.spec.whatwg.org/multipage/#link-type-preload
404                // When the type attribute of the link element of an external resource link that
405                // is already browsing-context connected, but was previously not obtained due to
406                // the type attribute specifying an unsupported type for the request destination,
407                // is set, removed, or changed.
408                if self.relations.get().contains(LinkRelations::PRELOAD) &&
409                    !self.previous_type_matched.get()
410                {
411                    self.handle_preload_url();
412                }
413            },
414            local_name!("media") => {
415                // https://html.spec.whatwg.org/multipage/#link-type-preload
416                // When the media attribute of the link element of an external resource link that
417                // is already browsing-context connected, but was previously not obtained due to
418                // the media attribute not matching the environment, is changed or removed.
419                if self.relations.get().contains(LinkRelations::PRELOAD) &&
420                    !self.previous_media_environment_matched.get()
421                {
422                    match mutation {
423                        AttributeMutation::Removed | AttributeMutation::Set(Some(_), _) => {
424                            self.handle_preload_url()
425                        },
426                        _ => {},
427                    };
428                } else if self.relations.get().contains(LinkRelations::STYLESHEET) &&
429                    let Some(ref stylesheet) = *self.stylesheet.borrow_mut()
430                {
431                    let document = self.owner_document();
432                    let shared_lock = document.style_shared_author_lock().clone();
433                    let mut guard = shared_lock.write();
434                    let media = stylesheet.media.write_with(&mut guard);
435                    match mutation {
436                        AttributeMutation::Set(..) => {
437                            *media = MediaList::parse_media_list(&attr.value(), document.window())
438                        },
439                        AttributeMutation::Removed => *media = StyleMediaList::empty(),
440                    };
441                    self.owner_document().invalidate_stylesheets(cx.no_gc());
442                }
443
444                let matches_media_environment =
445                    MediaList::matches_environment(&self.owner_document(), &attr.value());
446                self.previous_media_environment_matched
447                    .set(matches_media_environment);
448            },
449            _ => {},
450        }
451    }

463    fn bind_to_tree(&self, cx: &mut JSContext, context: &BindContext) {
464        if let Some(s) = self.super_type() {
465            s.bind_to_tree(cx, context);
466        }
467
468        let element = self.upcast::<Element>();
469        let href = element.get_attribute_string_value(&local_name!("href"));
470
471        if context.tree_connected &&
472            (href.as_ref().is_some_and(|x| !x.is_empty()) ||
473                element.has_attribute(&local_name!("imagesrcset")))
474        {
475            let relations = self.relations.get();
476            // https://html.spec.whatwg.org/multipage/#link-type-stylesheet:fetch-and-process-the-linked-resource
477            // > When the external resource link's link element becomes browsing-context connected.
478            if let Some(href) = href {
479                if relations.contains(LinkRelations::STYLESHEET) {
480                    self.handle_stylesheet_url(cx);
481                }
482
483                if relations.contains(LinkRelations::ICON) {
484                    self.handle_favicon_url(&href);
485                }
486
487                if relations.contains(LinkRelations::PREFETCH) {
488                    self.fetch_and_process_prefetch_link(&href);
489                }
490
491                // https://html.spec.whatwg.org/multipage/#link-type-modulepreload
492                if relations.contains(LinkRelations::MODULE_PRELOAD) {
493                    let link = DomRoot::from_ref(self);
494                    self.owner_document().add_delayed_task(
495                        task!(FetchModulePreload: |cx, link: DomRoot<HTMLLinkElement>| {
496                            link.fetch_and_process_modulepreload(cx);
497                        }),
498                    );
499                }
500            }
501
502            if relations.contains(LinkRelations::PRELOAD) {
503                self.handle_preload_url();
504            }
505        }
506    }
507
508    fn unbind_from_tree(&self, cx: &mut js::context::JSContext, context: &UnbindContext) {
509        if let Some(s) = self.super_type() {
510            s.unbind_from_tree(cx, context);
511        }
512
513        self.remove_stylesheet(cx.no_gc());
514    }

518    fn compute_destination_for_attribute(&self) -> Option<Destination> {
519        // Let destination be the result of translating the keyword
520        // representing the state of el's as attribute.
521        let element = self.upcast::<Element>();
522        element
523            .get_attribute_string_value(&local_name!("as"))
524            .and_then(|attr| LinkProcessingOptions::translate_a_preload_destination(&attr))
525    }

527    /// <https://html.spec.whatwg.org/multipage/#create-link-options-from-element>
528    fn processing_options(&self) -> LinkProcessingOptions {
529        let element = self.upcast::<Element>();
530
531        // Step 1. Let document be el's node document.
532        let document = self.upcast::<Node>().owner_doc();
533        let global = document.owner_global();
534
535        // Step 2. Let options be a new link processing options
536        let mut options = LinkProcessingOptions {
537            href: String::new(),
538            destination: Destination::None,
539            integrity: String::new(),
540            link_type: String::new(),
541            cryptographic_nonce_metadata: self.upcast::<Element>().nonce_value(),
542            cross_origin: cors_setting_for_element(element),
543            referrer_policy: referrer_policy_for_element(element),
544            policy_container: document.policy_container().to_owned(),
545            source_set: Some(self.source_set.borrow().clone()),
546            origin: document.borrow().origin().immutable().to_owned(),
547            base_url: document.borrow().base_url(),
548            request_client: global.request_client(None),
549            referrer: global.get_referrer(),
550        };
551
552        // Step 3. If el has an href attribute, then set options's href to the value of el's href attribute.
553        if let Some(href_attribute) = element.get_attribute_string_value(&local_name!("href")) {
554            options.href = href_attribute;
555        }
556
557        // Step 4. If el has an integrity attribute, then set options's integrity
558        //         to the value of el's integrity content attribute.
559        if let Some(integrity_attribute) =
560            element.get_attribute_string_value(&local_name!("integrity"))
561        {
562            options.integrity = integrity_attribute;
563        }
564
565        // Step 5. If el has a type attribute, then set options's type to the value of el's type attribute.
566        if let Some(type_attribute) = element.get_attribute_string_value(&local_name!("type")) {
567            options.link_type = type_attribute;
568        }
569
570        // Step 6. Assert: options's href is not the empty string, or options's source set is not null.
571        assert!(!options.href.is_empty() || options.source_set.is_some());
572
573        // Step 7. Return options.
574        options
575    }
576
577    /// <https://html.spec.whatwg.org/multipage/#default-fetch-and-process-the-linked-resource>
578    ///
579    /// This method does not implement Step 7 (fetching the request) and instead returns the [RequestBuilder],
580    /// as the fetch context that should be used depends on the link type.
581    fn default_fetch_and_process_the_linked_resource(&self) -> Option<RequestBuilder> {
582        // Step 1. Let options be the result of creating link options from el.
583        let options = self.processing_options();
584
585        // Step 2. Let request be the result of creating a link request given options.
586        let Some(request) = options.create_link_request(self.owner_window().webview_id()) else {
587            // Step 3. If request is null, then return.
588            return None;
589        };
590        // Step 4. Set request's synchronous flag.
591        let mut request = request.synchronous(true);
592
593        // Step 5. Run the linked resource fetch setup steps, given el and request. If the result is false, then return.
594        if !self.linked_resource_fetch_setup(&mut request) {
595            return None;
596        }
597
598        // TODO Step 6. Set request's initiator type to "css" if el's rel attribute
599        // contains the keyword stylesheet; "link" otherwise.
600
601        // Step 7. Fetch request with processResponseConsumeBody set to the following steps given response response and null,

646    /// The `fetch and process the linked resource` algorithm for [`rel="prefetch"`](https://html.spec.whatwg.org/multipage/#link-type-prefetch)
647    fn fetch_and_process_prefetch_link(&self, href: &str) {
648        // Step 1. If el's href attribute's value is the empty string, then return.
649        if href.is_empty() {
650            return;
651        }
652
653        // Step 2. Let options be the result of creating link options from el.
654        let mut options = self.processing_options();
655
656        // Step 3. Set options's destination to the empty string.
657        options.destination = Destination::None;
658
659        // Step 4. Let request be the result of creating a link request given options.
660        let Some(request) = options.create_link_request(self.owner_window().webview_id()) else {
661            // Step 5. If request is null, then return.
662            return;
663        };
664        let url = request.url.url();
665
666        // Step 6. Set request's initiator to "prefetch".
667        let request = request.initiator(Initiator::Prefetch);
668
669        // (Step 7, firing load/error events is handled in the FetchResponseListener impl for LinkFetchContext)
670
671        // Step 8. The user agent should fetch request, with processResponseConsumeBody set to processPrefetchResponse.
672        let document = self.upcast::<Node>().owner_doc();
673        let fetch_context = LinkFetchContext {
674            url,
675            link: Some(Trusted::new(self)),
676            global: Trusted::new(&document.global()),
677            type_: LinkFetchContextType::Prefetch,
678            response_body: BytesMut::new(),
679        };
680
681        document.fetch_background(request, fetch_context);
682    }

684    /// <https://html.spec.whatwg.org/multipage/#concept-link-obtain>
685    fn handle_stylesheet_url(&self, cx: &mut js::context::JSContext) {
686        let document = self.owner_document();
687        if document.browsing_context().is_none() {
688            return;
689        }
690
691        let element = self.upcast::<Element>();
692
693        // https://html.spec.whatwg.org/multipage/#processing-the-type-attribute
694        // > If the UA does not support the given MIME type for the given link relationship,
695        // > then the UA should not fetch and process the linked resource
696        //
697        // https://html.spec.whatwg.org/multipage/#link-type-stylesheet
698        // > The default type for resources given by the stylesheet keyword is text/css.
699        let type_ = element.get_string_attribute(&local_name!("type"));
700        if !type_.is_empty() && type_ != "text/css" {
701            return;
702        }
703
704        // Step 1.
705        let href = element.get_string_attribute(&local_name!("href"));
706        if href.is_empty() {
707            return;
708        }
709
710        // Step 2.
711        let link_url = match document.base_url().join(&href.str()) {
712            Ok(url) => url,
713            Err(e) => {
714                debug!("Parsing url {} failed: {}", href, e);
715                return;
716            },
717        };
718
719        // Step 3
720        let cors_setting = cors_setting_for_element(element);
721
722        let mq_str = element
723            .get_attribute_string_value(&local_name!("media"))
724            .unwrap_or_default();
725        let media = MediaList::parse_media_list(&mq_str, document.window());
726        let media = Arc::new(document.style_shared_author_lock().wrap(media));
727
728        let integrity_metadata = element
729            .get_attribute_string_value(&local_name!("integrity"))
730            .unwrap_or_default();
731
732        self.request_generation_id
733            .set(self.request_generation_id.get().increment());
734        self.pending_loads.set(0);
735
736        ElementStylesheetLoader::load_with_element(
737            cx,
738            self.upcast(),
739            StylesheetContextSource::LinkElement,
740            media,
741            link_url,
742            cors_setting,
743            integrity_metadata,
744        );
745    }

760    fn handle_favicon_url(&self, href: &str) {
761        // If el's href attribute's value is the empty string, then return.
762        if href.is_empty() {
763            return;
764        }
765
766        // The spec does not specify this, but we don't fetch favicons for iframes, as
767        // they won't be displayed anyways.
768        let window = self.owner_window();
769        if !window.is_top_level() {
770            return;
771        }
772        let Ok(href) = self.Href().parse() else {
773            return;
774        };
775
776        // Ignore all previous fetch operations
777        self.request_generation_id
778            .set(self.request_generation_id.get().increment());
779
780        let cache_result = window.image_cache().get_cached_image_status(
781            href,
782            window.origin().immutable().clone(),
783            cors_setting_for_element(self.upcast()),
784        );
785
786        match cache_result {
787            ImageCacheResult::Available(ImageOrMetadataAvailable::ImageAvailable {
788                image, ..
789            }) => {
790                self.process_favicon_response(image);
791            },
792            ImageCacheResult::Available(ImageOrMetadataAvailable::MetadataAvailable(_, id)) |
793            ImageCacheResult::Pending(id) => {
794                let sender = self.register_image_cache_callback(id);
795                window.image_cache().add_listener(ImageLoadListener::new(
796                    sender,
797                    window.pipeline_id(),
798                    id,
799                ));
800            },
801            ImageCacheResult::ReadyForRequest(id) => {
802                let Some(request) = self.default_fetch_and_process_the_linked_resource() else {
803                    return;
804                };
805
806                let sender = self.register_image_cache_callback(id);
807                window.image_cache().add_listener(ImageLoadListener::new(
808                    sender,
809                    window.pipeline_id(),
810                    id,
811                ));
812
813                let document = self.upcast::<Node>().owner_doc();
814                let fetch_context = FaviconFetchContext {
815                    url: self.owner_document().base_url(),
816                    image_cache: window.image_cache(),
817                    id,
818                    link: Trusted::new(self),
819                };
820                document.fetch_background(request, fetch_context);
821            },
822            ImageCacheResult::FailedToLoadOrDecode => {},
823        };
824    }

826    fn register_image_cache_callback(&self, id: PendingImageId) -> ImageCacheResponseCallback {
827        let trusted_node = Trusted::new(self);
828        let window = self.owner_window();
829        let request_generation_id = self.get_request_generation_id();
830        window.register_image_cache_listener(id, move |response, _| {
831            let trusted_node = trusted_node.clone();
832            let link_element = trusted_node.root();
833            let window = link_element.owner_window();
834
835            let ImageResponse::Loaded(image, _) = response.response else {
836                // We don't care about metadata and such for favicons.
837                return;
838            };
839
840            if request_generation_id != link_element.get_request_generation_id() {
841                // This load is no longer relevant.
842                return;
843            };
844
845            window
846                .as_global_scope()
847                .task_manager()
848                .networking_task_source()
849                .queue(task!(process_favicon_response: move || {
850                    let element = trusted_node.root();
851
852                    if request_generation_id != element.get_request_generation_id() {
853                        // This load is no longer relevant.
854                        return;
855                    };
856
857                    element.process_favicon_response(image);
858                }));
859        })
860    }

862    /// Rasterizes a loaded favicon file if necessary and notifies the embedder about it.
863    fn process_favicon_response(&self, image: Image) {
864        // TODO: Include the size attribute here
865        let window = self.owner_window();
866        let document = self.owner_document();
867
868        let send_rasterized_favicon_to_embedder = |raster_image: &pixels::RasterImage| {
869            // Let's not worry about animated favicons...
870            let frame = raster_image.first_frame();
871
872            let format = match raster_image.format {
873                PixelFormat::K8 => embedder_traits::PixelFormat::K8,
874                PixelFormat::KA8 => embedder_traits::PixelFormat::KA8,
875                PixelFormat::RGB8 => embedder_traits::PixelFormat::RGB8,
876                PixelFormat::RGBA8 => embedder_traits::PixelFormat::RGBA8,
877                PixelFormat::BGRA8 => embedder_traits::PixelFormat::BGRA8,
878            };
879
880            let embedder_image = embedder_traits::Image::new(
881                frame.width,
882                frame.height,
883                std::sync::Arc::new(GenericSharedMemory::from_arc_vec(
884                    raster_image.bytes.clone(),
885                )),
886                raster_image.frames[0].byte_range.clone(),
887                format,
888            );
889            document.set_favicon(embedder_image);
890        };
891
892        match image {
893            Image::Raster(raster_image) => send_rasterized_favicon_to_embedder(&raster_image),
894            Image::Vector(vector_image) => {
895                // This size is completely arbitrary.
896                let size = DeviceIntSize::new(250, 250);
897
898                let image_cache = window.image_cache();
899                if let Some(raster_image) =
900                    image_cache.rasterize_vector_image(vector_image.id, size, None)
901                {
902                    send_rasterized_favicon_to_embedder(&raster_image);
903                } else {
904                    // The rasterization callback will end up calling "process_favicon_response" again,
905                    // but this time with a raster image.
906                    let image_cache_sender = self.register_image_cache_callback(vector_image.id);
907                    image_cache.add_rasterization_complete_listener(
908                        window.pipeline_id(),
909                        vector_image.id,
910                        size,
911                        image_cache_sender,
912                    );
913                }
914            },
915        }
916    }

918    /// <https://html.spec.whatwg.org/multipage/#link-type-preload:fetch-and-process-the-linked-resource-2>
919    /// and type matching destination steps of <https://html.spec.whatwg.org/multipage/#preload>
920    fn handle_preload_url(&self) {
921        // Step 1. Update the source set for el.
922        self.source_set
923            .borrow_mut()
924            .update_source_set(self.upcast::<Element>());
925        // Step 2. Let options be the result of creating link options from el.
926        let mut options = self.processing_options();
927        // Step 3. Let destination be the result of translating the keyword
928        // representing the state of el's as attribute.
929        let Some(destination) = self.compute_destination_for_attribute() else {
930            // Step 4. If destination is null, then return.
931            return;
932        };
933        // Step 5. Set options's destination to destination.
934        options.destination = destination;
935        // Steps for https://html.spec.whatwg.org/multipage/#preload
936        {
937            // Step 1. If options's type doesn't match options's destination, then return.
938            let type_matches_destination = options.type_matches_destination();
939            self.previous_type_matched.set(type_matches_destination);
940            if !type_matches_destination {
941                return;
942            }
943        }
944        // Step 6. Preload options, with the following steps given a response response:
945        let document = self.upcast::<Node>().owner_doc();
946        options.preload(
947            self.owner_window().webview_id(),
948            Some(Trusted::new(self)),
949            &document,
950        );
951    }

953    /// <https://html.spec.whatwg.org/multipage/#link-type-preload:fetch-and-process-the-linked-resource-2>
954    pub(crate) fn fire_event_after_response(
955        &self,
956        cx: &mut JSContext,
957        response: Result<(), NetworkError>,
958    ) {
959        // Step 3.1 If response is a network error, fire an event named error at el.
960        // Otherwise, fire an event named load at el.
961        if response.is_err() {
962            self.upcast::<EventTarget>().fire_event(cx, atom!("error"));
963        } else {
964            self.upcast::<EventTarget>().fire_event(cx, atom!("load"));
965        }
966    }

968    /// <https://html.spec.whatwg.org/multipage/#link-type-modulepreload:fetch-and-process-the-linked-resource-2>
969    fn fetch_and_process_modulepreload(&self, cx: &mut JSContext) {
970        let el = self.upcast::<Element>();
971        let href_attribute_value = el.get_string_attribute(&local_name!("href"));
972
973        // Step 1. If el's href attribute's value is the empty string, then return.
974        if href_attribute_value.is_empty() {
975            return;
976        }
977
978        // Step 2. Let destination be the current state of el's as attribute (a destination), or "script" if it is in no state.
979        let destination = el
980            .get_attribute_string_value(&local_name!("as"))
981            .map(|value| value.to_ascii_lowercase())
982            .and_then(|value| match value.as_str() {
983                // `Destination::from_str` will map an empty string to `Destination::None`
984                "" => None,
985                // "fetch" is a valid preload destination, but not for modules
986                "fetch" => Some(Destination::None),
987                _ => Destination::from_str(&value).ok(),
988            })
989            .unwrap_or(Destination::Script);
990
991        let document = self.owner_document();
992        let global = document.global();
993
994        // A module preload destination is "json", "style", or a script-like destination.
995        let is_a_modulepreload_destination = match destination {
996            Destination::Json | Destination::Style => true,
997            // https://fetch.spec.whatwg.org/#ref-for-request-destination-script-like
998            // While "xslt" can cause script execution, it is not relevant here.
999            Destination::Xslt => false,
1000            d => d.is_script_like(),
1001        };
1002
1003        // Step 3. If destination is not a module preload destination, then queue an element task on the
1004        // networking task source given el to fire an event named error at el, and return.
1005        if !is_a_modulepreload_destination {
1006            return global
1007                .task_manager()
1008                .networking_task_source()
1009                .queue_simple_event(self.upcast(), atom!("error"));
1010        }
1011
1012        // Step 4. Let url be the result of encoding-parsing a URL given el's href attribute's value, relative to el's node document.
1013        // Step 5. If url is failure, then return.
1014        let Ok(url) = document.encoding_parse_a_url(&href_attribute_value.str()) else {
1015            return;
1016        };
1017        let url = ensure_blob_referenced_by_url_is_kept_alive(&global, url);
1018
1019        // Step 6. Let settings object be el's node document's relevant settings object.
1020
1021        // Step 7. Let credentials mode be the CORS settings attribute credentials mode for el's crossorigin attribute.
1022        let credentials_mode = cors_settings_attribute_credential_mode(el);
1023
1024        // Step 8. Let cryptographic nonce be el.[[CryptographicNonce]].
1025        let cryptographic_nonce = el.nonce_value();
1026
1027        // Step 9. Let integrity metadata be the value of el's integrity attribute, if it is specified, or the empty string otherwise.
1028        // Step 10. If el does not have an integrity attribute, then set integrity metadata to
1029        // the result of resolving a module integrity metadata with url and settings object.
1030        let integrity_metadata = el
1031            .get_attribute_string_value(&local_name!("integrity"))
1032            .unwrap_or_else(|| {
1033                global
1034                    .import_map()
1035                    .resolve_a_module_integrity_metadata(&url.url())
1036            });
1037
1038        // Step 11. Let referrer policy be the current state of el's referrerpolicy attribute.
1039        let referrer_policy = referrer_policy_for_element(el);
1040
1041        // TODO Step 12. Let fetch priority be the current state of el's fetchpriority attribute.
1042
1043        // Step 13. Let options be a script fetch options whose cryptographic nonce is cryptographic nonce,
1044        // integrity metadata is integrity metadata, parser metadata is "not-parser-inserted",
1045        // credentials mode is credentials mode, referrer policy is referrer policy, and fetch priority is fetch priority.
1046        let options = ScriptFetchOptions {
1047            cryptographic_nonce,
1048            integrity_metadata,
1049            parser_metadata: ParserMetadata::NotParserInserted,
1050            credentials_mode,
1051            referrer_policy,
1052            render_blocking: false,
1053        };
1054
1055        let link = DomRoot::from_ref(self);
1056
1057        // Step 14. Fetch a modulepreload module script graph given url, destination, settings object, options,
1058        // and with the following steps given result:
1059        fetch_a_modulepreload_module(
1060            cx,
1061            url,
1062            destination,
1063            &global,
1064            options,
1065            move |cx, fetch_failed| {
1066                // Step 1. If result is null, then fire an event named error at el, and return.
1067                // Step 2. Fire an event named load at el.
1068                let event = match fetch_failed {
1069                    true => atom!("error"),
1070                    false => atom!("load"),
1071                };
1072
1073                link.upcast::<EventTarget>().fire_event(cx, event);
1074            },
1075        );
1076    }
1077}

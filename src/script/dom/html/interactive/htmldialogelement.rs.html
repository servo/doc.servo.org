87    /// <https://html.spec.whatwg.org/multipage/#show-a-modal-dialog>
88    pub(crate) fn show_a_modal(
89        &self,
90        cx: &mut js::context::JSContext,
91        source: Option<DomRoot<Element>>,
92    ) -> ErrorResult {
93        let subject = self.upcast::<Element>();
94        // Step 1. If subject has an open attribute and is modal of subject is true, then return.
95        if subject.has_attribute(&local_name!("open")) &&
96            subject.state().contains(ElementState::MODAL)
97        {
98            return Ok(());
99        }
100
101        // Step 2. If subject has an open attribute, then throw an "InvalidStateError" DOMException.
102        if subject.has_attribute(&local_name!("open")) {
103            return Err(Error::InvalidState(Some(
104                "Cannot call showModal() on an already open dialog.".into(),
105            )));
106        }
107
108        // Step 3. If subject's node document is not fully active, then throw an "InvalidStateError" DOMException.
109        if !subject.owner_document().is_fully_active() {
110            return Err(Error::InvalidState(Some(
111                "Cannot call showModal() on a dialog whose document is not fully active.".into(),
112            )));
113        }
114
115        // Step 4. If subject is not connected, then throw an "InvalidStateError" DOMException.
116        if !subject.is_connected() {
117            return Err(Error::InvalidState(Some(
118                "Cannot call showModal() on a dialog that is not connected.".into(),
119            )));
120        }
121
122        // TODO: Step 5. If subject is in the popover showing state, then throw an "InvalidStateError" DOMException.
123
124        // Step 6. If the result of firing an event named beforetoggle, using ToggleEvent, with the cancelable attribute initialized to true, the oldState attribute initialized to "closed", the newState attribute initialized to "open", and the source attribute initialized to source at subject is false, then return.
125        let event = ToggleEvent::new(
126            cx,
127            &self.owner_window(),
128            atom!("beforetoggle"),
129            EventBubbles::DoesNotBubble,
130            EventCancelable::Cancelable,
131            DOMString::from_static("closed"),
132            DOMString::from_static("open"),
133            source.as_deref(),
134        );
135        let event = event.upcast::<Event>();
136        if !event.fire(cx, self.upcast::<EventTarget>()) {
137            return Ok(());
138        }
139
140        // Step 7. If subject has an open attribute, then return.
141        if subject.has_attribute(&local_name!("open")) {
142            return Ok(());
143        }
144
145        // Step 8. If subject is not connected, then return.
146        if !subject.is_connected() {
147            return Ok(());
148        }
149
150        // TODO: Step 9. If subject is in the popover showing state, then return.
151
152        // Step 10. Queue a dialog toggle event task given subject, "closed", "open", and source.
153        self.queue_dialog_toggle_event_task("closed", "open", source);
154
155        // Step 11. Add an open attribute to subject, whose value is the empty string.
156        subject.set_bool_attribute(cx, &local_name!("open"), true);
157        subject.set_open_state(cx.no_gc(), true);
158
159        // TODO: Step 12. Assert: subject's close watcher is not null.
160
161        // Step 13. Set is modal of subject to true.
162        self.upcast::<Element>().set_modal_state(cx.no_gc(), true);
163
164        // TODO: Step 14. Set subject's node document to be blocked by the modal dialog subject.
165
166        // TODO: Step 15. If subject's node document's top layer does not already contain subject, then add an element to the top layer given subject.
167
168        // Step 16. Set subject's previously focused element to the focused element.
169        self.upcast::<HTMLElement>().set_previously_focused_element(
170            self.owner_document()
171                .focus_handler()
172                .focused_area()
173                .element(),
174            cx.no_gc(),
175        );
176
177        // TODO: Step 17. Let document be subject's node document.
178
179        // TODO: Step 18. Let hideUntil be the result of running topmost popover ancestor given subject, document's showing hint popover list, null, and false.
180
181        // TODO: Step 19. If hideUntil is null, then set hideUntil to the result of running topmost popover ancestor given subject, document's showing auto popover list, null, and false.
182
183        // TODO: Step 20. If hideUntil is null, then set hideUntil to document.
184
185        // TODO: Step 21. Run hide all popovers until given hideUntil, false, and true.
186
187        // Step 22. Run the dialog focusing steps given subject.
188        self.run_dialog_focusing_steps(cx);
189        Ok(())
190    }

192    /// <https://html.spec.whatwg.org/multipage/#set-the-dialog-close-watcher>
193    pub(crate) fn set_the_close_watcher(&self, no_gc: &NoGC) {
194        // Step 1. Assert: dialog's close watcher is null.
195        assert!(self.close_watcher.borrow().is_none());
196
197        // Step 2. Assert: dialog has an open attribute and dialog's node document is fully active.
198        let dialog = self.upcast::<Element>();
199        assert!(
200            dialog.has_attribute(&local_name!("open")) && dialog.owner_document().is_fully_active()
201        );
202
203        // Step 3. Set dialog's close watcher to the result of establishing a close watcher given
204        // dialog's relevant global object, with:
205        // - cancelAction given canPreventClose being to return the result of firing an event named
206        //   cancel at dialog, with the cancelable attribute initialized to canPreventClose.
207        // - closeAction being to close the dialog given dialog, dialog's request close return
208        //   value, and dialog's request close source element.
209        // - getEnabledState being to return true if dialog's enable close watcher for request close
210        //   is true or dialog's computed closed-by state is not None; otherwise false.
211        let close_watcher = InternalCloseWatcher::establish(
212            no_gc,
213            dialog.owner_document().window(),
214            &InternalCloseWatcherOwner::Dialog(Dom::from_ref(self)),
215        );
216        self.close_watcher
217            .safe_borrow_mut(no_gc)
218            .replace(close_watcher);
219    }

221    /// <https://html.spec.whatwg.org/multipage/#close-the-dialog>
222    pub(crate) fn close_the_dialog(
223        &self,
224        cx: &mut js::context::JSContext,
225        result: Option<DOMString>,
226        source: Option<DomRoot<Element>>,
227    ) {
228        let subject = self.upcast::<Element>();
229        // Step 1. If subject does not have an open attribute, then return.
230        if !subject.has_attribute(&local_name!("open")) {
231            return;
232        }
233
234        // Step 2. Fire an event named beforetoggle, using ToggleEvent, with the oldState attribute initialized to "open", the newState attribute initialized to "closed", and the source attribute initialized to source at subject.
235        let event = ToggleEvent::new(
236            cx,
237            &self.owner_window(),
238            atom!("beforetoggle"),
239            EventBubbles::DoesNotBubble,
240            EventCancelable::NotCancelable,
241            DOMString::from_static("open"),
242            DOMString::from_static("closed"),
243            source.as_deref(),
244        );
245        let event = event.upcast::<Event>();
246        event.fire(cx, self.upcast::<EventTarget>());
247
248        // Step 3. If subject does not have an open attribute, then return.
249        if !subject.has_attribute(&local_name!("open")) {
250            return;
251        }
252
253        // Step 4. Queue a dialog toggle event task given subject, "open", "closed", and source.
254        self.queue_dialog_toggle_event_task("open", "closed", source);
255
256        // Step 5. Remove subject's open attribute.
257        subject.remove_attribute(cx, &ns!(), &local_name!("open"));
258        subject.set_open_state(cx.no_gc(), false);
259
260        // TODO: Step 6. If is modal of subject is true, then request an element to be removed from the top layer given subject.
261
262        // Step 7. Let wasModal be the value of subject's is modal flag.
263        let was_modal = subject.state().contains(ElementState::MODAL);
264
265        // Step 8. Set is modal of subject to false.
266        self.upcast::<Element>().set_modal_state(cx.no_gc(), false);
267
268        // Step 9. If result is not null, then set subject's returnValue attribute to result.
269        if let Some(new_value) = result {
270            *self.return_value.safe_borrow_mut(cx.no_gc()) = new_value.traced();
271        }
272
273        // Step 10. Set subject's request close return value to null.
274        *self.request_close_return_value.safe_borrow_mut(cx) = None;
275
276        // Step 11. Set subject's request close source element to null.
277        self.request_close_source_element.set(None);
278
279        // Step 12. If subject's previously focused element is not null, then:
280        if let Some(element) = self
281            .upcast::<HTMLElement>()
282            .previously_focused_element(cx.no_gc())
283        {
284            // Step 12.1. Let element be subject's previously focused element.
285            // Step 12.2. Set subject's previously focused element to null.
286            self.upcast::<HTMLElement>()
287                .set_previously_focused_element(None, cx.no_gc());
288
289            // Step 12.3. If subject's node document's focused area of the document's DOM anchor is
290            // a shadow-including inclusive descendant of subject, or wasModal is true, then run the
291            // focusing steps for element; the viewport should not be scrolled by doing this step.
292            let subject_node = subject.upcast::<Node>();
293            let document = subject.owner_document();
294            if document
295                .focus_handler()
296                .focused_area()
297                .dom_anchor(&document)
298                .traverse_preorder(ShadowIncluding::Yes)
299                .any(|node| &*node == subject_node) ||
300                was_modal
301            {
302                element
303                    .upcast::<Node>()
304                    .run_the_focusing_steps(cx, None, FocusTrigger::Script);
305            }
306        }
307
308        // Step 13. Queue an element task on the user interaction task source given the subject element to fire an event named close at subject.
309        let target = self.upcast::<EventTarget>();
310        self.owner_global()
311            .task_manager()
312            .user_interaction_task_source()
313            .queue_simple_event(target, atom!("close"));
314    }

316    /// <https://html.spec.whatwg.org/multipage/#dialog-request-close>
317    pub(crate) fn request_to_close(
318        &self,
319        cx: &mut JSContext,
320        return_value: Option<DOMString>,
321        source: Option<DomRoot<Element>>,
322    ) {
323        let subject = self.upcast::<Element>();
324
325        // Step 1. If subject does not have an open attribute, then return.
326        if !subject.has_attribute(&local_name!("open")) {
327            return;
328        }
329
330        // Step 2. If subject is not connected or subject's node document is not fully active, then
331        // return.
332        if !subject.is_connected() || !subject.owner_document().is_fully_active() {
333            return;
334        }
335
336        // Step 3. Assert: subject's close watcher is not null.
337        //
338        // Clone the handle so the cell isn't borrowed while script runs: a `close` listener can
339        // call `dialog.close()`, which runs `dialog_cleanup_steps` and nulls the close watcher.
340        let close_watcher = self
341            .close_watcher
342            .borrow()
343            .clone()
344            .expect("Close watcher must always exist");
345
346        // Step 4. Set subject's enable close watcher for request close to true.
347        self.enable_close_watcher_for_request_close.set(true);
348
349        // Step 5. Set subject's request close return value to returnValue.
350        *self.request_close_return_value.safe_borrow_mut(cx) = return_value.map(DOMString::traced);
351
352        // Step 6. Set subject's request close source element to source.
353        self.request_close_source_element.set(source.as_deref());
354
355        // Step 7. Request to close subject's close watcher with false.
356        close_watcher.request_to_close(cx, false);
357
358        // Step 8. Set subject's enable close watcher for request close to false.
359        self.enable_close_watcher_for_request_close.set(false);
360    }

404    /// <https://html.spec.whatwg.org/multipage/#dialog-focusing-steps>
405    fn run_dialog_focusing_steps(&self, cx: &mut JSContext) {
406        // TODO: Step 1. If the allow focus steps given subject's node document return false, then return.
407
408        // Step 2. Let control be null.
409        rooted!(&in(cx) let mut control = None);
410
411        // Step 3. If subject has the autofocus attribute, then set control to subject.
412        if self.upcast::<HTMLElement>().Autofocus() {
413            control.set(
414                self.upcast::<Node>()
415                    .get_the_focusable_area(cx, FocusTrigger::Script),
416            );
417        }
418
419        // Step 4. If control is null, then set control to the focus delegate of subject.
420        if control.is_none() {
421            control.set(self.upcast::<Node>().focus_delegate(cx, None));
422        }
423
424        // Step 5. If control is null, then set control to subject.
425        if control.is_none() {
426            control.set(
427                self.upcast::<Node>()
428                    .get_the_focusable_area(cx, FocusTrigger::Script),
429            );
430        }
431
432        // Step 6. Run the focusing steps for control.
433        // FIXME: Use the focusing step once they support a focusable area as an argument
434        if control.is_some() {
435            let document = self.owner_document();
436            document
437                .focus_handler()
438                .focus(cx, &control.take().unwrap(), FocusTrigger::Script);
439        }
440
441        // TODO: Step 7. Let topDocument be control's node navigable's top-level traversable's active document.
442        // TODO: Step 8. If control's node document's origin is not the same as the origin of topDocument, then return.
443        // TODO: Step 9. Empty topDocument's autofocus candidates.
444        // TODO: Step 10. Set topDocument's autofocus processed flag to true.
445    }

478        let return_value = self
479            .request_close_return_value

487    pub(crate) fn close_watcher_enabled_state(&self) -> bool {
488        self.enable_close_watcher_for_request_close.get() ||
489            self.computed_closed_by_state() != ComputedClosedByState::None
490    }

492    /// <https://html.spec.whatwg.org/multipage/#computed-closed-by-state>
493    fn computed_closed_by_state(&self) -> ComputedClosedByState {
494        let element = self.upcast::<Element>();
495
496        // Step 2. Return the state of dialog's closedby attribute.
497        if let Some(closed_by) = element.get_attribute_string_ref(&local_name!("closedby")) {
498            if closed_by.eq_ignore_ascii_case("any") {
499                return ComputedClosedByState::Any;
500            }
501            if closed_by.eq_ignore_ascii_case("closerequest") {
502                return ComputedClosedByState::CloseRequest;
503            }
504            if closed_by.eq_ignore_ascii_case("none") {
505                return ComputedClosedByState::None;
506            }
507        }
508
509        // Step 1. If the state of dialog's closedby attribute is Auto:
510        // https://html.spec.whatwg.org/multipage/#attr-dialog-closedby-auto-state
511        // > The closedby attribute's invalid value default and missing value default are both the
512        // > Auto state.
513
514        // Step 1.1. If dialog's is modal is true, then return Close Request.
515        if element.state().contains(ElementState::MODAL) {
516            return ComputedClosedByState::CloseRequest;
517        }
518
519        // Step 1.2. Return None.
520        ComputedClosedByState::None
521    }

556    /// <https://html.spec.whatwg.org/multipage/#dom-dialog-show>
557    fn Show(&self, cx: &mut js::context::JSContext) -> ErrorResult {
558        let element = self.upcast::<Element>();
559        // Step 1. If this has an open attribute and is modal of this is false, then return.
560        if element.has_attribute(&local_name!("open")) &&
561            !element.state().contains(ElementState::MODAL)
562        {

564        }

566        // Step 2. If this has an open attribute, then throw an "InvalidStateError" DOMException.
567        if element.has_attribute(&local_name!("open")) {
568            return Err(Error::InvalidState(Some(
569                "Cannot call show() on an already open dialog.".into(),
570            )));
571        }
572
573        // Step 3. If the result of firing an event named beforetoggle, using ToggleEvent, with the cancelable attribute initialized to true, the oldState attribute initialized to "closed", and the newState attribute initialized to "open" at this is false, then return.
574        let event = ToggleEvent::new(
575            cx,
576            &self.owner_window(),
577            atom!("beforetoggle"),
578            EventBubbles::DoesNotBubble,
579            EventCancelable::Cancelable,
580            DOMString::from_static("closed"),
581            DOMString::from_static("open"),
582            None,
583        );
584        let event = event.upcast::<Event>();
585        if !event.fire(cx, self.upcast::<EventTarget>()) {
586            return Ok(());
587        }
588
589        // Step 4. If this has an open attribute, then return.
590        if element.has_attribute(&local_name!("open")) {
591            return Ok(());
592        }
593
594        // Step 5. Queue a dialog toggle event task given this, "closed", "open", and null.
595        self.queue_dialog_toggle_event_task("closed", "open", None);
596
597        // Step 6. Add an open attribute to this, whose value is the empty string.
598        element.set_bool_attribute(cx, &local_name!("open"), true);
599        element.set_open_state(cx.no_gc(), true);
600
601        // Step 7. Set this's previously focused element to the focused element.
602        self.upcast::<HTMLElement>().set_previously_focused_element(
603            self.owner_document()
604                .focus_handler()
605                .focused_area()
606                .element(),
607            cx.no_gc(),
608        );
609
610        // TODO: Step 8. Let document be this's node document.
611
612        // TODO: Step 9. Let hideUntil be the result of running topmost popover ancestor given this, document's showing hint popover list, null, and false.
613
614        // TODO: Step 10. If hideUntil is null, then set hideUntil to the result of running topmost popover ancestor given this, document's showing auto popover list, null, and false.
615
616        // TODO: Step 11. If hideUntil is null, then set hideUntil to document.
617
618        // TODO: Step 12. Run hide all popovers until given hideUntil, false, and true.
619
620        // Step 13. Run the dialog focusing steps given this.
621        self.run_dialog_focusing_steps(cx);
622        Ok(())
623    }

651    /// <https://html.spec.whatwg.org/multipage/#the-dialog-element:is-valid-command-steps>
652    fn is_valid_command_steps(&self, command: CommandState) -> bool {
653        // Step 1. If command is in the Close state, the Request Close state, or the
654        // ShowModal state, then return true.
655        if command == CommandState::Close ||
656            command == CommandState::RequestClose ||
657            command == CommandState::ShowModal
658        {
659            return true;
660        }
661
662        // Step 2. Return false.
663        false
664    }

666    /// <https://html.spec.whatwg.org/multipage/#the-dialog-element:command-steps>
667    fn command_steps(
668        &self,
669        cx: &mut js::context::JSContext,
670        source: DomRoot<HTMLButtonElement>,
671        command: CommandState,
672    ) -> bool {
673        if self
674            .super_type()
675            .unwrap()
676            .command_steps(cx, source.clone(), command)
677        {
678            return true;
679        }
680
681        // TODO Step 1. If element is in the popover showing state, then return.
682        let element = self.upcast::<Element>();
683
684        // Step 2. If command is in the Close state and element has an open attribute, then
685        // close the dialog element with source's optional value and source.
686        if command == CommandState::Close && element.has_attribute(&local_name!("open")) {
687            let button_element = DomRoot::from_ref(source.upcast::<Element>());
688            self.close_the_dialog(cx, source.optional_value(), Some(button_element));
689            return true;
690        }
691
692        // Step 3. If command is in the Request Close state and element has an open attribute,
693        // then request to close the dialog element with source's optional value and source.
694        if command == CommandState::RequestClose && element.has_attribute(&local_name!("open")) {
695            let button_element = DomRoot::from_ref(source.upcast::<Element>());
696            self.request_to_close(cx, source.optional_value(), Some(button_element));
697            return true;
698        }
699
700        // Step 4. If command is the Show Modal state and element does not have an open attribute,
701        // then show a modal dialog given element and source.
702        if command == CommandState::ShowModal && !element.has_attribute(&local_name!("open")) {
703            let button_element = DomRoot::from_ref(source.upcast::<Element>());
704            let _ = self.show_a_modal(cx, Some(button_element));
705            return true;
706        }
707
708        false
709    }

746    fn attribute_mutated(
747        &self,
748        cx: &mut js::context::JSContext,
749        attr: AttrRef<'_>,
750        mutation: AttributeMutation,
751    ) {
752        self.super_type()
753            .unwrap()
754            .attribute_mutated(cx, attr, mutation);
755
756        // Step 1. If namespace is not null, then return.
757        if *attr.namespace() != ns!() {
758            return;
759        }
760
761        // Step 2. If localName is not open, then return.
762        if attr.local_name() != &local_name!("open") {
763            return;
764        }
765
766        // Step 3. If value is null and oldValue is not null, then run the dialog cleanup steps
767        // given element.
768        if matches!(mutation, AttributeMutation::Removed) {
769            self.dialog_cleanup_steps(cx.no_gc());
770            return;
771        }
772
773        // Step 4. If element's node document is not fully active, then return.
774        let element = self.upcast::<Element>();
775        if !element.owner_document().is_fully_active() {
776            return;
777        }
778
779        // Step 5. If element is not connected, then return.
780        if !element.is_connected() {
781            return;
782        }
783
784        // Step 6. If value is not null and oldValue is null, then run the dialog setup steps given
785        // element.
786        if matches!(mutation, AttributeMutation::Set(None)) {
787            self.dialog_setup_steps(cx.no_gc());
788        }
789    }

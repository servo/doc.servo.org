132    /// <https://html.spec.whatwg.org/multipage/#start-the-track-processing-model>
133    pub(crate) fn start_the_track_processing_model(&self, cx: &mut JSContext) {
134        // Step 1. If another occurrence of this algorithm is already running
135        // for this text track and its track element, return,
136        // letting that other algorithm take care of this element.
137        if self.is_running_processing_model_algorithm.get() {
138            return;
139        }
140        // Step 2. If the text track's text track mode is not set to one of hidden or showing, then return.
141        if !matches!(
142            self.track.Mode(),
143            TextTrackMode::Hidden | TextTrackMode::Showing
144        ) {
145            return;
146        }
147        // Step 3. If the text track's track element does not have a media element as a parent, return.
148        if self
149            .upcast::<Node>()
150            .GetParentElement()
151            .is_none_or(|parent| !parent.is::<HTMLMediaElement>())
152        {
153            return;
154        };
155        // Step 4. Run the remainder of these steps in parallel, allowing whatever caused these steps to run to continue.
156        // Step 5. Top: Await a stable state.
157        let task = TrackElementMicrotask::ProcessingModel {
158            elem: Dom::from_ref(self),
159        };
160        self.is_running_processing_model_algorithm.set(true);
161
162        ScriptThread::await_stable_state(cx, Box::new(task));
163    }

165    fn check_if_track_parent_element_changed(&self, cx: &mut JSContext) {
166        if let Some(parent) = self
167            .upcast::<Node>()
168            .GetParentNode()
169            .and_then(DomRoot::downcast::<HTMLMediaElement>)
170        {
171            // https://html.spec.whatwg.org/multipage/#sourcing-out-of-band-text-tracks
172            // > When a track element's parent element changes and the new parent is a media element,
173            // > then the user agent must add the track element's corresponding text track to
174            // > the media element's list of text tracks, and then queue a media element task
175            // > given the media element to fire an event named addtrack at the media element's
176            // > textTracks attribute's TextTrackList object, using TrackEvent,
177            // > with the track attribute initialized to the text track's TextTrack object.
178            parent.TextTracks(cx).add(cx, &self.track);
179
180            // https://html.spec.whatwg.org/multipage/#sourcing-out-of-band-text-tracks:start-the-track-processing-model
181            // > The track element's parent element changes and the new parent is a media element.
182            self.start_the_track_processing_model(cx);
183        }
184    }

192    /// <https://html.spec.whatwg.org/multipage/#dom-track-kind>
193    fn Kind(&self) -> DOMString {
194        let element = self.upcast::<Element>();
195        // Get the value of "kind" and transform all uppercase
196        // chars into lowercase.
197        let kind = element
198            .get_string_attribute(&local_name!("kind"))
199            .to_lowercase();
200        match &*kind {
201            "subtitles" | "captions" | "descriptions" | "chapters" | "metadata" => {
202                // The value of "kind" is valid. Return the lowercase version
203                // of it.
204                DOMString::from(kind)
205            },
206            _ if kind.is_empty() => {
207                // The default value should be "subtitles". If "kind" has not
208                // been set, the real value for "kind" is "subtitles"
209                DOMString::from_static("subtitles")
210            },
211            _ => {
212                // If "kind" has been set but it is not one of the valid
213                // values, return the default invalid value of "metadata"
214                DOMString::from_static("metadata")
215            },
216        }
217    }
218
219    // https://html.spec.whatwg.org/multipage/#dom-track-kind
220    // Do no transformations on the value of "kind" when setting it.
221    // All transformations should be done in the get method.
222    make_setter!(SetKind, "kind");

255impl VirtualMethods for HTMLTrackElement {
256    fn super_type(&self) -> Option<&dyn VirtualMethods> {
257        Some(self.upcast::<HTMLElement>() as &dyn VirtualMethods)
258    }
259
260    fn attribute_mutated(
261        &self,
262        cx: &mut JSContext,
263        attr: AttrRef<'_>,
264        mutation: AttributeMutation,
265    ) {
266        self.super_type()
267            .unwrap()
268            .attribute_mutated(cx, attr, mutation);
269        match *attr.local_name() {
270            local_name!("src") => {
271                // https://html.spec.whatwg.org/multipage/#attr-track-src
272                // > When the element's src attribute is set, run these steps:
273                if matches!(mutation, AttributeMutation::Set(..)) {
274                    // Step 2. Let value be the element's src attribute value.
275                    let value = &**attr.value();
276                    // Step 1. Let trackURL be failure.
277                    // Step 3. If value is not the empty string,
278                    // then set trackURL to the result of encoding-parsing-and-serializing
279                    // a URL given value, relative to the element's node document.
280                    // Step 4. Set the element's track URL to trackURL if it is not failure;
281                    // otherwise to the empty string.
282                    *self.track_url.borrow_mut() = if !value.is_empty() {
283                        self.owner_document().encoding_parse_a_url(value).ok()
284                    } else {
285                        None
286                    };
287                }
288                // https://html.spec.whatwg.org/multipage/#sourcing-out-of-band-text-tracks
289                // > Whenever a track element has its src attribute set, changed, or removed,
290                // > the user agent must immediately empty the element's text track's text track list of cues.
291                // > (This also causes the algorithm above to stop adding cues from the resource
292                // > being obtained using the previously given URL, if any.)
293                self.track.empty_cue_list();
294            },
295            _ => {},
296        }
297    }
298
299    fn moving_steps(&self, cx: &mut JSContext, context: &MoveContext) {
300        if let Some(super_type) = self.super_type() {
301            super_type.moving_steps(cx, context);
302        }
303
304        if let Some(parent) = context
305            .old_parent
306            .and_then(|node| node.downcast::<HTMLMediaElement>())
307        {
308            // https://html.spec.whatwg.org/multipage/#sourcing-out-of-band-text-tracks
309            // > When a track element's parent element changes and the old parent was a media element,
310            // > then the user agent must remove the track element's corresponding text track from
311            // > the media element's list of text tracks, and then queue a media element task
312            // > given the media element to fire an event named removetrack at the media element's
313            // > textTracks attribute's TextTrackList object, using TrackEvent,
314            // > with the track attribute initialized to the text track's TextTrack object.
315            parent.TextTracks(cx).remove(&self.track);
316        }
317
318        self.check_if_track_parent_element_changed(cx);
319    }
320
321    fn bind_to_tree(&self, cx: &mut JSContext, context: &BindContext) {
322        if let Some(super_type) = self.super_type() {
323            super_type.bind_to_tree(cx, context);
324        }
325
326        self.check_if_track_parent_element_changed(cx);
327    }
328
329    fn unbind_from_tree(&self, cx: &mut JSContext, context: &UnbindContext) {
330        if let Some(s) = self.super_type() {
331            s.unbind_from_tree(cx, context);
332        }
333
334        if let Some(parent) = context.parent.downcast::<HTMLMediaElement>() {
335            // https://html.spec.whatwg.org/multipage/#sourcing-out-of-band-text-tracks
336            // > When a track element's parent element changes and the old parent was a media element,
337            // > then the user agent must remove the track element's corresponding text track from
338            // > the media element's list of text tracks, and then queue a media element task
339            // > given the media element to fire an event named removetrack at the media element's
340            // > textTracks attribute's TextTrackList object, using TrackEvent,
341            // > with the track attribute initialized to the text track's TextTrack object.
342            parent.TextTracks(cx).remove(&self.track);
343        }
344    }

348pub(crate) enum TrackElementMicrotask {
349    ProcessingModel { elem: Dom<HTMLTrackElement> },
350}

352impl MicrotaskRunnable for TrackElementMicrotask {
353    fn handler(&self, cx: &mut JSContext) {
354        let _realm = match self {
355            TrackElementMicrotask::ProcessingModel { elem, .. } => enter_auto_realm(cx, &**elem),
356        };
357        match self {
358            // https://html.spec.whatwg.org/multipage/#start-the-track-processing-model
359            TrackElementMicrotask::ProcessingModel { elem } => {
360                // Not specced, but required for browser compatibility:
361                // https://github.com/whatwg/html/issues/12796
362                if elem.readiness_state.get() == TextTrackReadinessState::Loaded &&
363                    *elem.track_url.borrow() == *elem.last_successful_load.borrow()
364                {
365                    elem.is_running_processing_model_algorithm.set(false);
366                    return;
367                }
368
369                let media_parent = elem
370                    .upcast::<Node>()
371                    .GetParentNode()
372                    .and_then(DomRoot::downcast::<HTMLMediaElement>);
373
374                // The synchronous section consists of the following steps.
375                // (The steps in the synchronous section are marked with ⌛.)
376                // Step 6. ⌛ Set the text track readiness state to loading.
377                elem.readiness_state.set(TextTrackReadinessState::Loading);
378                // Step 7. ⌛ Let URL be the track URL of the track element.
379                let url = elem.track_url.borrow().clone();
380                // Step 8. ⌛ If the track element's parent is a media element,
381                // then let corsAttributeState be the state of the parent media element's
382                // crossorigin content attribute. Otherwise, let corsAttributeState be No CORS.
383                let cors_attribute_state =
384                    media_parent.and_then(|parent| cors_setting_for_element(parent.upcast()));
385                // Step 9. End the synchronous section, continuing the remaining steps in parallel.
386                // TODO
387                // Step 10. If URL is not the empty string:
388                if let Some(url) = url {
389                    // Step 10.1. Let request be the result of creating a potential-CORS request given URL,
390                    // "track", and corsAttributeState, and with the same-origin fallback flag set.
391                    let global = elem.global();
392                    let document = elem.owner_document();
393                    let request = create_a_potential_cors_request(
394                        Some(document.webview_id()),
395                        url.clone(),
396                        Destination::Track,
397                        cors_attribute_state,
398                        Some(true),
399                        global.get_referrer(),
400                    )
401                    // Step 10.2. Set request's client to the track element's node document's relevant
402                    // settings object.
403                    .with_global_scope(&global);
404                    // Step 10.3. Set request's initiator type to "track".
405                    //
406                    // Set in listener
407
408                    // Step 10.4. Fetch request.
409                    let listener = HTMLTrackElementFetchListener {
410                        element: Trusted::new(elem),
411                        url,
412                        payload: vec![],
413                    };
414                    document.fetch_background(request, listener);
415                } else {
416                    elem.is_running_processing_model_algorithm.set(false);
417                }
418                // Step 11. Wait until the text track readiness state is no longer set to loading.
419                // TODO
420                // Step 12. Wait until the track URL is no longer equal to URL,
421                // at the same time as the text track mode is set to hidden or showing.
422                // TODO
423                // Step 13. Jump to the step labeled top.
424                // TODO
425            },
426        }
427    }
428}

454impl FetchResponseListener for HTMLTrackElementFetchListener {
455    fn process_request_body(&mut self, _: RequestId) {}
456
457    fn process_response(
458        &mut self,
459        _: &mut JSContext,
460        _: RequestId,
461        _: Result<FetchMetadata, NetworkError>,
462    ) {
463    }
464
465    fn process_response_chunk(&mut self, _: &mut JSContext, _: RequestId, payload: Bytes) {
466        self.payload.extend_from_slice(&payload);
467    }
468
469    /// Step 10.4 of <https://html.spec.whatwg.org/multipage/#start-the-track-processing-model>
470    fn process_response_eof(
471        self,
472        cx: &mut JSContext,
473        _: RequestId,
474        status: Result<(), NetworkError>,
475        timing: ResourceFetchTiming,
476    ) {
477        let track = self.element.clone();
478        let element = self.element.root();
479        if status.is_err() {
480            // > If fetching fails for any reason (network error, the server returns an error code, CORS fails, etc.),
481            // > or if URL is the empty string, then queue an element task on the DOM manipulation task source
482            // > given the media element to first change the text track readiness state to failed to load
483            // > and then fire an event named error at the track element.
484            element
485                .global()
486                .task_manager()
487                .dom_manipulation_task_source()
488                .queue(task!(failed_to_load: move |cx| {
489                    let track = track.root();
490                    track.readiness_state.set(TextTrackReadinessState::FailedToLoad);
491                    track.upcast::<EventTarget>().fire_event(cx, atom!("error"));
492                }));
493        } else {
494            // > The tasks queued by the fetching algorithm on the networking task source to
495            // > process the data as it is being fetched must determine the type of the resource.
496            // > If the type of the resource is not a supported text track format, the load will fail,
497            // > as described below. Otherwise, the resource's data must be passed to the appropriate parser
498            // > (e.g., the WebVTT parser) as it is received, with the text track list of cues
499            // > being used for that parser's output. [WEBVTT]
500            let result = str::from_utf8(&self.payload)
501                .map_err(|str_error| debug!("WebVTT file contains non-utf8 data: {str_error}"))
502                .and_then(|payload| {
503                    let sink = TextTrackCueSink {
504                        track_element: track.clone(),
505                    };
506                    IncrementalWebVTTParser::new(sink)
507                        .parse_sync(cx, payload)
508                        .map_err(|parser_error| {
509                            debug!("Failed to parse WEBVTT file: {parser_error}")
510                        })
511                });
512            if result.is_ok() {
513                // > If fetching does not fail, and the file was successfully processed,
514                // > then the final task that is queued by the networking task source,
515                // > after it has finished parsing the data, must change the text track readiness state to loaded,
516                // > and fire an event named load at the track element.
517                let url = self.url.clone();
518                element
519                    .global()
520                    .task_manager()
521                    .networking_task_source()
522                    .queue(task!(successfully_loaded: move |cx| {
523                        let track = track.root();
524                        *track.last_successful_load.borrow_mut() = Some(url);
525                        track.readiness_state.set(TextTrackReadinessState::Loaded);
526                        track.upcast::<EventTarget>().fire_event(cx, atom!("load"));
527                    }));
528            } else {
529                // > If fetching does not fail, but the type of the resource is not a supported text track format,
530                // > or the file was not successfully processed (e.g., the format in question is an XML format
531                // > and the file contained a well-formedness error that XML requires be detected
532                // > and reported to the application), then the task that is queued on the networking task source
533                // > in which the aforementioned problem is found must change the text track readiness state
534                // > to failed to load and fire an event named error at the track element.
535                element
536                    .global()
537                    .task_manager()
538                    .networking_task_source()
539                    .queue(task!(failed_to_parse: move |cx| {
540                        let track = track.root();
541                        track.readiness_state.set(TextTrackReadinessState::FailedToLoad);
542                        track.upcast::<EventTarget>().fire_event(cx, atom!("error"));
543                    }));
544            }
545        }
546        element.is_running_processing_model_algorithm.set(false);
547        network_listener::submit_timing(cx, &self, &status, &timing);
548    }
549
550    fn process_csp_violations(
551        &mut self,
552        cx: &mut JSContext,
553        _: RequestId,
554        violations: Vec<Violation>,
555    ) {
556        let global = &self.resource_timing_global();
557        global.report_csp_violations(cx, violations, None, None);
558    }
559
560    fn should_invoke(&self) -> bool {
561        true
562    }
563}

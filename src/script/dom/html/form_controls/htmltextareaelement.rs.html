173    fn handle_focus_event(&self, event: &FocusEvent) {
174        let event_type = event.upcast::<Event>().type_();
175        if *event_type == *"blur" {
176            self.owner_document()
177                .embedder_controls()
178                .hide_embedder_control(self.upcast());
179        } else if *event_type == *"focus" {
180            self.owner_document()
181                .embedder_controls()
182                .show_embedder_control(
183                    ControlElement::Ime(Dom::from_ref(self.upcast())),
184                    EmbedderControlRequest::InputMethod(InputMethodRequest {
185                        input_method_type: InputMethodType::Text,
186                        text: String::from(self.Value()),
187                        insertion_point: self.GetSelectionEnd(),
188                        multiline: false,
189                        // We follow chromium's heuristic to show the virtual keyboard only if user had interacted before.
190                        allow_virtual_keyboard: self.owner_window().has_sticky_activation(),
191                    }),
192                    None,
193                );
194        }
195
196        // Focus changes can activate or deactivate a selection.
197        self.maybe_update_shared_selection();
198    }
199
200    fn handle_text_content_changed(&self, cx: &mut JSContext) {
201        self.validity_state(cx)
202            .perform_validation_and_update(cx, ValidationFlags::all());
203
204        let placeholder_shown =
205            self.textinput.borrow().is_empty() && !self.placeholder.borrow().is_empty();
206        self.upcast::<Element>()
207            .set_placeholder_shown_state(placeholder_shown);
208
209        self.text_input_widget.borrow().update_shadow_tree(cx, self);
210        self.text_input_widget
211            .borrow()
212            .update_placeholder_contents(cx, self);
213        self.maybe_update_shared_selection();
214    }
215
216    /// <https://w3c.github.io/selection-api/#dfn-schedule-a-selectionchange-event>
217    fn schedule_a_selection_change_event(&self) {
218        // Step 1. If target's has scheduled selectionchange event is true, abort these steps.
219        if self.has_scheduled_selectionchange_event.get() {
220            return;
221        }
222        // Step 2. Set target's has scheduled selectionchange event to true.
223        self.has_scheduled_selectionchange_event.set(true);
224        // Step 3. Queue a task on the user interaction task source to fire a selectionchange event on target.
225        let this = Trusted::new(self);
226        self.owner_global()
227            .task_manager()
228            .user_interaction_task_source()
229            .queue(
230                // https://w3c.github.io/selection-api/#firing-selectionchange-event
231                task!(selectionchange_task_steps: move |cx| {
232                    let this = this.root();
233                    // Step 1. Set target's has scheduled selectionchange event to false.
234                    this.has_scheduled_selectionchange_event.set(false);
235                    // Step 2. If target is an element, fire an event named selectionchange, which bubbles and not cancelable, at target.
236                    this.upcast::<EventTarget>().fire_event_with_params(
237                        cx,
238                        atom!("selectionchange"),
239                        EventBubbles::Bubbles,
240                        EventCancelable::NotCancelable,
241                        EventComposed::Composed,
242                    );
243                    // Step 3. Otherwise, if target is a document, fire an event named selectionchange,
244                    // which does not bubble and not cancelable, at target.
245                    //
246                    // n/a
247                }),
248            );
249    }

262        self.textinput.borrow().has_uncollapsed_selection()
263    }

266        self.value_dirty.set(value)
267    }
268
269    fn select_all(&self) {
270        self.textinput.borrow_mut().select_all();
271        self.maybe_update_shared_selection();
272    }
273
274    fn maybe_update_shared_selection(&self) {
275        let selection = {
276            let mut text_input = self.textinput.borrow_mut();
277            let selection_range = text_input.selection_start()..text_input.selection_end();
278            let enabled = self.upcast::<Element>().focus_state();
279
280            let range_remained_equal = selection_range == text_input.previous_selection_range;
281            if range_remained_equal && enabled == text_input.selection_for_layout.is_some() {
282                return;
283            }
284
285            if !range_remained_equal {
286                // https://w3c.github.io/selection-api/#selectionchange-event
287                // > When an input or textarea element provide a text selection and its selection changes
288                // > (in either extent or direction),
289                // > the user agent must schedule a selectionchange event on the element.
290                self.schedule_a_selection_change_event();
291            }
292
293            let selection = enabled.then(|| text_input.sorted_selection_character_offsets_range());
294            text_input.previous_selection_range = selection_range;
295            text_input.selection_for_layout = selection;
296            selection
297        };
298
299        if self
300            .text_input_widget
301            .borrow()
302            .set_text_run_selection(selection)
303        {
304            // Found an already laid out text run to update, so we only need to repaint:
305            self.owner_window().layout().set_needs_new_display_list();
306        } else {
307            // If there isn’t a text run, layout is pending to create it anyway
308        }
309    }
310
311    fn placeholder_text<'a>(&'a self) -> Ref<'a, DOMString> {
312        self.placeholder.borrow()
313    }
314
315    fn value_text(&self) -> DOMString {
316        self.Value()
317    }
318}

553impl HTMLTextAreaElement {
554    /// <https://w3c.github.io/webdriver/#ref-for-dfn-clear-algorithm-4>
555    /// Used by WebDriver to clear the textarea element.
556    pub(crate) fn clear(&self) {
557        self.value_dirty.set(false);
558        self.textinput.borrow_mut().set_content(DOMString::new());
559    }
560
561    pub(crate) fn reset(&self, cx: &mut JSContext) {
562        // https://html.spec.whatwg.org/multipage/#the-textarea-element:concept-form-reset-control
563        self.value_dirty.set(false);
564        self.textinput.borrow_mut().set_content(self.DefaultValue());
565        self.handle_text_content_changed(cx);
566    }
567
568    fn selection(&self) -> TextControlSelection<'_, Self> {
569        TextControlSelection::new(self, &self.textinput)
570    }
571
572    fn handle_key_reaction(&self, cx: &mut JSContext, action: KeyReaction, event: &Event) {
573        match action {
574            KeyReaction::TriggerDefaultAction => (),
575            KeyReaction::DispatchInput(text, is_composing, input_type) => {
576                if event.IsTrusted() {
577                    self.textinput.borrow().queue_input_event(
578                        self.upcast(),
579                        text,
580                        is_composing,
581                        input_type,
582                    );
583                }
584                self.value_dirty.set(true);
585                self.handle_text_content_changed(cx);
586                event.mark_as_handled();
587            },
588            KeyReaction::RedrawSelection => {
589                self.maybe_update_shared_selection();
590                event.mark_as_handled();
591            },
592            KeyReaction::Nothing => (),
593        }
594    }

597impl VirtualMethods for HTMLTextAreaElement {

602    fn attribute_mutated(
603        &self,
604        cx: &mut JSContext,
605        attr: AttrRef<'_>,
606        mutation: AttributeMutation,
607    ) {
608        self.super_type()
609            .unwrap()
610            .attribute_mutated(cx, attr, mutation);
611        match *attr.local_name() {
612            local_name!("disabled") => {
613                let el = self.upcast::<Element>();
614                match mutation {
615                    AttributeMutation::Set(..) => {
616                        el.set_disabled_state(true);
617                        el.set_enabled_state(false);
618
619                        el.set_read_write_state(false);
620                    },
621                    AttributeMutation::Removed => {
622                        el.set_disabled_state(false);
623                        el.set_enabled_state(true);
624                        el.check_ancestors_disabled_state_for_form_control();
625
626                        if !el.disabled_state() && !el.read_write_state() {
627                            el.set_read_write_state(true);
628                        }
629                    },
630                }
631            },
632            local_name!("maxlength") => match *attr.value() {
633                AttrValue::Int(_, value) => {
634                    let mut textinput = self.textinput.borrow_mut();
635
636                    if value < 0 {
637                        textinput.set_max_length(None);
638                    } else {
639                        textinput.set_max_length(Some(Utf16CodeUnits(value as usize)))
640                    }
641                },
642                _ => panic!("Expected an AttrValue::Int"),
643            },
644            local_name!("minlength") => match *attr.value() {
645                AttrValue::Int(_, value) => {
646                    let mut textinput = self.textinput.borrow_mut();
647
648                    if value < 0 {
649                        textinput.set_min_length(None);
650                    } else {
651                        textinput.set_min_length(Some(Utf16CodeUnits(value as usize)))
652                    }
653                },
654                _ => panic!("Expected an AttrValue::Int"),
655            },
656            local_name!("placeholder") => {
657                {
658                    let mut placeholder = self.placeholder.borrow_mut();
659                    match mutation {
660                        AttributeMutation::Set(..) => {
661                            let value = attr.value();
662                            let value_str: &str = value.as_ref();
663                            *placeholder =
664                                value_str.replace("\r\n", "\n").replace('\r', "\n").into();
665                        },
666                        AttributeMutation::Removed => placeholder.clear(),
667                    }
668                }
669                self.handle_text_content_changed(cx);
670            },
671            local_name!("readonly") => {
672                let el = self.upcast::<Element>();
673                match mutation {
674                    AttributeMutation::Set(..) => {
675                        el.set_read_write_state(false);
676                    },
677                    AttributeMutation::Removed => {
678                        el.set_read_write_state(!el.disabled_state());
679                    },
680                }
681            },
682            local_name!("form") => {
683                self.form_attribute_mutated(cx, mutation);
684            },
685            _ => {},
686        }
687
688        self.validity_state(cx)
689            .perform_validation_and_update(cx, ValidationFlags::all());
690    }

703    fn parse_plain_attribute(&self, name: &LocalName, value: DOMString) -> AttrValue {
704        match *name {
705            local_name!("cols") => AttrValue::from_limited_u32(value.into(), DEFAULT_COLS),
706            local_name!("rows") => AttrValue::from_limited_u32(value.into(), DEFAULT_ROWS),
707            local_name!("maxlength") => {

709            },
710            local_name!("minlength") => {
711                AttrValue::from_limited_i32(value.into(), DEFAULT_MIN_LENGTH)
712            },
713            _ => self
714                .super_type()
715                .unwrap()
716                .parse_plain_attribute(name, value),
717        }

720    fn unbind_from_tree(&self, cx: &mut JSContext, context: &UnbindContext) {
721        // Always attempt to hide IME when unbinding input elements from the tree.
722        self.owner_document()
723            .embedder_controls()
724            .hide_embedder_control(self.upcast());
725
726        self.super_type().unwrap().unbind_from_tree(cx, context);
727
728        let node = self.upcast::<Node>();
729        let el = self.upcast::<Element>();
730        if node
731            .ancestors()
732            .any(|ancestor| ancestor.is::<HTMLFieldSetElement>())
733        {
734            el.check_ancestors_disabled_state_for_form_control();
735        } else {
736            el.check_disabled_attribute();
737        }
738
739        self.validity_state(cx)
740            .perform_validation_and_update(cx, ValidationFlags::all());
741    }

743    // The cloning steps for textarea elements must propagate the raw value
744    // and dirty value flag from the node being cloned to the copy.
745    fn cloning_steps(
746        &self,
747        cx: &mut JSContext,
748        copy: &Node,
749        maybe_doc: Option<&Document>,
750        clone_children: CloneChildrenFlag,
751    ) {
752        if let Some(s) = self.super_type() {
753            s.cloning_steps(cx, copy, maybe_doc, clone_children);
754        }
755        let el = copy.downcast::<HTMLTextAreaElement>().unwrap();
756        el.value_dirty.set(self.value_dirty.get());
757        {
758            let mut textinput = el.textinput.borrow_mut();
759            textinput.set_content(self.textinput.borrow().get_content());
760        }
761        el.validity_state(cx)
762            .perform_validation_and_update(cx, ValidationFlags::all());
763    }

774    // copied and modified from htmlinputelement.rs
775    fn handle_event(&self, cx: &mut JSContext, event: &Event) {
776        if event.type_() == atom!("keydown") && !event.DefaultPrevented() {
777            if let Some(keyboard_event) = event.downcast::<KeyboardEvent>() {
778                // This can't be inlined, as holding on to textinput.borrow_mut()
779                // during self.implicit_submission will cause a panic.
780                let action = self.textinput.borrow_mut().handle_keydown(keyboard_event);
781                self.handle_key_reaction(cx, action, event);
782            }
783        } else if event.type_() == atom!("compositionstart") ||
784            event.type_() == atom!("compositionupdate") ||
785            event.type_() == atom!("compositionend")
786        {
787            if let Some(compositionevent) = event.downcast::<CompositionEvent>() {
788                if event.type_() == atom!("compositionend") {
789                    let action = self
790                        .textinput
791                        .borrow_mut()
792                        .handle_compositionend(compositionevent);
793                    self.handle_key_reaction(cx, action, event);
794                    self.upcast::<Node>().dirty(cx.no_gc(), NodeDamage::Other);
795                } else if event.type_() == atom!("compositionupdate") {
796                    let action = self
797                        .textinput
798                        .borrow_mut()
799                        .handle_compositionupdate(compositionevent);
800                    self.handle_key_reaction(cx, action, event);
801                    self.upcast::<Node>().dirty(cx.no_gc(), NodeDamage::Other);
802                }
803                self.maybe_update_shared_selection();
804                event.mark_as_handled();
805            }
806        } else if let Some(clipboard_event) = event.downcast::<ClipboardEvent>() {
807            let reaction = self
808                .textinput
809                .borrow_mut()
810                .handle_clipboard_event(clipboard_event);
811
812            let flags = reaction.flags;
813            if flags.contains(ClipboardEventFlags::FireClipboardChangedEvent) {
814                self.owner_document().event_handler().fire_clipboard_event(
815                    cx,
816                    None,
817                    ClipboardEventType::Change,
818                );
819            }
820            if flags.contains(ClipboardEventFlags::QueueInputEvent) {
821                self.textinput.borrow().queue_input_event(
822                    self.upcast(),
823                    reaction.text,
824                    IsComposing::NotComposing,
825                    reaction.input_type,
826                );
827            }
828            if !flags.is_empty() {
829                event.mark_as_handled();
830                self.handle_text_content_changed(cx);
831            }
832        } else if let Some(event) = event.downcast::<FocusEvent>() {
833            self.handle_focus_event(event);
834        }
835
836        self.validity_state(cx)
837            .perform_validation_and_update(cx, ValidationFlags::all());
838
839        if let Some(super_type) = self.super_type() {
840            super_type.handle_event(cx, event);
841        }
842    }
843
844    fn handle_mousedown_event(
845        &self,
846        cx: &mut JSContext,
847        mouse_event: &MouseEvent,
848        hit_test_result: &HitTestResult,
849    ) {
850        // If the placeholder is displayed, don't do any interactive mouse event handling.
851        if self.textinput.borrow().is_empty() {
852            if let Some(super_type) = self.super_type() {
853                super_type.handle_mousedown_event(cx, mouse_event, hit_test_result);
854            }
855            return;
856        }
857
858        if self.textinput.borrow_mut().handle_mousedown_event(
859            self.upcast(),
860            mouse_event,
861            hit_test_result,
862        ) {
863            self.maybe_update_shared_selection();
864            mouse_event.upcast::<Event>().mark_as_handled();
865        }
866    }
867
868    fn pop(&self, cx: &mut JSContext) {
869        self.super_type().unwrap().pop(cx);
870
871        // https://html.spec.whatwg.org/multipage/#the-textarea-element:stack-of-open-elements
872        self.reset(cx);
873    }

890impl Validatable for HTMLTextAreaElement {
891    fn as_element(&self) -> &Element {
892        self.upcast()
893    }
894
895    fn validity_state(&self, cx: &mut JSContext) -> DomRoot<ValidityState> {
896        self.validity_state
897            .or_init(|| ValidityState::new(cx, &self.owner_window(), self.upcast()))
898    }
899
900    fn is_instance_validatable(&self) -> bool {
901        // https://html.spec.whatwg.org/multipage/#enabling-and-disabling-form-controls%3A-the-disabled-attribute%3Abarred-from-constraint-validation
902        // https://html.spec.whatwg.org/multipage/#the-textarea-element%3Abarred-from-constraint-validation
903        // https://html.spec.whatwg.org/multipage/#the-datalist-element%3Abarred-from-constraint-validation
904        !self.upcast::<Element>().disabled_state() &&
905            !self.ReadOnly() &&
906            !is_barred_by_datalist_ancestor(self.upcast())
907    }
908
909    fn perform_validation(
910        &self,
911        _cx: &mut JSContext,
912        validate_flags: ValidationFlags,
913    ) -> ValidationFlags {
914        let mut failed_flags = ValidationFlags::empty();
915
916        let textinput = self.textinput.borrow();
917        let Utf16CodeUnits(value_len) = textinput.len_utf16();
918        let last_edit_by_user = !textinput.was_last_change_by_set_content();
919        let value_dirty = self.value_dirty.get();
920
921        // https://html.spec.whatwg.org/multipage/#suffering-from-being-missing
922        // https://html.spec.whatwg.org/multipage/#the-textarea-element%3Asuffering-from-being-missing
923        if validate_flags.contains(ValidationFlags::VALUE_MISSING) &&
924            self.Required() &&
925            self.is_mutable() &&
926            value_len == 0
927        {
928            failed_flags.insert(ValidationFlags::VALUE_MISSING);
929        }
930
931        if value_dirty && last_edit_by_user && value_len > 0 {
932            // https://html.spec.whatwg.org/multipage/#suffering-from-being-too-long
933            // https://html.spec.whatwg.org/multipage/#limiting-user-input-length%3A-the-maxlength-attribute%3Asuffering-from-being-too-long
934            if validate_flags.contains(ValidationFlags::TOO_LONG) {
935                let max_length = self.MaxLength();
936                if max_length != DEFAULT_MAX_LENGTH && value_len > (max_length as usize) {
937                    failed_flags.insert(ValidationFlags::TOO_LONG);
938                }
939            }
940
941            // https://html.spec.whatwg.org/multipage/#suffering-from-being-too-short
942            // https://html.spec.whatwg.org/multipage/#setting-minimum-input-length-requirements%3A-the-minlength-attribute%3Asuffering-from-being-too-short
943            if validate_flags.contains(ValidationFlags::TOO_SHORT) {
944                let min_length = self.MinLength();
945                if min_length != DEFAULT_MIN_LENGTH && value_len < (min_length as usize) {
946                    failed_flags.insert(ValidationFlags::TOO_SHORT);
947                }
948            }
949        }
950
951        failed_flags
952    }

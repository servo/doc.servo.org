36pub(crate) struct ImageInputType {
37    shadow_tree: DomRefCell<Option<ImageInputShadowTree>>,
38    /// <https://html.spec.whatwg.org/multipage/#concept-input-type-image-coordinate>
39    #[no_trace]
40    selected_coordinate: Cell<Point2D<i32, CSSPixel>>,
41}

46impl SpecificInputType for ImageInputType {
47    fn text_input_widget(&self) -> Option<&DomRefCell<TextInputWidget>> {
48        None
49    }
50
51    fn update_shadow_tree(&self, cx: &mut JSContext, input: &HTMLInputElement) {
52        self.ensure_shadow_tree(cx, input.upcast())
53            .update(cx, self, input)
54    }
55
56    fn attribute_mutated(
57        &self,
58        cx: &mut JSContext,
59        input: &HTMLInputElement,
60        attr: AttrRef<'_>,
61        _: AttributeMutation,
62    ) {
63        if matches!(*attr.local_name(), local_name!("src") | local_name!("alt")) {
64            self.update_shadow_tree(cx, input);
65        }
66    }
67}

69impl ImageInputType {
70    pub(crate) fn set_selected_coordinate(&self, input: &HTMLInputElement, event: &Event) {
71        if !event.IsTrusted() {
72            return;
73        }
74        let Some(mouse_event) = event.downcast::<MouseEvent>() else {
75            return;
76        };
77        let rect = input.upcast::<Node>().client_rect();
78        self.selected_coordinate.set(Point2D::new(
79            mouse_event.ClientX() - rect.origin.x,
80            mouse_event.ClientY() - rect.origin.y,
81        ));
82    }
83
84    pub(crate) fn selected_coordinate(&self) -> Point2D<i32, CSSPixel> {
85        self.selected_coordinate.get()
86    }
87
88    /// Step 2 and later of <https://html.spec.whatwg.org/multipage/#input-img-dimensions>
89    pub(crate) fn dimensions(&self, cx: &mut JSContext, input: &HTMLInputElement) -> (u32, u32) {
90        // Step 2. Let naturalDimensions be null.
91        // Step 3. If an image is available,
92        // then set naturalDimensions to the result of applying the default sizing algorithm
93        // with that image's natural width, natural height, and natural aspect ratio,
94        // if any, using a default object size of 300 by 150.
95        let natural_dimensions = self
96            .borrow_for_shadow_tree()
97            .as_ref()
98            .and_then(|shadow_tree| {
99                shadow_tree.image_element.image_data().map(|image_data| {
100                    let metadata = image_data.metadata();
101                    (metadata.width, metadata.height)
102                })
103            });
104        // Step 4. Return the result of determining the dimensions of input given naturalDimensions.
105        HTMLImageElement::determine_dimensions(cx, input.upcast(), natural_dimensions)
106    }
107}

109impl SpecificInputActivationType for ImageInputActivation {
110    /// <https://html.spec.whatwg.org/multipage/#image-button-state-(type=image):input-activation-behavior>
111    fn activation_behavior(
112        &self,
113        cx: &mut JSContext,
114        input: &HTMLInputElement,
115        event: &Event,
116        _target: &EventTarget,
117    ) {
118        // Step 1: If the element does not have a form owner, then return.
119        if let Some(form_owner) = input.form_owner() {
120            let document = input.owner_document();
121
122            // Step 2: If the element's node document is not fully active, then return.
123            if !document.is_fully_active() {
124                return;
125            }
126
127            // Step 3: If the user activated the control while explicitly selecting a coordinate,
128            // then set the element's selected coordinate to that coordinate.
129            if let InputType::Image(ref image_input_type) = *input.input_type() {
130                image_input_type.set_selected_coordinate(input, event);
131            }
132
133            // Step 4: Submit the element's form owner from the element with userInvolvement
134            // set to event's user navigation involvement.
135            form_owner.submit(
136                cx,
137                SubmittedFrom::NotFromForm,
138                FormSubmitterElement::Input(input),
139            )
140        }
141    }
142}

144#[derive(Clone, JSTraceable, MallocSizeOf, PartialEq)]
145#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
146pub(crate) struct ImageInputShadowTree {
147    image_element: Dom<HTMLImageElement>,
148}
149
150impl ImageInputShadowTree {
151    pub(crate) fn new(cx: &mut JSContext, shadow_root: &Node) -> Self {
152        let document = shadow_root.owner_document();
153        let img = Self::create_element_in_ua_shadowroot(cx, &document, local_name!("img"));
154        Node::replace_all(cx, Some(img.upcast()), shadow_root);
155        let img = img
156            .downcast::<HTMLImageElement>()
157            .expect("Must always create an image element");
158        Self {
159            image_element: Dom::from_ref(img),
160        }
161    }
162}
163
164impl SpecificShadowTree<HTMLInputElement, ImageInputType> for ImageInputShadowTree {
165    fn update(&self, cx: &mut JSContext, _: &ImageInputType, input_element: &HTMLInputElement) {
166        let input_element = input_element.upcast::<Element>();
167        for name in [&local_name!("src"), &local_name!("alt")] {
168            let new = input_element.get_attribute_string_value(name);
169            let image_element = self.image_element.upcast::<Element>();
170            if new == image_element.get_attribute_string_value(name) {
171                continue;
172            }
173            match new {
174                Some(value) => image_element.set_attribute(cx, name, value.into()),
175                None => {
176                    image_element.remove_attribute(cx, &ns!(), name);
177                },
178            }
179        }
180    }
181}

183impl UAShadowRoot<ImageInputShadowTree> for ImageInputType {
184    fn store_for_shadow_tree(&self, cx: &mut JSContext, shadow_root: DomRoot<ShadowRoot>) {
185        *self.shadow_tree.safe_borrow_mut(cx.no_gc()) =
186            Some(ImageInputShadowTree::new(cx, shadow_root.upcast()));
187    }
188
189    fn borrow_for_shadow_tree(&self) -> Ref<'_, Option<ImageInputShadowTree>> {
190        self.shadow_tree.borrow()
191    }

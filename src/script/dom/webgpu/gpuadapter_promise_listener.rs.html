5use js::jsapi::HandleObject;
6use script_bindings::cformat;
7use script_bindings::codegen::GenericBindings::WebGPUBinding::GPUDeviceLostReason;
8use webgpu_traits::{RequestDeviceError, WebGPUDeviceResponse};
9
10use crate::dom::bindings::error::Error;
11use crate::dom::bindings::reflector::DomGlobal;
12use crate::dom::gpuadapter::GPUAdapter;
13use crate::dom::promise::RootedPromise;
14use crate::dom::types::GPUDevice;
15use crate::routed_promise::RoutedPromiseListener;
16
17impl RoutedPromiseListener<WebGPUDeviceResponse> for GPUAdapter {
18    /// <https://www.w3.org/TR/webgpu/#dom-gpuadapter-requestdevice>
19    fn handle_response(
20        &self,
21        cx: &mut js::context::JSContext,
22        response: WebGPUDeviceResponse,
23        promise: &RootedPromise,
24    ) {
25        match response {
26            // 3.1 Let device be a new device with the capabilities described by descriptor.
27            (device_id, queue_id, Ok(descriptor)) => {
28                let device = GPUDevice::new(
29                    cx,
30                    &self.global(),
31                    self.channel(),
32                    self,
33                    HandleObject::null(),
34                    descriptor.required_features,
35                    descriptor.required_limits,
36                    device_id,
37                    queue_id,
38                    descriptor.label.unwrap_or_default(),
39                );
40                self.global().add_gpu_device(&device);
41                promise.resolve_native(cx, &device);
42            },
43            // 1. If features are not supported reject promise with a TypeError.
44            (_, _, Err(RequestDeviceError::UnsupportedFeature(f))) => promise.reject_error(
45                cx,
46                Error::Type(cformat!(
47                    "{}",
48                    wgpu_core::instance::RequestDeviceError::UnsupportedFeature(f)
49                )),
50            ),
51            // 2. If limits are not supported reject promise with an OperationError.
52            (_, _, Err(RequestDeviceError::LimitsExceeded(l))) => {
53                warn!(
54                    "{}",
55                    wgpu_core::instance::RequestDeviceError::LimitsExceeded(l)
56                );
57                promise.reject_error(
58                    cx,
59                    Error::Operation(Some("WebGPU Device Limit exceeded".to_string())),
60                )
61            },
62            // 3. user agent otherwise cannot fulfill the request
63            (device_id, queue_id, Err(RequestDeviceError::Other(e))) => {
64                // TODO(sagudev): firefox always says operation error,
65                // meanwhile we create "invalid" device that is not invalid in wgpu
66                // causing crashes when one tries to use it
67                // 1. Let device be a new device.
68                let device = GPUDevice::new(
69                    cx,
70                    &self.global(),
71                    self.channel(),
72                    self,
73                    HandleObject::null(),
74                    wgpu_types::Features::default(),
75                    wgpu_types::Limits::default(),
76                    device_id,
77                    queue_id,
78                    String::new(),
79                );
80                // 2. Lose the device(device, "unknown").
81                device.lose(GPUDeviceLostReason::Unknown, e);
82                promise.resolve_native(cx, &device);
83            },
84        }
85    }
86}

153impl WebGPUPromiseTrait<crate::DomTypeHolder> for RootedPromise {
154    fn callback_promise_adapter(
155        &self,
156        d: &GPUAdapter,
157    ) -> servo_base::generic_channel::GenericCallback<webgpu_traits::WebGPUDeviceResponse> {
158        let task_manager = <GPUAdapter as DomGlobal>::global(d).task_manager();
159        callback_promise(self, d, task_manager.dom_manipulation_task_source())
160    }
161
162    fn callback_promise_gpubuffer(
163        &self,
164        d: &GPUBuffer,
165    ) -> servo_base::generic_channel::GenericCallback<Result<Mapping, BufferAccessError>> {
166        let task_manager = <GPUBuffer as DomGlobal>::global(d).task_manager();
167        callback_promise(self, d, task_manager.dom_manipulation_task_source())
168    }
169
170    fn callback_promise_gpu(
171        &self,
172        d: &GPU,
173    ) -> servo_base::generic_channel::GenericCallback<webgpu_traits::WebGPUAdapterResponse> {
174        let task_manager = <GPU as DomGlobal>::global(d).task_manager();
175        callback_promise(self, d, task_manager.dom_manipulation_task_source())
176    }
177
178    fn callback_promise_gpushadermodule(
179        &self,
180        d: &script_webgpu::gpushadermodule::GPUShaderModule<crate::DomTypeHolder>,
181    ) -> servo_base::generic_channel::GenericCallback<Option<webgpu_traits::ShaderCompilationInfo>>
182    {
183        let task_manager = <GPUShaderModule as DomGlobal>::global(d).task_manager();
184        callback_promise(self, d, task_manager.dom_manipulation_task_source())
185    }
186}

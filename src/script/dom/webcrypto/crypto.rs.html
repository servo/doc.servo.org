23/// <https://w3c.github.io/webcrypto/#crypto-interface>
24#[dom_struct]
25pub(crate) struct Crypto {
26    reflector_: Reflector,
27    subtle: MutNullableDom<SubtleCrypto>,
28}
29
30impl Crypto {
31    fn new_inherited() -> Crypto {
32        Crypto {
33            reflector_: Reflector::new(),
34            subtle: MutNullableDom::default(),
35        }
36    }
37
38    pub(crate) fn new(cx: &mut JSContext, global: &GlobalScope) -> DomRoot<Crypto> {
39        reflect_dom_object(cx, Box::new(Crypto::new_inherited()), global)
40    }
41}

43impl CryptoMethods<crate::DomTypeHolder> for Crypto {
44    /// <https://w3c.github.io/webcrypto/#dfn-Crypto-attribute-subtle>
45    fn Subtle(&self, cx: &mut JSContext) -> DomRoot<SubtleCrypto> {
46        self.subtle
47            .or_init(|| SubtleCrypto::new(cx, &self.global()))
48    }
49
50    /// <https://w3c.github.io/webcrypto/#Crypto-method-getRandomValues>
51    fn GetRandomValues(
52        &self,
53        cx: &mut JSContext,
54        mut array: CustomAutoRooterGuard<ArrayBufferView>,
55    ) -> Fallible<RootedTraceableBox<HeapArrayBufferView>> {
56        // Step 1. If array is not an Int8Array, Uint8Array, Uint8ClampedArray, Int16Array,
57        // Uint16Array, Int32Array, Uint32Array, BigInt64Array, or BigUint64Array, then throw a
58        // TypeMismatchError and terminate the algorithm.
59        if !matches!(
60            array.get_array_type(),
61            Type::Int8 |
62                Type::Uint8 |
63                Type::Uint8Clamped |
64                Type::Int16 |
65                Type::Uint16 |
66                Type::Int32 |
67                Type::Uint32 |
68                Type::BigInt64 |
69                Type::BigUint64
70        ) {
71            return Err(Error::TypeMismatch(Some(
72                "The array must be an integer-based TypedArray".into(),
73            )));
74        }
75
76        // Step 2. Let byteLength be the byte length of array.
77        // Step 3. If byteLength is greater than 65536, throw a QuotaExceededError and terminate the
78        // algorithm.
79        // Step 4. Let bytes be a byte sequence of length byteLength.
80        let bytes = array.as_mut_slice_safe(cx.no_gc()).unwrap_or_default();
81        let byte_length = bytes.len();
82        if byte_length > 65536 {
83            return Err(Error::QuotaExceeded {
84                quota: None,
85                requested: None,
86            });
87        }
88
89        // Step 5. Fill bytes with cryptographically secure random bytes.
90        // Step 6. Write bytes into array.
91        rand::fill(bytes);
92
93        // Step 7. Return array.
94        let array = HeapBufferSource::from_view(cx, array);
95        array.get_typed_array().map_err(|_| Error::JSFailed)
96    }
97
98    /// <https://w3c.github.io/webcrypto/#Crypto-method-randomUUID>
99    fn RandomUUID(&self) -> DOMString {
100        // Step 1. Let bytes be a byte sequence of length 16.
101        // Step 2. Fill bytes with cryptographically secure random bytes.
102        // Step 3. Set the 4 most significant bits of bytes[6], which represent the UUID version, to 0100.
103        // Step 4. Set the 2 most significant bits of bytes[8], which represent the UUID variant, to 10.
104        // Step 5.
105        // Return the string concatenation of «
106        //     hexadecimal representation of bytes[0], hexadecimal representation of bytes[1],
107        //     hexadecimal representation of bytes[2], hexadecimal representation of bytes[3],
108        //     "-",
109        //     hexadecimal representation of bytes[4], hexadecimal representation of bytes[5],
110        //     "-",
111        //     hexadecimal representation of bytes[6], hexadecimal representation of bytes[7],
112        //     "-",
113        //     hexadecimal representation of bytes[8], hexadecimal representation of bytes[9],
114        //     "-",
115        //     hexadecimal representation of bytes[10], hexadecimal representation of bytes[11],
116        //     hexadecimal representation of bytes[12], hexadecimal representation of bytes[13],
117        //     hexadecimal representation of bytes[14], hexadecimal representation of bytes[15]
118        // ».
119        let uuid = Uuid::new_v4();
120        uuid.hyphenated()
121            .encode_lower(&mut Uuid::encode_buffer())
122            .to_owned()
123            .into()
124    }
125}

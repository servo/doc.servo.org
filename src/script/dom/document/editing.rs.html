47    pub(crate) fn editing_context(&self, no_gc: &NoGC, node: &Node) -> EditingContext {
48        if let Ok(editing_context) = EditingContext::try_from(node) {
49            return editing_context;
50        }
51
52        let mut current_node = UnrootedDom::from_ref(node, no_gc);
53        while let Some(parent) = current_node.parent_in_flat_tree(no_gc).into_parent() {
54            if let Ok(editing_context) = EditingContext::try_from(&**parent) {
55                return editing_context;
56            }
57            current_node = parent;
58        }
59
60        EditingContext::Document(DomRoot::from_ref(self))
61    }

63    /// <https://www.w3.org/TR/clipboard-apis/#clipboard-actions>
64    pub(crate) fn handle_clipboard_action(
65        &self,
66        cx: &mut JSContext,
67        editing_context: &EditingContext,
68        action: ClipboardAction,
69    ) -> InputEventResult {
70        let clipboard_event_type = match action {
71            ClipboardAction::Copy => ClipboardEventType::Copy,
72            ClipboardAction::Cut => ClipboardEventType::Cut,
73            ClipboardAction::Paste => ClipboardEventType::Paste,
74        };
75
76        // The script_triggered flag is set if the action runs because of a script, e.g. document.execCommand()
77        let script_triggered = false;
78
79        // The script_may_access_clipboard flag is set
80        // if action is paste and the script thread is allowed to read from clipboard or
81        // if action is copy or cut and the script thread is allowed to modify the clipboard
82        let script_may_access_clipboard = false;
83
84        // Step 1. If the script-triggered flag is set and the script-may-access-clipboard flag is unset
85        if script_triggered && !script_may_access_clipboard {
86            return InputEventResult::empty();
87        }
88
89        // Step 2. Fire a clipboard event
90        let event_target = editing_context.event_target();
91        let clipboard_event = self.fire_clipboard_event(cx, &event_target, clipboard_event_type);
92
93        let event = clipboard_event.upcast::<Event>();
94        if !event.DefaultPrevented() {
95            // Step 3. If the event was not canceled, then
96            match clipboard_event.clipboard_event_type() {
97                ClipboardEventType::Copy => {
98                    // Step 3.1. Copy the selected contents, if any, to the clipboard.
99                    // Implementations should create alternate text/html and text/plain
100                    // clipboard formats when content in a web page is selected.
101                    if let Some(selection) = editing_context.selection_content(cx) {
102                        self.send_to_embedder(EmbedderMsg::SetClipboardText(
103                            self.webview_id(),
104                            selection,
105                        ));
106                    }
107                    // Step 3.2. Fire a clipboard event named clipboardchange
108                    self.fire_clipboard_event(cx, &event_target, ClipboardEventType::Change);
109
110                    // This is how `true` is returned from this function.
111                    event.mark_as_handled();
112                },
113                ClipboardEventType::Cut => {
114                    if let Some(selection) = editing_context.selection_content(cx) &&
115                        editing_context.cutting_and_pasting_enabled()
116                    {
117                        // Step 3.1. If there is a selection in an editable context where
118                        // cutting is enabled, then
119                        // Step 3.1.1. Copy the selected contents, if any, to the clipboard.
120                        // Implementations should create alternate text/html and text/plain
121                        // clipboard formats when content in a web page is selected.
122                        self.send_to_embedder(EmbedderMsg::SetClipboardText(
123                            self.webview_id(),
124                            selection,
125                        ));
126
127                        // Step 3.1.2. Remove the contents of the selection from the document
128                        // and collapse the selection.
129                        editing_context.remove_the_contents_of_the_selection(cx);
130
131                        // Step 3.1.3. Fire a clipboard event named clipboardchange
132                        self.fire_clipboard_event(cx, &event_target, ClipboardEventType::Change);
133
134                        // Step 3.1.4. Queue tasks to fire any events that should fire due to
135                        // the modification, see §5.3 Integration with other scripts and
136                        // events for details.
137                        editing_context.fire_cut_events();
138
139                        // This is how `true` is returned from this function.
140                        event.mark_as_handled();
141                    }
142                },
143                ClipboardEventType::Paste => {
144                    if editing_context.has_selection_or_cursor() &&
145                        editing_context.cutting_and_pasting_enabled() &&
146                        let Some(text_content) = clipboard_event.text_content()
147                    {
148                        // Step 3.1. If there is a selection or cursor in an editable context
149                        // where pasting is enabled, then
150                        // Step 3.1.1. Insert the most suitable content found on the
151                        // clipboard, if any, into the context.
152                        editing_context.insert_content(cx, &text_content);
153
154                        // Step 3.1.2. Queue tasks to fire any events that should fire due to
155                        // the modification, see §5.3 Integration with other scripts and
156                        // events for details.
157                        editing_context.fire_paste_events(&text_content);
158
159                        // This is how `true` is returned from this function.
160                        event.mark_as_handled();
161                    }
162                },
163                _ => (),
164            }
165        } else {
166            // Step 4 If the event was canceled, then
167            match clipboard_event.clipboard_event_type() {
168                ClipboardEventType::Copy => {
169                    // Step 4.1 Call the write content to the clipboard algorithm,
170                    // passing on the DataTransferItemList items, a clear-was-called flag and a types-to-clear list.
171                    if let Some(clipboard_data) = clipboard_event.clipboard_data() {
172                        let drag_data_store =
173                            clipboard_data.data_store().expect("This shouldn't fail");
174                        self.write_content_to_the_clipboard(&drag_data_store);
175                    }
176                },
177                ClipboardEventType::Cut => {
178                    // Step 4.1 Call the write content to the clipboard algorithm,
179                    // passing on the DataTransferItemList items, a clear-was-called flag and a types-to-clear list.
180                    if let Some(clipboard_data) = clipboard_event.clipboard_data() {
181                        let drag_data_store =
182                            clipboard_data.data_store().expect("This shouldn't fail");
183                        self.write_content_to_the_clipboard(&drag_data_store);
184                    }
185
186                    // Step 4.2 Fire a clipboard event named clipboardchange
187                    self.fire_clipboard_event(cx, &event_target, ClipboardEventType::Change);
188                },
189                // Step 4.1 Return false.
190                ClipboardEventType::Paste => (),
191                _ => (),
192            }
193        }
194
195        // Step 5: Return true from the action.
196        // In this case we are returning the `InputEventResult` instead of true or false.
197        event.flags().into()
198    }

200    /// <https://www.w3.org/TR/clipboard-apis/#fire-a-clipboard-event>
201    fn fire_clipboard_event(
202        &self,
203        cx: &mut JSContext,
204        target: &EventTarget,
205        clipboard_event_type: ClipboardEventType,
206    ) -> DomRoot<ClipboardEvent> {
207        let clipboard_event = ClipboardEvent::new(
208            cx,
209            self.window(),
210            None,
211            clipboard_event_type,
212            EventBubbles::Bubbles,
213            EventCancelable::Cancelable,
214            None,
215        );
216
217        // Step 1 Let clear_was_called be false
218        // Step 2 Let types_to_clear an empty list
219        let mut drag_data_store = DragDataStore::new();
220
221        // Step 4 let clipboard-entry be the sequence number of clipboard content, null if the OS doesn't support it.
222
223        // Step 5 let trusted be true if the event is generated by the user agent, false otherwise
224        let trusted = true;
225
226        // Step 6 if the context is editable:
227        // Step 6.2 else TODO require Selection see https://github.com/w3c/clipboard-apis/issues/70
228        // Step 7
229        match clipboard_event.clipboard_event_type() {
230            ClipboardEventType::Copy | ClipboardEventType::Cut => {
231                // Step 7.2.1
232                drag_data_store.set_mode(Mode::ReadWrite);
233            },
234            ClipboardEventType::Paste => {
235                let (callback, receiver) =
236                    GenericCallback::new_blocking().expect("Could not create callback");
237                self.send_to_embedder(EmbedderMsg::GetClipboardText(self.webview_id(), callback));
238                let text_contents = receiver
239                    .recv()
240                    .map(Result::unwrap_or_default)
241                    .unwrap_or_default();
242
243                // Step 7.1.1
244                drag_data_store.set_mode(Mode::ReadOnly);
245                // Step 7.1.2 If trusted or the implementation gives script-generated events access to the clipboard
246                if trusted {
247                    // Step 7.1.2.1 For each clipboard-part on the OS clipboard:
248
249                    // Step 7.1.2.1.1 If clipboard-part contains plain text, then
250                    let data = DOMString::from(text_contents);
251                    let type_ = DOMString::from_static("text/plain");
252                    let _ = drag_data_store.add(Kind::Text { data, type_ });
253
254                    // Step 7.1.2.1.2 TODO If clipboard-part represents file references, then for each file reference
255                    // Step 7.1.2.1.3 TODO If clipboard-part contains HTML- or XHTML-formatted text then
256
257                    // Step 7.1.3 Update clipboard-event-data’s files to match clipboard-event-data’s items
258                    // Step 7.1.4 Update clipboard-event-data’s types to match clipboard-event-data’s items
259                }
260            },
261            ClipboardEventType::Change | ClipboardEventType::Other(..) => (),
262        }
263
264        // Step 3
265        let clipboard_event_data = DataTransfer::new(
266            cx,
267            self.window(),
268            Rc::new(RefCell::new(Some(drag_data_store))),
269        );
270
271        // Step 8
272        clipboard_event.set_clipboard_data(Some(&clipboard_event_data));
273
274        // Step 9
275        let event = clipboard_event.upcast::<Event>();
276        event.set_trusted(trusted);
277
278        // Step 10 Set event’s composed to true.
279        event.set_composed(true);
280
281        // Step 11
282        event.dispatch(cx, target, false);
283
284        DomRoot::from(clipboard_event)
285    }

287    /// <https://www.w3.org/TR/clipboard-apis/#write-content-to-the-clipboard>
288    fn write_content_to_the_clipboard(&self, drag_data_store: &DragDataStore) {
289        // Step 1
290        if drag_data_store.list_len() > 0 {
291            // Step 1.1 Clear the clipboard.
292            self.send_to_embedder(EmbedderMsg::ClearClipboard(self.webview_id()));
293            // Step 1.2
294            for item in drag_data_store.iter_item_list() {
295                match item {
296                    Kind::Text { data, .. } => {
297                        // Step 1.2.1.1 Ensure encoding is correct per OS and locale conventions
298                        // Step 1.2.1.2 Normalize line endings according to platform conventions
299                        // Step 1.2.1.3
300                        self.send_to_embedder(EmbedderMsg::SetClipboardText(
301                            self.webview_id(),
302                            data.to_string(),
303                        ));
304                    },
305                    Kind::File { .. } => {
306                        // Step 1.2.2 If data is of a type listed in the mandatory data types list, then
307                        // Step 1.2.2.1 Place part on clipboard with the appropriate OS clipboard format description
308                        // Step 1.2.3 Else this is left to the implementation
309                    },
310                }
311            }
312        } else {
313            // Step 2.1
314            if drag_data_store.clear_was_called {
315                // Step 2.1.1 If types-to-clear list is empty, clear the clipboard
316                self.send_to_embedder(EmbedderMsg::ClearClipboard(self.webview_id()));
317                // Step 2.1.2 Else remove the types in the list from the clipboard
318                // As of now this can't be done with Arboard, and it's possible that will be removed from the spec
319            }
320        }
321    }

323    /// <https://w3c.github.io/editing/docs/execCommand/#additional-requirements>
324    ///
325    /// This function does not do any checks for whether or not we are actually inside an
326    /// editing host, since those checks are performed by exec_command_for_command_id
327    /// either way.
328    pub(crate) fn perform_editing_action(
329        &self,
330        cx: &mut JSContext,
331        editing_action: &EditingAction,
332    ) -> bool {
333        if !servo_config::pref!(dom_exec_command_enabled) {
334            return false;
335        }
336
337        // This function does not do any checks for whether or not we are actually inside an
338        // editing host, since those checks are performed by exec_command_for_command_id either way.
339        let (command, argument) = match editing_action {
340            EditingAction::MoveCursor(..) => return false,
341            EditingAction::SelectAll | EditingAction::Clipboard(_) => {
342                unreachable!("Should have been handled before this point.")
343            },
344            EditingAction::InsertNewline => {
345                // > When the user instructs the user agent to insert a line break inside an
346                // > editing host without breaking out of the current block, such as by
347                // > pressing Shift-Enter or Option-Enter while the cursor is in an editable
348                // > node, the user agent must call execCommand("insertlinebreak") on the
349                // > relevant document.
350                (DOMString::from_static("insertlinebreak"), DOMString::new())
351            },
352            EditingAction::InsertParagraph => {
353                // > When the user instructs the user agent to insert a line break inside an
354                // > editing host, such as by pressing the Enter key while the cursor is in an
355                // > editable node, the user agent must call execCommand("insertparagraph") on
356                // > the relevant document.
357                (DOMString::from_static("insertparagraph"), DOMString::new())
358            },
359            // > When the user instructs the user agent to delete the previous character inside an
360            // > editing host, such as by pressing the Backspace key while the cursor is in an
361            // > editable node, the user agent must call execCommand("delete") on the relevant
362            // > document.
363            //
364            // TODO: Handle other types of motions here.
365            EditingAction::Backspace(..) => (DOMString::from_static("delete"), DOMString::new()),
366            // > When the user instructs the user agent to delete the next character inside an
367            // > editing host, such as by pressing the Delete key while the cursor is in an
368            // > editable node, the user agent must call execCommand("forwarddelete") on the
369            // > relevant document
370            EditingAction::Delete => (DOMString::from_static("forwarddelete"), DOMString::new()),
371            // > When the user instructs the user agent to insert text inside an editing host, such
372            // > as by typing on the keyboard while the cursor is in an editable node, the user
373            // > agent must call execCommand("inserttext", false, value) on the relevant document,
374            // > with value equal to the text the user provided. If the user inserts multiple
375            // > characters at once or in quick succession, this specification does not define
376            // > whether it is treated as one insertion or several consecutive insertions.
377            EditingAction::InsertText(text) => (
378                DOMString::from_static("inserttext"),
379                DOMString::from(text.as_str()),
380            ),
381        };
382
383        self.exec_command_for_command_id(cx, command, argument)
384    }

406impl TryFrom<&Node> for EditingContext {
407    type Error = ();
408
409    fn try_from(node: &Node) -> Result<Self, Self::Error> {
410        if let Some(text_area) = node.downcast::<HTMLTextAreaElement>() {
411            return Ok(EditingContext::TextControl(
412                TextControlElementEditingContext::TextArea(DomRoot::from_ref(text_area)),
413            ));
414        }
415        if let Some(input) = node.downcast::<HTMLInputElement>() &&
416            input.is_textual_or_password()
417        {
418            return Ok(EditingContext::TextControl(
419                TextControlElementEditingContext::Input(DomRoot::from_ref(input)),
420            ));
421        }
422        Err(())
423    }

558        match self {

563                };
564                if let Some(node) = document
565                    .GetBody()
566                    .map(DomRoot::upcast::<Node>)
567                    .or_else(|| document.GetDocumentElement().map(DomRoot::upcast::<Node>))
568                {
569                    let _ = selection.SelectAllChildren(cx, &node);
570                }
571            },
572        }
573    }

575    pub(crate) fn perform_editing_action(&self, cx: &mut JSContext, action: EditingAction) -> bool {
576        // A couple editing actions can be performed via the external interface of
577        // `EditingContext`. If that's the case we want to do that so that the same
578        // code path is used regardless of how the action was executed.
579        if let EditingAction::Clipboard(clipboard_action) = action {
580            return self
581                .document()
582                .handle_clipboard_action(cx, self, clipboard_action)
583                .contains(InputEventResult::Consumed);
584        }
585
586        if let EditingAction::SelectAll = action {
587            self.select_all(cx);
588            return true;
589        }
590
591        match self {
592            EditingContext::TextControl(element) => element
593                .text_control_element()
594                .perform_editing_action(cx, action),
595            EditingContext::Document(document) => document.perform_editing_action(cx, &action),
596        }
597    }

600pub(crate) fn editing_action_from_keyboard_event(
601    event: &KeyboardTypesEvent,
602) -> Option<EditingAction> {
603    let mut modifiers = event.modifiers;
604    if let Some(action) = ShortcutMatcher::new(event.state, event.key.clone(), modifiers)
605        .shortcut(CMD_OR_CONTROL, 'A', || Some(EditingAction::SelectAll))
606        .shortcut(CMD_OR_CONTROL, 'C', || {
607            Some(EditingAction::Clipboard(ClipboardAction::Copy))
608        })
609        .shortcut(CMD_OR_CONTROL, 'X', || {
610            Some(EditingAction::Clipboard(ClipboardAction::Cut))
611        })
612        .shortcut(CMD_OR_CONTROL, 'V', || {
613            Some(EditingAction::Clipboard(ClipboardAction::Paste))
614        })
615        // TODO: Figure out if the bit about Option+Enter works and whether or not this
616        //       ends up providing the correct behavior on Mac. (i.e. whether or not
617        //       Shift should be accepted in addition to Option.)
618        .shortcut(Modifiers::SHIFT, Key::Named(NamedKey::Enter), || {
619            Some(EditingAction::InsertNewline)
620        })
621        .shortcut(Modifiers::empty(), Key::Named(NamedKey::Enter), || {
622            Some(EditingAction::InsertParagraph)
623        })
624        .otherwise(|| None)
625        .flatten()
626    {
627        return Some(action);
628    };
629
630    let update_selection = if modifiers.contains(Modifiers::SHIFT) {
631        ModifySelection::Yes
632    } else {
633        ModifySelection::No
634    };
635    modifiers.remove(Modifiers::SHIFT);
636
637    ShortcutMatcher::new(KeyState::Down, event.key.clone(), modifiers)
638        .shortcut(Modifiers::empty(), Key::Named(NamedKey::ArrowLeft), || {
639            Some(EditingAction::MoveCursor(
640                EditingDirection::Backward,
641                EditingMotion::Grapheme,
642                update_selection,
643            ))
644        })
645        .shortcut(Modifiers::empty(), Key::Named(NamedKey::ArrowRight), || {
646            Some(EditingAction::MoveCursor(
647                EditingDirection::Forward,
648                EditingMotion::Grapheme,
649                update_selection,
650            ))
651        })
652        .shortcut(Modifiers::empty(), Key::Named(NamedKey::ArrowUp), || {
653            Some(EditingAction::MoveCursor(
654                EditingDirection::Backward,
655                EditingMotion::Line,
656                update_selection,
657            ))
658        })
659        .shortcut(Modifiers::empty(), Key::Named(NamedKey::ArrowDown), || {
660            Some(EditingAction::MoveCursor(
661                EditingDirection::Forward,
662                EditingMotion::Line,
663                update_selection,
664            ))
665        })
666        .shortcut(Modifiers::CONTROL | Modifiers::ALT, 'B', || {
667            Some(EditingAction::MoveCursor(
668                EditingDirection::Backward,
669                EditingMotion::Word,
670                update_selection,
671            ))
672        })
673        .shortcut(ALT_OR_CONTROL, Key::Named(NamedKey::ArrowLeft), || {
674            Some(EditingAction::MoveCursor(
675                EditingDirection::Backward,
676                EditingMotion::Word,
677                update_selection,
678            ))
679        })
680        .shortcut(Modifiers::CONTROL | Modifiers::ALT, 'F', || {
681            Some(EditingAction::MoveCursor(
682                EditingDirection::Forward,
683                EditingMotion::Word,
684                update_selection,
685            ))
686        })
687        .shortcut(ALT_OR_CONTROL, Key::Named(NamedKey::ArrowRight), || {
688            Some(EditingAction::MoveCursor(
689                EditingDirection::Forward,
690                EditingMotion::Word,
691                update_selection,
692            ))
693        })
694        .shortcut(Modifiers::empty(), Key::Named(NamedKey::Home), || {
695            Some(EditingAction::MoveCursor(
696                EditingDirection::Backward,
697                EditingMotion::LineStartOrEnd,
698                update_selection,
699            ))
700        })
701        .optional_shortcut(cfg!(target_os = "macos"), Modifiers::CONTROL, 'A', || {
702            Some(EditingAction::MoveCursor(
703                EditingDirection::Backward,
704                EditingMotion::LineStartOrEnd,
705                update_selection,
706            ))
707        })
708        .shortcut(Modifiers::empty(), Key::Named(NamedKey::End), || {
709            Some(EditingAction::MoveCursor(
710                EditingDirection::Forward,
711                EditingMotion::LineStartOrEnd,
712                update_selection,
713            ))
714        })
715        .optional_shortcut(cfg!(target_os = "macos"), Modifiers::CONTROL, 'E', || {
716            Some(EditingAction::MoveCursor(
717                EditingDirection::Backward,
718                EditingMotion::LineStartOrEnd,
719                update_selection,
720            ))
721        })
722        .optional_shortcut(
723            cfg!(target_os = "macos"),
724            Modifiers::META,
725            Key::Named(NamedKey::ArrowLeft),
726            || {
727                Some(EditingAction::MoveCursor(
728                    EditingDirection::Backward,
729                    EditingMotion::LineStartOrEnd,
730                    update_selection,
731                ))
732            },
733        )
734        .optional_shortcut(
735            cfg!(target_os = "macos"),
736            Modifiers::META,
737            Key::Named(NamedKey::ArrowRight),
738            || {
739                Some(EditingAction::MoveCursor(
740                    EditingDirection::Forward,
741                    EditingMotion::LineStartOrEnd,
742                    update_selection,
743                ))
744            },
745        )
746        .shortcut(Modifiers::empty(), Key::Named(NamedKey::PageUp), || {
747            Some(EditingAction::MoveCursor(
748                EditingDirection::Backward,
749                EditingMotion::Page,
750                update_selection,
751            ))
752        })
753        .shortcut(Modifiers::empty(), Key::Named(NamedKey::PageDown), || {
754            Some(EditingAction::MoveCursor(
755                EditingDirection::Forward,
756                EditingMotion::Page,
757                update_selection,
758            ))
759        })
760        .shortcut(Modifiers::empty(), Key::Named(NamedKey::Delete), || {
761            Some(EditingAction::Delete)
762        })
763        .shortcut(Modifiers::empty(), Key::Named(NamedKey::Backspace), || {
764            Some(EditingAction::Backspace(EditingMotion::Grapheme))
765        })
766        .shortcut(ALT_OR_CONTROL, Key::Named(NamedKey::Backspace), || {
767            Some(EditingAction::Backspace(EditingMotion::Word))
768        })
769        .otherwise(|| match &event.key {
770            Key::Character(character) if modifiers.is_empty() => {
771                Some(EditingAction::InsertText(character.to_string()))
772            },
773            _ => None,
774        })
775        .flatten()

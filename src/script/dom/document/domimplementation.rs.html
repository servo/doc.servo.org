32// https://dom.spec.whatwg.org/#domimplementation
33#[dom_struct]
34pub(crate) struct DOMImplementation {
35    reflector_: Reflector,
36    document: Dom<Document>,
37}

39impl DOMImplementation {
40    fn new_inherited(document: &Document) -> DOMImplementation {
41        DOMImplementation {
42            reflector_: Reflector::new(),
43            document: Dom::from_ref(document),
44        }
45    }
46
47    pub(crate) fn new(cx: &mut JSContext, document: &Document) -> DomRoot<DOMImplementation> {
48        let window = document.window();
49        reflect_dom_object(
50            cx,
51            Box::new(DOMImplementation::new_inherited(document)),
52            window,
53        )
54    }
55}

58impl DOMImplementationMethods<crate::DomTypeHolder> for DOMImplementation {
59    /// <https://dom.spec.whatwg.org/#dom-domimplementation-createdocumenttype>
60    fn CreateDocumentType(
61        &self,
62        cx: &mut js::context::JSContext,
63        qualified_name: DOMString,
64        pubid: DOMString,
65        sysid: DOMString,
66    ) -> Fallible<DomRoot<DocumentType>> {
67        // Step 1. If name is not a valid doctype name, then throw an
68        //      "InvalidCharacterError" DOMException.
69        if !is_valid_doctype_name(&qualified_name) {
70            debug!("Not a valid doctype name");
71            return Err(Error::InvalidCharacter(None));
72        }
73
74        Ok(DocumentType::new(
75            cx,
76            qualified_name,
77            Some(pubid),
78            Some(sysid),
79            &self.document,
80        ))
81    }

83    /// <https://dom.spec.whatwg.org/#dom-domimplementation-createdocument>
84    fn CreateDocument(
85        &self,
86        cx: &mut JSContext,
87        maybe_namespace: Option<DOMString>,
88        qname: DOMString,
89        maybe_doctype: Option<&DocumentType>,
90    ) -> Fallible<DomRoot<XMLDocument>> {
91        let win = self.document.window();
92        let loader = DocumentLoader::new(&self.document.loader());
93        let namespace = namespace_from_domstring(maybe_namespace.to_owned());
94
95        let content_type = match namespace {
96            ns!(html) => "application/xhtml+xml",
97            ns!(svg) => "image/svg+xml",
98            _ => "application/xml",
99        }
100        .parse()
101        .unwrap();
102
103        // Step 1. Let document be a new XMLDocument.
104        let doc = XMLDocument::new(
105            cx,
106            win,
107            HasBrowsingContext::No,
108            None,
109            self.document.origin().clone(),
110            IsHTMLDocument::NonHTMLDocument,
111            Some(content_type),
112            None,
113            DocumentActivity::Inactive,
114            loader,
115            Some(self.document.insecure_requests_policy()),
116            self.document.has_trustworthy_ancestor_or_current_origin(),
117            self.document.custom_element_reaction_stack(),
118            self.document.image_cache(),
119        );
120
121        // Step 2. Let element be null.
122        // Step 3. If qualifiedName is not the empty string, then set element to the result of running
123        // the internal createElementNS steps, given document, namespace, qualifiedName, and an empty dictionary.
124        let maybe_elem = if qname.is_empty() {
125            None
126        } else {
127            let options =
128                StringOrElementCreationOptions::ElementCreationOptions(ElementCreationOptions {
129                    is: None,
130                });
131            Some(
132                doc.upcast::<Document>()
133                    .CreateElementNS(cx, maybe_namespace, qname, options)?,
134            )
135        };
136
137        {
138            let doc_node = doc.upcast::<Node>();
139
140            // Step 4.
141            if let Some(doc_type) = maybe_doctype {
142                doc_node.AppendChild(cx, doc_type.upcast()).unwrap();
143            }
144
145            // Step 5.
146            if let Some(ref elem) = maybe_elem {
147                doc_node.AppendChild(cx, elem.upcast()).unwrap();
148            }
149        }
150
151        // Step 6.
152        // The origin is already set
153
154        // Step 7.
155        Ok(doc)
156    }

158    /// <https://dom.spec.whatwg.org/#dom-domimplementation-createhtmldocument>
159    fn CreateHTMLDocument(
160        &self,
161        cx: &mut JSContext,
162        title: Option<DOMString>,
163    ) -> DomRoot<Document> {
164        let win = self.document.window();
165        let loader = DocumentLoader::new(&self.document.loader());
166
167        // Step 1. Let doc be a new document that is an HTML document.
168        // Step 2. Set doc’s content type to "text/html".
169        let doc = Document::new(
170            cx,
171            win,
172            HasBrowsingContext::No,
173            None,
174            None,
175            // Step 8. doc’s origin is this’s associated document’s origin.
176            self.document.origin().clone(),
177            IsHTMLDocument::HTMLDocument,
178            None,
179            None,
180            DocumentActivity::Inactive,
181            loader,
182            None,
183            None,
184            Default::default(),
185            false,
186            self.document.allow_declarative_shadow_roots(),
187            Some(self.document.insecure_requests_policy()),
188            self.document.has_trustworthy_ancestor_or_current_origin(),
189            self.document.custom_element_reaction_stack(),
190            self.document.creation_sandboxing_flag_set(),
191            self.document.pipeline_id(),
192            self.document.image_cache(),
193        );
194
195        {
196            // Step 3. Append a new doctype, with "html" as its name and with its node document set to doc, to doc.
197            let doc_node = doc.upcast::<Node>();
198            let doc_type = DocumentType::new(cx, DOMString::from_static("html"), None, None, &doc);
199            doc_node.AppendChild(cx, doc_type.upcast()).unwrap();
200        }
201
202        {
203            // Step 4. Append the result of creating an element given doc, "html",
204            // and the HTML namespace, to doc.
205            let doc_node = doc.upcast::<Node>();
206            let doc_html = DomRoot::upcast::<Node>(Element::create(
207                cx,
208                QualName::new(None, ns!(html), local_name!("html")),
209                None,
210                &doc,
211                ElementCreator::ScriptCreated,
212                CustomElementCreationMode::Asynchronous,
213                None,
214            ));
215            doc_node
216                .AppendChild(cx, &doc_html)
217                .expect("Appending failed");
218
219            {
220                // Step 5. Append the result of creating an element given doc, "head",
221                // and the HTML namespace, to the html element created earlier.
222                let doc_head = DomRoot::upcast::<Node>(Element::create(
223                    cx,
224                    QualName::new(None, ns!(html), local_name!("head")),
225                    None,
226                    &doc,
227                    ElementCreator::ScriptCreated,
228                    CustomElementCreationMode::Asynchronous,
229                    None,
230                ));
231                doc_html.AppendChild(cx, &doc_head).unwrap();
232
233                // Step 6. If title is given:
234                if let Some(title_str) = title {
235                    // Step 6.1. Append the result of creating an element given doc, "title",
236                    // and the HTML namespace, to the head element created earlier.
237                    let doc_title = DomRoot::upcast::<Node>(Element::create(
238                        cx,
239                        QualName::new(None, ns!(html), local_name!("title")),
240                        None,
241                        &doc,
242                        ElementCreator::ScriptCreated,
243                        CustomElementCreationMode::Asynchronous,
244                        None,
245                    ));
246                    doc_head.AppendChild(cx, &doc_title).unwrap();
247
248                    // Step 6.2. Append a new Text node, with its data set to title (which could be the empty string)
249                    // and its node document set to doc, to the title element created earlier.
250                    let title_text = Text::new(cx, title_str, &doc);
251                    doc_title.AppendChild(cx, title_text.upcast()).unwrap();
252                }
253            }
254
255            // Step 7. Append the result of creating an element given doc, "body",
256            // and the HTML namespace, to the html element created earlier.
257            let doc_body = Element::create(
258                cx,
259                QualName::new(None, ns!(html), local_name!("body")),
260                None,
261                &doc,
262                ElementCreator::ScriptCreated,
263                CustomElementCreationMode::Asynchronous,
264                None,
265            );
266            doc_html.AppendChild(cx, doc_body.upcast()).unwrap();
267        }
268
269        // Step 9. Return doc.
270        doc
271    }

34/// The [`DocumentOverlayHandler`] is a structure responsible for overlays for
35/// the [`crate::Document`] and storing data related to overlays. It exists to
36/// decrease the size of the [`crate::Document`] structure.

82    pub(crate) fn fullscreen_element(&self) -> Option<DomRoot<Element>> {
83        // > The fullscreen element is the topmost element in the document’s top layer whose
84        // > fullscreen flag is set, if any, and null otherwise.
85        self.top_layer

110    /// <https://fullscreen.spec.whatwg.org/#fullscreen-an-element>
111    pub(crate) fn fullscreen_an_element(&self, cx: &mut JSContext, element: &Element) {
112        // Step 1. Let hideUntil be the result of running topmost popover ancestor given element,
113        // null, and false.
114        let hide_until = element.topmost_popover_ancestor(cx.no_gc(), None, false);
115
116        // Step 2. Run hide popovers until given element’s node document, hideUntil, false, and
117        // true.
118        element
119            .owner_document()
120            .overlay_handler()
121            .hide_popovers_until(cx, hide_until.as_deref(), false, true);
122
123        // Step 3. Set element’s fullscreen flag.
124        element.set_fullscreen_state(cx.no_gc(), true);
125
126        // Step 4. Remove from the top layer immediately given element.
127        self.remove_an_element_from_the_top_layer_immediately(element, cx.no_gc());
128
129        // Step 5. Add to the top layer given element.
130        self.add_an_element_to_the_top_layer(element, cx.no_gc());
131    }

133    /// <https://fullscreen.spec.whatwg.org/#unfullscreen-an-element>
134    pub(crate) fn unfullscreen_an_element(&self, no_gc: &NoGC, element: &Element) {
135        // > To unfullscreen an element, unset element’s fullscreen flag, iframe fullscreen flag
136        // > (if any), and keyboard lock flag, and remove from the top layer immediately given
137        // > element.
138        element.set_fullscreen_state(no_gc, false);
139        if let Some(iframe) = element.downcast::<HTMLIFrameElement>() {
140            iframe.set_iframe_fullscreen_flag(false);
141        }
142        // FIXME: Unset element’s keyboard lock flag.
143        self.remove_an_element_from_the_top_layer_immediately(element, no_gc);
144    }

146    /// <https://fullscreen.spec.whatwg.org/#unfullscreen-a-document>
147    pub(crate) fn unfullscreen_a_document(&self, no_gc: &NoGC) {
148        // > To unfullscreen a document, unfullscreen all elements, within document’s top layer,
149        // > whose fullscreen flag is set.
150        let fullscreened_elements = self
151            .top_layer
152            .borrow()
153            .iter()
154            .filter(|element| element.fullscreen_flag())
155            .map(|element| element.as_unrooted(no_gc))
156            .collect::<Vec<_>>();
157        for element in fullscreened_elements {
158            self.unfullscreen_an_element(no_gc, &element);
159        }
160    }

162    /// <https://fullscreen.spec.whatwg.org/#fully-exit-fullscreen>
163    pub(crate) fn fully_exit_fullscreen(&self, cx: &mut JSContext) {
164        // Step 1. If document’s fullscreen element is null, terminate these steps.
165        let Some(fullscreen_element) = self.fullscreen_element() else {
166            return;
167        };
168
169        // Step 2. Unfullscreen elements whose fullscreen flag is set, within document’s top
170        // layer, except for document’s fullscreen element.
171        let elements = self
172            .top_layer
173            .borrow()
174            .iter()
175            .filter(|element| element.fullscreen_flag() && ***element != *fullscreen_element)
176            .map(|element| element.as_unrooted(cx.no_gc()))
177            .collect::<Vec<_>>();
178        for element in elements {
179            self.unfullscreen_an_element(cx.no_gc(), &element);
180        }
181
182        // Step 3. Exit fullscreen document.
183        let _ = self.exit_fullscreen(cx);
184    }

193    /// <https://fullscreen.spec.whatwg.org/#run-the-fullscreen-steps>
194    pub(crate) fn run_the_fullscreen_steps(&self, cx: &mut JSContext) {
195        // Step 1. Let pendingEvents be a clone of document’s list of pending fullscreen events.
196        // Step 2. Empty document’s list of pending fullscreen events.
197        let pending_events: Vec<(Atom, DomRoot<Element>)> = self
198            .list_of_pending_fullscreen_events
199            .safe_borrow_mut(cx.no_gc())
200            .drain(..)
201            .map(|event| (event.event_type.clone(), event.element.as_rooted()))
202            .collect();
203
204        // Step 3. For each (type, element) in pendingEvents:
205        for (event_type, element) in pending_events {
206            // Step 3.1. Let target be element if element is connected and its node document is
207            // document, and otherwise let target be document.
208            let target: DomRoot<EventTarget> =
209                if element.is_connected() && *element.owner_document() == *self.document {
210                    DomRoot::upcast(element)
211                } else {
212                    DomRoot::upcast(self.document.as_rooted())
213                };
214
215            // Step 3.2. Fire an event named type, with its bubbles and composed attributes set to
216            // true, at target.
217            target.fire_event_with_params(
218                cx,
219                event_type,
220                EventBubbles::Bubbles,
221                EventCancelable::NotCancelable,
222                EventComposed::Composed,
223            );
224        }
225    }

227    /// <https://fullscreen.spec.whatwg.org/#simple-fullscreen-document>
228    pub(crate) fn is_simple_fullscreen_document(&self) -> bool {
229        // > A document is said to be a simple fullscreen document if there is exactly one element
230        // > in its top layer that has its fullscreen flag set.
231        self.top_layer
232            .borrow()
233            .iter()
234            .filter(|element| element.fullscreen_flag())
235            .count() ==
236            1
237    }

239    /// <https://fullscreen.spec.whatwg.org/#collect-documents-to-unfullscreen>
240    pub(crate) fn collect_documents_to_unfullscreen(&self) -> Vec<DomRoot<Document>> {
241        // Step 1. Let docs be an ordered set consisting of doc.
242        let mut docs = vec![self.document.as_rooted()];
243
244        // Step 2. While true:
245        loop {
246            // Step 2.1. Let lastDoc be docs’s last document.
247            let last_doc = docs
248                .last()
249                .expect("docs is initialized with an element")
250                .clone();
251
252            // Step 2.2. Assert: lastDoc’s fullscreen element is not null.
253            assert!(last_doc.overlay_handler().fullscreen_element().is_some());
254
255            // Step 2.3. If lastDoc is not a simple fullscreen document, break.
256            if !last_doc.overlay_handler().is_simple_fullscreen_document() {
257                break;
258            }
259
260            // Step 2.4. Let container be lastDoc’s node navigable’s container.
261            // Step 2.5. If container is null, then break.
262            //
263            // FIXME: Support cross-origin container
264            let window_proxy = last_doc.window().window_proxy();
265            let Some(container) = window_proxy.frame_element() else {
266                break;
267            };
268
269            // Also break when container's content navigable is no longer lastDoc's node navigable.
270            // This happens when that navigable is being destroyed. lastDoc's navigable then has no
271            // parent anymore, even though it still refers to its former container, so the lastDoc
272            // is treated as having no container.
273            if container
274                .downcast::<HTMLIFrameElement>()
275                .is_some_and(|iframe| {
276                    iframe.browsing_context_id() != Some(window_proxy.browsing_context_id())
277                })
278            {
279                break;
280            }
281
282            // Step 2.6. If container’s iframe fullscreen flag is set, break.
283            if container
284                .downcast::<HTMLIFrameElement>()
285                .is_some_and(|iframe| iframe.iframe_fullscreen_flag())
286            {
287                break;
288            }
289
290            // Step 2.7. Append container’s node document to docs.
291            docs.push(container.owner_document());
292        }
293
294        // Step 3. Return docs.
295        docs
296    }

298    /// <https://fullscreen.spec.whatwg.org/#exit-fullscreen>
299    pub(crate) fn exit_fullscreen(&self, cx: &mut JSContext) -> RootedPromise {
300        // Step 1. Let promise be a new promise.
301        let mut realm = CurrentRealm::assert(cx);
302        let promise = Promise::new_in_realm(&mut realm);
303
304        // Step 2. If doc is not fully active or doc’s fullscreen element is null, then reject
305        // promise with a TypeError exception and return promise.
306        let mut doc = self.document.as_rooted();
307        if !doc.is_fully_active() || self.fullscreen_element().is_none() {
308            promise.reject_error(
309                cx,
310                Error::Type(
311                    c"No fullscreen element to exit or document is not fully active".to_owned(),
312                ),
313            );
314            return promise;
315        }
316
317        // Step 3. Let resize be false.
318        let mut resize = false;
319
320        // Step 4. Let docs be the result of collecting documents to unfullscreen given doc.
321        let docs = self.collect_documents_to_unfullscreen();
322
323        // Step 5. Let topLevelDoc be doc’s node navigable’s top-level traversable’s active
324        // document.
325        // FIXME: Support cross-origin document
326        let top_level_doc = doc.window().top_level_document_if_local();
327
328        // Step 6. If topLevelDoc is in docs, and it is a simple fullscreen document, then set doc
329        // to topLevelDoc and resize to true.
330        if let Some(top_level_doc) = top_level_doc &&
331            docs.contains(&top_level_doc) &&
332            top_level_doc
333                .overlay_handler()
334                .is_simple_fullscreen_document()
335        {
336            doc = top_level_doc;
337            resize = true;
338        }
339
340        // Step 7. If doc’s fullscreen element is not connected:
341        if let Some(fullscreen_element) = doc.overlay_handler().fullscreen_element() &&
342            !fullscreen_element.is_connected()
343        {
344            // Step 7.1. Append (fullscreenchange, doc’s fullscreen element) to doc’s list of
345            // pending fullscreen events.
346            doc.overlay_handler()
347                .append_to_list_of_pending_fullscreen_events(
348                    cx.no_gc(),
349                    atom!("fullscreenchange"),
350                    &fullscreen_element,
351                );
352
353            // Step 7.2. Unfullscreen doc’s fullscreen element.
354            doc.overlay_handler()
355                .unfullscreen_an_element(cx.no_gc(), &fullscreen_element);
356        }
357
358        // Step 8. Return promise, and run the remaining steps in parallel.
359
360        // FIXME: Step 9. Run the fully unlock the screen orientation steps with doc.
361
362        // Step 10. If resize is true, resize doc’s viewport to its "normal" dimensions.
363        //
364        // This is sent right away instead of from the queued task below, as the task never runs
365        // when the document is being unloaded. The screen would then be stuck in fullscreen. The
366        // message is processed in parallel, so this aligns with the spec.
367        if resize {
368            doc.send_to_embedder(EmbedderMsg::NotifyFullscreenStateChanged(
369                doc.webview_id(),
370                false,
371            ));
372        }
373
374        let trusted_doc = Trusted::new(&*doc);
375        let trusted_promise = TrustedPromise::from(&promise);
376        doc.owner_global()
377            .task_manager()
378            .dom_manipulation_task_source()
379            .queue(task!(exit_fullscreen_steps: move |cx| {
380                let doc = trusted_doc.root();
381                let promise = trusted_promise.root(cx);
382
383                // Step 11. If doc’s fullscreen element is null, then resolve promise with undefined
384                // and terminate these steps.
385                if doc.overlay_handler().fullscreen_element().is_none() {
386                    promise.resolve_native(cx, &());
387                    return;
388                }
389
390                // Step 12. Let exitDocs be the result of collecting documents to unfullscreen given
391                // doc.
392                let exit_docs = doc.overlay_handler().collect_documents_to_unfullscreen();
393
394                // Step 13. Let descendantDocs be an ordered set consisting of doc’s descendant
395                // navigables' active documents whose fullscreen element is non-null, if any, in
396                // tree order.
397                let descendant_docs = LocalDescendantNavigablesIterator::new(&doc)
398                    .filter_map(|descendant_doc| {
399                        let element = descendant_doc.overlay_handler().fullscreen_element()?;
400                        Some((descendant_doc, element))
401                    })
402                    .collect::<Vec<_>>();
403
404                // Step 14. For each exitDoc in exitDocs:
405                for exit_doc in exit_docs {
406                    let fullscreen_element = exit_doc.overlay_handler().fullscreen_element().expect("The document is asserted to have a fullscreen element in collect_documents_to_unfullscreen");
407
408                    // Step 14.1. Append (fullscreenchange, exitDoc’s fullscreen element) to
409                    // exitDoc’s list of pending fullscreen events.
410                    exit_doc
411                        .overlay_handler()
412                        .append_to_list_of_pending_fullscreen_events(
413                            cx.no_gc(),
414                            atom!("fullscreenchange"),
415                            &fullscreen_element,
416                        );
417
418                    // Step 14.2. If resize is true, unfullscreen exitDoc.
419                    if resize {
420                        exit_doc.overlay_handler().unfullscreen_a_document(cx.no_gc());
421                    }
422                    // Step 14.3. Otherwise, unfullscreen exitDoc’s fullscreen element.
423                    else {
424                        exit_doc
425                            .overlay_handler()
426                            .unfullscreen_an_element(cx.no_gc(), &fullscreen_element);
427                    }
428                }
429
430                // Step 15. For each descendantDoc in descendantDocs:
431                for (descendant_doc, fullscreen_element) in descendant_docs {
432                    // Step 15.1. Append (fullscreenchange, descendantDoc’s fullscreen element) to
433                    // descendantDoc’s list of pending fullscreen events.
434                    descendant_doc
435                        .overlay_handler()
436                        .append_to_list_of_pending_fullscreen_events(
437                            cx.no_gc(),
438                            atom!("fullscreenchange"),
439                            &fullscreen_element,
440                        );
441
442                    // Step 15.2. Unfullscreen descendantDoc.
443                    descendant_doc
444                        .overlay_handler()
445                        .unfullscreen_a_document(cx.no_gc());
446                }
447
448                // Step 16. Resolve promise with undefined.
449                promise.resolve_native(cx, &());
450            }));
451
452        promise
453    }

455    /// <https://html.spec.whatwg.org/multipage/#allowed-to-use>
456    pub(crate) fn is_allowed_to_use_fullscreen(&self) -> bool {
457        // Step 1. If document's browsing context is null, then return false.
458        if self.document.browsing_context().is_none() {
459            return false;
460        };
461
462        // Step 2. If document is not fully active, then return false.
463        if !self.document.is_fully_active() {
464            return false;
465        }
466
467        // 3. If the result of running is feature enabled in document for origin on feature,
468        // document, and document's origin is "Enabled", then return true.
469        //
470        // FIXME: Properly check if feature is enabled, for now we do a makeshift attribute check
471        // for iFrames and otherwise assume it always is.
472        if self.document.window().is_top_level() {
473            true
474        } else {
475            self.document
476                .window()
477                .GetFrameElement()
478                .is_some_and(|el| el.has_attribute(&local_name!("allowfullscreen")))
479        }
480
481        // FIXME: Return false.
482    }

540    /// <https://drafts.csswg.org/css-position-4/#in-the-top-layer>
541    pub(crate) fn is_in_the_top_layer(&self, element: &Element) -> bool {
542        // > An element el is in the top layer if el is contained in its node document’s top layer
543        // > but not contained in its node document’s pending top layer removals.
544        self.top_layer_contains(element) && !self.pending_top_layer_removals_contains(element)
545    }

547    /// <https://drafts.csswg.org/css-position-4/#add-an-element-to-the-top-layer>
548    pub(crate) fn add_an_element_to_the_top_layer(&self, element: &Element, no_gc: &NoGC) {
549        // Step 1. Let doc be el’s node document.
550
551        // Step 2. If el is already contained in doc's top layer:
552        if self.top_layer_contains(element) {
553            // Step 2.1. Assert: el is also in doc's pending top layer removals. (Otherwise, this is
554            // a spec error.)
555            assert!(self.pending_top_layer_removals_contains(element));
556
557            // Step 2.2. Remove el from both doc's top layer and pending top layer removals.
558            self.top_layer
559                .safe_borrow_mut(no_gc)
560                .retain(|top_layer_element| &**top_layer_element != element);
561            self.pending_top_layer_removals
562                .safe_borrow_mut(no_gc)
563                .retain(|pending_element| &**pending_element != element);
564        }
565
566        // Step 3. Append el to doc's top layer.
567        self.top_layer
568            .safe_borrow_mut(no_gc)
569            .push(Dom::from_ref(element));
570
571        // TODO: Step 4. At the UA !important cascade origin, add a rule targeting el
572        // containing an overlay: auto declaration.
573
574        self.update_topmost_modal(no_gc);
575    }

577    /// <https://drafts.csswg.org/css-position-4/#request-an-element-to-be-removed-from-the-top-layer>
578    pub(crate) fn request_an_element_to_be_removed_from_the_top_layer(
579        &self,
580        element: &Element,
581        no_gc: &NoGC,
582    ) {
583        // Step 1. Let doc be el's node document.
584
585        // Step 2. If el is not contained doc's top layer, or el is already contained in doc's
586        // pending top layer removals, return.
587        if !self.top_layer_contains(element) || self.pending_top_layer_removals_contains(element) {
588            return;
589        }
590
591        // TODO: Step 3. Remove the UA !important overlay: auto rule targeting el.
592
593        // Step 4. Append el to doc's pending top layer removals.
594        self.pending_top_layer_removals
595            .safe_borrow_mut(no_gc)
596            .push(Dom::from_ref(element));
597
598        self.update_topmost_modal(no_gc);
599    }

601    /// <https://drafts.csswg.org/css-position-4/#remove-an-element-from-the-top-layer-immediately>
602    pub(crate) fn remove_an_element_from_the_top_layer_immediately(
603        &self,
604        element: &Element,
605        no_gc: &NoGC,
606    ) {
607        // Step 1. Let doc be el's node document.
608
609        // Step 2. Remove el from doc's top layer and pending top layer removals.
610        self.top_layer
611            .safe_borrow_mut(no_gc)
612            .retain(|top_layer_element| &**top_layer_element != element);
613        self.pending_top_layer_removals
614            .safe_borrow_mut(no_gc)
615            .retain(|pending_element| &**pending_element != element);
616
617        // TODO: Step 3. Remove the UA !important overlay: auto rule targeting el, if it exists.
618
619        self.update_topmost_modal(no_gc);
620    }

622    /// <https://drafts.csswg.org/css-position-4/#process-top-layer-removals>
623    pub(crate) fn process_top_layer_removals(&self, cx: &mut JSContext) {
624        // Step 1. For each element el in doc’s pending top layer removals: if el's computed value
625        // of overlay is none, or el is not rendered, remove from the top layer immediately el.
626        // TODO: Check `overlay` once it is supported.
627        let pending_removals: Vec<_> = self
628            .pending_top_layer_removals
629            .borrow()
630            .iter()
631            .map(|element| DomRoot::from_ref(&**element))
632            .collect();
633        for element in pending_removals {
634            if !element.has_css_layout_box_deprecated(cx) {
635                self.remove_an_element_from_the_top_layer_immediately(&element, cx)
636            }
637        }
638    }

640    /// <https://html.spec.whatwg.org/multipage/#blocked-by-a-modal-dialog>
641    pub fn update_topmost_modal(&self, no_gc: &NoGC) {
642        // > A Document document is blocked by a modal dialog subject if subject is the topmost
643        // > dialog element in document's top layer.
644        let new_topmost_modal = self
645            .top_layer
646            .borrow()
647            .iter()
648            .rev()
649            .find(|element| {
650                element.state().contains(ElementState::MODAL) &&
651                    !self.pending_top_layer_removals_contains(element)
652            })
653            .map(|element| DomRoot::from_ref(&**element));
654        let old_topmost_modal = self.topmost_modal.get();
655
656        if let Some(old_topmost_modal) = old_topmost_modal {
657            old_topmost_modal.set_state(no_gc, ElementState::TOPMOST_MODAL, false);
658        }
659        if let Some(new_topmost_modal) = &new_topmost_modal {
660            new_topmost_modal.set_state(no_gc, ElementState::TOPMOST_MODAL, true);
661        }
662
663        self.topmost_modal.set(new_topmost_modal.as_deref());
664
665        // > While document is so blocked, every node that is connected to document, with the
666        // > exception of the subject element and its flat tree descendants, must become inert.
667        if let Some(document_element) = self.document.GetDocumentElement() {
668            let is_inert = new_topmost_modal.is_some() ||
669                document_element.has_attribute(&local_name!("inert"));
670            document_element.set_state(no_gc, ElementState::INERT, is_inert);
671        }
672    }

675    pub(crate) fn is_blocked_by_a_modal_dialog(&self) -> bool {
676        self.topmost_modal.get().is_some()
677    }

679    /// <https://html.spec.whatwg.org/multipage/#hide-popovers-until>
680    pub(crate) fn hide_popovers_until(
681        &self,
682        cx: &mut JSContext,
683        endpoint: Option<&HTMLElement>,
684        focus_previous_element: bool,
685        fire_events: bool,
686    ) {
687        // Step 1. Let endpointIsHint be true if document's showing hint popover list contains
688        // endpoint; otherwise false.
689        let endpoint_is_hint = endpoint.is_some_and(|endpoint| {
690            self.showing_popover_list_contains(cx.no_gc(), PopoverState::Hint, endpoint)
691        });
692
693        // Step 2. Run hide popover stack until given document, endpoint, Hint,
694        // focusPreviousElement, and fireEvents.
695        self.hide_popover_stack_until(
696            cx,
697            endpoint,
698            PopoverState::Hint,
699            focus_previous_element,
700            fire_events,
701        );
702
703        // Step 3. Let autoEndpoint be endpoint.
704        //
705        // Need to keep the reference to the hint_parent alive such that we
706        // can dereference to obtain its endpoint
707        let hint_parent;
708        let mut auto_endpoint = endpoint;
709
710        // Step 4. If endpointIsHint is true, then set autoEndpoint to document's hint stack
711        // parent.
712        if endpoint_is_hint {
713            hint_parent = self.hint_stack_parent.get();
714            auto_endpoint = hint_parent.as_deref();
715        }
716
717        // Step 5. Run hide popover stack until given document, autoEndpoint, Auto,
718        // focusPreviousElement, and fireEvents.
719        self.hide_popover_stack_until(
720            cx,
721            auto_endpoint,
722            PopoverState::Auto,
723            focus_previous_element,
724            fire_events,
725        );
726    }

728    /// <https://html.spec.whatwg.org/multipage/#hide-popover-stack-until>
729    pub(crate) fn hide_popover_stack_until(
730        &self,
731        cx: &mut JSContext,
732        endpoint: Option<&HTMLElement>,
733        stack_type: PopoverState,
734        focus_previous_element: bool,
735        fire_events: bool,
736    ) {
737        // Step 1. Let popoverList be document's showing auto popover list if stackType is Auto,
738        // and document's showing hint popover list otherwise.
739        let popover_list: Vec<DomRoot<HTMLElement>> = self
740            .showing_popover_list_unrooted(cx.no_gc(), stack_type)
741            .iter()
742            .map(|popover| popover.as_rooted())
743            .collect();
744
745        // Step 2. Let lastHideIndex be 0 if popoverList does not contain endpoint; otherwise the
746        // index of endpoint in popoverList plus 1.
747        let last_hide_index = endpoint
748            .and_then(|endpoint| {
749                popover_list
750                    .iter()
751                    .position(|popover| &**popover == endpoint)
752            })
753            .map_or(0, |index| index + 1);
754
755        // Step 3. Let toHide be popoverList from lastHideIndex onwards, in reverse order.
756        let to_hide = popover_list[last_hide_index..].iter().rev();
757
758        // Step 4. Let toRemain be popoverList up to lastHideIndex.
759        let to_remain = &popover_list[..last_hide_index];
760
761        // Step 5. For each popover of toHide, run the hide popover algorithm given popover,
762        // focusPreviousElement, fireEvents, false, and null.
763        for popover in to_hide {
764            let _ = popover.hide_popover(cx, focus_previous_element, fire_events, false, None);
765        }
766
767        // Step 6. Let newPopoverList be document's showing auto popover list if stackType is
768        // Auto, and document's showing hint popover list otherwise.
769        let new_popover_list: Vec<DomRoot<HTMLElement>> = self
770            .showing_popover_list_unrooted(cx.no_gc(), stack_type)
771            .iter()
772            .map(|popover| popover.as_rooted())
773            .collect();
774
775        // Step 7. Let toCheck be newPopoverList in reverse order.
776        let to_check = new_popover_list.iter().rev();
777
778        // Step 8. For each popover of toCheck:
779        for popover in to_check {
780            // Step 8.1. If toRemain contains popover, then continue.
781            if to_remain.contains(popover) {
782                continue;
783            }
784
785            // Step 8.2. Run the hide popover algorithm given popover, focusPreviousElement, false,
786            // false, and null.
787            let _ = popover.hide_popover(cx, focus_previous_element, false, false, None);
788        }
789    }

791    /// <https://html.spec.whatwg.org/multipage/#topmost-auto-or-hint-popover>
792    pub(crate) fn find_the_topmost_auto_or_hint_popover(
793        &self,
794        no_gc: &NoGC,
795    ) -> Option<DomRoot<HTMLElement>> {
796        // Step 1. If document's showing hint popover list is not empty, then return its last
797        // element.
798        // Step 2. If document's showing auto popover list is not empty, then return its last
799        // element.
800        // Step 3. Return null.
801        //
802        // We use pop instead of last to take ownership of its last element. The Vec is temporary,
803        // so mutating it is fine.
804        self.showing_popover_list_unrooted(no_gc, PopoverState::Hint)
805            .pop()
806            .or_else(|| {
807                self.showing_popover_list_unrooted(no_gc, PopoverState::Auto)
808                    .pop()
809            })
810            .map(|popover| popover.as_rooted())
811    }

813    /// <https://html.spec.whatwg.org/multipage/#auto-popover-list>
814    /// <https://html.spec.whatwg.org/multipage/#showing-hint-popover-list>
815    ///
816    /// The spec lists these steps twice, once with "auto" and once with "hint" as the mode in step 2.1.
817    /// `mode` must be [`PopoverState::Auto`] or [`PopoverState::Hint`].
818    pub(crate) fn showing_popover_list_unrooted<'a>(
819        &self,
820        no_gc: &'a NoGC,
821        mode: PopoverState,
822    ) -> Vec<UnrootedDom<'a, HTMLElement>> {
823        debug_assert!(matches!(mode, PopoverState::Auto | PopoverState::Hint));
824
825        // Step 1. Let popovers be « ».
826        // Step 2. For each Element element of document's top layer:
827        // Step 2.1. If all of the following are true:
828        // - element is an HTML element;
829        // - element's opened in popover mode is mode; and
830        // - element's popover visibility state is showing,
831        // - then append element to popovers.
832        // Step 3. Return popovers.
833        self.top_layer
834            .borrow()
835            .iter()
836            .filter_map(|element| element.downcast::<HTMLElement>())
837            .filter(|element| {
838                element
839                    .upcast::<Element>()
840                    .popover_visibility_state_showing() &&
841                    element.opened_in_popover_mode() == Some(mode)
842            })
843            .map(|element| UnrootedDom::from_ref(element, no_gc))
844            .collect()
845    }

847    pub(crate) fn showing_popover_list_contains(
848        &self,
849        no_gc: &NoGC,
850        mode: PopoverState,
851        element: &HTMLElement,
852    ) -> bool {
853        self.showing_popover_list_unrooted(no_gc, mode)
854            .iter()
855            .any(|popover| ***popover == *element)

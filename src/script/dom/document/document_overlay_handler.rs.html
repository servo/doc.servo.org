107    /// <https://fullscreen.spec.whatwg.org/#dom-element-requestfullscreen>
108    pub(crate) fn enter_fullscreen(
109        &self,
110        cx: &mut CurrentRealm,
111        pending: &Element,
112    ) -> RootedPromise {
113        // Step 1
114        // > Let pendingDoc be this’s node document.
115        let document = pending.owner_document();
116
117        // Step 2
118        // > Let promise be a new promise.
119        let promise = Promise::new_in_realm(cx);
120
121        // Step 3
122        // > If pendingDoc is not fully active, then reject promise with a TypeError exception and return promise.
123        if !document.is_fully_active() {
124            promise.reject_error(cx, Error::Type(c"Document is not fully active".to_owned()));
125            return promise;
126        }
127
128        // Step 4
129        // > Let error be false.
130        let mut error = false;
131
132        // Step 5
133        // > If any of the following conditions are false, then set error to true:
134        {
135            // > - This’s namespace is the HTML namespace or this is an SVG svg or MathML math element. [SVG] [MATHML]
136            match *pending.namespace() {
137                ns!(mathml) => {
138                    if *pending.local_name() != local_name!("math") {
139                        error = true;
140                    }
141                },
142                ns!(svg) => {
143                    if *pending.local_name() != local_name!("svg") {
144                        error = true;
145                    }
146                },
147                ns!(html) => (),
148                _ => error = true,
149            }
150
151            // > - This is not a dialog element.
152            if pending.is::<HTMLDialogElement>() {
153                error = true;
154            }
155
156            // > - The fullscreen element ready check for this returns true.
157            if !pending.fullscreen_element_ready_check() {
158                error = true;
159            }
160
161            // > - Fullscreen is supported.
162            // <https://fullscreen.spec.whatwg.org/#fullscreen-is-supported>
163            // > Fullscreen is supported if there is no previously-established user preference, security risk, or platform limitation.
164            // TODO: Add checks for whether fullscreen is supported as definition.
165
166            // > - This’s relevant global object has transient activation or the algorithm is triggered by a user generated orientation change.
167            // TODO: implement screen orientation API
168            if !pending.owner_window().has_transient_activation() {
169                error = true;
170            }
171        }
172
173        if pref!(dom_fullscreen_test) {
174            // For reftests we just take over the current window,
175            // and don't try to really enter fullscreen.
176            info!("Tests don't really enter fullscreen.");
177        } else {
178            // TODO fullscreen is supported
179            // TODO This algorithm is allowed to request fullscreen.
180            warn!("Fullscreen not supported yet");
181        }
182
183        // Step 6
184        // > If error is false, then consume user activation given pendingDoc’s relevant global object.
185        if !error {
186            pending
187                .owner_window_unrooted(cx.no_gc())
188                .consume_user_activation();
189        }
190
191        // Step 8.
192        // > If error is false, then resize pendingDoc’s node navigable’s top-level traversable’s active document’s viewport’s dimensions,
193        // > optionally taking into account options["navigationUI"]:
194        // TODO(#21600): Improve spec compliance of steps 7-13 paralelism.
195        // TODO(#42064): Implement fullscreen options, and ensure that this is spec compliant for all embedder.
196        if !error {
197            let event = EmbedderMsg::NotifyFullscreenStateChanged(document.webview_id(), true);
198            document.send_to_embedder(event);
199        }
200
201        // Step 7
202        // > Return promise, and run the remaining steps in parallel.
203        let pipeline_id = self.document.window().pipeline_id();
204
205        let trusted_pending = Trusted::new(pending);
206        let trusted_pending_doc = Trusted::new(&*document);
207        let trusted_promise = TrustedPromise::from(&promise);
208        let handler = ElementPerformFullscreenEnter::new(
209            trusted_pending,
210            trusted_pending_doc,
211            trusted_promise,
212            error,
213        );
214        let script_msg = CommonScriptMsg::Task(
215            ScriptThreadEventCategory::EnterFullscreen,
216            handler,
217            Some(pipeline_id),
218            TaskSourceName::DOMManipulation,
219        );
220        let msg = MainThreadScriptMsg::Common(script_msg);
221        self.document
222            .window()
223            .main_thread_script_chan()
224            .send(msg)
225            .unwrap();
226
227        promise
228    }

230    /// <https://fullscreen.spec.whatwg.org/#exit-fullscreen>
231    pub(crate) fn exit_fullscreen(&self, cx: &mut JSContext) -> RootedPromise {
232        let document = &*self.document;
233
234        // Step 1
235        // > Let promise be a new promise
236        let mut realm = CurrentRealm::assert(cx);
237        let promise = Promise::new_in_realm(&mut realm);
238
239        // Step 2
240        // > If doc is not fully active or doc’s fullscreen element is null, then reject promise with a TypeError exception and return promise.
241        if !document.is_fully_active() || self.fullscreen_element().is_none() {
242            promise.reject_error(
243                cx,
244                Error::Type(
245                    c"No fullscreen element to exit or document is not fully active".to_owned(),
246                ),
247            );
248            return promise;
249        }
250
251        // TODO(#42067): Implement step 3-7, handling fullscreen's propagation across navigables.
252
253        let element = self.fullscreen_element().unwrap();
254        // Step 10
255        // > If resize is true, resize doc’s viewport to its "normal" dimensions.
256        // TODO(#21600): Improve spec compliance of steps 8-15 paralelism.
257        let event = EmbedderMsg::NotifyFullscreenStateChanged(document.webview_id(), false);
258        document.send_to_embedder(event);
259
260        // Step 8
261        // > Return promise, and run the remaining steps in parallel.
262        let trusted_element = Trusted::new(&*element);
263        let trusted_promise = TrustedPromise::from(&promise);
264        let handler = ElementPerformFullscreenExit::new(trusted_element, trusted_promise);
265        let pipeline_id = Some(self.document.window().pipeline_id());
266        let script_msg = CommonScriptMsg::Task(
267            ScriptThreadEventCategory::ExitFullscreen,
268            handler,
269            pipeline_id,
270            TaskSourceName::DOMManipulation,
271        );
272        let msg = MainThreadScriptMsg::Common(script_msg);
273        self.document
274            .window()
275            .main_thread_script_chan()
276            .send(msg)
277            .unwrap();
278
279        promise
280    }

358    /// <https://drafts.csswg.org/css-position-4/#in-the-top-layer>
359    pub(crate) fn is_in_the_top_layer(&self, element: &Element) -> bool {
360        // > An element el is in the top layer if el is contained in its node document’s top layer
361        // > but not contained in its node document’s pending top layer removals.
362        self.top_layer_contains(element) && !self.pending_top_layer_removals_contains(element)
363    }

365    /// <https://drafts.csswg.org/css-position-4/#add-an-element-to-the-top-layer>
366    pub(crate) fn add_an_element_to_the_top_layer(&self, element: &Element, no_gc: &NoGC) {
367        // Step 1. Let doc be el’s node document.
368
369        // Step 2. If el is already contained in doc's top layer:
370        if self.top_layer_contains(element) {
371            // Step 2.1. Assert: el is also in doc's pending top layer removals. (Otherwise, this is
372            // a spec error.)
373            assert!(self.pending_top_layer_removals_contains(element));
374
375            // Step 2.2. Remove el from both doc's top layer and pending top layer removals.
376            self.top_layer
377                .safe_borrow_mut(no_gc)
378                .retain(|top_layer_element| &**top_layer_element != element);
379            self.pending_top_layer_removals
380                .safe_borrow_mut(no_gc)
381                .retain(|pending_element| &**pending_element != element);
382        }
383
384        // Step 3. Append el to doc's top layer.
385        self.top_layer
386            .safe_borrow_mut(no_gc)
387            .push(Dom::from_ref(element));
388
389        // TODO: Step 4. At the UA !important cascade origin, add a rule targeting el
390        // containing an overlay: auto declaration.
391
392        self.update_topmost_modal(no_gc);
393    }

395    /// <https://drafts.csswg.org/css-position-4/#request-an-element-to-be-removed-from-the-top-layer>
396    pub(crate) fn request_an_element_to_be_removed_from_the_top_layer(
397        &self,
398        element: &Element,
399        no_gc: &NoGC,
400    ) {
401        // Step 1. Let doc be el's node document.
402
403        // Step 2. If el is not contained doc's top layer, or el is already contained in doc's
404        // pending top layer removals, return.
405        if !self.top_layer_contains(element) || self.pending_top_layer_removals_contains(element) {
406            return;
407        }
408
409        // TODO: Step 3. Remove the UA !important overlay: auto rule targeting el.
410
411        // Step 4. Append el to doc's pending top layer removals.
412        self.pending_top_layer_removals
413            .safe_borrow_mut(no_gc)
414            .push(Dom::from_ref(element));
415
416        self.update_topmost_modal(no_gc);
417    }

419    /// <https://drafts.csswg.org/css-position-4/#remove-an-element-from-the-top-layer-immediately>
420    pub(crate) fn remove_an_element_from_the_top_layer_immediately(
421        &self,
422        element: &Element,
423        no_gc: &NoGC,

433            .retain(|pending_element| &**pending_element != element);
434
435        // TODO: Step 3. Remove the UA !important overlay: auto rule targeting el, if it exists.
436
437        self.update_topmost_modal(no_gc);

440    /// <https://drafts.csswg.org/css-position-4/#process-top-layer-removals>
441    pub(crate) fn process_top_layer_removals(&self, cx: &mut JSContext) {
442        // Step 1. For each element el in doc’s pending top layer removals: if el's computed value
443        // of overlay is none, or el is not rendered, remove from the top layer immediately el.
444        // TODO: Check `overlay` once it is supported.
445        let pending_removals: Vec<_> = self
446            .pending_top_layer_removals
447            .borrow()
448            .iter()
449            .map(|element| DomRoot::from_ref(&**element))
450            .collect();
451        for element in pending_removals {
452            if !element.has_css_layout_box(cx) {
453                self.remove_an_element_from_the_top_layer_immediately(&element, cx)
454            }
455        }
456    }

458    /// <https://html.spec.whatwg.org/multipage/#blocked-by-a-modal-dialog>
459    pub fn update_topmost_modal(&self, no_gc: &NoGC) {
460        // > A Document document is blocked by a modal dialog subject if subject is the topmost
461        // > dialog element in document's top layer.
462        let new_topmost_modal = self
463            .top_layer
464            .borrow()
465            .iter()
466            .rev()
467            .find(|element| {
468                element.state().contains(ElementState::MODAL) &&
469                    !self.pending_top_layer_removals_contains(element)
470            })
471            .map(|element| DomRoot::from_ref(&**element));
472        let old_topmost_modal = self.topmost_modal.get();
473
474        if let Some(old_topmost_modal) = old_topmost_modal {
475            old_topmost_modal.set_state(no_gc, ElementState::TOPMOST_MODAL, false);
476        }
477        if let Some(new_topmost_modal) = &new_topmost_modal {
478            new_topmost_modal.set_state(no_gc, ElementState::TOPMOST_MODAL, true);
479        }
480
481        self.topmost_modal.set(new_topmost_modal.as_deref());
482
483        // > While document is so blocked, every node that is connected to document, with the
484        // > exception of the subject element and its flat tree descendants, must become inert.
485        if let Some(document_element) = self.document.GetDocumentElement() {
486            let is_inert = new_topmost_modal.is_some() ||
487                document_element.has_attribute(&local_name!("inert"));
488            document_element.set_state(no_gc, ElementState::INERT, is_inert);
489        }
490    }

497    /// <https://html.spec.whatwg.org/multipage/#hide-popovers-until>
498    pub(crate) fn hide_popovers_until(
499        &self,
500        cx: &mut JSContext,
501        endpoint: Option<&HTMLElement>,
502        focus_previous_element: bool,
503        fire_events: bool,
504    ) {
505        // Step 1. Let endpointIsHint be true if document's showing hint popover list contains
506        // endpoint; otherwise false.
507        let endpoint_is_hint = endpoint.is_some_and(|endpoint| {
508            self.showing_popover_list_contains(cx.no_gc(), PopoverState::Hint, endpoint)
509        });
510
511        // Step 2. Run hide popover stack until given document, endpoint, Hint,
512        // focusPreviousElement, and fireEvents.
513        self.hide_popover_stack_until(
514            cx,
515            endpoint,
516            PopoverState::Hint,
517            focus_previous_element,
518            fire_events,
519        );
520
521        // Step 3. Let autoEndpoint be endpoint.
522        //
523        // Need to keep the reference to the hint_parent alive such that we
524        // can dereference to obtain its endpoint
525        let hint_parent;
526        let mut auto_endpoint = endpoint;
527
528        // Step 4. If endpointIsHint is true, then set autoEndpoint to document's hint stack
529        // parent.
530        if endpoint_is_hint {
531            hint_parent = self.hint_stack_parent.get();
532            auto_endpoint = hint_parent.as_deref();
533        }
534
535        // Step 5. Run hide popover stack until given document, autoEndpoint, Auto,
536        // focusPreviousElement, and fireEvents.
537        self.hide_popover_stack_until(
538            cx,
539            auto_endpoint,
540            PopoverState::Auto,
541            focus_previous_element,
542            fire_events,
543        );
544    }

546    /// <https://html.spec.whatwg.org/multipage/#hide-popover-stack-until>
547    pub(crate) fn hide_popover_stack_until(
548        &self,
549        cx: &mut JSContext,
550        endpoint: Option<&HTMLElement>,
551        stack_type: PopoverState,
552        focus_previous_element: bool,
553        fire_events: bool,
554    ) {
555        // Step 1. Let popoverList be document's showing auto popover list if stackType is Auto,
556        // and document's showing hint popover list otherwise.
557        let popover_list: Vec<DomRoot<HTMLElement>> = self
558            .showing_popover_list_unrooted(cx.no_gc(), stack_type)
559            .iter()
560            .map(|popover| popover.as_rooted())
561            .collect();
562
563        // Step 2. Let lastHideIndex be 0 if popoverList does not contain endpoint; otherwise the
564        // index of endpoint in popoverList plus 1.
565        let last_hide_index = endpoint
566            .and_then(|endpoint| {
567                popover_list
568                    .iter()
569                    .position(|popover| &**popover == endpoint)
570            })
571            .map_or(0, |index| index + 1);
572
573        // Step 3. Let toHide be popoverList from lastHideIndex onwards, in reverse order.
574        let to_hide = popover_list[last_hide_index..].iter().rev();
575
576        // Step 4. Let toRemain be popoverList up to lastHideIndex.
577        let to_remain = &popover_list[..last_hide_index];
578
579        // Step 5. For each popover of toHide, run the hide popover algorithm given popover,
580        // focusPreviousElement, fireEvents, false, and null.
581        for popover in to_hide {
582            let _ = popover.hide_popover(cx, focus_previous_element, fire_events, false, None);
583        }
584
585        // Step 6. Let newPopoverList be document's showing auto popover list if stackType is
586        // Auto, and document's showing hint popover list otherwise.
587        let new_popover_list: Vec<DomRoot<HTMLElement>> = self
588            .showing_popover_list_unrooted(cx.no_gc(), stack_type)
589            .iter()
590            .map(|popover| popover.as_rooted())
591            .collect();
592
593        // Step 7. Let toCheck be newPopoverList in reverse order.
594        let to_check = new_popover_list.iter().rev();
595
596        // Step 8. For each popover of toCheck:
597        for popover in to_check {
598            // Step 8.1. If toRemain contains popover, then continue.
599            if to_remain.contains(popover) {
600                continue;
601            }
602
603            // Step 8.2. Run the hide popover algorithm given popover, focusPreviousElement, false,
604            // false, and null.
605            let _ = popover.hide_popover(cx, focus_previous_element, false, false, None);
606        }
607    }
608
609    /// <https://html.spec.whatwg.org/multipage/#topmost-auto-or-hint-popover>
610    pub(crate) fn find_the_topmost_auto_or_hint_popover(
611        &self,
612        no_gc: &NoGC,
613    ) -> Option<DomRoot<HTMLElement>> {
614        // Step 1. If document's showing hint popover list is not empty, then return its last
615        // element.
616        // Step 2. If document's showing auto popover list is not empty, then return its last
617        // element.
618        // Step 3. Return null.
619        //
620        // We use pop instead of last to take ownership of its last element. The Vec is temporary,
621        // so mutating it is fine.
622        self.showing_popover_list_unrooted(no_gc, PopoverState::Hint)
623            .pop()
624            .or_else(|| {
625                self.showing_popover_list_unrooted(no_gc, PopoverState::Auto)
626                    .pop()
627            })
628            .map(|popover| popover.as_rooted())
629    }
630
631    /// <https://html.spec.whatwg.org/multipage/#auto-popover-list>
632    /// <https://html.spec.whatwg.org/multipage/#showing-hint-popover-list>
633    ///
634    /// The spec lists these steps twice, once with "auto" and once with "hint" as the mode in step 2.1.
635    /// `mode` must be [`PopoverState::Auto`] or [`PopoverState::Hint`].
636    pub(crate) fn showing_popover_list_unrooted<'a>(
637        &self,
638        no_gc: &'a NoGC,
639        mode: PopoverState,
640    ) -> Vec<UnrootedDom<'a, HTMLElement>> {
641        debug_assert!(matches!(mode, PopoverState::Auto | PopoverState::Hint));
642
643        // Step 1. Let popovers be « ».
644        // Step 2. For each Element element of document's top layer:
645        // Step 2.1. If all of the following are true:
646        // - element is an HTML element;
647        // - element's opened in popover mode is mode; and
648        // - element's popover visibility state is showing,
649        // - then append element to popovers.
650        // Step 3. Return popovers.
651        self.top_layer
652            .borrow()
653            .iter()
654            .filter_map(|element| element.downcast::<HTMLElement>())
655            .filter(|element| {
656                element
657                    .upcast::<Element>()
658                    .popover_visibility_state_showing() &&
659                    element.opened_in_popover_mode() == Some(mode)
660            })
661            .map(|element| UnrootedDom::from_ref(element, no_gc))
662            .collect()
663    }
664
665    pub(crate) fn showing_popover_list_contains(
666        &self,
667        no_gc: &NoGC,
668        mode: PopoverState,
669        element: &HTMLElement,
670    ) -> bool {
671        self.showing_popover_list_unrooted(no_gc, mode)
672            .iter()
673            .any(|popover| ***popover == *element)
674    }

722    ) -> Box<ElementPerformFullscreenEnter> {

727            error,
728        })
729    }

731
732impl TaskOnce for ElementPerformFullscreenEnter {
733    /// Step 9-14 of <https://fullscreen.spec.whatwg.org/#dom-element-requestfullscreen>
734    fn run_once(self, cx: &mut js::context::JSContext) {
735        let element = self.element.root();
736        let promise = self.promise.root(cx);
737        let document = element.owner_document();
738
739        // Step 9
740        // > If any of the following conditions are false, then set error to true:
741        // > - This’s node document is pendingDoc.
742        // > - The fullscreen element ready check for this returns true.
743        // Step 10
744        // > If error is true:
745        // > - Append (fullscreenerror, this) to pendingDoc’s list of pending fullscreen events.
746        // > - Reject promise with a TypeError exception and terminate these steps.
747        if self.document.root() != document ||
748            !element.fullscreen_element_ready_check() ||
749            self.error
750        {
751            // TODO(#31866): we should queue this and fire them in update the rendering.
752            document
753                .upcast::<EventTarget>()
754                .fire_event(cx, atom!("fullscreenerror"));
755            promise.reject_error(cx, Error::Type(c"fullscreen is not connected".to_owned()));
756            return;
757        }
758
759        // TODO(#42067): Implement step 11-13
760        // The following operations is based on the old version of the specs.
761        element.set_fullscreen_state(cx.no_gc(), true);
762        document
763            .overlay_handler()
764            .set_fullscreen_element(Some(&element));
765        document.upcast::<EventTarget>().fire_event_with_params(
766            cx,
767            atom!("fullscreenchange"),
768            EventBubbles::Bubbles,
769            EventCancelable::NotCancelable,
770            EventComposed::Composed,
771        );
772
773        // Step 14.
774        // > Resolve promise with undefined.
775        promise.resolve_native(cx, &());
776    }
777}

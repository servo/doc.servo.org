48/// The set of animations for a document.
49#[derive(Default, JSTraceable, MallocSizeOf)]
50#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
51pub(crate) struct AnimationManager {
52    /// The map of nodes to their animation states.
53    #[no_trace]
54    sets: DocumentAnimationSet,
55
56    /// The set of [`AnimatingImages`] which is used to communicate the addition
57    /// and removal of animating images from layout.
58    #[no_trace]
59    #[conditional_malloc_size_of]
60    animating_images: Arc<RwLock<AnimatingImages>>,
61
62    /// Whether or not we have animations that are running.
63    has_running_animations: Cell<bool>,
64
65    /// A map of nodes with in-progress CSS transitions or pending events.
66    rooted_animation_nodes: DomRefCell<FxHashMap<NoTrace<OpaqueNode>, Dom<Node>>>,
67
68    /// A map of nodes with in-progress image animations. This is kept outside
69    /// of [`Self::animating_images`] as that data structure is shared with layout.
70    rooted_image_nodes: DomRefCell<FxHashMap<NoTrace<OpaqueNode>, Dom<Node>>>,
71
72    /// A list of pending animation-related events.
73    pending_events: DomRefCell<Vec<TransitionOrAnimationEvent>>,
74
75    /// The timeline value at the last time all animations were marked dirty.
76    /// This is used to prevent marking animations dirty when the timeline
77    /// has not changed.
78    timeline_value_at_last_dirty: Cell<f64>,
79
80    /// The [`TimerId`] of the currently scheduled animated image update callback.
81    #[no_trace]
82    image_callback_timer_id: Cell<Option<TimerId>>,
83}

115    // Mark all animations dirty, if they haven't been marked dirty since the
116    // specified `current_timeline_value`. Returns true if animations were marked
117    // dirty or false otherwise.
118    pub(crate) fn mark_animating_nodes_as_dirty(
119        &self,
120        no_gc: &NoGC,
121        current_timeline_value: f64,
122    ) -> bool {
123        if current_timeline_value <= self.timeline_value_at_last_dirty.get() {
124            return false;
125        }
126        self.timeline_value_at_last_dirty
127            .set(current_timeline_value);
128
129        let sets = self.sets.sets.read();
130        let rooted_nodes = self.rooted_animation_nodes.borrow();
131        for node in sets
132            .keys()
133            .filter_map(|key| rooted_nodes.get(&NoTrace(key.node)))
134        {
135            node.dirty(no_gc, NodeDamage::Style);
136        }
137
138        true
139    }

141    pub(crate) fn update_for_new_timeline_value(&self, window: &Window, now: f64) {
142        let pipeline_id = window.pipeline_id();
143        let mut sets = self.sets.sets.write();
144
145        for (key, set) in sets.iter_mut() {
146            self.start_pending_animations(key, set, now, pipeline_id);
147
148            // When necessary, iterate our running animations to the next iteration.
149            for animation in set.animations.iter_mut() {
150                if animation.iterate_if_necessary(now) {
151                    self.add_animation_event(
152                        key,
153                        animation,
154                        TransitionOrAnimationEventType::AnimationIteration,
155                        now,
156                        pipeline_id,
157                    );
158                }
159            }
160
161            self.finish_running_animations(key, set, now, pipeline_id);
162        }
163
164        self.unroot_unused_nodes(&sets);
165    }

167    /// Cancel animations for the given node, if any exist.
168    pub(crate) fn cancel_animations_for_node(&self, node: &Node) {
169        let opaque_node = node.to_opaque();
170
171        let animation_keys = [
172            AnimationSetKey::new_for_non_pseudo(opaque_node),
173            AnimationSetKey::new_for_pseudo(opaque_node, PseudoElement::Before),
174            AnimationSetKey::new_for_pseudo(opaque_node, PseudoElement::After),
175        ];
176
177        let mut animations = self.sets.sets.write();
178        for key in animation_keys {
179            if let Some(set) = animations.get_mut(&key) {
180                set.cancel_all_animations();
181            }
182        }
183
184        self.animating_images().write().remove(opaque_node);
185        self.rooted_image_nodes
186            .borrow_mut()
187            .remove(&NoTrace(opaque_node));
188    }

190    /// This does five things:
191    ///  - Cancel animations for any nodes that are no longer being rendered or delegating rendering.
192    ///  - Process any new animations that were discovered after reflow.
193    ///  - Collect pending events for any animations that changed state.
194    ///  - Root any nodes with newly animating images
195    ///  - Schedule an image update for newly animating images
196    pub(crate) fn do_post_reflow_update(&self, window: &Window, now: f64) {
197        let mut sets = self.sets.sets.write();
198        {
199            let rooted_nodes = self.rooted_animation_nodes.borrow();
200            for (key, set) in sets.iter_mut() {
201                if rooted_nodes.get(&NoTrace(key.node)).is_some_and(|node| {
202                    !node.is_being_rendered_or_delegates_rendering(key.pseudo_element)
203                }) {
204                    set.cancel_all_animations();
205                }
206            }
207        }
208
209        let pipeline_id = window.pipeline_id();
210        self.root_newly_animating_dom_nodes(&sets);
211
212        for (key, set) in sets.iter_mut() {
213            self.handle_canceled_animations(key, set, now, pipeline_id);
214            self.handle_new_animations(key, set, now, pipeline_id);
215        }
216
217        // Remove empty states from our collection of states in order to free
218        // up space as soon as we are no longer tracking any animations for
219        // a node.
220        sets.retain(|_, state| !state.is_empty());
221        let have_running_animations = sets.values().any(|state| state.needs_animation_ticks());
222
223        self.update_running_animations_presence(window, have_running_animations);
224
225        // Cancel animations for any images that are no longer rendering.
226        self.rooted_image_nodes
227            .borrow_mut()
228            .retain(|opaque_node, node| {
229                if node.is_being_rendered(None) {
230                    return true;
231                }
232                self.animating_images.write().remove(opaque_node.0);
233                false
234            });
235
236        if self.animating_images().write().clear_dirty() {
237            self.root_nodes_with_newly_animating_images();
238            self.maybe_schedule_image_animation_update(window, now);
239        }
240    }

242    pub(crate) fn update_active_image_animation_frames(&self, window: &Window, now: f64) {
243        if self.animating_images.read().is_empty() {
244            return;
245        }
246
247        let updates = self
248            .animating_images
249            .write()
250            .node_to_state_map
251            .values_mut()
252            .filter_map(|state| {
253                if !state.update_frame_for_animation_timeline_value(now) {
254                    return None;
255                }
256
257                let image = &state.image;
258                let frame = image
259                    .frame_data(state.active_frame)
260                    .expect("No frame found")
261                    .clone();
262                if let Some(mut descriptor) =
263                    image.webrender_image_descriptor_and_offset_for_frame()
264                {
265                    descriptor.offset = frame.byte_range.start as i32;
266                    Some(ImageUpdate::UpdateImageForAnimation(
267                        image.id.unwrap(),
268                        descriptor,
269                    ))
270                } else {
271                    error!("Doing normal image update which will be slow!");
272                    None
273                }
274            })
275            .collect();
276        window
277            .paint_api()
278            .update_images(window.webview_id().into(), updates);
279
280        self.maybe_schedule_image_animation_update(window, now);
281    }

290    fn maybe_schedule_image_animation_update(&self, window: &Window, now: f64) {
291        with_script_thread(|script_thread| {
292            if let Some(current_timer_id) = self.image_callback_timer_id.take() {
293                self.image_callback_timer_id.set(None);
294                script_thread.cancel_timer(current_timer_id);
295            }
296
297            if let Some(duration) = self.duration_to_next_image_animation_frame(now) {
298                let trusted_window = Trusted::new(window);
299                let timer_id = script_thread.schedule_timer(timers::TimerEventRequest {
300                    callback: Box::new(move || {
301                        let window = trusted_window.root();
302                        window.Document().set_has_pending_animated_image_update();
303                    }),
304                    duration,
305                });
306                self.image_callback_timer_id.set(Some(timer_id));
307            }
308        })
309    }

320    fn update_running_animations_presence(&self, window: &Window, new_value: bool) {
321        let had_running_animations = self.has_running_animations.get();
322        if new_value == had_running_animations {
323            return;
324        }
325
326        self.has_running_animations.set(new_value);
327        self.handle_animation_presence_or_pending_events_change(window);
328    }
329
330    fn handle_animation_presence_or_pending_events_change(&self, window: &Window) {
331        let has_running_animations = self.has_running_animations.get();
332        let has_pending_events = !self.pending_events.borrow().is_empty();
333
334        // Do not send the AnimationCallbacksAbsent state until all pending
335        // animation events are delivered.
336        let state = match has_running_animations || has_pending_events {
337            true => AnimationsPresentState::AnimationsPresent,
338            false => AnimationsPresentState::NoAnimationsPresent,
339        };
340        window.send_to_constellation(ScriptToConstellationMessage::ChangeRunningAnimationsState(
341            state,
342        ));
343    }

354    /// Walk through the list of pending animations and start all of the ones that
355    /// have left the delay phase.
356    fn start_pending_animations(
357        &self,
358        key: &AnimationSetKey,
359        set: &mut ElementAnimationSet,
360        now: f64,
361        pipeline_id: PipelineId,
362    ) {
363        for animation in set.animations.iter_mut() {
364            if animation.state == AnimationState::Pending && animation.started_at <= now {
365                animation.state = AnimationState::Running;
366                self.add_animation_event(
367                    key,
368                    animation,
369                    TransitionOrAnimationEventType::AnimationStart,
370                    now,
371                    pipeline_id,
372                );
373            }
374        }
375
376        for transition in set.transitions.iter_mut() {
377            if transition.state == AnimationState::Pending && transition.start_time <= now {
378                transition.state = AnimationState::Running;
379                self.add_transition_event(
380                    key,
381                    transition,
382                    TransitionOrAnimationEventType::TransitionStart,
383                    now,
384                    pipeline_id,
385                );
386            }
387        }
388    }

390    /// Walk through the list of running animations and remove all of the ones that
391    /// have ended.
392    fn finish_running_animations(
393        &self,
394        key: &AnimationSetKey,
395        set: &mut ElementAnimationSet,
396        now: f64,
397        pipeline_id: PipelineId,
398    ) {
399        for animation in set.animations.iter_mut() {
400            if animation.state == AnimationState::Running && animation.has_ended(now) {
401                animation.state = AnimationState::Finished;
402                self.add_animation_event(
403                    key,
404                    animation,
405                    TransitionOrAnimationEventType::AnimationEnd,
406                    now,
407                    pipeline_id,
408                );
409            }
410        }
411
412        for transition in set.transitions.iter_mut() {
413            if transition.state == AnimationState::Running && transition.has_ended(now) {
414                transition.state = AnimationState::Finished;
415                self.add_transition_event(
416                    key,
417                    transition,
418                    TransitionOrAnimationEventType::TransitionEnd,
419                    now,
420                    pipeline_id,
421                );
422            }
423        }
424    }

426    /// Send events for canceled animations. Currently this only handles canceled
427    /// transitions, but eventually this should handle canceled CSS animations as
428    /// well.
429    fn handle_canceled_animations(
430        &self,
431        key: &AnimationSetKey,
432        set: &mut ElementAnimationSet,
433        now: f64,
434        pipeline_id: PipelineId,
435    ) {
436        for transition in &set.transitions {
437            if transition.state == AnimationState::Canceled {
438                self.add_transition_event(
439                    key,
440                    transition,
441                    TransitionOrAnimationEventType::TransitionCancel,
442                    now,
443                    pipeline_id,
444                );
445            }
446        }
447
448        for animation in &set.animations {
449            if animation.state == AnimationState::Canceled {
450                self.add_animation_event(
451                    key,
452                    animation,
453                    TransitionOrAnimationEventType::AnimationCancel,
454                    now,
455                    pipeline_id,
456                );
457            }
458        }
459
460        set.clear_canceled_animations();
461    }

463    fn handle_new_animations(
464        &self,
465        key: &AnimationSetKey,
466        set: &mut ElementAnimationSet,
467        now: f64,
468        pipeline_id: PipelineId,
469    ) {
470        for animation in set.animations.iter_mut() {
471            animation.is_new = false;
472        }
473
474        for transition in set.transitions.iter_mut() {
475            if transition.is_new {
476                self.add_transition_event(
477                    key,
478                    transition,
479                    TransitionOrAnimationEventType::TransitionRun,
480                    now,
481                    pipeline_id,
482                );
483                transition.is_new = false;
484            }
485        }
486    }

488    /// Ensure that all nodes with new animations are rooted. This should be called
489    /// immediately after a restyle, to ensure that these addresses are still valid.
490    fn root_newly_animating_dom_nodes(
491        &self,
492        sets: &FxHashMap<AnimationSetKey, ElementAnimationSet>,
493    ) {
494        let mut rooted_nodes = self.rooted_animation_nodes.borrow_mut();
495        for (key, set) in sets.iter() {
496            let opaque_node = key.node;
497            if rooted_nodes.contains_key(&NoTrace(opaque_node)) {
498                continue;
499            }
500
501            if set.animations.iter().any(|animation| animation.is_new) ||
502                set.transitions.iter().any(|transition| transition.is_new)
503            {
504                root_node(&mut rooted_nodes, opaque_node);
505            }
506        }
507    }

509    // Unroot any nodes that we have rooted but are no longer tracking animations for.
510    fn unroot_unused_nodes(&self, sets: &FxHashMap<AnimationSetKey, ElementAnimationSet>) {
511        let pending_events = self.pending_events.borrow();
512        let nodes: FxHashSet<OpaqueNode> = sets.keys().map(|key| key.node).collect();
513        self.rooted_animation_nodes.borrow_mut().retain(|node, _| {
514            nodes.contains(&node.0) || pending_events.iter().any(|event| event.node == node.0)
515        });
516    }

518    fn add_transition_event(
519        &self,
520        key: &AnimationSetKey,
521        transition: &Transition,
522        event_type: TransitionOrAnimationEventType,
523        now: f64,
524        pipeline_id: PipelineId,
525    ) {
526        // Calculate the `elapsed-time` property of the event and take the absolute
527        // value to prevent -0 values.
528        let elapsed_time = match event_type {
529            TransitionOrAnimationEventType::TransitionRun |
530            TransitionOrAnimationEventType::TransitionStart => transition
531                .property_animation
532                .duration
533                .min((-transition.delay).max(0.)),
534            TransitionOrAnimationEventType::TransitionEnd => transition.property_animation.duration,
535            TransitionOrAnimationEventType::TransitionCancel => {
536                (now - transition.start_time).max(0.)
537            },
538            _ => unreachable!(),
539        }
540        .abs();
541
542        self.pending_events
543            .borrow_mut()
544            .push(TransitionOrAnimationEvent {
545                pipeline_id,
546                event_type,
547                node: key.node,
548                pseudo_element: key.pseudo_element,
549                property_or_animation_name: transition
550                    .property_animation
551                    .property_id()
552                    .name()
553                    .into(),
554                elapsed_time,
555            });
556    }

558    fn add_animation_event(
559        &self,
560        key: &AnimationSetKey,
561        animation: &Animation,
562        event_type: TransitionOrAnimationEventType,
563        now: f64,
564        pipeline_id: PipelineId,
565    ) {
566        let iteration_index = match animation.iteration_state {
567            KeyframesIterationState::Finite(current, _) |
568            KeyframesIterationState::Infinite(current) => current,
569        };
570
571        let active_duration = match animation.iteration_state {
572            KeyframesIterationState::Finite(_, max) => max * animation.duration,
573            KeyframesIterationState::Infinite(_) => f64::MAX,
574        };
575
576        // Calculate the `elapsed-time` property of the event and take the absolute
577        // value to prevent -0 values.
578        let elapsed_time = match event_type {
579            TransitionOrAnimationEventType::AnimationStart => {
580                (-animation.delay).max(0.).min(active_duration)
581            },
582            TransitionOrAnimationEventType::AnimationIteration => {
583                iteration_index * animation.duration
584            },
585            TransitionOrAnimationEventType::AnimationEnd => {
586                (iteration_index * animation.duration) + animation.current_iteration_duration()
587            },
588            TransitionOrAnimationEventType::AnimationCancel => {
589                (iteration_index * animation.duration) + (now - animation.started_at).max(0.)
590            },
591            _ => unreachable!(),
592        }
593        .abs();
594
595        self.pending_events
596            .borrow_mut()
597            .push(TransitionOrAnimationEvent {
598                pipeline_id,
599                event_type,
600                node: key.node,
601                pseudo_element: key.pseudo_element,
602                property_or_animation_name: animation.name.to_string(),
603                elapsed_time,
604            });
605    }

607    /// An implementation of the final steps of
608    /// <https://drafts.csswg.org/web-animations-1/#update-animations-and-send-events>.
609    pub(crate) fn send_pending_events(&self, window: &Window, cx: &mut js::context::JSContext) {
610        // > 4. Let events to dispatch be a copy of doc’s pending animation event queue.
611        // > 5. Clear doc’s pending animation event queue.
612        //
613        // Take all of the events here, in case sending one of these events
614        // triggers adding new events by forcing a layout.
615        let events = std::mem::take(&mut *self.pending_events.safe_borrow_mut(cx.no_gc()));
616        if events.is_empty() {
617            return;
618        }
619
620        // > 6. Perform a stable sort of the animation events in events to dispatch as follows:
621        // >    1. Sort the events by their scheduled event time such that events that were
622        // >       scheduled to occur earlier sort before events scheduled to occur later, and
623        // >       events whose scheduled event time is unresolved sort before events with a
624        // >       resolved scheduled event time.
625        // >    2. Within events with equal scheduled event times, sort by their composite
626        // >       order.
627        //
628        // TODO: Sorting of animation events isn't done yet.
629
630        // 7. Dispatch each of the events in events to dispatch at their corresponding
631        // target using the order established in the previous step.
632        for event in events.into_iter() {
633            // We root the node here to ensure that sending this event doesn't
634            // unroot it as a side-effect.
635            let node = match self
636                .rooted_animation_nodes
637                .borrow()
638                .get(&NoTrace(event.node))
639            {
640                Some(node) => DomRoot::from_ref(&**node),
641                None => {
642                    warn!("Tried to send an event for an unrooted node");
643                    continue;
644                },
645            };
646
647            let event_atom = match event.event_type {
648                TransitionOrAnimationEventType::AnimationEnd => atom!("animationend"),
649                TransitionOrAnimationEventType::AnimationStart => atom!("animationstart"),
650                TransitionOrAnimationEventType::AnimationCancel => atom!("animationcancel"),
651                TransitionOrAnimationEventType::AnimationIteration => atom!("animationiteration"),
652                TransitionOrAnimationEventType::TransitionCancel => atom!("transitioncancel"),
653                TransitionOrAnimationEventType::TransitionEnd => atom!("transitionend"),
654                TransitionOrAnimationEventType::TransitionRun => atom!("transitionrun"),
655                TransitionOrAnimationEventType::TransitionStart => atom!("transitionstart"),
656            };
657            let parent = EventInit {
658                bubbles: true,
659                cancelable: false,
660                composed: false,
661            };
662
663            let property_or_animation_name =
664                DOMString::from(event.property_or_animation_name.clone());
665            let pseudo_element = event
666                .pseudo_element
667                .map_or_else(DOMString::new, |pseudo_element| {
668                    DOMString::from(pseudo_element.to_css_string())
669                });
670            let elapsed_time = Finite::new(event.elapsed_time as f32).unwrap();
671            let window = node.owner_window();
672
673            if event.event_type.is_transition_event() {
674                let event_init = TransitionEventInit {
675                    parent,
676                    propertyName: property_or_animation_name,
677                    elapsedTime: elapsed_time,
678                    pseudoElement: pseudo_element,
679                };
680                TransitionEvent::new(cx, &window, event_atom, &event_init)
681                    .upcast::<Event>()
682                    .fire(cx, node.upcast());
683            } else {
684                let event_init = AnimationEventInit {
685                    parent,
686                    animationName: property_or_animation_name,
687                    elapsedTime: elapsed_time,
688                    pseudoElement: pseudo_element,
689                };
690                AnimationEvent::new(cx, &window, event_atom, &event_init)
691                    .upcast::<Event>()
692                    .fire(cx, node.upcast());
693            }
694        }
695
696        if self.pending_events.borrow().is_empty() {
697            self.handle_animation_presence_or_pending_events_change(window);
698        }
699    }
700}

702/// The type of transition event to trigger. These are defined by
703/// CSS Transitions § 6.1 and CSS Animations § 4.2
704#[derive(Clone, Debug, Deserialize, JSTraceable, MallocSizeOf, Serialize)]
705pub(crate) enum TransitionOrAnimationEventType {
706    /// "The transitionrun event occurs when a transition is created (i.e., when it
707    /// is added to the set of running transitions)."
708    TransitionRun,
709    /// "The transitionstart event occurs when a transition’s delay phase ends."
710    TransitionStart,
711    /// "The transitionend event occurs at the completion of the transition. In the
712    /// case where a transition is removed before completion, such as if the
713    /// transition-property is removed, then the event will not fire."
714    TransitionEnd,
715    /// "The transitioncancel event occurs when a transition is canceled."
716    TransitionCancel,
717    /// "The animationstart event occurs at the start of the animation. If there is
718    /// an animation-delay then this event will fire once the delay period has expired."
719    AnimationStart,
720    /// "The animationiteration event occurs at the end of each iteration of an
721    /// animation, except when an animationend event would fire at the same time."
722    AnimationIteration,
723    /// "The animationend event occurs when the animation finishes"
724    AnimationEnd,
725    /// "The animationcancel event occurs when the animation stops running in a way
726    /// that does not fire an animationend event..."
727    AnimationCancel,
728}
729
730impl TransitionOrAnimationEventType {
731    /// Whether or not this event is a transition-related event.
732    pub(crate) fn is_transition_event(&self) -> bool {
733        match *self {
734            Self::TransitionRun |
735            Self::TransitionEnd |
736            Self::TransitionCancel |
737            Self::TransitionStart => true,
738            Self::AnimationEnd |
739            Self::AnimationIteration |
740            Self::AnimationStart |
741            Self::AnimationCancel => false,
742        }
743    }
744}
745
746#[derive(Deserialize, JSTraceable, MallocSizeOf, Serialize)]
747/// A transition or animation event.
748pub(crate) struct TransitionOrAnimationEvent {
749    /// The pipeline id of the layout task that sent this message.
750    #[no_trace]
751    pub(crate) pipeline_id: PipelineId,
752    /// The type of transition event this should trigger.
753    pub(crate) event_type: TransitionOrAnimationEventType,
754    /// The address of the node which owns this transition.
755    #[no_trace]
756    pub(crate) node: OpaqueNode,
757    /// The pseudo element for this transition or animation, if applicable.
758    #[no_trace]
759    pub(crate) pseudo_element: Option<PseudoElement>,
760    /// The name of the property that is transitioning (in the case of a transition)
761    /// or the name of the animation (in the case of an animation).
762    pub(crate) property_or_animation_name: String,
763    /// The elapsed time property to send with this transition event.
764    pub(crate) elapsed_time: f64,
765}

767/// Inserts [opaque_node] into [rooted_nodes], if it isn't already present. This should be
768/// called immediately after a restyle, to ensure that these addresses are still valid.
769fn root_node(
770    rooted_nodes: &mut FxHashMap<NoTrace<OpaqueNode>, Dom<Node>>,
771    opaque_node: OpaqueNode,
772) {
773    #[expect(unsafe_code)]
774    rooted_nodes.entry(NoTrace(opaque_node)).or_insert_with(|| {
775        // SAFETY: This should be safe as this method is run directly after layout,
776        // which should not remove any nodes.
777        let address = UntrustedNodeAddress(opaque_node.0 as *const c_void);
778        unsafe { Dom::from_ref(&*from_untrusted_node_address(address)) }
779    });

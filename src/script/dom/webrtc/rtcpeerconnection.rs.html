180    fn new(
181        cx: &mut JSContext,
182        window: &Window,
183        proto: Option<HandleObject>,
184        config: &RTCConfiguration,
185    ) -> DomRoot<RTCPeerConnection> {
186        let this = reflect_weak_referenceable_dom_object_with_proto(
187            cx,
188            Rc::new(RTCPeerConnection::new_inherited()),
189            window,
190            proto,
191        );
192        let signaller = this.make_signaller();
193        *this.controller.borrow_mut() = Some(ServoMedia::get().create_webrtc(signaller));
194        if let Some(ref servers) = config.iceServers &&
195            let Some(server) = servers.first()
196        {
197            let server = match server.urls {
198                StringOrStringSequence::String(ref s) => Some(s.clone()),
199                StringOrStringSequence::StringSequence(ref s) => s.first().cloned(),
200            };
201            if let Some(server) = server {
202                let policy = match config.bundlePolicy {
203                    RTCBundlePolicy::Balanced => BundlePolicy::Balanced,
204                    RTCBundlePolicy::Max_compat => BundlePolicy::MaxCompat,
205                    RTCBundlePolicy::Max_bundle => BundlePolicy::MaxBundle,
206                };
207                this.controller
208                    .borrow()
209                    .as_ref()
210                    .unwrap()
211                    .configure(String::from(server), policy);
212            }
213        }
214        this
215    }

229    fn on_ice_candidate(&self, cx: &mut JSContext, candidate: IceCandidate) {

282    fn on_data_channel_event(
283        &self,
284        cx: &mut CurrentRealm,
285        channel_id: DataChannelId,
286        event: DataChannelEvent,
287    ) {
288        if self.closed.get() {
289            return;
290        }
291
292        match event {
293            DataChannelEvent::NewChannel => {
294                let channel = RTCDataChannel::new(
295                    cx,
296                    &self.global(),
297                    self,
298                    USVString::from("".to_owned()),
299                    &RTCDataChannelInit::empty(),
300                    Some(channel_id),
301                );
302
303                let event = RTCDataChannelEvent::new(
304                    cx,
305                    self.global().as_window(),
306                    atom!("datachannel"),
307                    false,
308                    false,
309                    &channel,
310                );
311                event.upcast::<Event>().fire(cx, self.upcast());
312            },
313            _ => {
314                let channel: DomRoot<RTCDataChannel> =
315                    if let Some(channel) = self.data_channels.borrow().get(&channel_id) {
316                        DomRoot::from_ref(&**channel)
317                    } else {
318                        warn!(
319                            "Got an event for an unregistered data channel {:?}",
320                            channel_id
321                        );
322                        return;
323                    };
324
325                match event {
326                    DataChannelEvent::Open => channel.on_open(cx),
327                    DataChannelEvent::Close => channel.on_close(cx),
328                    DataChannelEvent::Error(error) => channel.on_error(cx, error),
329                    DataChannelEvent::OnMessage(message) => channel.on_message(cx, message),
330                    DataChannelEvent::StateChange(state) => channel.on_state_change(cx, state),
331                    DataChannelEvent::NewChannel => unreachable!(),
332                }
333            },
334        };
335    }

352    /// <https://www.w3.org/TR/webrtc/#update-ice-gathering-state>
353    fn update_gathering_state(&self, cx: &mut JSContext, state: GatheringState) {
354        // step 1
355        if self.closed.get() {
356            return;
357        }
358
359        // step 2 (state derivation already done by gstreamer)
360        let state: RTCIceGatheringState = state.convert();
361
362        // step 3
363        if state == self.gathering_state.get() {
364            return;
365        }
366
367        // step 4
368        self.gathering_state.set(state);
369
370        // step 5
371        let event = Event::new(
372            cx,
373            &self.global(),
374            atom!("icegatheringstatechange"),
375            EventBubbles::DoesNotBubble,
376            EventCancelable::NotCancelable,
377        );
378        event.upcast::<Event>().fire(cx, self.upcast());
379
380        // step 6
381        if state == RTCIceGatheringState::Complete {
382            let event = RTCPeerConnectionIceEvent::new(
383                cx,
384                self.global().as_window(),
385                atom!("icecandidate"),
386                None,
387                None,
388                true,
389            );
390            event.upcast::<Event>().fire(cx, self.upcast());
391        }
392    }

446    fn create_offer(&self) {
447        let generation = self.offer_answer_generation.get();
448        let task_source = self
449            .global()
450            .task_manager()
451            .networking_task_source()
452            .to_sendable();
453        let this = Trusted::new(self);
454        self.controller
455            .borrow_mut()
456            .as_ref()
457            .unwrap()
458            .create_offer(Box::new(move |desc: SessionDescription| {
459                task_source.queue(task!(offer_created: move |cx| {
460                    let this = this.root();
461                    if this.offer_answer_generation.get() != generation {
462                        // the state has changed since we last created the offer,
463                        // create a fresh one
464                        this.create_offer();
465                    } else {
466                        let init: RTCSessionDescriptionInit = desc.convert();
467                        for promise in this.offer_promises.borrow().iter() {
468                            promise.resolve_native(cx, &init);
469                        }
470                        this.offer_promises.safe_borrow_mut(cx.no_gc()).clear();
471                    }
472                }));
473            }));
474    }

555    /// <https://w3c.github.io/webrtc-pc/#dom-rtcpeerconnection-addicecandidate>
556    fn AddIceCandidate(
557        &self,
558        current_realm: &mut CurrentRealm,
559        candidate: &RTCIceCandidateInit,
560    ) -> Rc<Promise> {
561        let p = Promise::new_in_realm(current_realm);
562        if candidate.sdpMid.is_none() && candidate.sdpMLineIndex.is_none() {
563            p.reject_error(
564                current_realm,
565                Error::Type(c"one of sdpMid and sdpMLineIndex must be set".to_owned()),
566            );
567            return p;
568        }
569
570        // XXXManishearth add support for sdpMid
571        if candidate.sdpMLineIndex.is_none() {
572            p.reject_error(
573                current_realm,
574                Error::Type(c"servo only supports sdpMLineIndex right now".to_owned()),
575            );
576            return p;
577        }
578
579        // XXXManishearth this should be enqueued
580        // https://w3c.github.io/webrtc-pc/#enqueue-an-operation
581
582        self.controller
583            .borrow_mut()
584            .as_ref()
585            .unwrap()
586            .add_ice_candidate(IceCandidate {
587                sdp_mline_index: candidate.sdpMLineIndex.unwrap() as u32,
588                candidate: candidate.candidate.to_string(),
589            });
590
591        // XXXManishearth add_ice_candidate should have a callback
592        p.resolve_native(current_realm, &());
593        p
594    }

638    /// <https://w3c.github.io/webrtc-pc/#dom-rtcpeerconnection-setlocaldescription>
639    fn SetLocalDescription(
640        &self,
641        current_realm: &mut CurrentRealm,
642        desc: &RTCSessionDescriptionInit,
643    ) -> Rc<Promise> {
644        // XXXManishearth validate the current state
645        let p = Promise::new_in_realm(current_realm);
646        let this = Trusted::new(self);
647        let desc: SessionDescription = desc.convert();
648        let trusted_promise = TrustedPromise::new(p.clone());
649        let task_source = self
650            .global()
651            .task_manager()
652            .networking_task_source()
653            .to_sendable();
654        self.controller
655            .borrow()
656            .as_ref()
657            .unwrap()
658            .set_local_description(
659                desc.clone(),
660                Box::new(move || {
661                    task_source.queue(task!(local_description_set: move |current_realm| {
662                        // XXXManishearth spec actually asks for an intricate
663                        // dance between pending/current local/remote descriptions
664                        let this = this.root();
665                        let desc = desc.convert();
666                        let desc = RTCSessionDescription::new(
667                            current_realm,
668                            this.global().as_window(),
669                            None,
670                            desc.type_,
671                            desc.sdp,
672                        );
673                        this.local_description.set(Some(&desc));
674                        trusted_promise.root().resolve_native(current_realm, &())
675                    }));
676                }),
677            );
678        p
679    }

681    /// <https://w3c.github.io/webrtc-pc/#dom-rtcpeerconnection-setremotedescription>
682    fn SetRemoteDescription(
683        &self,
684        current_realm: &mut CurrentRealm,
685        desc: &RTCSessionDescriptionInit,
686    ) -> Rc<Promise> {
687        // XXXManishearth validate the current state
688        let p = Promise::new_in_realm(current_realm);
689        let this = Trusted::new(self);
690        let desc: SessionDescription = desc.convert();
691        let trusted_promise = TrustedPromise::new(p.clone());
692        let task_source = self
693            .global()
694            .task_manager()
695            .networking_task_source()
696            .to_sendable();
697        self.controller
698            .borrow()
699            .as_ref()
700            .unwrap()
701            .set_remote_description(
702                desc.clone(),
703                Box::new(move || {
704                    task_source.queue(task!(remote_description_set: move |current_realm| {
705                        // XXXManishearth spec actually asks for an intricate
706                        // dance between pending/current local/remote descriptions
707                        let this = this.root();
708                        let desc = desc.convert();
709                        let desc = RTCSessionDescription::new(
710                            current_realm,
711                            this.global().as_window(),
712                            None,
713                            desc.type_,
714                            desc.sdp,
715                        );
716                        this.remote_description.set(Some(&desc));
717                        trusted_promise.root().resolve_native(current_realm, &())
718                    }));
719                }),
720            );
721        p
722    }

750    /// <https://www.w3.org/TR/webrtc/#dom-rtcpeerconnection-close>
751    fn Close(&self, cx: &mut JSContext) {
752        // Step 1
753        if self.closed.get() {
754            return;
755        }
756        // Step 2
757        self.closed.set(true);
758
759        // Step 4
760        self.signaling_state.set(RTCSignalingState::Closed);
761
762        // Step 5 handled by backend
763        self.controller.borrow_mut().as_ref().unwrap().quit();
764
765        // Step 6
766        for val in self.data_channels.borrow().values() {
767            val.on_state_change(cx, DataChannelState::Closed);
768        }
769
770        // Step 7-10
771        // (no current support for transports, etc)
772
773        // Step 11
774        self.ice_connection_state.set(RTCIceConnectionState::Closed);
775
776        // Step 11
777        // (no current support for connection state)
778    }

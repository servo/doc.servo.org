42/// Represents a `BufferSource` as defined in the WebIDL specification.
43///
44/// A `BufferSource` is either an `ArrayBuffer` or an `ArrayBufferView`, which
45/// provides a view onto an `ArrayBuffer`.
46///
47/// See: <https://webidl.spec.whatwg.org/#BufferSource>
48#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
49pub(crate) enum BufferSource {
50    /// Represents an `ArrayBufferView` (e.g., `Uint8Array`, `DataView`).
51    /// See: <https://webidl.spec.whatwg.org/#ArrayBufferView>
52    ArrayBufferView(Box<Heap<*mut JSObject>>),
53
54    /// Represents an `ArrayBuffer`, a fixed-length binary data buffer.
55    /// See: <https://webidl.spec.whatwg.org/#idl-ArrayBuffer>
56    ArrayBuffer(Box<Heap<*mut JSObject>>),
57}
58
59impl Clone for BufferSource {
60    fn clone(&self) -> Self {
61        match self {
62            BufferSource::ArrayBufferView(heap) => {
63                BufferSource::ArrayBufferView(Heap::boxed(heap.get()))
64            },
65            BufferSource::ArrayBuffer(heap) => BufferSource::ArrayBuffer(Heap::boxed(heap.get())),
66        }
67    }
68}

100/// <https://webidl.spec.whatwg.org/#dfn-get-buffer-source-copy>
101///
102/// Spec steps and how they're covered:
103///
104/// - **Convert to JS value**: Handled by WebIDL bindings before this function
105///   receives the typed `ArrayBufferViewOrArrayBuffer` union.
106///
107/// - **ArrayBufferView offset/length**: `view.to_vec()` (mozjs) calls
108///   `GetArrayBufferViewLengthAndData`, which respects the view's
109///   `[[ByteOffset]]` and `[[ByteLength]]` — only the viewed bytes are copied.
110///
111/// - **ArrayBuffer**: `buffer.to_vec()` (mozjs) calls
112///   `GetArrayBufferLengthAndData`, copying the entire buffer contents.
113///
114/// - **Detached buffer**: When a buffer is detached, SpiderMonkey's
115///   `GetArrayBuffer(LengthAndData|ViewLengthAndData)` returns a null pointer
116///   and zero length. `to_vec()` thus produces an empty `Vec<u8>`.
117///
118/// - **SharedArrayBuffer**: Not applicable — `ArrayBufferViewOrArrayBuffer`
119///   does not include a SharedArrayBuffer variant.
120pub(crate) fn get_buffer_source_copy(source: ArrayBufferViewOrArrayBufferRef<'_>) -> Vec<u8> {
121    match source {
122        ArrayBufferViewOrArrayBufferRef::ArrayBufferView(view) => view.to_vec(),
123        ArrayBufferViewOrArrayBufferRef::ArrayBuffer(buffer) => buffer.to_vec(),
124    }
125    .unwrap_or(vec![])
126}
127
128/// Returns a slice referencing the bytes in the buffer source, without copying.
129///
130/// Use this instead of [`get_buffer_source_copy`] when the data is consumed
131/// synchronously — it avoids the allocation of a `Vec<u8>`.
132pub(crate) fn get_buffer_source_slice<'a>(
133    source: &'a ArrayBufferViewOrArrayBuffer,
134    no_gc: &'a NoGC,
135) -> &'a [u8] {
136    match source {
137        ArrayBufferViewOrArrayBuffer::ArrayBufferView(view) => view.as_slice_safe(no_gc),
138        ArrayBufferViewOrArrayBuffer::ArrayBuffer(buffer) => buffer.as_slice_safe(no_gc),
139    }
140    .unwrap_or(&[])
141}

418    /// <https://tc39.es/ecma262/#sec-clonearraybuffer>
419    pub(crate) fn clone_array_buffer(
420        &self,
421        cx: &mut JSContext,
422        byte_offset: usize,
423        byte_length: usize,
424    ) -> Fallible<RootedTraceableBox<HeapBufferSource<ArrayBufferU8>>> {
425        let result = match &self.buffer_source {
426            BufferSource::ArrayBufferView(buffer) => {
427                let mut is_shared = false;
428                rooted!(&in(cx) let view_buffer =
429                    unsafe { JS_GetArrayBufferViewBuffer(cx, Handle::from_raw(buffer.handle()), &mut is_shared) });
430                debug_assert!(!is_shared);
431
432                unsafe { ArrayBufferClone(cx, view_buffer.handle(), byte_offset, byte_length) }
433            },
434            BufferSource::ArrayBuffer(buffer) => unsafe {
435                ArrayBufferClone(
436                    cx,
437                    Handle::from_raw(buffer.handle()),
438                    byte_offset,
439                    byte_length,
440                )
441            },
442        };
443
444        rooted!(&in(cx) let result = result);
445
446        if result.is_null() {
447            // Normalize SpiderMonkey failure: consume pending exception and
448            // map it to a DOM Error.
449            rooted!(&in(cx) let mut _ex = UndefinedValue());
450            unsafe {
451                // If SpiderMonkey set an exception, clear it so callers see a clean cx.
452                if JS_GetPendingException(cx, _ex.handle_mut()) {
453                    JS_ClearPendingException(cx);
454                }
455            }
456
457            Err(Error::Type(c"can't clone array buffer".to_owned()))
458        } else {
459            Ok(RootedTraceableBox::new(
460                HeapBufferSource::<ArrayBufferU8>::new(result.handle()),
461            ))
462        }
463    }
464    /// <https://streams.spec.whatwg.org/#abstract-opdef-cloneasuint8array>

466    pub(crate) fn clone_as_uint8_array(
467        &self,
468        cx: &mut JSContext,
469    ) -> Fallible<RootedTraceableBox<HeapBufferSource<ArrayBufferViewU8>>> {
470        match &self.buffer_source {
471            BufferSource::ArrayBufferView(buffer) => {
472                // Assert: O is an Object.
473                // Assert: O has an [[ViewedArrayBuffer]] internal slot.
474                assert!(unsafe { JS_IsArrayBufferViewObject(*buffer.handle()) });
475
476                // Assert: ! IsDetachedBuffer(O.[[ViewedArrayBuffer]]) is false.
477                assert!(!self.is_detached_buffer(cx));
478
479                // Let buffer be ? CloneArrayBuffer(O.[[ViewedArrayBuffer]],
480                // O.[[ByteOffset]], O.[[ByteLength]], %ArrayBuffer%).
481                let byte_offset = self.get_byte_offset();
482                let byte_length = self.byte_length();
483
484                let buffer = self.clone_array_buffer(cx, byte_offset, byte_length)?;
485
486                // Let array be ! Construct(%Uint8Array%, « buffer »).
487                // Return array.
488                construct_typed_array(cx, &Type::Uint8, &buffer, 0, byte_length as i64)
489            },
490            BufferSource::ArrayBuffer(_buffer) => {
491                unreachable!("BufferSource::ArrayBuffer does not have a view buffer.")
492            },
493        }
494    }

510    pub(crate) fn acquire_data(&self, cx: &mut JSContext) -> Result<Vec<T::Element>, ()> {
511        assert!(self.is_initialized());
512
513        typedarray!(&in(cx) let array: TypedArray = match &self.buffer_source {
514            BufferSource::ArrayBufferView(buffer) | BufferSource::ArrayBuffer(buffer)
515            => {
516                buffer.get()
517            },
518        });
519        let data = if let Ok(array) =
520            array as Result<CustomAutoRooterGuard<'_, TypedArray<T, *mut JSObject>>, &mut ()>
521        {
522            if let Some(data) = array.to_vec() {
523                let _ = self.detach_buffer(cx);
524                Ok(data)
525            } else {
526                Err(())
527            }
528        } else {
529            Err(())
530        };
531
532        match &self.buffer_source {
533            BufferSource::ArrayBufferView(buffer) | BufferSource::ArrayBuffer(buffer) => {
534                buffer.set(ptr::null_mut());
535            },
536        }
537        data
538    }

602    /// <https://streams.spec.whatwg.org/#abstract-opdef-cancopydatablockbytes>
603    // CanCopyDataBlockBytes(descriptorBuffer, destStart, queueBuffer, queueByteOffset, bytesToCopy)
604    pub(crate) fn can_copy_data_block_bytes(
605        &self,
606        cx: &mut JSContext,
607        to_index: usize,
608        from_buffer: &HeapBufferSource<ArrayBufferU8>,
609        from_index: usize,
610        bytes_to_copy: usize,
611    ) -> bool {
612        // Assert: toBuffer is an Object.
613        // Assert: toBuffer has an [[ArrayBufferData]] internal slot.
614        assert!(self.is_array_buffer_object());
615
616        // Assert: fromBuffer is an Object.
617        // Assert: fromBuffer has an [[ArrayBufferData]] internal slot.
618        assert!(from_buffer.is_array_buffer_object());
619
620        // If toBuffer is fromBuffer, return false.
621        match &self.buffer_source {
622            BufferSource::ArrayBufferView(heap) | BufferSource::ArrayBuffer(heap) => {
623                match &from_buffer.buffer_source {
624                    BufferSource::ArrayBufferView(from_heap) |
625                    BufferSource::ArrayBuffer(from_heap) => {
626                        if std::ptr::eq(heap.get(), from_heap.get()) {
627                            return false;
628                        }
629                    },
630                }
631            },
632        }
633
634        // If ! IsDetachedBuffer(toBuffer) is true, return false.
635        if self.is_detached_buffer(cx) {
636            return false;
637        }
638
639        // If ! IsDetachedBuffer(fromBuffer) is true, return false.
640        if from_buffer.is_detached_buffer(cx) {
641            return false;
642        }
643
644        // If toIndex + count > toBuffer.[[ArrayBufferByteLength]], return false.
645        if to_index + bytes_to_copy > self.byte_length() {
646            return false;
647        }
648
649        // If fromIndex + count > fromBuffer.[[ArrayBufferByteLength]], return false.
650        if from_index + bytes_to_copy > from_buffer.byte_length() {
651            return false;
652        }
653
654        // Return true.
655        true
656    }

658    pub(crate) fn copy_data_block_bytes(
659        &self,
660        cx: &mut JSContext,
661        dest_start: usize,
662        from_buffer: &HeapBufferSource<ArrayBufferU8>,
663        from_byte_offset: usize,
664        bytes_to_copy: usize,
665    ) -> bool {
666        match &self.buffer_source {
667            BufferSource::ArrayBufferView(heap) | BufferSource::ArrayBuffer(heap) => unsafe {
668                match &from_buffer.buffer_source {
669                    BufferSource::ArrayBufferView(from_heap) |
670                    BufferSource::ArrayBuffer(from_heap) => ArrayBufferCopyData(
671                        cx,
672                        Handle::from_raw(heap.handle()),
673                        dest_start,
674                        Handle::from_raw(from_heap.handle()),
675                        from_byte_offset,
676                        bytes_to_copy,
677                    ),
678                }
679            },
680        }
681    }

683    /// <https://streams.spec.whatwg.org/#can-transfer-array-buffer>
684    pub(crate) fn can_transfer_array_buffer(&self, cx: &mut JSContext) -> bool {
685        // Assert: O is an Object.
686        // Assert: O has an [[ArrayBufferData]] internal slot.
687        assert!(self.is_array_buffer_object());
688
689        // If ! IsDetachedBuffer(O) is true, return false.
690        if self.is_detached_buffer(cx) {
691            return false;
692        }
693
694        // If SameValue(O.[[ArrayBufferDetachKey]], undefined) is false, return false.
695        // Return true.
696        let mut is_defined = false;
697        match &self.buffer_source {
698            BufferSource::ArrayBufferView(heap) | BufferSource::ArrayBuffer(heap) => unsafe {
699                if !HasDefinedArrayBufferDetachKey(
700                    cx,
701                    Handle::from_raw(heap.handle()),
702                    &mut is_defined,
703                ) {
704                    return false;
705                }
706            },
707        }
708
709        !is_defined
710    }

712    /// <https://streams.spec.whatwg.org/#transfer-array-buffer>
713    pub(crate) fn transfer_array_buffer(
714        &self,
715        cx: &mut JSContext,
716    ) -> Fallible<RootedTraceableBox<HeapBufferSource<ArrayBufferU8>>> {
717        assert!(self.is_array_buffer_object());
718
719        // Assert: ! IsDetachedBuffer(O) is false.
720        assert!(!self.is_detached_buffer(cx));
721
722        // Let arrayBufferByteLength be O.[[ArrayBufferByteLength]].
723        // Step 3 (Reordered)
724        let buffer_length = self.byte_length();
725
726        // Let arrayBufferData be O.[[ArrayBufferData]].
727        // Step 2 (Reordered)
728        let buffer_data = match &self.buffer_source {
729            BufferSource::ArrayBufferView(buffer) | BufferSource::ArrayBuffer(buffer) => unsafe {
730                StealArrayBufferContents(cx, Handle::from_raw(buffer.handle()))
731            },
732        };
733
734        // Perform ? DetachArrayBuffer(O).
735        // This will throw an exception if O has an [[ArrayBufferDetachKey]] that is not undefined,
736        // such as a WebAssembly.Memory’s buffer. [WASM-JS-API-1]
737        if !self.detach_buffer(cx) {
738            rooted!(&in(cx) let mut rval = UndefinedValue());
739            unsafe {
740                assert!(JS_GetPendingException(cx, rval.handle_mut()));
741                JS_ClearPendingException(cx)
742            };
743
744            Err(Error::Type(c"can't transfer array buffer".to_owned()))
745        } else {
746            // Return a new ArrayBuffer object, created in the current Realm,
747            // whose [[ArrayBufferData]] internal slot value is arrayBufferData and
748            // whose [[ArrayBufferByteLength]] internal slot value is arrayBufferByteLength.
749            rooted!(&in(cx) let result = unsafe {
750                NewArrayBufferWithContents(cx, buffer_length, buffer_data)
751            });
752            if result.is_null() {
753                return Err(Error::JSFailed);
754            }
755            Ok(RootedTraceableBox::new(
756                HeapBufferSource::<ArrayBufferU8>::new(result.handle()),
757            ))
758        }
759    }
760}

812pub(crate) fn byte_size(byte_type: Type) -> u64 {
813    match byte_type {
814        Type::Int8 | Type::Uint8 | Type::Uint8Clamped => 1,
815        Type::Int16 | Type::Uint16 | Type::Float16 => 2,
816        Type::Int32 | Type::Uint32 | Type::Float32 => 4,
817        Type::Int64 | Type::Float64 | Type::BigInt64 | Type::BigUint64 => 8,
818        Type::Simd128 => 16,
819        _ => unreachable!("invalid scalar type"),
820    }
821}

833pub(crate) fn create_buffer_source_with_constructor(
834    cx: &mut JSContext,
835    constructor: &Constructor,
836    buffer_source: &HeapBufferSource<ArrayBufferU8>,
837    byte_offset: usize,
838    byte_length: usize,
839) -> Fallible<RootedTraceableBox<HeapBufferSource<ArrayBufferViewU8>>> {
840    match &buffer_source.buffer_source {
841        BufferSource::ArrayBuffer(heap) => match constructor {
842            Constructor::DataView => {
843                rooted!(&in(cx) let view = unsafe {
844                    JS_NewDataView(
845                        cx,
846                        Handle::from_raw(heap.handle()),
847                        byte_offset,
848                        byte_length,
849                    )
850                });
851                if view.is_null() {
852                    return Err(Error::JSFailed);
853                }
854                Ok(RootedTraceableBox::new(HeapBufferSource::new(
855                    view.handle(),
856                )))
857            },
858            Constructor::Name(name_type) => construct_typed_array(
859                cx,
860                name_type,
861                buffer_source,
862                byte_offset,
863                byte_length as i64,
864            ),
865        },
866        BufferSource::ArrayBufferView(_) => {
867            unreachable!("Can not create a new ArrayBufferView from an existing ArrayBufferView");
868        },
869    }

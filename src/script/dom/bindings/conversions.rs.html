46/// Get a `DomRoot<T>` for the given DOM object, unwrapping any wrapper
47/// around it first, and checking if the object is of the correct type.
48///
49/// Returns Err(()) if `obj` is an opaque security wrapper or if the object is
50/// not a reflector for a DOM object of the given type (as defined by the
51/// proto_id and proto_depth).
52pub(crate) fn root_from_object_static<T>(obj: *mut JSObject) -> Result<DomRoot<T>, ()>
53where
54    T: DomObject + IDLInterface,
55{
56    unsafe { native_from_object_static(obj).map(|ptr| DomRoot::from_ref(&*ptr)) }
57}
58
59/// Get a `DomRoot<T>` for a DOM object accessible from a `HandleObject`.
60pub(crate) fn root_from_handleobject<T>(
61    cx: &mut JSContext,
62    obj: HandleObject,
63) -> Result<DomRoot<T>, ()>
64where
65    T: DomObject + IDLInterface,
66{
67    unsafe { root_from_object(cx, obj.get()) }

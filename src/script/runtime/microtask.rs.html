78pub(crate) trait MicrotaskRunnable: JSTraceable + MallocSizeOf {
79    // must also take care of entering the realm
80    fn handler(&self, _cx: &mut JSContext) {}
81}

83/// A microtask that comes from a queueMicrotask() Javascript call
84#[derive(JSTraceable, MallocSizeOf)]
85#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
86pub(crate) struct UserMicrotask {
87    #[conditional_malloc_size_of]
88    pub(crate) callback: Rc<VoidFunction>,
89    pub(crate) global: Dom<GlobalScope>,
90}
91
92impl MicrotaskRunnable for UserMicrotask {
93    fn handler(&self, cx: &mut JSContext) {
94        let mut realm = enter_auto_realm(cx, &*self.global);
95        let cx = &mut realm;
96        let _ = self
97            .callback
98            .Call_(cx, &*self.global, ExceptionHandling::Report);
99    }
100}

102fn microtask_from_jsval(val: JSVal) -> *mut Box<dyn MicrotaskRunnable> {
103    val.to_private() as *const Box<dyn MicrotaskRunnable> as *mut Box<dyn MicrotaskRunnable>

106impl MicrotaskQueue {
107    /// Add a new microtask to this queue. It will be invoked as part of the next
108    /// microtask checkpoint.
109    #[expect(unsafe_code)]
110    pub(crate) fn enqueue(&self, cx: &JSContext, task: Box<dyn MicrotaskRunnable>) {
111        let task = Box::new(task);
112        let raw = Box::into_raw(task);
113        unsafe {
114            EnqueueMicroTask(cx, &PrivateValue(raw as *const c_void));
115        }
116    }
117
118    /// <https://html.spec.whatwg.org/multipage/#perform-a-microtask-checkpoint>
119    /// Perform a microtask checkpoint, executing all queued microtasks until the queue is empty.
120    #[expect(unsafe_code)]
121    pub(crate) fn checkpoint(&self, cx: &mut JSContext, globalscopes: Vec<DomRoot<GlobalScope>>) {
122        // Step 1. If the event loop's performing a microtask checkpoint is true, then return.
123        if self.performing_a_microtask_checkpoint.get() {
124            return;
125        }
126
127        // Step 2. Set the event loop's performing a microtask checkpoint to true.
128        self.performing_a_microtask_checkpoint.set(true);
129
130        debug!("Now performing a microtask checkpoint");
131
132        rooted!(&in(cx) let mut generic_task: js::jsapi::GenericMicroTask);
133        rooted!(&in(cx) let mut js_micro_task: *mut js::jsapi::JSMicroTask);
134        rooted!(&in(cx) let mut execution_global: *mut js::jsapi::JSObject);
135        rooted!(&in(cx) let mut incumbent_global: *mut js::jsapi::JSObject);
136        rooted!(&in(cx) let mut data: *mut js::jsapi::JSObject);
137
138        // Step 3. While the event loop's microtask queue is not empty:
139        // based on https://spidermonkey.dev/blog/2026/01/15/job-responsibility.html#running-micro-tasks
140        // and https://searchfox.org/firefox-main/rev/7ae92e67d094086cd3e09918ec94b6278a948535/xpcom/base/CycleCollectedJSContext.cpp#1176
141        // and its helper functions
142        while unsafe { HasAnyMicroTasks(cx) } {
143            unsafe { JS_DequeueNextMicroTask(cx, generic_task.handle_mut()) };
144
145            // https://searchfox.org/firefox-main/rev/50691777d300fffc7d1f7844b59769109bc76f3e/xpcom/base/CycleCollectedJSContext.cpp#916
146            if !unsafe { IsJSMicroTask(generic_task.as_ptr()) } {
147                rooted!(&in(cx) let task = unsafe {
148                    Box::from_raw(
149                        microtask_from_jsval(*generic_task),
150                    )
151                });
152                task.handler(cx);
153                continue;
154            }
155
156            js_micro_task.set(unsafe { ToMaybeWrappedJSMicroTask(generic_task.as_ptr()) });
157            execution_global.set(unsafe { GetExecutionGlobalFromJSMicroTask(js_micro_task.get()) });
158            if !unsafe {
159                MaybeGetHostDefinedDataFromJSMicroTask(
160                    js_micro_task.get(),
161                    incumbent_global.handle_mut(),
162                    data.handle_mut(),
163                )
164            } {
165                continue;
166            }
167
168            let interaction = if let Some(promise) =
169                NonNull::new(unsafe { MaybeGetPromiseFromJSMicroTask(js_micro_task.get()) })
170            {
171                unsafe { GetPromiseUserInputEventHandlingState(promise.as_ptr()) }
172            } else {
173                PromiseUserInputEventHandlingState::DontCare
174            };
175            let _maybe_user_interacting_guard = if interaction ==
176                PromiseUserInputEventHandlingState::HadUserInteractionAtCreation
177            {
178                Some(ScriptThread::user_interacting_guard())
179            } else {
180                None
181            };
182            let global_scope = unsafe { GlobalScope::from_object(execution_global.get()) };
183            run_a_script::<DomTypeHolder, _, _>(cx, &global_scope, |cx| {
184                let mut r = || {
185                    let mut realm = AutoRealm::new_from_handle(cx, execution_global.handle());
186                    let _ = unsafe { RunJSMicroTask(&mut realm, js_micro_task.handle()) };
187                };
188                if incumbent_global.get().is_null() {
189                    r();
190                } else {
191                    let global_scope = unsafe { GlobalScope::from_object(incumbent_global.get()) };
192                    run_a_callback::<DomTypeHolder, _>(&global_scope, r);
193                }
194            });
195        }
196
197        // Step 4. For each environment settings object settingsObject whose responsible
198        // event loop is this event loop, notify about rejected promises given
199        // settingsObject's global object.
200        for global in globalscopes.clone().into_iter() {
201            notify_about_rejected_promises(cx, &global);
202        }
203
204        // https://html.spec.whatwg.org/multipage/#perform-a-microtask-checkpoint
205        // Step 5. Cleanup Indexed Database transactions.
206        // https://w3c.github.io/IndexedDB/#cleanup-indexed-database-transactions
207        // “These steps are invoked by [HTML]. They ensure that transactions created by a script call
208        // to transaction() are deactivated once the task that invoked the script has completed.”
209        for global in globalscopes.iter() {
210            if let Some(factory) = global.indexeddb_factory() {
211                let _ = factory.cleanup_indexeddb_transactions(cx);
212            }
213        }
214
215        // TODO: Step 6. Perform ClearKeptObjects().
216
217        // Step 7. Set the event loop's performing a microtask checkpoint to false.
218        self.performing_a_microtask_checkpoint.set(false);
219        // TODO: Step 8. Record timing info for microtask checkpoint.
220    }
221
222    #[expect(unsafe_code)]
223    pub(crate) fn clear(&self, cx: &JSContext) {
224        rooted!(&in(cx) let mut generic_task: js::jsapi::GenericMicroTask);
225        while unsafe { HasAnyMicroTasks(cx) } {
226            unsafe { JS_DequeueNextMicroTask(cx, generic_task.handle_mut()) };
227            if !unsafe { IsJSMicroTask(generic_task.as_ptr()) } {
228                let task = unsafe { Box::from_raw(microtask_from_jsval(*generic_task)) };
229                drop(task);
230            }
231        }
232    }
233}

235#[expect(unsafe_code)]
236pub(crate) unsafe extern "C" fn trace_non_gc_things_micro_task(
237    trc: *mut JSTracer,
238    val: *mut JSVal,
239) {
240    wrap_panic(&mut || {
241        let task = microtask_from_jsval(unsafe { *val });
242        unsafe { (**task).trace(trc) };
243    })

21#[derive(JSTraceable, MallocSizeOf)]
22#[cfg_attr(crown, crown::unrooted_must_root_lint::must_root)]
23pub(crate) struct DocumentSelectionDragHandler {
24    /// The node with `user-select: contain` that this selection must not leave, if any.
25    user_select_contain_node_for_selection_anchor: Option<Dom<Node>>,
26}
27
28impl DocumentSelectionDragHandler {
29    pub(crate) fn new(user_select_contain_node: Option<&Node>) -> Self {
30        Self {
31            user_select_contain_node_for_selection_anchor: user_select_contain_node
32                .map(Dom::from_ref),
33        }
34    }
35
36    pub(crate) fn still_connected(&self) -> bool {
37        self.user_select_contain_node_for_selection_anchor
38            .as_ref()
39            .is_none_or(|node| node.is_connected())
40    }
41
42    /// Process a mouse move event on this [`DocumentSelectionDragHandler`].
43    ///
44    /// Returns `true` if the drag should continue and `false` otherwise.
45    pub(crate) fn moved(&self, cx: &mut JSContext, hit_test_result: &HitTestResult) -> bool {
46        let Some((container, offset)) = hit_test_result.dom_position_for_selection.as_ref() else {
47            return true;
48        };
49        let Some(selection) = container.owner_document().selection() else {
50            return true;
51        };
52        let (container, offset) = adjust_focus_for_user_select(
53            cx,
54            selection.composed_anchor_position(),
55            container.clone(),
56            *offset,
57            self.user_select_contain_node_for_selection_anchor
58                .as_deref(),
59        );
60        selection.collapse_or_extend_to_dom_position(cx, &container, offset);
61        true
62    }
63}

65/// Adjust the range boundary for the selection anchor (start of a drag gesture)
66/// based on [`user-select`] of the container node and of its ancestors.
67///
68/// Returns `None` if no new drag gesture should be started (`user-select: none`), or:
69///
70/// * The new anchor container
71/// * The new anchor offset
72/// * The node with `user-select: contain` that the selection must not leave, if any
73///
74/// [`user-select`]: https://drafts.csswg.org/css-ui-4/#content-selection
75pub(crate) fn adjust_anchor_for_user_select(
76    no_gc: &NoGC,
77    mut anchor_container_candidate: DomRoot<Node>,
78    mut anchor_offset_candidate: Utf32CodeUnitsOrNodeOffset,
79) -> Option<(
80    DomRoot<Node>,
81    Utf32CodeUnitsOrNodeOffset,
82    Option<DomRoot<Node>>,
83)> {
84    let mut cache = Default::default();
85    if anchor_container_candidate.used_user_select(no_gc, &mut cache) == UsedUserSelect::None {
86        return None;
87    }
88
89    // The nearest inclusive ancestor with `user-select: contain`
90    let mut nearest_with_user_select_contain = None;
91
92    // The furthest inclusive ancestor with `user-select: all`, as long as all intermediate
93    // inclusive ancestors also have `user-select: all`.
94    let mut furthest_with_user_select_all = None;
95    let mut each_so_far_has_user_select_all = true;
96
97    for ancestor in anchor_container_candidate.inclusive_ancestors_in_flat_tree_unrooted(no_gc) {
98        match ancestor.used_user_select(no_gc, &mut cache) {
99            UsedUserSelect::Text | UsedUserSelect::None => {
100                each_so_far_has_user_select_all = false;
101            },
102            UsedUserSelect::Contain => {
103                nearest_with_user_select_contain = Some(ancestor.as_rooted());
104                // `each_so_far_has_user_select_all` should be set to false but will no
105                // longer be used after this break.
106                break;
107            },
108            UsedUserSelect::All => {
109                if each_so_far_has_user_select_all {
110                    furthest_with_user_select_all = Some(ancestor);
111                }
112            },
113        }
114    }
115
116    if let Some(atomic) = furthest_with_user_select_all {
117        // TODO: to properly implement `user-select: all` we should keep track of two potential
118        // start boundaries: one with offset zero as is done here, and another with offset
119        // `Node::len(atomic)` representing the end of that node. The latter would be used
120        // when the range is backwards (if the end boundary is outside of and before `atomic`)
121        // so that `atomic` would be entirely selected regardless of the range direction.
122        anchor_container_candidate = atomic.as_rooted();
123        anchor_offset_candidate = Utf32CodeUnitsOrNodeOffset(0);
124    }
125    Some((
126        anchor_container_candidate,
127        anchor_offset_candidate,
128        nearest_with_user_select_contain,
129    ))
130}

132/// Adjust the range boundary for the selection focus (end of a drag gesture)
133/// based on [`user-select`] of the container node and of its ancestors.
134///
135/// [`user-select`]: https://drafts.csswg.org/css-ui-4/#content-selection
136pub(crate) fn adjust_focus_for_user_select(
137    no_gc: &NoGC,
138    anchor: Option<(DomRoot<Node>, u32)>,
139    mut focus_container_candidate: DomRoot<Node>,
140    mut focus_offset_candidate: Utf32CodeUnitsOrNodeOffset,
141    user_select_contain_node_for_anchor: Option<&Node>,
142) -> (DomRoot<Node>, Utf32CodeUnitsOrNodeOffset) {
143    let mut cache = Default::default();
144    let mut user_select_contain_node_for_anchor_is_inclusive_ancestor = false;
145    let mut each_so_far_has_user_select_all = true;
146    let mut furthest_node_with_user_select_all = None;
147    let mut each_so_far_has_user_select_none = true;
148    // Either `user-select: contain` the selection starts outside of, or `user-select: none`
149    let mut furthest_node_to_avoid = None;
150
151    for inclusive_ancestor in
152        focus_container_candidate.inclusive_ancestors_in_flat_tree_unrooted(no_gc)
153    {
154        if user_select_contain_node_for_anchor == Some(&**inclusive_ancestor) {
155            user_select_contain_node_for_anchor_is_inclusive_ancestor = true
156        }
157        match inclusive_ancestor.used_user_select(no_gc, &mut cache) {
158            UsedUserSelect::Text => {
159                each_so_far_has_user_select_all = false;
160                each_so_far_has_user_select_none = false;
161            },
162            UsedUserSelect::None => {
163                each_so_far_has_user_select_all = false;
164                if each_so_far_has_user_select_none {
165                    furthest_node_to_avoid = Some(inclusive_ancestor)
166                }
167            },
168            UsedUserSelect::Contain => {
169                each_so_far_has_user_select_all = false;
170                each_so_far_has_user_select_none = false;
171                if !user_select_contain_node_for_anchor_is_inclusive_ancestor {
172                    furthest_node_to_avoid = Some(inclusive_ancestor)
173                }
174            },
175            UsedUserSelect::All => {
176                each_so_far_has_user_select_none = false;
177                if each_so_far_has_user_select_all {
178                    furthest_node_with_user_select_all = Some(inclusive_ancestor)
179                }
180            },
181        }
182        if !each_so_far_has_user_select_all &&
183            !each_so_far_has_user_select_none &&
184            user_select_contain_node_for_anchor_is_inclusive_ancestor
185        {
186            // Nothing else to find in ancestors
187            break;
188        }
189    }
190    let selection_is_backward = || {
191        anchor
192            .as_ref()
193            .is_some_and(|(anchor_container, anchor_offset)| {
194                let (ordering, _containment) = compare_dom_positions::<
195                    FlatTreeForSelectionNoGcTraversal,
196                >(
197                    no_gc,
198                    anchor_container,
199                    *anchor_offset,
200                    &focus_container_candidate,
201                    focus_container_candidate.to_sibling_or_utf16_offset(focus_offset_candidate),
202                );
203                ordering == Some(Ordering::Greater)
204            })
205    };
206    if let Some(contain_for_anchor) = user_select_contain_node_for_anchor &&
207        !user_select_contain_node_for_anchor_is_inclusive_ancestor
208    {
209        // `focus_container` is outside of `contain_for_anchor`:
210        // find the closest position within `contain_for_anchor`: either its start or end
211        focus_offset_candidate = if selection_is_backward() {
212            Utf32CodeUnitsOrNodeOffset(0)
213        } else {
214            Utf32CodeUnitsOrNodeOffset(Node::len(contain_for_anchor))
215        };
216        focus_container_candidate = DomRoot::from_ref(contain_for_anchor);
217    } else if let Some(focus_ancestor_to_avoid) = furthest_node_to_avoid {
218        focus_offset_candidate = if selection_is_backward() {
219            Utf32CodeUnitsOrNodeOffset(Node::len(&focus_ancestor_to_avoid))
220        } else {
221            Utf32CodeUnitsOrNodeOffset(0)
222        };
223        focus_container_candidate = focus_ancestor_to_avoid.as_rooted();
224    } else if let Some(with_user_select_all) = furthest_node_with_user_select_all {
225        // anchor was snapped to the start of the `user-select: all` element,
226        // so snap the focus to the end unconditionally
227        focus_offset_candidate = Utf32CodeUnitsOrNodeOffset(Node::len(&with_user_select_all));
228        focus_container_candidate = with_user_select_all.as_rooted();
229    }
230
231    (focus_container_candidate, focus_offset_candidate)
232}

234impl Node {
235    /// Returns the used value of <https://drafts.csswg.org/css-ui-4/#propdef-user-select>.
236    ///
237    /// It is the caller’s responsibility to ensure that style is up to date for this node and
238    /// its (flat tree) ancestors.
239    ///
240    /// `cache` can be initialized with `Default::default()`, and should be shared across calls
241    /// for nodes that may share some (flat tree) ancestors.
242    pub(crate) fn used_user_select<'no_gc>(
243        &self,
244        no_gc: &'no_gc NoGC,
245        cache: &mut FxHashMap<UnrootedDom<'no_gc, Node>, UsedUserSelect>,
246    ) -> UsedUserSelect {
247        let cache_key = UnrootedDom::from_ref(self, no_gc);
248        if let Some(&used_value) = cache.get(&cache_key) {
249            return used_value;
250        }
251        // > The used value is the same as the computed value, except:
252        // >
253        // > 1. on editable elements where the used value is always `contain`
254        // >    regardless of the computed value
255        // > 2. when the computed value is `auto`, in which case the used value
256        // >    is one of the other values as defined below
257        // >
258        // > For the purpose of this specification, an editable element is
259        // > either an editing host or a mutable form control with textual
260        // > content, such as textarea.
261        //
262        // For form controls, we handle selection separately without looking at
263        // `user-select`.
264        if self.is_editing_host() {
265            let used_value = UsedUserSelect::Contain;
266            cache.insert(cache_key, used_value);
267            return used_value;
268        }
269        let computed_value = self
270            .downcast()
271            .and_then(Element::computed_user_select)
272            // Non-element nodes and unstyled elements: use the initial value
273            .unwrap_or(UserSelect::Auto);
274        let used_value = match computed_value {
275            UserSelect::Text => UsedUserSelect::Text,
276            UserSelect::None => UsedUserSelect::None,
277            UserSelect::Contain => UsedUserSelect::Contain,
278            UserSelect::All => UsedUserSelect::All,
279            UserSelect::Auto => {
280                let parent_used_value = self
281                    .parent_in_flat_tree(no_gc)
282                    .into_parent()
283                    .map(|parent| parent.used_user_select(no_gc, cache));
284                // > The used value of `auto` is determined as follows:
285                // > * On the `::before` and `::after` pseudo-elements, the used value is `none`
286                // > * If the element is an editable element, the used value is `contain`
287                // > * Otherwise, if the used value of `user-select` on the parent of this element
288                // >   is `all`, the used value is `all`
289                // > * Otherwise, if the used value of `user-select` on the parent of this element
290                // >   is `none`, the used value is `none`
291                // > * Otherwise, the used value is `text`
292                match parent_used_value {
293                    Some(UsedUserSelect::All) => UsedUserSelect::All,
294                    Some(UsedUserSelect::None) => UsedUserSelect::None,
295                    _ => UsedUserSelect::Text,
296                }
297            },
298        };
299        cache.insert(cache_key, used_value);
300        used_value
301    }

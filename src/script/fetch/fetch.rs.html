65/// Fetch canceller object. By default initialized to having a
66/// request associated with it, which can be aborted or terminated.
67/// Calling `ignore` will sever the relationship with the request,
68/// meaning it cannot be cancelled through this canceller from that point on.
69#[derive(Default, JSTraceable, MallocSizeOf)]
70pub(crate) struct FetchCanceller {
71    #[no_trace]
72    request_id: Option<RequestId>,
73    #[no_trace]
74    core_resource_thread: Option<CoreResourceThread>,
75    keep_alive: bool,
76}
77
78impl FetchCanceller {
79    /// Create a FetchCanceller associated with a request,
80    /// and a particular(public vs private) resource thread.
81    pub(crate) fn new(
82        request_id: RequestId,
83        keep_alive: bool,
84        core_resource_thread: CoreResourceThread,
85    ) -> Self {
86        Self {
87            request_id: Some(request_id),
88            core_resource_thread: Some(core_resource_thread),
89            keep_alive,
90        }
91    }
92
93    pub(crate) fn keep_alive(&self) -> bool {
94        self.keep_alive
95    }
96
97    fn cancel(&mut self) {
98        if let Some(request_id) = self.request_id.take() {
99            // stop trying to make fetch happen
100            // it's not going to happen
101
102            if let Some(ref core_resource_thread) = self.core_resource_thread {
103                // No error handling here. Cancellation is a courtesy call,
104                // we don't actually care if the other side heard.
105                cancel_async_fetch(vec![request_id], core_resource_thread);
106            }
107        }
108    }
109
110    /// Use this if you don't want it to send a cancellation request
111    /// on drop (e.g. if the fetch completes)
112    pub(crate) fn ignore(&mut self) {
113        let _ = self.request_id.take();
114    }
115
116    /// <https://fetch.spec.whatwg.org/#fetch-controller-abort>
117    pub(crate) fn abort(&mut self) {
118        self.cancel();
119    }
120
121    /// <https://fetch.spec.whatwg.org/#fetch-controller-terminate>
122    pub(crate) fn terminate(&mut self) {
123        self.cancel();
124    }
125}

153/// <https://fetch.spec.whatwg.org/#concept-fetch-group>
154#[derive(MallocSizeOf)]
155pub(crate) struct FetchGroup {
156    /// The [`CoreResourceThread`] for this [`FetchGroup`].
157    core_resource_thread: CoreResourceThread,
158    /// <https://fetch.spec.whatwg.org/#fetch-group-deferred-fetch-records>
159    #[conditional_malloc_size_of]
160    pub(crate) deferred_fetch_records: FxHashMap<DeferredFetchRecordId, QueuedDeferredFetchRecord>,
161    /// <https://fetch.spec.whatwg.org/#concept-fetch-record>
162    pub(crate) fetch_records: FxHashMap<RequestId, FetchRecord>,
163}
164
165impl FetchGroup {
166    pub(crate) fn new(core_resource_thread: CoreResourceThread) -> Self {
167        Self {
168            core_resource_thread,
169            deferred_fetch_records: Default::default(),
170            fetch_records: Default::default(),
171        }
172    }
173
174    pub(crate) fn fetch<Listener: FetchResponseListener>(
175        &mut self,
176        request: RequestBuilder,
177        listener: NetworkListener<Listener>,
178    ) {
179        self.fetch_records.insert(
180            request.id,
181            FetchRecord {
182                controller: Some(FetchCanceller::new(
183                    request.id,
184                    request.keep_alive,
185                    self.core_resource_thread.clone(),
186                )),
187                done: false,
188            },
189        );
190        fetch_async(
191            &self.core_resource_thread,
192            request,
193            None,
194            listener.into_callback(),
195        );
196    }
197
198    pub(crate) fn deferred_fetches(&self) -> Vec<QueuedDeferredFetchRecord> {
199        self.deferred_fetch_records.values().cloned().collect()
200    }
201
202    fn append_deferred_fetch(
203        &mut self,
204        deferred_record: DeferredFetchRecord,
205    ) -> DeferredFetchRecordId {
206        let deferred_fetch_record_id = DeferredFetchRecordId::default();
207        self.deferred_fetch_records
208            .insert(deferred_fetch_record_id, Rc::new(deferred_record));
209        deferred_fetch_record_id
210    }
211
212    pub(crate) fn deferred_fetch_record_for_id(
213        &self,
214        deferred_fetch_record_id: &DeferredFetchRecordId,
215    ) -> QueuedDeferredFetchRecord {
216        self.deferred_fetch_records
217            .get(deferred_fetch_record_id)
218            .expect("Should always use a generated fetch_record_id instead of passing your own")
219            .clone()
220    }
221
222    pub(crate) fn fetch_controller(
223        &mut self,
224        request_id: &RequestId,
225    ) -> Option<&mut FetchCanceller> {
226        self.fetch_records.get_mut(request_id)?.controller.as_mut()
227    }
228
229    /// <https://fetch.spec.whatwg.org/#concept-fetch-group-terminate>
230    ///
231    /// Returns `true` if any fetches were cancelled and `false` otherwise.
232    pub(crate) fn terminate(&mut self, global: &GlobalScope) -> bool {
233        // Step 1. For each fetch record record of fetchGroup’s fetch records,
234        // if record’s controller is non-null and record’s request’s done flag
235        // is unset and keepalive is false, terminate record’s controller.
236        let mut cancelled_any = false;
237        self.fetch_records.retain(|_, fetch_record| {
238            let Some(controller) = fetch_record.controller.as_mut() else {
239                return false;
240            };
241            if fetch_record.done {
242                return false;
243            }
244            if !controller.keep_alive() {
245                controller.terminate();
246                cancelled_any = true;
247                return false;
248            }
249            true
250        });
251
252        // Step 2. Process deferred fetches for fetchGroup.
253        self.process_deferred_fetches(global);
254
255        cancelled_any
256    }
257
258    /// <https://fetch.spec.whatwg.org/#process-deferred-fetches>
259    pub(crate) fn process_deferred_fetches(&mut self, global: &GlobalScope) {
260        // Step 1. For each deferred fetch record deferredRecord of fetchGroup’s
261        // deferred fetch records, process a deferred fetch deferredRecord.
262        for deferred_fetch in self.deferred_fetches() {
263            self.process_a_deferred_fetch(global, &deferred_fetch);
264        }
265    }
266
267    /// <https://fetch.spec.whatwg.org/#process-a-deferred-fetch>
268    pub(crate) fn process_a_deferred_fetch(
269        &mut self,
270        global: &GlobalScope,
271        deferred_fetch: &DeferredFetchRecord,
272    ) {
273        // Step 1. If deferredRecord’s invoke state is not "pending", then return.
274        if deferred_fetch.invoke_state.get() != DeferredFetchRecordInvokeState::Pending {
275            return;
276        }
277        // Step 2. Set deferredRecord’s invoke state to "sent".
278        deferred_fetch
279            .invoke_state
280            .set(DeferredFetchRecordInvokeState::Sent);
281        // Step 3. Fetch deferredRecord’s request.
282        let fetch_later_listener = FetchLaterListener {
283            url: deferred_fetch.request.url(),
284            global: Trusted::new(global),
285        };
286        let task_source = global.task_manager().networking_task_source().to_sendable();
287        self.fetch(
288            request_init_from_request(deferred_fetch.request.clone(), global),
289            NetworkListener::new(fetch_later_listener, task_source, global),
290        );
291        // Step 4 is handled by caller
292    }
293
294    pub(crate) fn mark_fetch_request_as_done(&mut self, request_id: &RequestId) {
295        if let Some(fetch_record) = self.fetch_records.get_mut(request_id) {
296            fetch_record.done = true;
297        }
298    }
299}

301fn request_init_from_request(request: NetTraitsRequest, global: &GlobalScope) -> RequestBuilder {
302    let mut builder = RequestBuilder::new(
303        request.target_webview_id,
304        request.url_with_blob_claim(),
305        request.referrer,
306    )
307    .method(request.method)
308    .headers(request.headers)
309    .unsafe_request(request.unsafe_request)
310    .body(request.body)
311    .destination(request.destination)
312    .synchronous(request.synchronous)
313    .mode(request.mode)
314    .cache_mode(request.cache_mode)
315    .use_cors_preflight(request.use_cors_preflight)
316    .credentials_mode(request.credentials_mode)
317    .use_url_credentials(request.use_url_credentials)
318    .referrer_policy(request.referrer_policy)
319    .pipeline_id(request.pipeline_id)
320    .redirect_mode(request.redirect_mode)
321    .integrity_metadata(request.integrity_metadata)
322    .cryptographic_nonce_metadata(request.cryptographic_nonce_metadata)
323    .parser_metadata(request.parser_metadata)
324    .initiator(request.initiator)
325    .client(global.request_client(None))
326    .response_tainting(request.response_tainting);
327    builder.id = request.id;
328    builder.reload_navigation = request.reload_navigation;
329    builder.history_navigation = request.history_navigation;
330    builder
331}

333/// <https://fetch.spec.whatwg.org/#abort-fetch>
334fn abort_fetch_call(
335    promise: Rc<Promise>,
336    request: &Request,
337    response_object: Option<&Response>,
338    abort_reason: HandleValue,
339    global: &GlobalScope,
340    cx: &mut JSContext,
341) {
342    // Step 1. Reject promise with error.
343    promise.reject(cx, abort_reason);
344    // Step 2. If request’s body is non-null and is readable, then cancel request’s body with error.
345    if let Some(body) = request.body() &&
346        body.is_readable()
347    {
348        body.cancel(cx, global, abort_reason);
349    }
350    // Step 3. If responseObject is null, then return.
351    // Step 4. Let response be responseObject’s response.
352    let Some(response) = response_object else {
353        return;
354    };
355    // Step 5. If response’s body is non-null and is readable, then error response’s body with error.
356    if let Some(body) = response.body() &&
357        body.is_readable()
358    {
359        body.error(cx, abort_reason);
360    }
361}

363/// <https://fetch.spec.whatwg.org/#dom-global-fetch>
364#[expect(non_snake_case)]
365pub(crate) fn Fetch(
366    global: &GlobalScope,
367    input: RequestInfo,
368    init: RootedTraceableBox<RequestInit>,
369    cx: &mut CurrentRealm,
370) -> Rc<Promise> {
371    // Step 1. Let p be a new promise.
372    let promise = Promise::new_in_realm(cx);
373
374    // Step 7. Let responseObject be null.
375    // NOTE: We do initialize the object earlier so we can use it to track errors.
376    let response = Response::new(cx, global);
377    response.Headers(cx).set_guard(Guard::Immutable);
378
379    // Step 2. Let requestObject be the result of invoking the initial value of Request as constructor
380    //         with input and init as arguments. If this throws an exception, reject p with it and return p.
381    let request_object = match Request::Constructor(cx, global, None, input, init) {
382        Err(e) => {
383            response.error_stream(cx, e.clone());
384            promise.reject_error(cx, e);
385            return promise;
386        },
387        Ok(r) => r,
388    };
389    // Step 3. Let request be requestObject’s request.
390    let request = request_object.request().clone();
391
392    // Step 4. If requestObject’s signal is aborted, then:
393    let signal = request_object.Signal();
394    if signal.aborted() {
395        // Step 4.1. Abort the fetch() call with p, request, null, and requestObject’s signal’s abort reason.
396        rooted!(&in(cx) let mut abort_reason = UndefinedValue());
397        signal.Reason(abort_reason.handle_mut());
398        abort_fetch_call(
399            promise.clone(),
400            &request_object,
401            None,
402            abort_reason.handle(),
403            global,
404            cx,
405        );
406        // Step 4.2. Return p.
407        return promise;
408    }
409
410    // Step 5. Let globalObject be request’s client’s global object.
411    // NOTE:   We already get the global object as an argument
412    let mut request_builder = request_init_from_request(request, global);
413
414    // Step 6. If globalObject is a ServiceWorkerGlobalScope object, then set request’s
415    //         service-workers mode to "none".
416    if global.is::<ServiceWorkerGlobalScope>() {
417        request_builder.service_workers_mode = ServiceWorkersMode::None;
418    }
419
420    // Step 8. Let relevantRealm be this’s relevant realm.
421    //
422    // Is `comp` as argument
423
424    // Step 9. Let locallyAborted be false.
425    // Step 10. Let controller be null.
426    let fetch_context = FetchContext {
427        fetch_promise: Some(TrustedPromise::new(promise.clone())),
428        response_object: Trusted::new(&*response),
429        request: Trusted::new(&*request_object),
430        global: Trusted::new(global),
431        locally_aborted: false,
432        url: request_builder.url.url(),
433    };
434    let network_listener = NetworkListener::new(
435        fetch_context,
436        global.task_manager().networking_task_source().to_sendable(),
437        global,
438    );
439    let fetch_context = network_listener.context.clone();
440
441    // Step 11. Add the following abort steps to requestObject’s signal:
442    signal.add(&AbortAlgorithm::Fetch(fetch_context));
443
444    // Step 12. Set controller to the result of calling fetch given request and
445    // processResponse given response being these steps:
446    global
447        .fetch_group_mut()
448        .fetch(request_builder, network_listener);
449
450    // Step 13. Return p.
451    promise
452}

454/// <https://fetch.spec.whatwg.org/#queue-a-deferred-fetch>
455fn queue_deferred_fetch(
456    request: NetTraitsRequest,
457    activate_after: Finite<f64>,
458    global: &GlobalScope,
459) -> DeferredFetchRecordId {
460    let trusted_global = Trusted::new(global);
461    let mut request = request;
462    // Step 1. Populate request from client given request.
463    request.client = Some(global.request_client(None));
464    request.populate_request_from_client();
465    // Step 2. Set request’s service-workers mode to "none".
466    request.service_workers_mode = ServiceWorkersMode::None;
467    // Step 3. Set request’s keepalive to true.
468    request.keep_alive = true;
469    // Step 4. Let deferredRecord be a new deferred fetch record whose request is request, and whose notify invoked is onActivatedWithoutTermination.
470    let deferred_record = DeferredFetchRecord {
471        request,
472        invoke_state: Cell::new(DeferredFetchRecordInvokeState::Pending),
473        activated: Cell::new(false),
474    };
475
476    // Step 5. Append deferredRecord to request’s client’s fetch group’s deferred fetch records.
477    let deferred_fetch_record_id = global
478        .fetch_group_mut()
479        .append_deferred_fetch(deferred_record);
480
481    // Step 6. If activateAfter is non-null, then run the following steps in parallel:
482    global.schedule_timer(TimerEventRequest {
483        callback: Box::new(move || {
484            // Step 6.2. Process deferredRecord.
485            let global = trusted_global.root();
486            let mut fetch_group = global.fetch_group_mut();
487            let deferred_fetch_record =
488                fetch_group.deferred_fetch_record_for_id(&deferred_fetch_record_id);
489            fetch_group.process_a_deferred_fetch(&global, &deferred_fetch_record);
490
491            // Last step of https://fetch.spec.whatwg.org/#process-a-deferred-fetch
492            //
493            // Step 4. Queue a global task on the deferred fetch task source with
494            // deferredRecord’s request’s client’s global object to run deferredRecord’s notify invoked.
495            let trusted_global = trusted_global.clone();
496            global.task_manager().deferred_fetch_task_source().queue(
497                task!(notify_deferred_record: move || {
498                    trusted_global.root().fetch_group().deferred_fetch_record_for_id(&deferred_fetch_record_id).activate();
499                }),
500            );
501        }),
502        // Step 6.1. The user agent should wait until any of the following conditions is met:
503        duration: Duration::from_millis(*activate_after as u64),
504    });
505    // Step 7. Return deferredRecord.
506    deferred_fetch_record_id
507}

509/// <https://fetch.spec.whatwg.org/#dom-window-fetchlater>
510#[expect(non_snake_case, unsafe_code)]
511pub(crate) fn FetchLater(
512    cx: &mut JSContext,
513    window: &Window,
514    input: RequestInfo,
515    init: RootedTraceableBox<DeferredRequestInit>,
516) -> Fallible<DomRoot<FetchLaterResult>> {
517    let global_scope = window.upcast();
518    let document = window.Document();
519    // Step 1. Let requestObject be the result of invoking the initial value
520    // of Request as constructor with input and init as arguments.
521    let request_object = Request::constructor(cx, global_scope, None, input, &init.parent)?;
522    // Step 2. If requestObject’s signal is aborted, then throw signal’s abort reason.
523    let signal = request_object.Signal();
524    if signal.aborted() {
525        rooted!(&in(cx) let mut abort_reason = UndefinedValue());
526        signal.Reason(abort_reason.handle_mut());
527        unsafe {
528            assert!(!JS_IsExceptionPending(cx));
529            JS_SetPendingException(cx, abort_reason.handle(), ExceptionStackBehavior::Capture)
530        };
531        return Err(Error::JSFailed);
532    }
533    // Step 3. Let request be requestObject’s request.
534    let request = request_object.request();
535    // Step 4. Let activateAfter be null.
536    let mut activate_after = Finite::wrap(0_f64);
537    // Step 5. If init is given and init["activateAfter"] exists, then set
538    // activateAfter to init["activateAfter"].
539    if let Some(init_activate_after) = init.activateAfter.as_ref() {
540        activate_after = *init_activate_after;
541    }
542    // Step 6. If activateAfter is less than 0, then throw a RangeError.
543    if *activate_after < 0.0 {
544        return Err(Error::Range(c"activateAfter must be at least 0".to_owned()));
545    }
546    // Step 7. If this’s relevant global object’s associated document is not fully active, then throw a TypeError.
547    if !document.is_fully_active() {
548        return Err(Error::Type(c"Document is not fully active".to_owned()));
549    }
550    let url = request.url();
551    // Step 8. If request’s URL’s scheme is not an HTTP(S) scheme, then throw a TypeError.
552    if !matches!(url.scheme(), "http" | "https") {
553        return Err(Error::Type(c"URL is not http(s)".to_owned()));
554    }
555    // Step 9. If request’s URL is not a potentially trustworthy URL, then throw a SecurityError.
556    if !url.is_potentially_trustworthy() {
557        return Err(Error::Type(c"URL is not trustworthy".to_owned()));
558    }
559    // Step 10. If request’s body is not null, and request’s body length is null, then throw a TypeError.
560    if request
561        .body
562        .as_ref()
563        .is_some_and(|body| body.len().is_none())
564    {
565        return Err(Error::Type(c"Body is empty".to_owned()));
566    }
567    // Step 11. If the available deferred-fetch quota given request’s client and request’s URL’s
568    // origin is less than request’s total request length, then throw a "QuotaExceededError" DOMException.
569    let quota = document.available_deferred_fetch_quota(request.url().origin());
570    let requested = request.total_request_length() as isize;
571    if quota < requested {
572        return Err(Error::QuotaExceeded {
573            quota: Some(Finite::wrap(quota as f64)),
574            requested: Some(Finite::wrap(requested as f64)),
575        });
576    }
577    // Step 12. Let activated be false.
578    // Step 13. Let deferredRecord be the result of calling queue a deferred fetch given request,
579    // activateAfter, and the following step: set activated to true.
580    let deferred_record_id = queue_deferred_fetch(request.clone(), activate_after, global_scope);
581    // Step 14. Add the following abort steps to requestObject’s signal: Set deferredRecord’s invoke state to "aborted".
582    signal.add(&AbortAlgorithm::FetchLater(deferred_record_id));
583    // Step 15. Return a new FetchLaterResult whose activated getter steps are to return activated.
584    Ok(FetchLaterResult::new(cx, window, deferred_record_id))
585}

605impl DeferredFetchRecord {
606    /// Part of step 13 of <https://fetch.spec.whatwg.org/#dom-window-fetchlater>
607    fn activate(&self) {
608        // and the following step: set activated to true.
609        self.activated.set(true);
610    }
611    /// Part of step 14 of <https://fetch.spec.whatwg.org/#dom-window-fetchlater>

621    }

624#[derive(JSTraceable, MallocSizeOf)]
625pub(crate) struct FetchContext {
626    #[ignore_malloc_size_of = "unclear ownership semantics"]
627    fetch_promise: Option<TrustedPromise>,
628    response_object: Trusted<Response>,
629    request: Trusted<Request>,
630    global: Trusted<GlobalScope>,
631    locally_aborted: bool,
632    #[no_trace]
633    url: ServoUrl,
634}
635
636impl FetchContext {
637    /// Step 11 of <https://fetch.spec.whatwg.org/#dom-global-fetch>
638    pub(crate) fn abort_fetch(&mut self, abort_reason: HandleValue, cx: &mut JSContext) {
639        // Step 11.1. Set locallyAborted to true.
640        self.locally_aborted = true;
641
642        // Step 11.2. Assert: controller is non-null.
643        //
644        // Note: We currently prune fetch records that have finished (behaviorally
645        // equivalent to the specification and better for memory usage), so it might be
646        // the case that the `FetchRecord` is gone and the controller inaccessible.
647
648        // Step 11.3. Abort controller with requestObject’s signal’s abort reason.
649        let global = self.global.root();
650        let request = self.request.root();
651        if let Some(controller) = global
652            .fetch_group_mut()
653            .fetch_controller(&request.request().id)
654        {
655            controller.abort();
656        }
657
658        // Step 11.4. Abort the fetch() call with p, request, responseObject,
659        // and requestObject’s signal’s abort reason.
660        let promise = self
661            .fetch_promise
662            .take()
663            .expect("fetch promise is missing")
664            .root();
665        abort_fetch_call(
666            promise,
667            &request,
668            Some(&self.response_object.root()),
669            abort_reason,
670            &global,
671            cx,
672        );
673    }
674}

676/// Step 12 of <https://fetch.spec.whatwg.org/#dom-global-fetch>
677impl FetchResponseListener for FetchContext {
678    fn process_request_body(&mut self, _: RequestId) {
679        // TODO
680    }
681
682    fn process_response(
683        &mut self,
684        cx: &mut JSContext,
685        _: RequestId,
686        fetch_metadata: Result<FetchMetadata, NetworkError>,
687    ) {
688        // Step 12.1. If locallyAborted is true, then abort these steps.
689        if self.locally_aborted {
690            return;
691        }
692        let promise = self
693            .fetch_promise
694            .take()
695            .expect("fetch promise is missing")
696            .root();
697
698        let mut realm = enter_auto_realm(cx, &*promise);
699        let cx = &mut realm.current_realm();
700        match fetch_metadata {
701            // Step 12.3. If response is a network error, then reject
702            // p with a TypeError and abort these steps.
703            Err(error) => {
704                promise.reject_error(cx, Error::Type(cformat!("Network error: {:?}", error)));
705                self.fetch_promise = Some(TrustedPromise::new(promise));
706                let response = self.response_object.root();
707                response.set_type(cx, DOMResponseType::Error);
708                response.error_stream(cx, Error::Type(c"Network error occurred".to_owned()));
709                return;
710            },
711            // Step 12.4. Set responseObject to the result of creating a Response object,
712            // given response, "immutable", and relevantRealm.
713            Ok(metadata) => match metadata {
714                FetchMetadata::Unfiltered(m) => {
715                    let r = self.response_object.root();
716                    fill_headers_with_metadata(cx, &r, m);
717                    r.set_type(cx, DOMResponseType::Default);
718                },
719                FetchMetadata::Filtered { filtered, .. } => match filtered {
720                    FilteredMetadata::Basic(m) => {
721                        let r = self.response_object.root();
722                        fill_headers_with_metadata(cx, &r, m);
723                        r.set_type(cx, DOMResponseType::Basic);
724                    },
725                    FilteredMetadata::Cors(m) => {
726                        let r = self.response_object.root();
727                        fill_headers_with_metadata(cx, &r, m);
728                        r.set_type(cx, DOMResponseType::Cors);
729                    },
730                    FilteredMetadata::Opaque => {
731                        self.response_object
732                            .root()
733                            .set_type(cx, DOMResponseType::Opaque);
734                    },
735                    FilteredMetadata::OpaqueRedirect(url) => {
736                        let r = self.response_object.root();
737                        r.set_type(cx, DOMResponseType::Opaqueredirect);
738                        r.set_final_url(url);
739                    },
740                },
741            },
742        }
743
744        // Step 12.5. Resolve p with responseObject.
745        promise.resolve_native(cx, &self.response_object.root());
746        self.fetch_promise = Some(TrustedPromise::new(promise));
747    }
748
749    fn process_response_chunk(&mut self, cx: &mut JSContext, _: RequestId, chunk: Bytes) {
750        let response = self.response_object.root();
751        response.stream_chunk(cx, chunk);
752    }
753
754    fn process_response_eof(
755        self,
756        cx: &mut JSContext,
757        _: RequestId,
758        response: Result<(), NetworkError>,
759        timing: ResourceFetchTiming,
760    ) {
761        let response_object = self.response_object.root();
762        let mut realm = enter_auto_realm(cx, &*response_object);
763        let cx = &mut realm.current_realm();
764        if let Err(ref error) = response &&
765            *error == NetworkError::DecompressionError
766        {
767            response_object.error_stream(cx, Error::Type(c"Network error occurred".to_owned()));
768        }
769        response_object.finish(cx);
770        // TODO
771        // ... trailerObject is not supported in Servo yet.
772
773        // navigation submission is handled in servoparser/mod.rs
774        network_listener::submit_timing(cx, &self, &response, &timing);
775    }
776
777    fn process_csp_violations(
778        &mut self,
779        cx: &mut JSContext,
780        _request_id: RequestId,
781        violations: Vec<Violation>,
782    ) {
783        let global = &self.resource_timing_global();
784        global.report_csp_violations(cx, violations, None, None);
785    }
786}

863/// Convenience function for synchronously loading a whole resource.
864pub(crate) fn load_whole_resource(
865    request: RequestBuilder,
866    core_resource_thread: &CoreResourceThread,
867    global: &GlobalScope,
868    csp_violations_processor: &dyn CspViolationsProcessor,
869    cx: &mut JSContext,
870) -> Result<(Metadata, Vec<u8>, bool), NetworkError> {
871    let (action_sender, action_receiver) = GenericCallback::new_blocking().unwrap();
872    let url = request.url.url();
873    core_resource_thread
874        .send(CoreResourceMsg::Fetch(
875            request,
876            FetchChannels::ResponseMsg(action_sender),
877        ))
878        .unwrap();
879
880    let mut buf = vec![];
881    let mut metadata = None;
882    let mut muted_errors = false;
883    loop {
884        match action_receiver.recv().unwrap() {
885            FetchResponseMsg::ProcessRequestBody(..) => {},
886            FetchResponseMsg::ProcessResponse(_, Ok(m)) => {
887                muted_errors = m.is_cors_cross_origin();
888                metadata = Some(match m {
889                    FetchMetadata::Unfiltered(m) => m,
890                    FetchMetadata::Filtered { unsafe_, .. } => unsafe_,
891                })
892            },
893            FetchResponseMsg::ProcessResponseChunk(_, data) => buf.extend_from_slice(&data),
894            FetchResponseMsg::ProcessResponseEOF(_, Ok(_), _) => {
895                let metadata = metadata.unwrap();
896                if let Some(timing) = &metadata.timing {
897                    submit_timing_data(cx, global, url, InitiatorType::Other, timing);
898                }
899                return Ok((metadata, buf, muted_errors));
900            },
901            FetchResponseMsg::ProcessResponse(_, Err(e)) |
902            FetchResponseMsg::ProcessResponseEOF(_, Err(e), _) => return Err(e),
903            FetchResponseMsg::ProcessCspViolations(_, violations) => {
904                csp_violations_processor.process_csp_violations(cx, violations);
905            },
906            FetchResponseMsg::ProcessContentLength(_request_id, size) => {
907                buf.reserve(size - buf.len())
908            },
909        }
910    }
911}

946/// <https://html.spec.whatwg.org/multipage/#create-a-potential-cors-request>
947#[allow(clippy::too_many_arguments)]
948pub(crate) fn create_a_potential_cors_request_with_claim(
949    webview_id: Option<WebViewId>,
950    url: UrlWithBlobClaim,
951    destination: Destination,
952    cors_setting: Option<CorsSettings>,
953    same_origin_fallback: Option<bool>,
954    referrer: Referrer,
955) -> RequestBuilder {
956    RequestBuilder::new(webview_id, url, referrer)
957        // Step 1. Let mode be "no-cors" if corsAttributeState is No CORS, and "cors" otherwise.
958        .mode(match cors_setting {
959            Some(_) => RequestMode::CorsMode,
960            // Step 2. If same-origin fallback flag is set and mode is "no-cors", set mode to "same-origin".
961            None if same_origin_fallback == Some(true) => RequestMode::SameOrigin,
962            None => RequestMode::NoCors,
963        })
964        .credentials_mode(match cors_setting {
965            // Step 4. If corsAttributeState is Anonymous, set credentialsMode to "same-origin".
966            Some(CorsSettings::Anonymous) => CredentialsMode::CredentialsSameOrigin,
967            // Step 3. Let credentialsMode be "include".
968            _ => CredentialsMode::Include,
969        })
970        // Step 5. Return a new request whose URL is url, destination is destination,
971        // mode is mode, credentials mode is credentialsMode, and whose use-URL-credentials flag is set.
972        .destination(destination)
973        .use_url_credentials(true)

1//! The [Diplomat](https://rust-diplomat.github.io/diplomat/) macro crate.
2//!
3//! This crate provides the `#[diplomat::bridge]` proc macro, which can be used to
4//! generate FFI hooks for use with bindings generated by the `diplomat-tool` library.
5//!
6//! Be sure to also include the [`diplomat_runtime`](https://docs.rs/diplomat_runtime) crate
7//! when using the `#[diplomat::bridge]` macro.
8

18fn cfgs_to_stream(attrs: &[Attribute]) -> proc_macro2::TokenStream {
19    attrs
20        .iter()
21        .fold(quote!(), |prev, attr| quote!(#prev #attr))
22}

24fn param_ty(param_ty: &ast::TypeName) -> syn::Type {
25    match &param_ty {
26        ast::TypeName::StrReference(lt @ Some(_lt), encoding, _) => {
27            // At the param boundary we MUST use FFI-safe diplomat slice types,
28            // not Rust stdlib types (which are not FFI-safe and must be converted)
29            encoding.get_diplomat_slice_type(lt)
30        }
31        ast::TypeName::StrReference(None, encoding, _) => encoding.get_diplomat_slice_type(&None),
32        ast::TypeName::StrSlice(encoding, _) => {
33            // At the param boundary we MUST use FFI-safe diplomat slice types,
34            // not Rust stdlib types (which are not FFI-safe and must be converted)
35            let inner = encoding.get_diplomat_slice_type(&Some(ast::Lifetime::Anonymous));
36            syn::parse_quote_spanned!(Span::call_site() => diplomat_runtime::DiplomatSlice<#inner>)
37        }
38        ast::TypeName::PrimitiveSlice(ltmt, prim, _) => {
39            // At the param boundary we MUST use FFI-safe diplomat slice types,
40            // not Rust stdlib types (which are not FFI-safe and must be converted)
41            prim.get_diplomat_slice_type(ltmt)
42        }
43        ast::TypeName::Option(..) if !param_ty.is_ffi_safe() => {
44            param_ty.ffi_safe_version().to_syn()
45        }
46        _ => param_ty.to_syn(),
47    }
48}

50fn param_conversion(
51    name: &ast::Ident,
52    param_type: &ast::TypeName,
53    cast_to: Option<&syn::Type>,
54) -> Option<proc_macro2::TokenStream> {
55    match &param_type {
56        // conversion only needed for slices that are specified as Rust types rather than diplomat_runtime types
57        ast::TypeName::StrReference(.., StdlibOrDiplomat::Stdlib)
58        | ast::TypeName::StrSlice(.., StdlibOrDiplomat::Stdlib)
59        | ast::TypeName::PrimitiveSlice(.., StdlibOrDiplomat::Stdlib)
60        | ast::TypeName::Result(..) => Some(if let Some(cast_to) = cast_to {
61            quote!(let #name: #cast_to = #name.into();)
62        } else {
63            quote!(let #name = #name.into();)
64        }),
65        // Convert Option<struct/enum/primitive> and DiplomatOption<opaque>
66        // simplify the check by just checking is_ffi_safe()
67        ast::TypeName::Option(inner, _stdlib) => {
68            let mut tokens = TokenStream::new();
69
70            if !param_type.is_ffi_safe() {
71                let inner_ty = inner.ffi_safe_version().to_syn();
72                tokens.extend(quote!(let #name : Option<#inner_ty> = #name.into();));
73            }
74            if !inner.is_ffi_safe() {
75                tokens.extend(quote!(let #name = #name.map(|v| v.into());));
76            }
77
78            if !tokens.is_empty() {
79                Some(tokens)
80            } else {
81                None
82            }
83        }
84        ast::TypeName::Function(in_types, out_type, mutability) => {
85            let cb_wrap_ident = &name;
86            let mut cb_param_list = vec![];
87            let mut cb_params_and_types_list = vec![];
88            let mut cb_arg_type_list = vec![];
89            let mut all_params_conversion = vec![];
90            for (index, in_ty) in in_types.iter().enumerate() {
91                let param_ident_str = format!("arg{index}");
92                let orig_type = in_ty.to_syn();
93                let param_converted_type = param_ty(in_ty);
94                if let Some(conversion) = param_conversion(
95                    &ast::Ident::from(param_ident_str.clone()),
96                    in_ty,
97                    Some(&param_converted_type),
98                ) {
99                    all_params_conversion.push(conversion);
100                }
101                let param_ident = Ident::new(&param_ident_str, Span::call_site());
102                cb_arg_type_list.push(param_converted_type);
103                cb_params_and_types_list.push(quote!(#param_ident: #orig_type));
104                cb_param_list.push(param_ident);
105            }
106
107            let (ret_type, conversion) = if !out_type.is_ffi_safe() {
108                (out_type.ffi_safe_version(), quote! { .into() })
109            } else {
110                (*out_type.clone(), TokenStream::new())
111            };
112
113            let cb_ret_type = ret_type.to_syn();
114
115            let mutability = match mutability {
116                ast::Mutability::Immutable => quote!(const),
117                ast::Mutability::Mutable => quote!(mut),
118            };
119            let tokens = quote! {
120                let #cb_wrap_ident = move | #(#cb_params_and_types_list,)* | unsafe {
121                    #(#all_params_conversion)*
122                    let _ = &#cb_wrap_ident; // Force the lambda to capture the full object, see https://doc.rust-lang.org/edition-guide/rust-2021/disjoint-capture-in-closures.html
123                    std::mem::transmute::<unsafe extern "C" fn (*mut c_void, ...) -> #cb_ret_type, unsafe extern "C" fn (*#mutability c_void, #(#cb_arg_type_list,)*) -> #cb_ret_type>
124                        (#cb_wrap_ident.run_callback)(#cb_wrap_ident.data, #(#cb_param_list,)*) #conversion
125                };
126            };
127            Some(parse2(tokens).unwrap())
128        }
129        _ => None,
130    }
131}

133fn gen_custom_vtable(custom_trait: &ast::Trait, custom_trait_vtable_type: &Ident) -> Item {
134    let mut method_sigs: Vec<proc_macro2::TokenStream> = vec![];
135    method_sigs.push(quote!(
136        pub destructor: Option<unsafe extern "C" fn(*const c_void)>,
137        pub size: usize,
138        pub alignment: usize,
139    ));
140    for m in &custom_trait.methods {
141        // TODO check that this is the right conversion, it might be the wrong direction
142        let mut param_types: Vec<syn::Type> = m.params.iter().map(|p| param_ty(&p.ty)).collect();
143        let method_name = Ident::new(&format!("run_{}_callback", m.name), Span::call_site());
144        let return_tokens = match &m.output_type {
145            Some(ret_ty) => {
146                let conv_ret_ty = ret_ty.to_syn();
147                quote!( -> #conv_ret_ty)
148            }
149            None => {
150                quote! {}
151            }
152        };
153        param_types.insert(0, syn::parse_quote!(*const c_void));
154        method_sigs.push(quote!(
155            pub #method_name: unsafe extern "C" fn (#(#param_types),*) #return_tokens,
156
157        ));
158    }
159    syn::parse_quote!(
160        #[repr(C)]
161        #[allow(deprecated)]
162        pub struct #custom_trait_vtable_type {
163            #(#method_sigs)*
164        }
165    )
166}

168fn gen_custom_trait_impl(custom_trait: &ast::Trait, custom_trait_struct_name: &Ident) -> Item {
169    let mut methods: Vec<Item> = vec![];
170    for m in &custom_trait.methods {
171        let param_names: Vec<proc_macro2::TokenStream> = m
172            .params
173            .iter()
174            .map(|p| {
175                let p_name = &p.name;
176                quote! {, #p_name}
177            })
178            .collect();
179        let mut all_params_conversion = vec![];
180        let mut param_names_and_types: Vec<proc_macro2::TokenStream> = m
181            .params
182            .iter()
183            .map(|p| {
184                let orig_type = p.ty.to_syn();
185                let p_ty = param_ty(&p.ty);
186                if let Some(conversion) = param_conversion(&p.name.clone(), &p.ty, Some(&p_ty)) {
187                    all_params_conversion.push(conversion);
188                }
189                let p_name = &p.name;
190                quote!(#p_name : #orig_type)
191            })
192            .collect();
193        let method_name = &m.name;
194        let (return_tokens, end_token) = match &m.output_type {
195            Some(ret_ty) => {
196                let conv_ret_ty = ret_ty.to_syn();
197                (quote!( -> #conv_ret_ty), quote! {})
198            }
199            None => (quote! {}, quote! {;}),
200        };
201        if let Some(self_param) = &m.self_param {
202            let mut self_modifier = quote! {};
203            if let Some((lifetime, mutability)) = &self_param.reference {
204                let lifetime_mod = if *lifetime == ast::Lifetime::Anonymous {
205                    quote! { & }
206                } else {
207                    let prime = "'".to_string();
208                    let lifetime = lifetime.to_syn();
209                    quote! { & #prime #lifetime }
210                };
211                let mutability_mod = if *mutability == ast::Mutability::Mutable {
212                    quote! {mut}
213                } else {
214                    quote! {}
215                };
216                self_modifier = quote! { #lifetime_mod #mutability_mod }
217            }
218            param_names_and_types.insert(0, quote!(#self_modifier self));
219        }
220
221        let lifetimes = {
222            let lifetime_env = &m.lifetimes;
223            if lifetime_env.is_empty() {
224                quote! {}
225            } else {
226                quote! { <#lifetime_env> }
227            }
228        };
229        let runner_method_name =
230            Ident::new(&format!("run_{method_name}_callback"), Span::call_site());
231        methods.push(syn::Item::Fn(syn::parse_quote!(
232            fn #method_name #lifetimes (#(#param_names_and_types),*) #return_tokens {
233                unsafe {
234                    #(#all_params_conversion)*
235                    ((self.vtable).#runner_method_name)(self.data #(#param_names)*)#end_token
236                }
237            }
238
239        )));
240    }
241    let trait_name = &custom_trait.name;
242    syn::parse_quote!(
243        impl #trait_name for #custom_trait_struct_name {
244            #(#methods)*
245        }
246    )
247}

249fn gen_custom_type_method(strct: &ast::CustomType, m: &ast::Method) -> Item {
250    let self_ident = Ident::new(strct.name().as_str(), Span::call_site());
251    let method_ident = Ident::new(m.name.as_str(), Span::call_site());
252    let extern_ident = Ident::new(m.abi_name.as_str(), Span::call_site());
253    gen_custom_function(FuncGen {
254        self_ident: Some(self_ident),
255        func_ident: method_ident,
256        extern_ident,
257        self_param: m.self_param.as_ref(),
258        params: &m.params,
259        return_type: m.return_type.as_ref(),
260        lifetime_env: &m.lifetime_env,
261        attrs: &m.attrs,
262    })
263}

265struct FuncGen<'a> {
266    self_ident: Option<Ident>,
267    func_ident: Ident,
268    extern_ident: Ident,
269    self_param: Option<&'a crate::ast::SelfParam>,
270    params: &'a [crate::ast::Param],
271    return_type: Option<&'a crate::ast::TypeName>,
272    lifetime_env: &'a crate::ast::LifetimeEnv,
273    attrs: &'a crate::ast::Attrs,
274}

276fn gen_custom_function(func_info: FuncGen) -> Item {
277    let mut all_params = vec![];
278
279    let mut all_params_conversion = vec![];
280    let mut all_params_names = vec![];
281    func_info.params.iter().for_each(|p| {
282        let ty = param_ty(&p.ty);
283        let name = &p.name;
284        all_params_names.push(name);
285        all_params.push(syn::parse_quote!(#name: #ty));
286        if let Some(conversion) = param_conversion(&p.name, &p.ty, None) {
287            all_params_conversion.push(conversion);
288        }
289    });
290
291    let this_ident = Pat::Ident(PatIdent {
292        attrs: vec![],
293        by_ref: None,
294        mutability: None,
295        ident: Ident::new("this", Span::call_site()),
296        subpat: None,
297    });
298
299    if let Some(self_param) = func_info.self_param {
300        all_params.insert(
301            0,
302            FnArg::Typed(PatType {
303                attrs: vec![],
304                pat: Box::new(this_ident.clone()),
305                colon_token: syn::token::Colon(Span::call_site()),
306                ty: Box::new(self_param.to_typename().to_syn()),
307            }),
308        );
309    }
310
311    let lifetimes = {
312        let lifetime_env = func_info.lifetime_env;
313        if lifetime_env.is_empty() {
314            quote! {}
315        } else {
316            quote! { <#lifetime_env> }
317        }
318    };
319
320    let func_ident = func_info.func_ident;
321
322    let method_invocation = if func_info.self_param.is_some() {
323        quote! { #this_ident.#func_ident }
324    } else if let Some(self_ident) = func_info.self_ident {
325        quote! { #self_ident::#func_ident }
326    } else {
327        quote! { #func_ident }
328    };
329
330    let (return_tokens, maybe_into) = if let Some(return_type) = func_info.return_type {
331        if let ast::TypeName::Result(ok, err, StdlibOrDiplomat::Stdlib) = return_type {
332            let ok = ok.to_syn();
333            let err = err.to_syn();
334            (
335                quote! { -> diplomat_runtime::DiplomatResult<#ok, #err> },
336                quote! { .into() },
337            )
338        } else if let ast::TypeName::StrReference(_, _, StdlibOrDiplomat::Stdlib)
339        | ast::TypeName::StrSlice(.., StdlibOrDiplomat::Stdlib)
340        | ast::TypeName::PrimitiveSlice(_, _, StdlibOrDiplomat::Stdlib) = return_type
341        {
342            let return_type_syn = return_type.ffi_safe_version().to_syn();
343            (quote! { -> #return_type_syn }, quote! { .into() })
344        } else if let ast::TypeName::Ordering = return_type {
345            let return_type_syn = return_type.to_syn();
346            (quote! { -> #return_type_syn }, quote! { as i8 })
347        } else if let ast::TypeName::Option(ty, is_std_option) = return_type {
348            match ty.as_ref() {
349                // pass by reference, Option becomes null
350                ast::TypeName::Box(..) | ast::TypeName::Reference(..) => {
351                    let return_type_syn = return_type.to_syn();
352                    let conversion = if *is_std_option == StdlibOrDiplomat::Stdlib {
353                        quote! {}
354                    } else {
355                        quote! {.into()}
356                    };
357                    (quote! { -> #return_type_syn }, conversion)
358                }
359                // anything else goes through DiplomatResult
360                _ => {
361                    let ty = ty.to_syn();
362                    let conversion = if *is_std_option == StdlibOrDiplomat::Stdlib {
363                        quote! { .ok_or(()).into() }
364                    } else {
365                        quote! {}
366                    };
367                    (
368                        quote! { -> diplomat_runtime::DiplomatResult<#ty, ()> },
369                        conversion,
370                    )
371                }
372            }
373        } else {
374            let return_type_syn = return_type.to_syn();
375            (quote! { -> #return_type_syn }, quote! {})
376        }
377    } else {
378        (quote! {}, quote! {})
379    };
380
381    let write_flushes = func_info
382        .params
383        .iter()
384        .filter(|p| p.is_write())
385        .map(|p| {
386            let p = &p.name;
387            quote! { #p.flush(); }
388        })
389        .collect::<Vec<_>>();
390
391    let extern_ident = func_info.extern_ident;
392
393    let cfg = cfgs_to_stream(&func_info.attrs.cfg);
394    if write_flushes.is_empty() {
395        Item::Fn(syn::parse_quote! {
396            #[no_mangle]
397            #cfg
398            #[allow(deprecated)]
399            extern "C" fn #extern_ident #lifetimes(#(#all_params),*) #return_tokens {
400                #(#all_params_conversion)*
401                #method_invocation(#(#all_params_names),*) #maybe_into
402            }
403        })
404    } else {
405        Item::Fn(syn::parse_quote! {
406            #[no_mangle]
407            #cfg
408            #[allow(deprecated)]
409            extern "C" fn #extern_ident #lifetimes(#(#all_params),*) #return_tokens {
410                #(#all_params_conversion)*
411                let ret = #method_invocation(#(#all_params_names),*);
412                #(#write_flushes)*
413                ret #maybe_into
414            }
415        })
416    }
417}

419struct AttributeInfo {
420    repr: bool,
421    opaque: bool,
422    #[allow(unused)]
423    is_out: bool,
424}
425
426impl AttributeInfo {
427    fn extract(attrs: &mut Vec<Attribute>) -> Self {
428        let mut repr = false;
429        let mut opaque = false;
430        let mut is_out = false;
431        attrs.retain(|attr| {
432            let ident = &attr.path().segments.iter().next().unwrap().ident;
433            if ident == "repr" {
434                repr = true;
435                // don't actually extract repr attrs, just detect them
436                return true;
437            } else if ident == "diplomat" {
438                if attr.path().segments.len() == 2 {
439                    let seg = &attr.path().segments.iter().nth(1).unwrap().ident;
440                    if seg == "opaque" {
441                        opaque = true;
442                        return false;
443                    } else if seg == "out" {
444                        is_out = true;
445                        return false;
446                    } else if seg == "rust_link"
447                        || seg == "out"
448                        || seg == "attr"
449                        || seg == "abi_rename"
450                        || seg == "demo"
451                    {
452                        // diplomat-tool reads these, not diplomat::bridge.
453                        // throw them away so rustc doesn't complain about unknown attributes
454                        return false;
455                    } else if seg == "enum_convert" || seg == "transparent_convert" {
456                        // diplomat::bridge doesn't read this, but it's handled separately
457                        // as an attribute
458                        return true;
459                    } else if seg == "config" {
460                        panic!("#[diplomat::config] is restricted to top level types in lib.rs.");
461                    } else {
462                        panic!("Only #[diplomat::opaque] and #[diplomat::rust_link] are supported: {seg:?}")
463                    }
464                } else {
465                    panic!("#[diplomat::foo] attrs have a single-segment path name")
466                }
467            }
468            true
469        });
470
471        Self {
472            repr,
473            opaque,
474            is_out,
475        }
476    }
477}

479fn gen_bridge(mut input: ItemMod) -> ItemMod {
480    let module = ast::Module::from_syn(&input, true);
481    // Clean out any diplomat attributes so Rust doesn't get mad
482    let _attrs = AttributeInfo::extract(&mut input.attrs);
483    let (brace, mut new_contents) = input.content.unwrap();
484
485    new_contents.push(parse2(quote! { use diplomat_runtime::*; }).unwrap());
486    new_contents.push(parse2(quote! { use core::ffi::c_void; }).unwrap());
487
488    new_contents.iter_mut().for_each(|c| match c {
489        Item::Struct(s) => {
490            let info = AttributeInfo::extract(&mut s.attrs);
491
492            if !info.opaque {
493                // This is validated by HIR, but it's also nice to validate it in the macro so that there
494                // are early error messages
495                for field in s.fields.iter_mut() {
496                    let _attrs = AttributeInfo::extract(&mut field.attrs);
497                    let ty = ast::TypeName::from_syn(&field.ty, None);
498                    if !ty.is_ffi_safe() {
499                        let ffisafe = ty.ffi_safe_version();
500                        panic!("Found non-FFI safe type inside struct: {ty}, try {ffisafe}");
501                    }
502                }
503            }
504
505            // Normal opaque types don't need repr(transparent) because the inner type is
506            // never referenced. #[diplomat::transparent_convert] handles adding repr(transparent)
507            // on its own
508            if !info.opaque {
509                let repr = if !info.repr {
510                    quote!(#[repr(C)])
511                } else {
512                    quote!()
513                };
514
515                *s = syn::parse_quote! {
516                    #repr
517                    #s
518                }
519            }
520        }
521
522        Item::Enum(e) => {
523            let info = AttributeInfo::extract(&mut e.attrs);
524
525            for v in &mut e.variants {
526                let info = AttributeInfo::extract(&mut v.attrs);
527                if info.opaque {
528                    panic!("#[diplomat::opaque] not allowed on enum variants");
529                }
530            }
531
532            // Normal opaque types don't need repr(transparent) because the inner type is
533            // never referenced.
534            if !info.opaque {
535                *e = syn::parse_quote! {
536                    #[repr(C)]
537                    #[derive(Clone, Copy)]
538                    #e
539                };
540            }
541        }
542
543        Item::Impl(i) => {
544            for item in &mut i.items {
545                if let syn::ImplItem::Fn(ref mut m) = *item {
546                    let info = AttributeInfo::extract(&mut m.attrs);
547                    if info.opaque {
548                        panic!("#[diplomat::opaque] not allowed on methods")
549                    }
550                    for i in m.sig.inputs.iter_mut() {
551                        let _attrs = match i {
552                            syn::FnArg::Receiver(s) => AttributeInfo::extract(&mut s.attrs),
553                            syn::FnArg::Typed(t) => AttributeInfo::extract(&mut t.attrs),
554                        };
555                    }
556                }
557            }
558        }
559        _ => (),
560    });
561
562    for custom_type in module.declared_types.values() {
563        custom_type.methods().iter().for_each(|m| {
564            let gen_m = gen_custom_type_method(custom_type, m);
565            new_contents.push(gen_m);
566        });
567
568        if let ast::CustomType::Opaque(opaque) = custom_type {
569            let destroy_ident = Ident::new(opaque.dtor_abi_name.as_str(), Span::call_site());
570
571            let type_ident = custom_type.name().to_syn();
572
573            let (lifetime_defs, lifetimes) = if let Some(lifetime_env) = custom_type.lifetimes() {
574                (
575                    quote! { <#lifetime_env> },
576                    lifetime_env.lifetimes_to_tokens(),
577                )
578            } else {
579                (quote! {}, quote! {})
580            };
581
582            let cfg = cfgs_to_stream(&custom_type.attrs().cfg);
583
584            // for now, body is empty since all we need to do is drop the box
585            // TODO(#13): change to take a `*mut` and handle DST boxes appropriately
586            new_contents.push(Item::Fn(syn::parse_quote! {
587                #[no_mangle]
588                #cfg
589                #[allow(deprecated)]
590                extern "C" fn #destroy_ident #lifetime_defs(this: Box<#type_ident #lifetimes>) {}
591            }));
592        }
593    }
594
595    for custom_trait in module.declared_traits.values() {
596        let custom_trait_name = Ident::new(
597            &format!("DiplomatTraitStruct_{}", custom_trait.name),
598            Span::call_site(),
599        );
600        let custom_trait_vtable_type =
601            Ident::new(&format!("{}_VTable", custom_trait.name), Span::call_site());
602
603        // vtable
604        new_contents.push(gen_custom_vtable(custom_trait, &custom_trait_vtable_type));
605
606        // trait struct
607        new_contents.push(syn::parse_quote! {
608            #[repr(C)]
609            pub struct #custom_trait_name {
610                data: *const c_void,
611                pub vtable: #custom_trait_vtable_type,
612            }
613        });
614        if custom_trait.is_send {
615            new_contents.push(syn::parse_quote! {
616                unsafe impl std::marker::Send for #custom_trait_name {}
617            });
618        }
619        if custom_trait.is_sync {
620            new_contents.push(syn::parse_quote! {
621                unsafe impl std::marker::Sync for #custom_trait_name {}
622            });
623        }
624
625        // trait struct wrapper for all methods
626        new_contents.push(gen_custom_trait_impl(custom_trait, &custom_trait_name));
627
628        // destructor
629        new_contents.push(syn::parse_quote! {
630            impl Drop for #custom_trait_name {
631                fn drop(&mut self) {
632                    if let Some(destructor) = self.vtable.destructor {
633                        unsafe {
634                            (destructor)(self.data);
635                        }
636                    }
637                }
638            }
639        })
640    }
641
642    for func in module.declared_functions.values() {
643        let func_ident = Ident::new(func.name.as_str(), Span::call_site());
644        let extern_ident = Ident::new(func.abi_name.as_str(), Span::call_site());
645        new_contents.push(gen_custom_function(FuncGen {
646            self_ident: None,
647            func_ident,
648            extern_ident,
649            self_param: None,
650            params: &func.params,
651            return_type: func.output_type.as_ref(),
652            lifetime_env: &func.lifetimes,
653            attrs: &func.attrs,
654        }))
655    }
656
657    ItemMod {
658        attrs: input.attrs,
659        vis: input.vis,
660        mod_token: input.mod_token,
661        ident: input.ident,
662        content: Some((brace, new_contents)),
663        semi: input.semi,
664        unsafety: None,
665    }
666}

687/// Generate From and Into implementations for a Diplomat enum
688///
689/// This is invoked as `#[diplomat::enum_convert(OtherEnumName)]`
690/// on a Diplomat enum. It will assume the other enum has exactly the same variants
691/// and generate From and Into implementations using those. In case that enum is `#[non_exhaustive]`,
692/// you may use `#[diplomat::enum_convert(OtherEnumName, needs_wildcard)]` to generate a panicky wildcard
693/// branch. It is up to the library author to ensure the enums are kept in sync. You may use the `#[non_exhaustive_omitted_patterns]`
694/// lint to enforce this.
695#[proc_macro_attribute]
696pub fn enum_convert(
697    attr: proc_macro::TokenStream,
698    input: proc_macro::TokenStream,
699) -> proc_macro::TokenStream {
700    // proc macros handle compile errors by using special error tokens.
701    // In case of an error, we don't want the original code to go away too
702    // (otherwise that will cause more errors) so we hold on to it and we tack it in
703    // with no modifications below
704    let input_cached: proc_macro2::TokenStream = input.clone().into();
705    let expanded =
706        enum_convert::gen_enum_convert(parse_macro_input!(attr), parse_macro_input!(input));
707
708    let full = quote! {
709        #expanded
710        #input_cached
711    };
712    proc_macro::TokenStream::from(full.to_token_stream())
713}

715/// Generate conversions from inner types for opaque Diplomat types with a single field
716///
717/// This is invoked as `#[diplomat::transparent_convert]`
718/// on an opaque Diplomat type. It will add `#[repr(transparent)]` and implement `pub(crate) fn transparent_convert()`
719/// which allows constructing an `&Self` from a reference to the inner field.
720#[proc_macro_attribute]
721pub fn transparent_convert(
722    _attr: proc_macro::TokenStream,
723    input: proc_macro::TokenStream,
724) -> proc_macro::TokenStream {
725    // proc macros handle compile errors by using special error tokens.
726    // In case of an error, we don't want the original code to go away too
727    // (otherwise that will cause more errors) so we hold on to it and we tack it in
728    // with no modifications below
729    let input_cached: proc_macro2::TokenStream = input.clone().into();
730    let expanded = transparent_convert::gen_transparent_convert(parse_macro_input!(input));
731
732    let full = quote! {
733        #expanded
734        #[repr(transparent)]
735        #input_cached
736    };
737    proc_macro::TokenStream::from(full.to_token_stream())
738}

741pub fn macro_rules(
742    _attr: proc_macro::TokenStream,
743    input: proc_macro::TokenStream,
744) -> proc_macro::TokenStream {
745    // proc macros handle compile errors by using special error tokens.
746    // In case of an error, we don't want the original code to go away too
747    // (otherwise that will cause more errors) so we hold on to it and we tack it in
748    // with no modifications below

756}

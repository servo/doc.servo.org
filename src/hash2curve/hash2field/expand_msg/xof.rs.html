12/// Implements `expand_message_xof` via the [`ExpandMsg`] trait:
13/// <https://www.rfc-editor.org/rfc/rfc9380.html#name-expand_message_xof>
14///
15/// # Errors
16///
17/// See [`ExpandMsgXofError`] for details.
18pub struct ExpandMsgXof<HashT>
19where
20    HashT: Default + ExtendableOutput + Update + HashMarker,
21{
22    reader: <HashT as ExtendableOutput>::Reader,
23    remaining: u16,
24}
25
26impl<HashT> fmt::Debug for ExpandMsgXof<HashT>
27where
28    HashT: Default + ExtendableOutput + Update + HashMarker,
29    <HashT as ExtendableOutput>::Reader: fmt::Debug,
30{
31    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
32        f.debug_struct("ExpandMsgXof")
33            .field("reader", &self.reader)
34            .finish()
35    }
36}

38impl<HashT, K> ExpandMsg<K> for ExpandMsgXof<HashT>
39where
40    HashT: Default + ExtendableOutput + Update + HashMarker,
41    // If DST is larger than 255 bytes, the length of the computed DST is calculated by `K * 2`.
42    // https://www.rfc-editor.org/rfc/rfc9380.html#section-5.3.1-2.1
43    K: Mul<U2, Output: ArraySize>,
44    // The collision resistance of `HashT` MUST be at least `K` bits.
45    // https://www.rfc-editor.org/rfc/rfc9380.html#section-5.3.2-2.1
46    HashT: CollisionResistance<CollisionResistance: IsGreaterOrEqual<K, Output = True>>,
47{
48    type Hash = HashT;
49    type Expander<'dst> = Self;
50    type Error = ExpandMsgXofError;
51
52    fn expand_message<'dst>(
53        msg: &[&[u8]],
54        dst: &'dst [&[u8]],
55        len_in_bytes: NonZero<u16>,
56    ) -> Result<Self::Expander<'dst>, ExpandMsgXofError> {
57        let len_in_bytes = len_in_bytes.get();
58
59        let domain = Domain::<Prod<K, U2>>::xof::<HashT>(dst)?;
60        let mut reader = HashT::default();
61
62        for msg in msg {
63            reader = reader.chain(msg);
64        }
65
66        reader.update(&len_in_bytes.to_be_bytes());
67        domain.update_hash(&mut reader);
68        reader.update(&[domain.len()]);
69        let reader = reader.finalize_xof();
70        Ok(Self {
71            reader,
72            remaining: len_in_bytes,
73        })
74    }
75}
76
77impl<HashT> Expander for ExpandMsgXof<HashT>
78where
79    HashT: Default + ExtendableOutput + Update + HashMarker,
80{
81    fn fill_bytes(&mut self, okm: &mut [u8]) -> Result<usize, Error> {
82        if self.remaining == 0 {
83            return Err(Error);
84        }
85
86        let bytes_to_read = self.remaining.min(okm.len().try_into().unwrap_or(u16::MAX));
87        self.reader.read(&mut okm[..bytes_to_read.into()]);
88        self.remaining -= bytes_to_read;
89        Ok(bytes_to_read.into())
90    }
91}

93/// Error type for [`ExpandMsgXof`].
94#[derive(Debug, Clone, Copy, PartialEq, Eq)]
95pub enum ExpandMsgXofError {
96    /// The domain separation tag is invalid because it is empty.
97    EmptyDst,
98    /// The target security level (`K`) must not be greater then `127`
99    /// if the domain separation tag is longer than `255`.
100    DstSecurityLevel,
101}
102
103impl core::fmt::Display for ExpandMsgXofError {
104    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
105        match self {
106            Self::EmptyDst => write!(f, "the domain separation tag is empty"),
107            Self::DstSecurityLevel => write!(f, "target security level is too large"),
108        }
109    }
110}

148    fn assert_message(msg: &[u8], domain: &Domain<'_, U32>, len_in_bytes: u16, bytes: &[u8]) {
149        let msg_len = msg.len();
150        assert_eq!(msg, &bytes[..msg_len]);
151
152        let len_in_bytes_len = msg_len + size_of::<u16>();
153        assert_eq!(
154            len_in_bytes.to_be_bytes(),
155            &bytes[msg_len..len_in_bytes_len]
156        );
157
158        let dst = len_in_bytes_len + usize::from(domain.len());
159        domain.assert(&bytes[len_in_bytes_len..dst]);
160
161        let dst_len = dst + size_of::<u8>();
162        assert_eq!([domain.len()], &bytes[dst..dst_len]);
163
164        assert_eq!(dst_len, bytes.len());
165    }

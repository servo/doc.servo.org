5//! Implementation of public domain matching.
6//!
7//! The list is a file located on the `resources` folder and loaded once on first need.
8//!
9//! The list can be updated with `./mach update-pub-domains` from this source:
10//! <https://publicsuffix.org/list/>
11//!
12//! This implementation is not strictly following the specification of the list. Wildcards are not
13//! restricted to appear only in the leftmost position, but the current list has no such cases so
14//! we don't need to make the code more complex for it. The `mach` update command makes sure that
15//! those cases are not present.
16

27// We can use FxHash here.
28// The list is given by publicsuffix.org so an attack is highly unlikely
29#[derive(Clone, Debug, Default, MallocSizeOf)]
30pub struct PubDomainRules {
31    rules: FxHashSet<String>,
32    wildcards: FxHashSet<String>,
33    exceptions: FxHashSet<String>,
34}
35
36static PUB_DOMAINS: LazyLock<PubDomainRules> = LazyLock::new(load_pub_domains);
37
38pub fn public_suffix_list_size_of(ops: &mut MallocSizeOfOps) -> usize {
39    PUB_DOMAINS.size_of(ops)
40}

42impl<'a> FromIterator<&'a str> for PubDomainRules {
43    fn from_iter<T>(iter: T) -> Self
44    where
45        T: IntoIterator<Item = &'a str>,
46    {
47        let mut result = PubDomainRules::default();
48        for item in iter {
49            if let Some(stripped) = item.strip_prefix('!') {
50                result.exceptions.insert(String::from(stripped));
51            } else if let Some(stripped) = item.strip_prefix("*.") {
52                result.wildcards.insert(String::from(stripped));
53            } else {
54                result.rules.insert(String::from(item));
55            }
56        }
57        result
58    }
59}

61impl PubDomainRules {
62    pub fn parse(content: &str) -> PubDomainRules {
63        content
64            .lines()
65            .map(str::trim)
66            .filter(|s| !s.is_empty())
67            .filter(|s| !s.starts_with("//"))
68            .collect()
69    }
70    fn suffix_pair<'a>(&self, domain: &'a str) -> (&'a str, &'a str) {
71        let domain = domain.trim_start_matches('.');
72        // <https://url.spec.whatwg.org/#host-public-suffix>
73        // A trailing dot does not take part in the public suffix list lookup, but it is
74        // kept in the result, so that "example.com." and "example.com" stay distinct.
75        let trailing_dot = usize::from(domain.ends_with('.'));
76        let (public, registrable) =
77            self.suffix_pair_without_trailing_dot(&domain[..domain.len() - trailing_dot]);
78        (
79            &domain[domain.len() - trailing_dot - public.len()..],
80            &domain[domain.len() - trailing_dot - registrable.len()..],
81        )
82    }
83    fn suffix_pair_without_trailing_dot<'a>(&self, domain: &'a str) -> (&'a str, &'a str) {
84        let mut suffix = domain;
85        let mut prev_suffix = domain;
86        for (index, _) in domain.match_indices('.') {
87            let next_suffix = &domain[index + 1..];
88            if self.exceptions.contains(suffix) {
89                return (next_suffix, suffix);
90            }
91            if self.wildcards.contains(next_suffix) || self.rules.contains(suffix) {
92                return (suffix, prev_suffix);
93            }
94            prev_suffix = suffix;
95            suffix = next_suffix;
96        }
97        (suffix, prev_suffix)
98    }
99    pub fn public_suffix<'a>(&self, domain: &'a str) -> &'a str {
100        let (public, _) = self.suffix_pair(domain);
101        public
102    }
103    pub fn registrable_suffix<'a>(&self, domain: &'a str) -> &'a str {
104        let (_, registrable) = self.suffix_pair(domain);
105        registrable
106    }
107    pub fn is_public_suffix(&self, domain: &str) -> bool {
108        // Speeded-up version of
109        // domain != "" &&
110        // self.public_suffix(domain) == domain.
111        let domain = domain.trim_start_matches('.');
112        let domain = domain.strip_suffix('.').unwrap_or(domain);
113        match domain.find('.') {
114            None => !domain.is_empty(),
115            Some(index) => {
116                !self.exceptions.contains(domain) && self.wildcards.contains(&domain[index + 1..]) ||
117                    self.rules.contains(domain)
118            },
119        }
120    }
121    pub fn is_registrable_suffix(&self, domain: &str) -> bool {
122        // Speeded-up version of
123        // self.public_suffix(domain) != domain &&
124        // self.registrable_suffix(domain) == domain.
125        let domain = domain.trim_start_matches('.');
126        let domain = domain.strip_suffix('.').unwrap_or(domain);
127        match domain.find('.') {
128            None => false,
129            Some(index) => {
130                self.exceptions.contains(domain) ||
131                    !self.wildcards.contains(&domain[index + 1..]) &&
132                        !self.rules.contains(domain) &&
133                        self.is_public_suffix(&domain[index + 1..])
134            },
135        }
136    }
137}

159/// <https://html.spec.whatwg.org/multipage/#same-site>
160pub fn is_same_site(site_a: &ImmutableOrigin, site_b: &ImmutableOrigin) -> bool {
161    // First steps are for
162    // https://html.spec.whatwg.org/multipage/#concept-site-same-site
163    //
164    // Step 1. If A and B are the same opaque origin, then return true.
165    if !site_a.is_tuple() && !site_b.is_tuple() && site_a == site_b {
166        return true;
167    }
168
169    // Step 2. If A or B is an opaque origin, then return false.
170    let ImmutableOrigin::Tuple(scheme_a, host_a, _) = site_a else {
171        return false;
172    };
173    let ImmutableOrigin::Tuple(scheme_b, host_b, _) = site_b else {
174        return false;
175    };
176
177    // Step 3. If A's and B's scheme values are different, then return false.
178    if scheme_a != scheme_b {
179        return false;
180    }
181
182    // Step 4. If A's and B's host values are not equal, then return false.
183    // Includes the steps of https://html.spec.whatwg.org/multipage/#obtain-a-site
184    if let (Host::Domain(domain_a), Host::Domain(domain_b)) = (host_a, host_b) {
185        if reg_suffix(domain_a) != reg_suffix(domain_b) {
186            return false;
187        }
188    } else if host_a != host_b {
189        return false;
190    }
191
192    // Step 5. Return true.
193    true
194}

196/// The registered domain name (aka eTLD+1) for a URL.
197/// Returns None if the URL has no host name.
198/// Returns the registered suffix for the host name if it is a domain.
199/// Leaves the host name alone if it is an IP address.
200pub fn registered_domain_name(url: &ServoUrl) -> Option<Host> {
201    match url.origin() {
202        ImmutableOrigin::Tuple(_, Host::Domain(domain), _) => {
203            Some(Host::Domain(String::from(reg_suffix(&domain))))
204        },
205        ImmutableOrigin::Tuple(_, ip, _) => Some(ip),
206        ImmutableOrigin::Opaque(_) => None,
207    }
208}

210/// A [site](https://html.spec.whatwg.org/multipage/#site).
211#[derive(Clone, Debug, Deserialize, Eq, Hash, MallocSizeOf, PartialEq, Serialize)]
212pub enum Site {
213    Opaque(ImmutableOrigin),
214    Tuple { scheme: String, host: Host },
215}
216
217impl Site {
218    /// <https://html.spec.whatwg.org/multipage/#obtain-a-site>
219    pub fn from_origin(origin: &ImmutableOrigin) -> Self {
220        match origin {
221            // Step 1. If origin is an opaque origin, then return origin.
222            ImmutableOrigin::Opaque(_) => Site::Opaque(origin.clone()),
223            // Step 3. Return (origin's scheme, origin's host's registrable domain).
224            ImmutableOrigin::Tuple(scheme, Host::Domain(domain), _) => Site::Tuple {
225                scheme: scheme.clone(),
226                host: Host::Domain(reg_suffix(domain).to_owned()),
227            },
228            // Step 2. If origin's host's registrable domain is null, then return (origin's scheme, origin's host).
229            ImmutableOrigin::Tuple(scheme, host, _) => Site::Tuple {
230                scheme: scheme.clone(),
231                host: host.clone(),
232            },
233        }
234    }

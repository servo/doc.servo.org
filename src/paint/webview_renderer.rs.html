82/// A renderer for a libservo `WebView`. This is essentially the [`ServoRenderer`]'s interface to a
83/// libservo `WebView`, but the code here cannot depend on libservo in order to prevent circular
84/// dependencies, which is why we store a `dyn WebViewTrait` here instead of the `WebView` itself.
85pub(crate) struct WebViewRenderer {
86    /// The [`WebViewId`] of the `WebView` associated with this [`WebViewDetails`].
87    pub id: WebViewId,
88    /// The renderer's view of the embedding layer `WebView` as a trait implementation,
89    /// so that the renderer doesn't need to depend on the embedding layer. This avoids
90    /// a dependency cycle.
91    pub webview: Box<dyn WebViewTrait>,
92    /// The root [`PipelineId`] of the currently displayed page in this WebView.
93    pub root_pipeline_id: Option<PipelineId>,
94    /// The rectangle of the [`WebView`] in device pixels, which is the viewport.
95    pub rect: DeviceRect,
96    /// Tracks details about each active pipeline that `Paint` knows about.
97    pub pipelines: FxHashMap<PipelineId, PipelineDetails>,
98    /// Pending scroll/zoom events.
99    pending_scroll_zoom_events: Vec<ScrollZoomEvent>,
100    /// A map of pending wheel events. These are events that have been sent to script,
101    /// but are waiting for processing. When they are handled by script, they may trigger
102    /// scroll events depending on whether `preventDefault()` was called on the event.
103    pending_wheel_events: FxHashMap<InputEventId, WheelEvent>,
104    /// Touch input state machine
105    touch_handler: TouchHandler,
106    /// "Desktop-style" zoom that resizes the viewport to fit the window.
107    pub page_zoom: Scale<f32, CSSPixel, DeviceIndependentPixel>,
108    /// "Mobile-style" zoom that does not reflow the page. When there is no [`PinchZoom`] a
109    /// zoom factor of 1.0 is implied and the [`PinchZoom::transform`] will be the identity.
110    pinch_zoom: PinchZoom,
111    /// The HiDPI scale factor for the `WebView` associated with this renderer. This is controlled
112    /// by the embedding layer.
113    hidpi_scale_factor: Scale<f32, DeviceIndependentPixel, DevicePixel>,
114    /// Whether or not this [`WebViewRenderer`] is hidden.
115    hidden: bool,
116    /// Whether or not this [`WebViewRenderer`] isn't throttled and has a pipeline with
117    /// active animations or animation frame callbacks.
118    animating: bool,
119    /// A [`ViewportDescription`] for this [`WebViewRenderer`], which contains the limitations
120    /// and initial values for zoom derived from the `viewport` meta tag in web content.
121    viewport_description: ViewportDescription,
122
123    /// The dimensions of the screen on which this WebView is rendering.
124    screen_size: Size2D<f32, DevicePixel>,
125
126    //
127    // Data that is shared with the parent renderer.
128    //
129    /// The channel on which messages can be sent to the constellation.
130    embedder_to_constellation_sender: Sender<EmbedderToConstellationMessage>,
131    /// The [`BaseRefreshDriver`] which manages the painting of `WebView`s during animations.
132    refresh_driver: Rc<BaseRefreshDriver>,
133    /// The active webrender document.
134    webrender_document: DocumentId,
135}

210    /// Record that `source` has reported `pipeline_id` as exited. Returns `true`
211    /// once every source has done so and the pipeline details have been discarded,
212    /// so that the caller can also drop the pipeline's scene state in WebRender.
213    pub(crate) fn pipeline_exited(
214        &mut self,
215        pipeline_id: PipelineId,
216        source: PipelineExitSource,
217    ) -> bool {
218        let pipeline = self.pipelines.entry(pipeline_id);
219        let Entry::Occupied(mut pipeline) = pipeline else {
220            return false;
221        };
222
223        pipeline.get_mut().exited.insert(source);
224
225        // Do not remove pipeline details until both the Constellation and Script have
226        // finished processing the pipeline shutdown. This prevents any followup messges
227        // from re-adding the pipeline details and creating a zombie.
228        if !pipeline.get().exited.is_all() {
229            return false;
230        }
231
232        // Flush the LCP candidates when exiting pipeline.
233        pipeline.get_mut().lcp_candidates.clear();
234
235        pipeline.remove_entry();
236        true
237    }

253    pub(crate) fn send_scroll_positions_to_layout_for_pipeline(
254        &self,
255        pipeline_id: PipelineId,
256        scrolled_node: ExternalScrollId,
257    ) {
258        let Some(details) = self.pipelines.get(&pipeline_id) else {
259            return;
260        };
261
262        let offsets = details.scroll_tree.scroll_offsets();
263
264        // This might be true if we have not received a display list from the layout
265        // associated with this pipeline yet. In that case, the layout is not ready to
266        // receive scroll offsets anyway, so just save time and prevent other issues by
267        // not sending them.
268        if offsets.is_empty() {
269            return;
270        }
271
272        let _ = self.embedder_to_constellation_sender.send(
273            EmbedderToConstellationMessage::SetScrollStates(
274                pipeline_id,
275                ScrollStateUpdate {
276                    scrolled_node,
277                    offsets,
278                },
279            ),
280        );
281    }

283    pub(crate) fn set_frame_tree_on_pipeline_details(
284        &mut self,
285        frame_tree: &SendableFrameTree,
286        parent_pipeline_id: Option<PipelineId>,
287    ) {
288        let pipeline_id = frame_tree.pipeline.id;

290        pipeline_details.pipeline = Some(frame_tree.pipeline.clone());
291        pipeline_details.parent_pipeline_id = parent_pipeline_id;
292        pipeline_details.children = frame_tree
293            .children
294            .iter()
295            .map(|frame_tree| frame_tree.pipeline.id)
296            .collect();
297
298        for kid in &frame_tree.children {
299            self.set_frame_tree_on_pipeline_details(kid, Some(pipeline_id));
300        }

303    /// Sets or unsets the animations-running flag for the given pipeline. Returns
304    /// true if the pipeline has started animating.
305    pub(crate) fn change_pipeline_running_animations_state(
306        &mut self,
307        pipeline_id: PipelineId,
308        animation_state: AnimationState,
309    ) -> bool {
310        let pipeline_details = self.ensure_pipeline_details(pipeline_id);
311        let was_animating = pipeline_details.animating();
312        match animation_state {
313            AnimationState::AnimationsPresent => {
314                pipeline_details.animations_running = true;
315            },
316            AnimationState::AnimationCallbacksPresent => {
317                pipeline_details.animation_callbacks_running = true;
318            },
319            AnimationState::NoAnimationsPresent => {
320                pipeline_details.animations_running = false;
321            },
322            AnimationState::AnimationCallbacksAbsent => {
323                pipeline_details.animation_callbacks_running = false;
324            },
325        }
326        let started_animating = !was_animating && pipeline_details.animating();
327
328        self.update_animation_state();
329
330        // It's important that an animation tick is triggered even if the
331        // WebViewRenderer's overall animation state hasn't changed. It's possible that
332        // the WebView was animating, but not producing new display lists. In that case,
333        // no repaint will happen and thus no repaint will trigger the next animation tick.
334        started_animating
335    }

337    /// Sets or unsets the throttled flag for the given pipeline. Returns
338    /// true if the pipeline has started animating.
339    pub(crate) fn set_throttled(&mut self, pipeline_id: PipelineId, throttled: bool) -> bool {
340        let pipeline_details = self.ensure_pipeline_details(pipeline_id);
341        let was_animating = pipeline_details.animating();
342        pipeline_details.throttled = throttled;
343        let started_animating = !was_animating && pipeline_details.animating();
344
345        // Throttling a pipeline can cause it to be taken into the "not-animating" state.
346        self.update_animation_state();
347
348        // It's important that an animation tick is triggered even if the
349        // WebViewRenderer's overall animation state hasn't changed. It's possible that
350        // the WebView was animating, but not producing new display lists. In that case,
351        // no repaint will happen and thus no repaint will trigger the next animation tick.
352        started_animating
353    }

380    /// Update touch-based animations (currently just fling) during a `RefreshDriver`-based
381    /// frame tick. Returns `true` if we should continue observing frames (the fling is ongoing)
382    /// or `false` if we should stop observing frames (the fling has finished).
383    pub(crate) fn update_touch_handling_at_new_frame_start(&mut self) -> bool {

385            return false;
386        };
387
388        self.on_scroll_window_event(
389            Scroll::Delta((-fling_action.delta).into()),
390            fling_action.cursor,
391            // Fling is a continuation of a touch pan, so it must respect
392            // `touch-action` like the originating touch gesture.
393            ScrollType::Touch,
394        );
395        true

398    fn dispatch_input_event_with_hit_testing(
399        &mut self,
400        render_api: &RenderApi,
401        event: InputEventAndId,
402    ) -> bool {
403        let event_point = event
404            .event
405            .point()
406            .map(|point| point.as_device_point(self.device_pixels_per_page_pixel()));
407        let is_touch_down = matches!(
408            &event.event,
409            InputEvent::Touch(TouchEvent {
410                event_type: TouchEventType::Down,
411                ..
412            })
413        );
414        let hit_test_result = match event_point {
415            Some(point) => {
416                let hit_test_result = match event.event {
417                    InputEvent::Touch(_) => self.touch_handler.get_hit_test_result_cache_value(),
418                    _ => None,
419                }
420                .or_else(|| self.hit_test(render_api, point).into_iter().nth(0));
421                if hit_test_result.is_none() {
422                    warn!("Empty hit test result for input event, ignoring.");
423                    return false;
424                }
425                hit_test_result
426            },
427            None => None,
428        };
429
430        // For touch-down, capture the hit node's `touch-action` and scrollable
431        // axes from the scroll tree so the pan axis-lock policy can be decided
432        // at pan-start.
433        if is_touch_down && let Some(hit) = &hit_test_result {
434            let policy_input = self
435                .pipelines
436                .get(&hit.pipeline_id)
437                .and_then(|pipeline_details| {
438                    pipeline_details
439                        .scroll_tree
440                        .touch_action_and_scrollable_axes_for(hit.external_scroll_id)
441                })
442                .map(
443                    |(touch_action, scrollable_x, scrollable_y)| PanPolicyInput {
444                        touch_action,
445                        scrollable_x,
446                        scrollable_y,
447                    },
448                );
449            if let Some(input) = policy_input {
450                self.touch_handler.set_pan_policy_input(input);
451            }
452        }
453
454        if let Err(error) = self.embedder_to_constellation_sender.send(
455            EmbedderToConstellationMessage::ForwardInputEvent(self.id, event, hit_test_result),
456        ) {
457            warn!("Sending event to constellation failed ({error:?}).");
458            false
459        } else {
460            true
461        }
462    }

464    pub(crate) fn notify_input_event(
465        &mut self,
466        render_api: &RenderApi,
467        repaint_reason: &Cell<RepaintReason>,
468        event_and_id: InputEventAndId,
469    ) -> bool {
470        if let InputEvent::Touch(touch_event) = event_and_id.event {
471            return self.on_touch_event(render_api, repaint_reason, touch_event, event_and_id.id);
472        }
473
474        if let InputEvent::Wheel(wheel_event) = event_and_id.event {
475            self.pending_wheel_events
476                .insert(event_and_id.id, wheel_event);
477        }
478
479        self.dispatch_input_event_with_hit_testing(render_api, event_and_id)
480    }

482    fn send_touch_event(
483        &mut self,
484        render_api: &RenderApi,
485        event: TouchEvent,
486        id: InputEventId,
487    ) -> bool {
488        let cancelable = event.is_cancelable();
489        let event_type = event.event_type;
490
491        let input_event_and_id = InputEventAndId {
492            event: InputEvent::Touch(event),
493            id,
494        };
495
496        let result = self.dispatch_input_event_with_hit_testing(render_api, input_event_and_id);
497
498        // We only post-process events that are actually cancelable. Uncancelable ones
499        // are processed immediately and can be ignored once they have been sent to the
500        // Constellation.
501        if cancelable && result {
502            self.touch_handler
503                .add_pending_touch_input_event(id, event.touch_id, event_type);
504        }
505
506        result
507    }

509    pub(crate) fn on_touch_event(
510        &mut self,
511        render_api: &RenderApi,
512        repaint_reason: &Cell<RepaintReason>,
513        event: TouchEvent,
514        id: InputEventId,
515    ) -> bool {
516        let result = match event.event_type {
517            TouchEventType::Down => self.on_touch_down(render_api, event, id),
518            TouchEventType::Move => self.on_touch_move(render_api, event, id),
519            TouchEventType::Up => self.on_touch_up(render_api, event, id),
520            TouchEventType::Cancel => self.on_touch_cancel(render_api, event, id),
521        };
522
523        self.touch_handler
524            .add_touch_move_refresh_observer_if_necessary(
525                self.refresh_driver.clone(),
526                repaint_reason,
527            );
528        result
529    }

531    fn on_touch_down(
532        &mut self,
533        render_api: &RenderApi,
534        event: TouchEvent,
535        id: InputEventId,
536    ) -> bool {
537        let point = event
538            .point
539            .as_device_point(self.device_pixels_per_page_pixel());
540        self.touch_handler.on_touch_down(event.touch_id, point);
541        self.send_touch_event(render_api, event, id)
542    }

544    fn on_touch_move(
545        &mut self,
546        render_api: &RenderApi,
547        mut event: TouchEvent,
548        id: InputEventId,
549    ) -> bool {
550        let point = event
551            .point
552            .as_device_point(self.device_pixels_per_page_pixel());
553        let action = self.touch_handler.on_touch_move(
554            event.touch_id,
555            point,
556            self.device_pixels_per_page_pixel_not_including_pinch_zoom()
557                .get(),
558        );
559        if let Some(action) = action {
560            // if first move processed and allowed, we directly process the move event,
561            // without waiting for the script handler.
562            if self
563                .touch_handler
564                .move_allowed(self.touch_handler.current_sequence_id)
565            {
566                // https://w3c.github.io/touch-events/#cancelability
567                event.disable_cancelable();
568                self.pending_scroll_zoom_events.push(action);
569            }
570        }
571        let mut reached_constellation = false;
572        // When the event is touchmove, if the script thread is processing the touch
573        // move event, we skip sending the event to the script thread.
574        // This prevents the script thread from stacking up for a large amount of time.
575        if !self.touch_handler.is_handling_touch_move_for_touch_id(
576            self.touch_handler.current_sequence_id,
577            event.touch_id,
578        ) {
579            reached_constellation = self.send_touch_event(render_api, event, id);
580            if reached_constellation && event.is_cancelable() {
581                self.touch_handler.set_handling_touch_move_for_touch_id(
582                    self.touch_handler.current_sequence_id,
583                    event.touch_id,
584                    TouchIdMoveTracking::Track,
585                );
586            }
587        }
588        reached_constellation
589    }

612    fn on_touch_event_processed(
613        &mut self,
614        render_api: &RenderApi,
615        pending_touch_input_event: PendingTouchInputEvent,
616        result: InputEventResult,
617    ) {
618        let PendingTouchInputEvent {
619            sequence_id,
620            event_type,
621            touch_id,
622        } = pending_touch_input_event;
623
624        if result.contains(InputEventResult::DefaultPrevented) {
625            debug!(
626                "Touch event {:?} in sequence {:?} prevented!",
627                event_type, sequence_id
628            );
629            match event_type {
630                TouchEventType::Down => {
631                    // prevents both click and move
632                    self.touch_handler.prevent_click(sequence_id);
633                    self.touch_handler.prevent_move(sequence_id);
634                    self.touch_handler
635                        .remove_pending_touch_move_actions(sequence_id);
636                },
637                TouchEventType::Move => {
638                    // script thread processed the touch move event, mark this false.
639                    if let Some(info) = self.touch_handler.get_touch_sequence_mut(sequence_id) {
640                        info.prevent_move = TouchMoveAllowed::Prevented;
641                        if let TouchSequenceState::PendingFling { .. } = info.state {
642                            info.state = TouchSequenceState::Finished;
643                        }
644                        self.touch_handler.set_handling_touch_move_for_touch_id(
645                            self.touch_handler.current_sequence_id,
646                            touch_id,
647                            TouchIdMoveTracking::Remove,
648                        );
649                        self.touch_handler
650                            .remove_pending_touch_move_actions(sequence_id);
651                    }
652                },
653                TouchEventType::Up => {
654                    // Note: We don't have to consider PendingFling here, since we handle that
655                    // in the DefaultAllowed case of the touch_move event.
656                    // Note: Removing can and should fail, if we still have an active Fling,
657                    let Some(info) = &mut self.touch_handler.get_touch_sequence_mut(sequence_id)
658                    else {
659                        // The sequence ID could already be removed, e.g. if Fling finished,
660                        // before the touch_up event was handled (since fling can start
661                        // immediately if move was previously allowed, and clicks are anyway not
662                        // happening from fling).
663                        return;
664                    };
665                    match info.state {
666                        TouchSequenceState::PendingClick(_) => {
667                            info.state = TouchSequenceState::Finished;
668                            self.touch_handler.remove_touch_sequence(sequence_id);
669                        },
670                        TouchSequenceState::Flinging { .. } => {
671                            // We can't remove the touch sequence yet
672                        },
673                        TouchSequenceState::Finished => {
674                            self.touch_handler.remove_touch_sequence(sequence_id);
675                        },
676                        TouchSequenceState::Touching |
677                        TouchSequenceState::Panning { .. } |
678                        TouchSequenceState::Pinching |
679                        TouchSequenceState::MultiTouch |
680                        TouchSequenceState::PendingFling { .. } => {
681                            // It's possible to transition from Pinch to pan, Which means that
682                            // a touch_up event for a pinch might have arrived here, but we
683                            // already transitioned to pan or even PendingFling.
684                            // We don't need to do anything in these cases though.
685                        },
686                    }
687                },
688                TouchEventType::Cancel => {
689                    // We could still have pending event handlers, so we remove the pending
690                    // actions, and try to remove the touch sequence.
691                    self.touch_handler
692                        .remove_pending_touch_move_actions(sequence_id);
693                    self.touch_handler.try_remove_touch_sequence(sequence_id);
694                },
695            }
696        } else {
697            debug!(
698                "Touch event {:?} in sequence {:?} allowed",
699                event_type, sequence_id
700            );
701            match event_type {
702                TouchEventType::Down => {},
703                TouchEventType::Move => {
704                    self.pending_scroll_zoom_events.extend(
705                        self.touch_handler
706                            .take_pending_touch_move_actions(sequence_id),
707                    );
708                    self.touch_handler.set_handling_touch_move_for_touch_id(
709                        self.touch_handler.current_sequence_id,
710                        touch_id,
711                        TouchIdMoveTracking::Remove,
712                    );
713                    if let Some(info) = self.touch_handler.get_touch_sequence_mut(sequence_id) &&
714                        info.prevent_move == TouchMoveAllowed::Pending
715                    {
716                        info.prevent_move = TouchMoveAllowed::Allowed;
717                        if let TouchSequenceState::PendingFling { velocity, point } = info.state {
718                            info.state = TouchSequenceState::Flinging { velocity, point }
719                        }
720                    }
721                },
722                TouchEventType::Up => {
723                    let Some(info) = self.touch_handler.get_touch_sequence_mut(sequence_id) else {
724                        // The sequence was already removed because there is no default action.
725                        return;
726                    };
727                    match info.state {
728                        TouchSequenceState::PendingClick(point) => {
729                            info.state = TouchSequenceState::Finished;
730                            // PreventDefault from touch_down may have been processed after
731                            // touch_up already occurred.
732                            if !info.prevent_click {
733                                self.simulate_mouse_click(render_api, point);
734                            }
735                            self.touch_handler.remove_touch_sequence(sequence_id);
736                        },
737                        TouchSequenceState::Flinging { .. } => {
738                            // We can't remove the touch sequence yet
739                        },
740                        TouchSequenceState::Finished => {
741                            self.touch_handler.remove_touch_sequence(sequence_id);
742                        },
743                        TouchSequenceState::Panning { .. } |
744                        TouchSequenceState::Pinching |
745                        TouchSequenceState::PendingFling { .. } => {
746                            // It's possible to transition from Pinch to pan, Which means that
747                            // a touch_up event for a pinch might have arrived here, but we
748                            // already transitioned to pan or even PendingFling.
749                            // We don't need to do anything in these cases though.
750                        },
751                        TouchSequenceState::MultiTouch | TouchSequenceState::Touching => {
752                            // We transitioned to touching from multi-touch or pinching.
753                        },
754                    }
755                },
756                TouchEventType::Cancel => {
757                    self.touch_handler
758                        .remove_pending_touch_move_actions(sequence_id);
759                    self.touch_handler.try_remove_touch_sequence(sequence_id);
760                },
761            }
762        }
763    }

811    /// Process pending scroll events for this [`WebViewRenderer`]. Returns a tuple containing:
812    ///
813    ///  - A boolean that is true if a zoom occurred.
814    ///  - An optional [`ScrollResult`] if a scroll occurred.
815    ///
816    /// It is up to the caller to ensure that these events update the rendering appropriately.
817    pub(crate) fn process_pending_scroll_and_pinch_zoom_events(
818        &mut self,
819        render_api: &RenderApi,
820    ) -> (PinchZoomResult, Option<ScrollResult>) {
821        if self.pending_scroll_zoom_events.is_empty() {
822            return (PinchZoomResult::DidNotPinchZoom, None);
823        }
824
825        // Batch up all scroll events and changes to pinch zoom into a single change, or
826        // else we'll do way too much painting.
827        let mut combined_scroll_event: Option<ScrollEvent> = None;
828        let mut new_pinch_zoom = self.pinch_zoom;
829        let device_pixels_per_page_pixel = self.device_pixels_per_page_pixel();
830
831        for scroll_event in self.pending_scroll_zoom_events.drain(..) {
832            match scroll_event {
833                ScrollZoomEvent::PinchZoom(magnification, center) => {
834                    let new_factor = self
835                        .viewport_description
836                        .clamp_zoom(self.pinch_zoom.zoom_factor().0 * magnification);
837                    new_pinch_zoom.set_zoom(new_factor, center);
838                },
839                ScrollZoomEvent::Scroll(scroll_event_info) => {
840                    let combined_event = match combined_scroll_event.as_mut() {
841                        None => {
842                            combined_scroll_event = Some(scroll_event_info);
843                            continue;
844                        },
845                        Some(combined_event) => combined_event,
846                    };
847
848                    match (combined_event.scroll, scroll_event_info.scroll) {
849                        (Scroll::Delta(old_delta), Scroll::Delta(new_delta)) => {
850                            let old_delta =
851                                old_delta.as_device_vector(device_pixels_per_page_pixel);
852                            let new_delta =
853                                new_delta.as_device_vector(device_pixels_per_page_pixel);
854                            combined_event.scroll = Scroll::Delta((old_delta + new_delta).into());
855                        },
856                        (Scroll::Start, _) | (Scroll::End, _) => {
857                            // Once we see Start or End, we shouldn't process any more events.
858                            break;
859                        },
860                        (_, Scroll::Start) | (_, Scroll::End) => {
861                            // If this is an event which is scrolling to the start or end of the page,
862                            // disregard other pending events and exit the loop.
863                            *combined_event = scroll_event_info;
864                            break;
865                        },
866                    }
867                },
868            }
869        }
870
871        // When zoomed in via pinch zoom, first try to move the center of the zoom and use the rest
872        // of the delta for scrolling. This allows moving the zoomed into viewport around in the
873        // unzoomed viewport before actually scrolling the underlying layers.
874        if let Some(combined_scroll_event) = combined_scroll_event.as_mut() {
875            new_pinch_zoom.pan(
876                &mut combined_scroll_event.scroll,
877                self.device_pixels_per_page_pixel(),
878            )
879        }
880
881        let scroll_result = combined_scroll_event.and_then(|combined_event| {
882            self.scroll_node_at_device_point(
883                render_api,
884                combined_event.point.to_f32(),
885                combined_event.scroll,
886                combined_event.scroll_type,
887            )
888        });
889        if let Some(ref scroll_result) = scroll_result {
890            self.send_scroll_positions_to_layout_for_pipeline(
891                scroll_result.hit_test_result.pipeline_id,
892                scroll_result.external_scroll_id,
893            );
894        } else {
895            self.touch_handler.stop_fling_if_needed();
896        }
897
898        // Additionally notify pinch zoom update to the script.
899        let pinch_zoom_result = self.set_pinch_zoom(new_pinch_zoom);
900        if pinch_zoom_result == PinchZoomResult::DidPinchZoom {
901            self.send_pinch_zoom_infos_to_script();
902            // Pinch zoom changes the viewport transform without touching the pipeline, so no reflow
903            // will occur. Notify the embedding layer so it can refresh the accessibility root node,
904            // whose transform scales by the pinch zoom.
905            self.webview.notify_viewport_updated();
906        }
907
908        (pinch_zoom_result, scroll_result)
909    }

911    /// Perform a hit test at the given [`DevicePoint`] and apply the [`Scroll`]
912    /// scrolling to the applicable scroll node under that point. If a scroll was
913    /// performed, returns the hit test result contains [`PipelineId`] of the node
914    /// scrolled, the id, and the final scroll delta.
915    fn scroll_node_at_device_point(
916        &mut self,
917        render_api: &RenderApi,
918        cursor: DevicePoint,
919        scroll: Scroll,
920        scroll_type: ScrollType,
921    ) -> Option<ScrollResult> {
922        let scroll_location = match scroll {
923            Scroll::Delta(delta) => {
924                let device_pixels_per_page = self.device_pixels_per_page_pixel();
925                let calculate_delta =
926                    delta.as_device_vector(device_pixels_per_page) / device_pixels_per_page;
927                ScrollLocation::Delta(calculate_delta.cast_unit())
928            },
929            Scroll::Start => ScrollLocation::Start,
930            Scroll::End => ScrollLocation::End,
931        };
932
933        let hit_test_results: Vec<_> = self
934            .touch_handler
935            .get_hit_test_result_cache_value()
936            .map(|result| vec![result])
937            .unwrap_or_else(|| self.hit_test(render_api, cursor));
938
939        // Iterate through all hit test results, processing only the first node of each pipeline.
940        // This is needed to propagate the scroll events from a pipeline representing an iframe to
941        // its ancestor pipelines.
942        let mut previous_pipeline_id = None;
943        for hit_test_result in hit_test_results {
944            let pipeline_details = self.pipelines.get_mut(&hit_test_result.pipeline_id)?;
945            if previous_pipeline_id.replace(hit_test_result.pipeline_id) !=
946                Some(hit_test_result.pipeline_id)
947            {
948                let scroll_result = pipeline_details.scroll_tree.scroll_node_or_ancestor(
949                    hit_test_result.external_scroll_id,
950                    scroll_location,
951                    scroll_type,
952                );
953                if let Some((external_scroll_id, offset)) = scroll_result {
954                    // We would like to cache the hit test for the node that that actually scrolls
955                    // while panning, which we don't know until right now (as some nodes
956                    // might be at the end of their scroll area). In particular, directionality of
957                    // scroll matters. That's why this is done here and not as soon as the touch
958                    // starts.
959                    self.touch_handler.set_hit_test_result_cache_value(
960                        hit_test_result.clone(),
961                        self.device_pixels_per_page_pixel(),
962                    );
963                    return Some(ScrollResult {
964                        hit_test_result,
965                        external_scroll_id,
966                        offset,
967                    });
968                }
969            }
970        }
971        None
972    }

974    /// Scroll the viewport (root pipeline, root scroll node) of this WebView, but first
975    /// attempting to pan the pinch zoom viewport. This is called when processing
976    /// key-based scrolling from script.
977    pub(crate) fn scroll_viewport_by_delta(
978        &mut self,
979        delta: LayoutVector2D,
980    ) -> (PinchZoomResult, Vec<ScrollResult>) {
981        let device_pixels_per_page_pixel = self.device_pixels_per_page_pixel();
982        let delta_in_device_pixels = delta.cast_unit() * device_pixels_per_page_pixel;
983        let remaining = self.pinch_zoom.pan_with_device_scroll(
984            Scroll::Delta(delta_in_device_pixels.into()),
985            device_pixels_per_page_pixel,
986        );
987
988        let pinch_zoom_result = match remaining == delta_in_device_pixels {
989            true => PinchZoomResult::DidNotPinchZoom,
990            false => PinchZoomResult::DidPinchZoom,
991        };
992        if remaining == Vector2D::zero() {
993            return (pinch_zoom_result, vec![]);
994        }
995
996        let Some(root_pipeline_id) = self.root_pipeline_id else {
997            return (pinch_zoom_result, vec![]);
998        };
999        let Some(root_pipeline) = self.pipelines.get_mut(&root_pipeline_id) else {
1000            return (pinch_zoom_result, vec![]);
1001        };
1002
1003        let remaining = remaining / device_pixels_per_page_pixel;
1004        let Some((external_scroll_id, offset)) = root_pipeline.scroll_tree.scroll_node_or_ancestor(
1005            ExternalScrollId(0, root_pipeline_id.into()),
1006            ScrollLocation::Delta(remaining.cast_unit()),
1007            // These are initiated only by keyboard events currently.
1008            ScrollType::InputEvents,
1009        ) else {
1010            return (pinch_zoom_result, vec![]);
1011        };
1012
1013        let hit_test_result = PaintHitTestResult {
1014            pipeline_id: root_pipeline_id,
1015            // It's difficult to get a good value for this as it needs to be piped
1016            // all the way through script and back here.
1017            point_in_viewport: Default::default(),
1018            external_scroll_id,
1019        };
1020
1021        self.send_scroll_positions_to_layout_for_pipeline(root_pipeline_id, external_scroll_id);
1022
1023        if pinch_zoom_result == PinchZoomResult::DidPinchZoom {
1024            self.send_pinch_zoom_infos_to_script();
1025        }
1026
1027        let scroll_result = ScrollResult {
1028            hit_test_result,
1029            external_scroll_id,
1030            offset,
1031        };
1032        (pinch_zoom_result, vec![scroll_result])
1033    }

1078    /// The scale to use when displaying this [`WebViewRenderer`] in WebRender
1079    /// including both viewport scale (page zoom and hidpi scale) as well as any
1080    /// pinch zoom applied. This is based on the latest display list received,
1081    /// as page zoom changes are applied asynchronously and the rendered view
1082    /// should reflect the latest display list.
1083    pub(crate) fn device_pixels_per_page_pixel(&self) -> Scale<f32, CSSPixel, DevicePixel> {
1084        let viewport_scale = self
1085            .root_pipeline_id
1086            .and_then(|pipeline_id| self.pipelines.get(&pipeline_id))
1087            .and_then(|pipeline| pipeline.viewport_scale)
1088            .unwrap_or_else(|| self.page_zoom * self.hidpi_scale_factor);
1089        viewport_scale * self.pinch_zoom.zoom_factor()
1090    }

1112    fn send_window_size_message(&self) {
1113        // The device pixel ratio used by the style system should include the scale from page pixels
1114        // to device pixels, but not including any pinch zoom.
1115        let device_pixel_ratio = self.device_pixels_per_page_pixel_not_including_pinch_zoom();
1116        // From <https://www.w3.org/TR/css-viewport-1/#actual-viewport>:
1117        // This is the viewport you get after processing the viewport <meta> tag.
1118        let layout_viewport = self.rect.size().to_f32() /
1119            (device_pixel_ratio * Scale::new(self.viewport_description.initial_scale.get()));
1120        let _ = self.embedder_to_constellation_sender.send(
1121            EmbedderToConstellationMessage::ChangeViewportDetails(
1122                self.id,
1123                ViewportDetails {
1124                    hidpi_scale_factor: device_pixel_ratio,
1125                    size: layout_viewport,
1126                    device_size: self.screen_size,
1127                },
1128                WindowSizeType::Resize,
1129            ),
1130        );
1131    }

1188    pub(crate) fn notify_input_event_handled(
1189        &mut self,
1190        render_api: &RenderApi,
1191        repaint_reason: &Cell<RepaintReason>,
1192        id: InputEventId,
1193        result: InputEventResult,
1194    ) {
1195        if let Some(pending_touch_input_event) =
1196            self.touch_handler.take_pending_touch_input_event(id)
1197        {
1198            self.on_touch_event_processed(render_api, pending_touch_input_event, result);
1199            self.touch_handler
1200                .add_touch_move_refresh_observer_if_necessary(
1201                    self.refresh_driver.clone(),
1202                    repaint_reason,
1203                );
1204        }
1205
1206        if let Some(wheel_event) = self.pending_wheel_events.remove(&id) &&
1207            !result.contains(InputEventResult::DefaultPrevented)
1208        {
1209            // A scroll delta for a wheel event is the inverse of the wheel delta.
1210            let scroll_delta =
1211                DeviceVector2D::new(-wheel_event.delta.x as f32, -wheel_event.delta.y as f32);
1212            self.notify_scroll_event(Scroll::Delta(scroll_delta.into()), wheel_event.point);
1213        }
1214    }

16
17use crate::painter::PaintMetricState;
18use crate::web_content_animation::PipelineAnimations;
19
20pub(crate) struct PipelineDetails {
21    /// The pipeline associated with this PipelineDetails object.
22    pub pipeline: Option<CompositionPipeline>,
23
24    /// The id of the parent pipeline, if any.
25    pub parent_pipeline_id: Option<PipelineId>,
26
27    /// The ids of the child pipelines for this pipeline.
28    pub children: Vec<PipelineId>,
29
30    /// Whether animations are running
31    pub animations_running: bool,
32
33    /// Whether there are animation callbacks
34    pub animation_callbacks_running: bool,
35
36    /// Whether to use less resources by stopping animations.
37    pub throttled: bool,
38
39    /// The `Paint`-side [ScrollTree]. This is used to allow finding and scrolling
40    /// nodes in `Paint` before forwarding new offsets to WebRender.
41    pub scroll_tree: ScrollTree,
42
43    /// The paint metric status of the first paint.
44    pub first_paint_metric: PaintMetricState,
45
46    /// The paint metric status of the first contentful paint.
47    pub first_contentful_paint_metric: PaintMetricState,
48
49    /// LCP candidates waiting to be presented, in order by [WebRenderEpoch].
50    pub lcp_candidates: VecDeque<(WebRenderEpoch, LCPCandidate)>,
51
52    /// The CSS pixel to device pixel scale of the viewport of this pipeline, including
53    /// page zoom, but not including any pinch zoom amount. This is used to detect
54    /// situations where the current display list is for an old scale.
55    pub viewport_scale: Option<Scale<f32, CSSPixel, DevicePixel>>,
56
57    /// Which parts of Servo have reported that this `Pipeline` has exited. Only when all
58    /// have done so will it be discarded.
59    pub exited: PipelineExitSource,
60
61    /// The [`Epoch`] of the latest display list received for this `Pipeline` or `None` if no
62    /// display list has been received.
63    pub display_list_epoch: Option<Epoch>,
64
65    /// Paint-driven animations associated with this [`PipelineDetails`]. Currently only text caret
66    /// is handled this way.
67    ///
68    /// Note: This does not manage animations and transitions from CSS or for user input
69    /// interaction.
70    pub animations: PipelineAnimations,
71}
72
73impl PipelineDetails {
74    pub(crate) fn animation_callbacks_running(&self) -> bool {
75        self.animation_callbacks_running
76    }
77
78    pub(crate) fn animating(&self) -> bool {
79        !self.throttled && (self.animation_callbacks_running || self.animations_running)
80    }
81}
82
83impl PipelineDetails {
84    pub(crate) fn new() -> PipelineDetails {
85        PipelineDetails {
86            pipeline: None,
87            parent_pipeline_id: None,
88            children: Default::default(),
89            viewport_scale: None,
90            animations_running: false,
91            animation_callbacks_running: false,
92            throttled: false,
93            scroll_tree: ScrollTree::default(),
94            first_paint_metric: PaintMetricState::Waiting,
95            first_contentful_paint_metric: PaintMetricState::Waiting,
96            lcp_candidates: VecDeque::new(),
97            exited: PipelineExitSource::empty(),
98            display_list_epoch: None,
99            animations: Default::default(),
100        }
101    }
102
103    pub(crate) fn install_new_scroll_tree(&mut self, new_scroll_tree: ScrollTree) {
104        let old_scroll_offsets = self.scroll_tree.scroll_offsets();
105        self.scroll_tree = new_scroll_tree;
106        self.scroll_tree.set_all_scroll_offsets(&old_scroll_offsets);
107    }
108}

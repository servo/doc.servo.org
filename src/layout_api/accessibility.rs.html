124/// The inverse of the map above, i.e. a map from supported AccessKit role to the corresponding ARIA
125/// role.
126///
127/// This is created by inverting the map above to ensure consistency, then for roles where multiple
128/// ARIA roles map to the same AccessKit role an explicit entry is added.
129pub static ACCESSKIT_ROLE_TO_ARIA_ROLE: LazyLock<FxHashMap<Role, Atom>> = LazyLock::new(|| {
130    let mut accesskit_to_aria: FxHashMap<Role, Atom> = ARIA_ROLE_TO_ACCESSKIT_ROLE
131        .iter()
132        .map(|(aria_role, accesskit_role)| (*accesskit_role, aria_role.clone()))
133        .collect();
134    // Note: AccessKit doesn't expose nodes with the `GenericContainer` role by default, so it acts
135    // more like ARIA's `none`/`presentation` role.
136    accesskit_to_aria.insert(Role::GenericContainer, Atom::from("none"));
137    accesskit_to_aria
138});

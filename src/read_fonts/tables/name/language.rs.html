1/// Converts a Macintosh or Windows name-table language ID to a BCP 47 tag.
2///
3/// `platform_id` is 1 for Macintosh or 3 for Windows. Other platforms do not
4/// use the language IDs in this table and return `None`, as do unknown IDs.
5pub fn language_id_to_bcp47(platform_id: u16, language_id: u16) -> Option<&'static str> {
6    match platform_id {
7        1 if language_id <= 151 => {}
8        3 if language_id >= 0x0401 => {}
9        _ => return None,
10    }
11    match LANGUAGE_ID_TO_BCP47.binary_search_by(|entry| entry.0.cmp(&language_id)) {
12        Ok(ix) => LANGUAGE_ID_TO_BCP47.get(ix).map(|entry| entry.1),
13        _ => None,
14    }
15}

371#[cfg(test)]
372mod tests {
373    use super::language_id_to_bcp47;
374
375    #[test]
376    fn respects_platform_and_unknown_ids() {
377        assert_eq!(language_id_to_bcp47(1, 0), Some("en"));
378        assert_eq!(language_id_to_bcp47(3, 0x0409), Some("en-US"));
379        assert_eq!(language_id_to_bcp47(1, 0x0409), None);
380        assert_eq!(language_id_to_bcp47(3, 0), None);
381        assert_eq!(language_id_to_bcp47(0, 0), None);
382        assert_eq!(language_id_to_bcp47(3, 0xffff), None);
383    }

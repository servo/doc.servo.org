1//! Detection of fonts that need their TrueType instructions to draw correct
2//! contours.
3//!
4//! These are often called "tricky" fonts. They are recognized by family
5//! name or by the checksums of the `cvt`, `fpgm` and `prep` tables. The
6//! lists of names and checksums come from FreeType.
7
8use crate::{
9    tables::{name::Name, os2::SelectionFlags},
10    types::{BigEndian, NameId},
11    TableProvider,
12};
13
14/// Returns `true` if the font needs its instructions to draw correct
15/// contours.
16///
17/// Only fonts with TrueType outlines qualify, so a font with neither a
18/// `glyf` nor a `GLYF` table returns `false`.
19///
20/// This reads the family name and the hinting tables. A caller that checks
21/// more than once should cache the result, as
22/// [`Font::is_hint_reliant`](crate::model::Font::is_hint_reliant) does.
23pub fn is_hint_reliant<'a>(font: &impl TableProvider<'a>) -> bool {
24    if font.glyf().is_err() && font.glyf_extended().is_err() {
25        return false;
26    }
27    is_hint_reliant_name(font) || matches_hint_reliant_id_list(FontId::from_font(font))
28}

30/// Returns `true` if the family name that FreeType reports for the font is
31/// on the list.
32fn is_hint_reliant_name<'a>(font: &impl TableProvider<'a>) -> bool {
33    let Ok(name) = font.name() else {
34        return false;
35    };
36    // With the OS/2 WWS bit set the typographic family comes first, and
37    // without it the WWS family does.
38    //
39    // <https://gitlab.freedesktop.org/freetype/freetype/-/blob/80a507a6b8e3d2906ad2c8ba69329bd2fb2a85ef/src/sfnt/sfobjs.c#L1045>
40    let wws = font
41        .os2()
42        .is_ok_and(|os2| os2.fs_selection().contains(SelectionFlags::WWS));
43    let ids: &[NameId] = if wws {
44        &[NameId::TYPOGRAPHIC_FAMILY_NAME, NameId::FAMILY_NAME]
45    } else {
46        &[
47            NameId::WWS_FAMILY_NAME,
48            NameId::TYPOGRAPHIC_FAMILY_NAME,
49            NameId::FAMILY_NAME,
50        ]
51    };
52    ids.iter()
53        .find_map(|id| AsciiName::read(&name, *id))
54        .is_some_and(|family| matches_hint_reliant_name_list(family.bytes()))
55}

57/// A name decoded to ASCII the same way as FreeType.
58///
59/// <https://gitlab.freedesktop.org/freetype/freetype/-/blob/80a507a6b8e3d2906ad2c8ba69329bd2fb2a85ef/src/sfnt/sfobjs.c#L136>
60struct AsciiName<'a> {
61    data: &'a [u8],
62    utf16: bool,
63}
64
65impl<'a> AsciiName<'a> {
66    /// Reads the record that FreeType chooses for `id`.
67    ///
68    /// A Windows record is used if it is English or if there is no Macintosh
69    /// record. Otherwise, a Macintosh English or Roman record is used. A
70    /// Unicode record is used if there is neither.
71    fn read(name: &Name<'a>, id: NameId) -> Option<Self> {
72        let (mut apple_english, mut apple_roman, mut windows, mut unicode) =
73            (None, None, None, None);
74        let mut windows_is_english = false;
75        let records = name.name_record().iter();
76        for record in records.filter(|record| record.name_id() == id && record.length() > 0) {
77            match record.platform_id() {
78                0 | 2 => unicode = Some(record),
79                1 if record.language_id() == 0 => apple_english = Some(record),
80                1 if record.encoding_id() == 0 => apple_roman = Some(record),
81                3 => {
82                    let english = record.language_id() & 0x3ff == 0x009;
83                    if (windows.is_none() || english) && matches!(record.encoding_id(), 0 | 1 | 10)
84                    {
85                        windows = Some(record);
86                        windows_is_english = english;
87                    }
88                }
89                _ => {}
90            }
91        }
92        let apple = apple_english.or(apple_roman);
93        let (record, utf16) = match (windows, apple) {
94            (Some(windows), _) if apple.is_none() || windows_is_english => (windows, true),
95            (_, Some(apple)) => (apple, false),
96            _ => (unicode?, true),
97        };
98        let start = record.string_offset().to_u32() as usize;
99        let data = name
100            .string_data()
101            .as_bytes()
102            .get(start..start + record.length() as usize)?;
103        Some(Self { data, utf16 })
104    }
105
106    /// Returns the name's characters up to the first NUL, with `?` in place
107    /// of each character outside the range 32 to 127.
108    fn bytes(&self) -> impl Iterator<Item = u8> + '_ {
109        let step = if self.utf16 { 2 } else { 1 };
110        self.data
111            .chunks_exact(step)
112            .map(|unit| {
113                unit.iter()
114                    .fold(0u32, |code, byte| code << 8 | *byte as u32)
115            })
116            .take_while(|code| *code != 0)
117            .map(|code| {
118                if (32..=127).contains(&code) {
119                    code as u8
120                } else {
121                    b'?'
122                }
123            })
124    }
125}

127/// Returns `true` if `name` contains a family name that FreeType knows to
128/// rely on hinting.
129///
130/// The random tag of a font embedded in a PDF is skipped. Matching follows
131/// FreeType's `strstr` search.
132///
133/// <https://gitlab.freedesktop.org/freetype/freetype/-/blob/80a507a6b8e3d2906ad2c8ba69329bd2fb2a85ef/src/truetype/ttobjs.c#L153>
134fn matches_hint_reliant_name_list(mut name: impl Iterator<Item = u8>) -> bool {
135    let mut head = [0u8; 8];
136    let mut head_len = 0;
137    for byte in name.by_ref().take(head.len()) {
138        head[head_len] = byte;
139        head_len += 1;
140    }
141    // The tag is six capitals and a plus, in a name of at least eight.
142    let tagged =
143        head_len == head.len() && head[6] == b'+' && head[..6].iter().all(u8::is_ascii_uppercase);
144    let skip = if tagged { 7 } else { 0 };
145    // The trailing bytes, enough to hold the longest name on the list.
146    let mut window = [0u8; MAX_HINT_RELIANT_NAME_LEN];
147    let mut seen = 0;
148    head[skip..head_len]
149        .iter()
150        .copied()
151        .chain(name)
152        .any(|byte| {
153            window.copy_within(1.., 0);
154            window[MAX_HINT_RELIANT_NAME_LEN - 1] = byte;
155            seen += 1;
156            HINT_RELIANT_NAMES
157                .iter()
158                .any(|tricky| tricky.len() <= seen && window.ends_with(tricky.as_bytes()))
159        })
160}

204#[derive(Copy, Clone, PartialEq, Default, Debug)]
205struct TableId {
206    checksum: u32,
207    len: u32,
208}
209
210impl TableId {
211    fn from_data(data: Option<&[u8]>) -> Self {
212        let Some(data) = data else {
213            return Self::default();
214        };
215        Self {
216            // Note: FreeType always just computes the checksum
217            // <https://gitlab.freedesktop.org/freetype/freetype/-/blob/80a507a6b8e3d2906ad2c8ba69329bd2fb2a85ef/src/truetype/ttobjs.c#L281>
218            checksum: crate::tables::compute_checksum(data),
219            len: data.len() as u32,
220        }
221    }
222
223    /// Creates the identity of a control value table.
224    ///
225    /// The checksum sums the entries the same way that
226    /// [`compute_checksum`](crate::tables::compute_checksum) sums bytes. Two
227    /// big-endian entries form a word, and a trailing entry is padded.
228    fn from_cvt(cvt: Option<&[BigEndian<i16>]>) -> Self {
229        let Some(cvt) = cvt else {
230            return Self::default();
231        };
232        let mut checksum = 0u32;
233        for pair in cvt.chunks(2) {
234            let high = pair[0].get() as u16 as u32;
235            let low = pair.get(1).map_or(0, |value| value.get() as u16 as u32);
236            checksum = checksum.wrapping_add((high << 16) | low);
237        }
238        Self {
239            checksum,
240            len: (cvt.len() * 2) as u32,
241        }
242    }
243}

246struct FontId {

466    // The identity of a cvt is summed from its entries, which has to agree
467    // with the byte checksum FreeType takes.
468    #[test]
469    fn cvt_identity_matches_the_byte_checksum() {
470        for data in [font_test_data::TINOS_SUBSET, font_test_data::CVAR] {
471            let font = FontRef::new(data).unwrap();
472            let cvt = font.cvt().unwrap();
473            let bytes = font.data_for_tag(crate::types::Tag::new(b"cvt ")).unwrap();
474            assert_eq!(
475                TableId::from_cvt(Some(cvt)),
476                TableId::from_data(Some(bytes.as_bytes()))
477            );
478        }
479        // An odd entry count pads the last word.
480        let cvt = [0x0102i16, 0x0304, 0x0506].map(BigEndian::from);
481        assert_eq!(
482            TableId::from_cvt(Some(&cvt)),
483            TableId::from_data(Some(&[1, 2, 3, 4, 5, 6]))
484        );
485    }

506    /// `data` with its `glyf` and `loca` tables renamed `GLYF` and `LOCA`.
507    fn uppercase_glyf(data: &[u8]) -> Vec<u8> {
508        let count = u16::from_be_bytes([data[4], data[5]]) as usize;
509        let mut records: Vec<[u8; 16]> = data[12..12 + 16 * count]
510            .chunks_exact(16)
511            .map(|record| record.try_into().unwrap())
512            .collect();
513        for record in &mut records {
514            match &record[..4] {
515                b"glyf" => record[..4].copy_from_slice(b"GLYF"),
516                b"loca" => record[..4].copy_from_slice(b"LOCA"),
517                _ => {}
518            }
519        }
520        // The directory stays sorted by tag.
521        records.sort();
522        let mut font = data.to_vec();
523        font[12..12 + 16 * count].copy_from_slice(&records.concat());
524        font
525    }

535    /// Whether `data`, with a name table holding `records`, is hint reliant.
536    fn with_names(data: &[u8], records: &[(u16, u16, u16, &str)]) -> bool {
537        let strings: Vec<Vec<u8>> = records
538            .iter()
539            .map(|(platform, _, _, value)| match platform {
540                1 => value.bytes().collect(),
541                _ => value.encode_utf16().flat_map(u16::to_be_bytes).collect(),
542            })
543            .collect();
544        let count = records.len() as u16;
545        let mut name = alloc::vec![0, 0];
546        name.extend_from_slice(&count.to_be_bytes());
547        name.extend_from_slice(&(6 + 12 * count).to_be_bytes());
548        let mut offset = 0u16;
549        for ((platform, language, id, _), string) in records.iter().zip(&strings) {
550            let encoding = if *platform == 1 { 0u16 } else { 1 };
551            for field in [
552                *platform,
553                encoding,
554                *language,
555                *id,
556                string.len() as u16,
557                offset,
558            ] {
559                name.extend_from_slice(&field.to_be_bytes());
560            }
561            offset += string.len() as u16;
562        }
563        strings
564            .iter()
565            .for_each(|string| name.extend_from_slice(string));
566        is_hint_reliant(&WithName(FontRef::new(data).unwrap(), &name))
567    }

569    /// A font's tables with a replacement `name` table.
570    struct WithName<'a>(FontRef<'a>, &'a [u8]);
571
572    impl<'a> TableProvider<'a> for WithName<'a> {
573        fn data_for_tag(&self, tag: crate::types::Tag) -> Option<crate::FontData<'a>> {
574            if tag == crate::types::Tag::new(b"name") {
575                return Some(crate::FontData::new(self.1));
576            }
577            self.0.data_for_tag(tag)
578        }
579    }

581    // FreeType's family name: the whole string, its non-ASCII characters
582    // read as `?`, from the record and name id FreeType picks.
583    #[test]
584    fn family_names_are_read_as_freetype_reads_them() {
585        const FAMILY: u16 = 1;
586        const TYPOGRAPHIC: u16 = 16;
587        assert!(!hint_reliant_with_names(&[(3, 0x409, FAMILY, "Arial")]));
588        assert!(hint_reliant_with_names(&[(3, 0x409, FAMILY, "MingLiU")]));
589        // Longer than any name on the list.
590        assert!(hint_reliant_with_names(&[(
591            3,
592            0x409,
593            FAMILY,
594            "ABCDEF+DFKaiShu-SB-Estd-BF"
595        )]));
596        // The list's "HuaTianKaiTi?" ends in a character FreeType could not
597        // read as ASCII.
598        assert!(hint_reliant_with_names(&[(
599            3,
600            0x409,
601            FAMILY,
602            "HuaTianKaiTi体"
603        )]));
604        assert!(!hint_reliant_with_names(&[(
605            3,
606            0x409,
607            FAMILY,
608            "HuaTianKaiTi"
609        )]));
610        // The typographic family wins over the family.
611        assert!(!hint_reliant_with_names(&[
612            (3, 0x409, FAMILY, "MingLiU"),
613            (3, 0x409, TYPOGRAPHIC, "Arial"),
614        ]));
615        // A Macintosh English record wins over a non-English Windows one,
616        assert!(!hint_reliant_with_names(&[
617            (3, 0x404, FAMILY, "MingLiU"),
618            (1, 0, FAMILY, "Arial"),
619        ]));
620        // and loses to an English one.
621        assert!(hint_reliant_with_names(&[
622            (3, 0x409, FAMILY, "MingLiU"),
623            (1, 0, FAMILY, "Arial"),
624        ]));
625    }

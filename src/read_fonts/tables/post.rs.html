5impl<'a> Post<'a> {
6    /// The number of glyph names covered by this table
7    pub fn num_names(&self) -> usize {
8        match self.version() {
9            Version16Dot16::VERSION_1_0 => DEFAULT_GLYPH_NAMES.len(),
10            Version16Dot16::VERSION_2_0 => self.num_glyphs().unwrap_or_default() as usize,
11            _ => 0,
12        }
13    }
14
15    /// Returns the name for the given glyph.
16    ///
17    /// Note that this is a relatively expensive operation, as it may require
18    /// a linear scan through the string data to find the target name. If you
19    /// need to iterate over all glyph names or collect them into a map for
20    /// faster access, use [`Self::glyph_names`] instead.
21    pub fn glyph_name(&self, glyph_id: GlyphId16) -> Option<&'a str> {
22        let glyph_id = glyph_id.to_u16() as usize;
23        match self.version() {
24            Version16Dot16::VERSION_1_0 => DEFAULT_GLYPH_NAMES.get(glyph_id).copied(),
25            Version16Dot16::VERSION_2_0 => {
26                let idx = self.glyph_name_index()?.get(glyph_id)?.get() as usize;
27                if idx < DEFAULT_GLYPH_NAMES.len() {
28                    return DEFAULT_GLYPH_NAMES.get(idx).copied();
29                }
30                let idx = idx - DEFAULT_GLYPH_NAMES.len();
31                self.string_data()?.get(idx)?.ok().map(|s| s.0)
32            }
33            _ => None,
34        }
35    }
36
37    /// Return an iterator over the glyph names in this table.
38    pub fn glyph_names(&self) -> GlyphNames<'a> {
39        let num_names = self.num_names() as u32;
40        let kind = match self.version() {
41            Version16Dot16::VERSION_1_0 => GlyphNameIterKind::V1(self.clone(), 0),
42            Version16Dot16::VERSION_2_0 => GlyphNameIterKind::V2 {
43                post: self.clone(),
44                idx: 0,
45                checkpoint_stride: (num_names as usize).div_ceil(NUM_CHECKPOINTS + 1).max(1),
46                checkpoints: [UNSET_CHECKPOINT; NUM_CHECKPOINTS],
47                last_actual_idx: None,
48                last_offset: 0,
49                strings_end: usize::MAX,
50            },
51            _ => GlyphNameIterKind::None,
52        };
53        GlyphNames { num_names, kind }
54    }
55
56    //FIXME: how do we want to traverse this? I want to stop needing to
57    // add special cases for things...
58}

90impl<'a> FontRead<'a> for PString<'a> {
91    fn read_with_args(data: FontData<'a>, _: ()) -> Result<Self, ReadError> {
92        let len: u8 = data.read_at(0)?;
93        let pstring = data
94            .as_bytes()
95            .get(1..len as usize + 1)
96            .ok_or(ReadError::OutOfBounds)?;
97
98        if pstring.is_ascii() {
99            Ok(PString(std::str::from_utf8(pstring).unwrap()))
100        } else {
101            //FIXME not really sure how we want to handle this?
102            Err(ReadError::MalformedData("Must be valid ascii"))
103        }
104    }

111const NUM_CHECKPOINTS: usize = 16;
112const UNSET_CHECKPOINT: u32 = u32::MAX;
113
114/// Returns the array slot that holds the offset of `index`, if `index` is one
115/// of the checkpoints we keep.
116///
117/// Slot `n` is read back as the start of index `(n + 1) * stride`, so an index
118/// past the last checkpoint has to be dropped rather than clamped into it.
119fn checkpoint_slot(index: usize, stride: usize) -> Option<usize> {
120    (index % stride == 0)
121        .then(|| index / stride)
122        .filter(|slot| (1..=NUM_CHECKPOINTS).contains(slot))
123        .map(|slot| slot - 1)
124}

126/// Iterator over the glyph names in a post table.
127#[derive(Clone)]
128pub struct GlyphNames<'a> {
129    num_names: u32,
130    kind: GlyphNameIterKind<'a>,
131}
132
133#[derive(Clone)]
134enum GlyphNameIterKind<'a> {
135    None,
136    V1(Post<'a>, u32),
137    V2 {
138        post: Post<'a>,
139        idx: u32,
140        // The number of indices between checkpoints
141        checkpoint_stride: usize,
142        // The offset of each checkpoint; checkpoint 0 is implicit
143        checkpoints: [u32; NUM_CHECKPOINTS],
144        // The last actual index and that was scanned, for monotonic fast path
145        last_actual_idx: Option<usize>,
146        // The offset associated with the last scanned index
147        last_offset: usize,
148        // The first string index the data cannot reach. Strings are read in
149        // order, so no later one can be read either.
150        strings_end: usize,
151    },
152}
153
154impl<'a> Iterator for GlyphNames<'a> {
155    type Item = (GlyphId, &'a str);
156
157    /// Returns the next glyph that has a readable name.
158    ///
159    /// Glyphs whose names cannot be read are skipped, so the glyph ids can
160    /// have gaps.
161    fn next(&mut self) -> Option<Self::Item> {
162        loop {
163            let (gid, name) = self.next_entry()?;
164            if let Some(name) = name {
165                return Some((gid.into(), name));
166            }
167        }
168    }
169}

171impl<'a> GlyphNames<'a> {
172    /// Returns the next glyph and its name, or `None` if there are no more
173    /// glyphs.
174    fn next_entry(&mut self) -> Option<(GlyphId16, Option<&'a str>)> {
175        match &mut self.kind {
176            GlyphNameIterKind::None => None,
177            GlyphNameIterKind::V1(post, idx) => {
178                if *idx >= self.num_names {
179                    return None;
180                }
181                let gid = GlyphId16::new(*idx as u16);
182                *idx += 1;
183                Some((gid, post.glyph_name(gid)))
184            }
185            GlyphNameIterKind::V2 {
186                post,
187                idx,
188                checkpoint_stride,
189                checkpoints,
190                last_actual_idx,
191                last_offset,
192                strings_end,
193            } => {
194                if *idx >= self.num_names {
195                    return None;
196                }
197                let stride = *checkpoint_stride;
198                let gid = GlyphId16::new(*idx as u16);
199                let index = *idx as usize;
200                *idx += 1;
201                let mut name = || -> Option<&'a str> {
202                    let mut actual_idx = post.glyph_name_index()?.get(index)?.get() as usize;
203                    Some(if actual_idx < DEFAULT_GLYPH_NAMES.len() {
204                        DEFAULT_GLYPH_NAMES.get(actual_idx).copied()?
205                    } else {
206                        actual_idx -= DEFAULT_GLYPH_NAMES.len();
207                        if actual_idx >= *strings_end {
208                            return None;
209                        }
210                        let string_data =
211                            post.offset_data().slice(post.string_data_byte_range())?;
212                        // Checkpoint 0 is implicit and always at offset 0; the
213                        // array stores logical checkpoints 1..=NUM_CHECKPOINTS.
214                        let target_slot = (actual_idx / stride).min(NUM_CHECKPOINTS);
215                        // Find the the starting location for our scan
216                        let (mut scan_idx, mut offset) = {
217                            // Search backward from the target slot to find the
218                            // nearest checkpoint that has been set
219                            let mut slot = target_slot;
220                            while slot > 0 && checkpoints[slot - 1] == UNSET_CHECKPOINT {
221                                slot -= 1;
222                            }
223                            if slot == 0 {
224                                // Fallback to implicit checkpoint 0
225                                (0, 0)
226                            } else {
227                                // Otherwise, start scanning from the nearest
228                                // checkpoint
229                                (slot * stride, checkpoints[slot - 1] as usize)
230                            }
231                        };
232                        // See if we can use the monotonic fast path.
233                        if let Some(last_idx) = *last_actual_idx {
234                            // Start scanning from the last index if that provides
235                            // a smaller search space than the nearest checkpoint
236                            if last_idx <= actual_idx && last_idx > scan_idx {
237                                scan_idx = last_idx;
238                                offset = *last_offset;
239                            }
240                        }
241                        // Now do the linear scan over the string data
242                        while scan_idx < actual_idx {
243                            let Some(next) = PString::read_len_at(string_data, offset)
244                                .and_then(|len| offset.checked_add(len))
245                            else {
246                                *strings_end = scan_idx;
247                                return None;
248                            };
249                            offset = next;
250                            scan_idx += 1;
251                            // If this index is a checkpoint, record the offset
252                            // for future scans
253                            if let Some(slot) = checkpoint_slot(scan_idx, stride) {
254                                checkpoints[slot] = u32::try_from(offset).ok()?;
255                            }
256                        }
257                        if let Some(slot) = checkpoint_slot(actual_idx, stride) {
258                            checkpoints[slot] = u32::try_from(offset).ok()?;
259                        }
260                        // Record the last index and offset for future scans
261                        *last_actual_idx = Some(actual_idx);
262                        *last_offset = offset;
263                        PString::read(string_data.split_off(offset)?).ok()?.0
264                    })
265                };
266                Some((gid, name()))
267            }
268        }
269    }

464    /// A post table may hold more strings than it has glyphs, which puts a
465    /// name index past the last checkpoint the iterator keeps. Clamping such
466    /// an index into the final slot makes a later scan resume from the wrong
467    /// offset.
468    #[test]
469    fn glyph_names_matches_naive_past_the_last_checkpoint() {
470        // Two glyphs give a stride of one, so only the first NUM_CHECKPOINTS
471        // string indices get a checkpoint of their own. The higher index is
472        // read first, so the clamped slot is written before it is read back.
473        let indices = [258u16 + 50, 258 + 30];
474        let mut buf = BeBuffer::new()
475            .push(Version16Dot16::VERSION_2_0)
476            .push(Fixed::from_i32(0))
477            .extend([FWord::new(0), FWord::new(0)])
478            .push(0u32)
479            .extend([0u32, 0, 0, 0])
480            .push(indices.len() as u16);
481        for index in indices {
482            buf = buf.push(index);
483        }
484        for i in 0..70u8 {
485            buf = buf.push(2u8).extend([b'a' + i / 26, b'a' + i % 26]);
486        }
487        let post = Post::read(buf.data().into()).unwrap();
488        let from_naive: Vec<_> = (0..indices.len() as u16)
489            .map(|gid| post.glyph_name(GlyphId16::new(gid)).unwrap())
490            .collect();
491        assert_eq!(from_naive, ["by", "be"]);
492        let from_iter: Vec<_> = post.glyph_names().map(|(_, name)| name).collect();
493        assert_eq!(from_iter, from_naive);
494    }

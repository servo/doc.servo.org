10/// Converts a measurement in font units into a caller's own units.
11///
12/// Written by whatever owns scaling in a text stack: a shaper, or the bridge
13/// between one and this crate. Callers that want font units read them from
14/// [`GlyphMetrics`], [`Metrics`] and [`StyleMetrics`] and never implement this.
15///
16/// An implementation chooses the number format, the rounding, and the order
17/// of operations for each axis. This crate defines how measurements combine
18/// into a metric.
19///
20/// The arithmetic is part of the trait for the same reason: an
21/// implementation decides what an overflowing sum does and which way a
22/// halved odd number rounds.
23pub trait Scale {
24    /// What a scaled measurement is expressed as.
25    type Value: Copy;
26
27    /// Returns `a` plus `b`.
28    ///
29    /// This and the two below take no scale. They describe
30    /// [`Value`](Self::Value) itself, so every scale of that type answers
31    /// them alike.
32    fn add(a: Self::Value, b: Self::Value) -> Self::Value;
33
34    /// Returns `a` less `b`.
35    fn sub(a: Self::Value, b: Self::Value) -> Self::Value;
36
37    /// Returns half of `value`.
38    fn half(value: Self::Value) -> Self::Value;
39
40    /// Scales a measurement along the x axis.
41    fn scale_x(&self, value: F48Dot16) -> Self::Value;
42
43    /// Scales a measurement along the y axis.
44    ///
45    /// Separate from [`scale_x`](Self::scale_x) because the two axes need
46    /// not agree, and because a caller whose y runs down the page says so
47    /// here.
48    fn scale_y(&self, value: F48Dot16) -> Self::Value;
49
50    /// Scales where a glyph's ink sits.
51    ///
52    /// One call rather than four, so that a scale running the other way up
53    /// settles in one place what that means for a bearing and a size. The
54    /// scale also decides whether to round the box's corners before scaling.
55    fn scale_glyph_extents(&self, extents: GlyphExtents<F48Dot16>) -> GlyphExtents<Self::Value>;
56
57    /// Scales a region given by its corners.
58    ///
59    /// One call for the same reason as [`scale_glyph_extents`](Self::scale_glyph_extents):
60    /// a scale running the other way up decides here which corner ends up
61    /// least.
62    fn scale_rect(&self, bounds: BoundingBox<F48Dot16>) -> BoundingBox<Self::Value>;
63
64    /// Returns the size in pixels per em, if the scale has one.
65    ///
66    /// Embedded bitmaps are measured from the strike selected for this size.
67    /// `None`, the default, selects the largest strike.
68    fn ppem(&self) -> Option<f32> {
69        None
70    }
71}

73/// Scales design-unit metrics to 26.6 pixels with 16.16 fixed-point
74/// arithmetic.
75///
76/// Sums saturate and halves round toward negative infinity. A measurement
77/// with a fraction is rounded to a whole unit before it is scaled;
78/// [`ScaleF32`] keeps the fraction.
79#[derive(Copy, Clone, Debug)]
80pub struct Scale26Dot6 {
81    x: Fixed,
82    y: Fixed,
83    ppem: f32,
84}
85
86impl Scale26Dot6 {
87    /// Creates a scale with a separate size for each axis.
88    pub fn new(x_ppem: f32, y_ppem: f32, units_per_em: u16) -> Self {
89        Self {
90            x: Self::factor(x_ppem, units_per_em),
91            y: Self::factor(y_ppem, units_per_em),
92            ppem: x_ppem.max(y_ppem),
93        }
94    }
95
96    /// Creates a scale that maps `units_per_em` font units to `ppem` pixels.
97    pub fn from_ppem(ppem: f32, units_per_em: u16) -> Self {
98        Self::new(ppem, ppem, units_per_em)
99    }
100
101    /// Returns the factor that converts whole font units to 26.6 pixels.
102    ///
103    /// The factor includes the conversion to 1/64 pixel, so a 16.16 multiply
104    /// of a whole design unit produces 26.6 bits.
105    fn factor(ppem: f32, units_per_em: u16) -> Fixed {
106        Fixed::from_bits((ppem * 64.0) as i32) / Fixed::from_bits(units_per_em.max(1) as i32)
107    }
108
109    fn scale(value: F48Dot16, factor: Fixed) -> F26Dot6 {
110        F26Dot6::from_bits((Fixed::from_bits(value.to_i32()) * factor).to_bits())
111    }
112}
113
114impl Scale for Scale26Dot6 {
115    type Value = F26Dot6;
116
117    fn add(a: F26Dot6, b: F26Dot6) -> F26Dot6 {
118        a.saturating_add(b)
119    }
120
121    fn sub(a: F26Dot6, b: F26Dot6) -> F26Dot6 {
122        a.saturating_sub(b)
123    }
124
125    fn half(value: F26Dot6) -> F26Dot6 {
126        F26Dot6::from_bits(value.to_bits() >> 1)
127    }
128
129    fn scale_x(&self, value: F48Dot16) -> F26Dot6 {
130        Self::scale(value, self.x)
131    }
132
133    fn scale_y(&self, value: F48Dot16) -> F26Dot6 {
134        Self::scale(value, self.y)
135    }
136
137    fn scale_glyph_extents(&self, e: GlyphExtents<F48Dot16>) -> GlyphExtents<F26Dot6> {
138        if e.width <= F48Dot16::ZERO || e.height <= F48Dot16::ZERO {
139            return GlyphExtents::default();
140        }
141        // Match FreeType's FT_RoundFix: ties round away from zero.
142        let round = |value: F48Dot16| {
143            let bits = value.to_bits();
144            F48Dot16::from_bits(bits.wrapping_add(0x8000 - i64::from(bits < 0)) & !0xffff)
145        };
146        let left = round(e.x_bearing);
147        let top = round(e.y_bearing);
148        let right = round(e.x_bearing + e.width);
149        let bottom = round(e.y_bearing - e.height);
150        GlyphExtents {
151            x_bearing: self.scale_x(left),
152            y_bearing: self.scale_y(top),
153            width: self.scale_x(right - left),
154            height: self.scale_y(top - bottom),
155        }
156    }
157
158    fn scale_rect(&self, b: BoundingBox<F48Dot16>) -> BoundingBox<F26Dot6> {
159        BoundingBox {
160            x_min: self.scale_x(b.x_min),
161            y_min: self.scale_y(b.y_min),
162            x_max: self.scale_x(b.x_max),
163            y_max: self.scale_y(b.y_max),
164        }
165    }
166
167    fn ppem(&self) -> Option<f32> {
168        Some(self.ppem)
169    }
170}

574        &self,
575        convert: impl Fn(S::Value) -> V,
576        glyphs: impl Iterator<Item = (GlyphId, &'o mut V)>,
577    ) {
578        self.metrics
579            .h_advance_batched(|value| convert(self.scale.scale_x(value)), glyphs);
580    }

582    /// Returns the advance height of `glyph`.
583    ///
584    /// A font with no vertical metrics stacks its glyphs by the line, the
585    /// only measurement here a caller can supply. One that has them is
586    /// measured by what it says, and a supplied line changes nothing.
587    #[inline]
588    pub fn v_advance(&self, glyph: GlyphId) -> S::Value {
589        // Only a font with no vertical metrics stacks by the line; one
590        // that has them is read below, supplied line or not.
591        match self.line_height_fallback() {
592            Some(height) => height,
593            None => self.scale.scale_y(self.metrics.v_advance_exact(glyph)),
594        }
595    }
596
597    /// Writes the advance height of each glyph to its slot, in order.
598    #[inline]
599    pub fn v_advance_batched<'o, V: 'o>(
600        &self,
601        convert: impl Fn(S::Value) -> V,
602        glyphs: impl Iterator<Item = (GlyphId, &'o mut V)>,
603    ) {
604        // Settled before the run, as the table to read is. Only a font
605        // with no vertical metrics reaches this, and then all its glyphs
606        // advance the same; one that has them is read below, supplied line
607        // or not.
608        if let Some(height) = self.line_height_fallback() {
609            for (_, out) in glyphs {
610                *out = convert(height);
611            }
612            return;
613        }
614        self.metrics
615            .v_advance_batched(|value| convert(self.scale.scale_y(value)), glyphs);
616    }

618    /// Returns the y coordinate of `glyph`'s vertical origin.
619    ///
620    /// This places the glyph relative to the pen in vertical text. The x
621    /// coordinate is half the advance width, so only y is returned.
622    ///
623    /// The origin comes from the first available of: `VORG`, the `glyf` top
624    /// phantom point when the font has vertical metrics, the ink centered in
625    /// the line, and the ascender. The last two use the supplied ink
626    /// measurement.
627    pub fn v_origin_y(&self, glyph: GlyphId) -> S::Value {
628        let mut origin = self.scale.scale_y(F48Dot16::ZERO);
629        self.v_origin_y_batched(|value| value, core::iter::once((glyph, &mut origin)));
630        origin
631    }

633    /// Writes the vertical origin of each glyph to its slot, in order.
634    ///
635    /// The line is measured only if a glyph reaches the extents fallback, and
636    /// then reused for the rest of the run.
637    pub fn v_origin_y_batched<'o, V: 'o>(
638        &self,
639        convert: impl Fn(S::Value) -> V,
640        glyphs: impl Iterator<Item = (GlyphId, &'o mut V)>,
641    ) {
642        let mut line = None;
643        self.metrics.v_origin_y_batched_with(
644            |glyph, origin| {
645                let value = match origin {
646                    Some(origin) => self.scale.scale_y(origin),
647                    None => self
648                        .origin_against_ink(glyph, *line.get_or_insert_with(|| self.origin_line())),
649                };
650                convert(value)
651            },
652            |glyphs| {
653                let line = self.origin_line();
654                self.extents_batched(
655                    |extents| convert(self.origin_from_extents(extents, line)),
656                    glyphs,
657                );
658            },
659            glyphs,
660        );
661    }

663    /// Returns where `glyph`'s ink sits.
664    ///
665    /// An empty glyph has zero extents. `None` means the glyph is unavailable
666    /// or its extents could not be read. A supplied extents function may use
667    /// `None` for any glyph it does not measure.
668    #[inline]
669    pub fn extents(&self, glyph: GlyphId) -> Option<GlyphExtents<S::Value>> {
670        match self.extents {
671            Some(extents) => extents(glyph),
672            None => {
673                let mut extents = None;
674                self.metrics.extents_batched(
675                    |value| value.map(|extents| self.scale.scale_glyph_extents(extents)),
676                    core::iter::once((glyph, &mut extents)),
677                );
678                extents
679            }
680        }
681    }
682
683    /// Writes where each glyph's ink sits to its slot, in order.
684    #[inline]
685    pub fn extents_batched<'o, V: 'o>(
686        &self,
687        convert: impl Fn(Option<GlyphExtents<S::Value>>) -> V,
688        glyphs: impl Iterator<Item = (GlyphId, &'o mut V)>,
689    ) {
690        // Settled before the run, as the source to read is: a supplied
691        // function answers for every glyph, and the font is never read.
692        if let Some(extents) = self.extents {
693            for (glyph, out) in glyphs {
694                *out = convert(extents(glyph));
695            }
696            return;
697        }
698        self.metrics.extents_batched(
699            |extents| convert(extents.map(|extents| self.scale.scale_glyph_extents(extents))),
700            glyphs,
701        );
702    }

704    /// The height a glyph takes from the line, where it takes one.
705    ///
706    /// `None` for two reasons that mean the same thing here: the font has
707    /// vertical metrics of its own, so the line is not what its glyphs stack
708    /// by; or it has no line either, which the unscaled metrics answer with
709    /// an em. A supplied line applies only in the first case, so supplying
710    /// one never overrides what a font says about itself.
711    ///
712    /// Each end is scaled before the two are subtracted, so a scale that
713    /// rounds rounds them as a caller supplying its own line would, and the
714    /// answer does not turn on whether one was supplied.
715    #[inline]
716    fn line_height_fallback(&self) -> Option<S::Value> {
717        if self.metrics.states_vertical_advances() {
718            return None;
719        }
720        let line = self.origin_line();
721        Some(S::sub(line.ascender, line.descender))
722    }
723
724    /// The line a glyph stacks by, in the units this type reports.
725    ///
726    /// The supplied line if there is one, and the font's otherwise. A font
727    /// with no line metrics gets one proportioned to its em.
728    fn origin_line(&self) -> LineExtents<S::Value> {
729        // A supplied line is reported as it was given: a caller stating its
730        // own has already decided which way its ends run.
731        if let Some(line) = self.line {
732            return line;
733        }
734        let line = self.metrics.origin_line();
735        // Each end is scaled, rather than the height between them, so a scale
736        // that rounds reports ends a caller can lay out against.
737        LineExtents {
738            ascender: self.scale.scale_y(line.ascender),
739            descender: self.scale.scale_y(line.descender),
740        }
741    }
742
743    /// The last two rungs of [`v_origin_y`](Self::v_origin_y), which measure
744    /// against the ink.
745    fn origin_against_ink(&self, glyph: GlyphId, line: LineExtents<S::Value>) -> S::Value {
746        self.origin_from_extents(self.extents(glyph), line)
747    }
748
749    fn origin_from_extents(
750        &self,
751        extents: Option<GlyphExtents<S::Value>>,
752        line: LineExtents<S::Value>,
753    ) -> S::Value {
754        let Some(extents) = extents else {
755            return line.ascender;
756        };
757        let line_height = S::sub(line.ascender, line.descender);
758        S::add(
759            extents.y_bearing,
760            S::half(S::sub(line_height, extents.height)),
761        )
762    }

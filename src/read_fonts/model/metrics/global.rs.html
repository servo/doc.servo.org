13/// The two ends of a line, in whatever unit a caller uses.
14///
15/// Separate from [`LineBox`] because the gap after a line is not part of
16/// where it begins and ends, and the metrics built on a line use only the
17/// two ends.
18#[derive(Copy, Clone, Default, PartialEq, Eq, Debug)]
19pub struct LineExtents<T> {
20    /// Height of the line above the baseline. Never negative.
21    pub ascender: T,
22    /// Depth of the line below the baseline, as a negative value. Never
23    /// positive.
24    pub descender: T,
25}
26
27/// Ascender, descender and line gap from one table.
28///
29/// In font units unless a caller names another unit. The ascender is above
30/// the baseline and the descender below it, whatever sign a font stores
31/// them with.
32#[derive(Copy, Clone, Default, PartialEq, Eq, Debug)]
33pub struct LineBox<T = F48Dot16> {
34    /// Height of the line above the baseline. Never negative.
35    pub ascender: T,
36    /// Depth of the line below the baseline, as a negative value. Never
37    /// positive.
38    pub descender: T,
39    /// Extra space between the bottom of one line and the top of the next.
40    pub line_gap: T,
41}
42
43impl<T: Copy> LineBox<T> {
44    /// Returns where this line begins and ends, without the gap after it.
45    ///
46    /// Text stacks by the two ends; the gap separates one run of stacked
47    /// text from the next.
48    pub fn extents(&self) -> LineExtents<T> {
49        LineExtents {
50            ascender: self.ascender,
51            descender: self.descender,
52        }
53    }
54}
55
56impl LineBox {
57    /// Returns `true` if either end of the line is away from the baseline.
58    ///
59    /// Some fonts leave both ends at zero, and the line is then not used.
60    pub(crate) fn has_nonzero_extent(&self) -> bool {
61        self.ascender != F48Dot16::ZERO || self.descender != F48Dot16::ZERO
62    }
63}

65/// Font-wide metrics.
66///
67/// Read from `head`, `maxp`, `hhea`, `vhea` and `OS/2`, which together are a
68/// few hundred bytes.
69///
70/// A font has three sets of line metrics, and they often differ.
71/// [`hhea_line`](Self::hhea_line) is what macOS reads,
72/// [`win_line`](Self::win_line) is what Windows reads, and [`typo_line`](Self::typo_line) is what the
73/// font asks for when [`use_typo_metrics`](Self::use_typo_metrics) is set.
74/// Choosing between them is left to the caller.
75///
76/// A field missing from the font is `None`, which is distinct from a zero
77/// value.
78///
79/// `MVAR` varies most of these, so a set belongs to the location it was read
80/// at. The units per em, the counts, the bounding box, the widest and tallest
81/// advance and the average width cannot vary.
82///
83/// # Units
84///
85/// Measurements are in font units. They are [`F48Dot16`] because a
86/// location can move them off a whole unit, and rounding here would decide
87/// for every caller how that fraction is spent.
88///
89/// The two axes do not scale alike: [`vhea_line`](Self::vhea_line) is
90/// measured across the vertical baseline rather than along it, so it scales
91/// on x where the horizontal sets scale on y.
92#[derive(Copy, Clone, Default, PartialEq, Eq, Debug)]
93pub struct Metrics {
94    /// Design units per em, which every measurement of the font uses.
95    ///
96    /// Always between 16 and 16384. A font with a value outside that range,
97    /// or with no `head`, uses 1000.
98    pub units_per_em: u16,
99    /// Design units per em as written in `head`, or zero if the font has no
100    /// `head`.
101    ///
102    /// Differs from [`units_per_em`](Self::units_per_em) only for a font with
103    /// an unusable value, which a caller may choose to reject.
104    pub raw_units_per_em: u16,
105    /// Number of glyphs in the font.
106    pub num_glyphs: u32,
107    /// Box enclosing every glyph in the font.
108    pub bounds: BoundingBox<F48Dot16>,
109    /// Line metrics from `hhea`.
110    pub hhea_line: Option<LineBox>,
111    /// Widest advance of any glyph.
112    pub max_h_advance: Option<F48Dot16>,
113    /// Line metrics from `vhea`, measured across the vertical baseline.
114    pub vhea_line: Option<LineBox>,
115    /// Tallest advance of any glyph.
116    pub max_v_advance: Option<F48Dot16>,
117    /// Typographic line metrics, from `OS/2`.
118    pub typo_line: Option<LineBox>,
119    /// Whether the font asks for [`typo_line`](Self::typo_line) to be used
120    /// in preference to [`hhea_line`](Self::hhea_line).
121    pub use_typo_metrics: bool,
122    /// The line outside which the font asks not to be clipped.
123    ///
124    /// `OS/2` gives the descent as a positive number below the baseline.
125    /// This negates it, so the pair describes a line like every other pair
126    /// here.
127    pub win_line: Option<LineExtents<F48Dot16>>,
128    /// Height of a lowercase x.
129    pub x_height: Option<F48Dot16>,
130    /// Height of a capital letter.
131    pub cap_height: Option<F48Dot16>,
132    /// Average glyph width, as the font reports it.
133    pub average_char_width: Option<F48Dot16>,
134}

136impl Metrics {
137    /// Reads the metrics from `tables` at `normalized_coords`.
138    ///
139    /// A missing or unreadable table leaves the metrics that come from it
140    /// unset rather than failing.
141    ///
142    /// Pass an empty location for the font's defaults. At any other location
143    /// `MVAR` is read and its deltas applied.
144    pub fn from_sfnt<'a>(
145        tables: &impl TableProvider<'a>,
146        normalized_coords: &[NormalizedCoord],
147    ) -> Self {
148        let coords = normalized_coords;
149        // Only at a location, so that the common case does not reach for a
150        // table it has nothing to ask.
151        let mvar = (!coords.is_empty()).then(|| tables.mvar().ok()).flatten();
152        let deltas = mvar.as_ref().and_then(|mvar| mvar.at(coords));
153        let deltas = deltas.as_ref();
154        let mut metrics = Self::default();
155        let head = tables.head();
156        metrics.raw_units_per_em = head.as_ref().map_or(0, |head| head.units_per_em());
157        metrics.units_per_em = units_per_em(metrics.raw_units_per_em);
158        if let Ok(head) = head {
159            metrics.bounds = BoundingBox {
160                x_min: F48Dot16::from_i32(head.x_min() as i32),
161                y_min: F48Dot16::from_i32(head.y_min() as i32),
162                x_max: F48Dot16::from_i32(head.x_max() as i32),
163                y_max: F48Dot16::from_i32(head.y_max() as i32),
164            };
165        }
166        if let Ok(maxp) = tables.selected_maxp() {
167            metrics.num_glyphs = maxp.num_glyphs();
168        }
169        if let Ok(hhea) = tables.selected_hhea() {
170            // `MVAR` names one horizontal ascender, descender and line gap
171            // between them, so the same delta applies to whichever of these
172            // two sets a caller goes on to believe.
173            metrics.hhea_line = Some(line_box(
174                metric(hhea.ascender().to_i16() as i32, deltas, tags::HASC),
175                metric(hhea.descender().to_i16() as i32, deltas, tags::HDSC),
176                metric(hhea.line_gap().to_i16() as i32, deltas, tags::HLGP),
177            ));
178            metrics.max_h_advance =
179                Some(F48Dot16::from_i32(hhea.advance_width_max().to_u16() as i32));
180        }
181        if let Ok(vhea) = tables.selected_vhea() {
182            metrics.vhea_line = Some(line_box(
183                metric(vhea.ascender().to_i16() as i32, deltas, tags::VASC),
184                metric(vhea.descender().to_i16() as i32, deltas, tags::VDSC),
185                metric(vhea.line_gap().to_i16() as i32, deltas, tags::VLGP),
186            ));
187            metrics.max_v_advance =
188                Some(F48Dot16::from_i32(vhea.advance_height_max().to_u16() as i32));
189        }
190        if let Ok(os2) = tables.os2() {
191            metrics.typo_line = Some(line_box(
192                metric(os2.s_typo_ascender() as i32, deltas, tags::HASC),
193                metric(os2.s_typo_descender() as i32, deltas, tags::HDSC),
194                metric(os2.s_typo_line_gap() as i32, deltas, tags::HLGP),
195            ));
196            metrics.win_line = Some(LineExtents {
197                ascender: metric(os2.us_win_ascent() as i32, deltas, tags::HCLA),
198                descender: -metric(os2.us_win_descent() as i32, deltas, tags::HCLD),
199            });
200            metrics.use_typo_metrics = os2
201                .fs_selection()
202                .contains(SelectionFlags::USE_TYPO_METRICS);
203            metrics.x_height = os2
204                .sx_height()
205                .map(|value| metric(value as i32, deltas, tags::XHGT));
206            metrics.cap_height = os2
207                .s_cap_height()
208                .map(|value| metric(value as i32, deltas, tags::CPHT));
209            metrics.average_char_width = Some(F48Dot16::from_i32(os2.x_avg_char_width() as i32));
210        }
211        metrics
212    }
213
214    /// Returns the ascender, descender and gap for horizontal text.
215    ///
216    /// The line is the first of these that applies:
217    ///
218    /// 1. The typographic line, if `OS/2` sets `USE_TYPO_METRICS`.
219    /// 2. The `hhea` line, if either end is nonzero.
220    /// 3. The typographic line, if either end is nonzero.
221    /// 4. The `OS/2` clipping line, with no gap.
222    ///
223    /// Returns `None` if the font has no line metrics.
224    pub fn h_line(&self) -> Option<LineBox> {
225        match (self.hhea_line, self.typo_line) {
226            (_, Some(typo)) if self.use_typo_metrics => Some(typo),
227            (Some(hhea), _) if hhea.has_nonzero_extent() => Some(hhea),
228            (_, Some(typo)) if typo.has_nonzero_extent() => Some(typo),
229            // The clipping line has no gap between one line and the next, so
230            // there is none to report.
231            (hhea, _) => self
232                .win_line
233                .map(|win| LineBox {
234                    ascender: win.ascender,
235                    descender: win.descender,
236                    line_gap: F48Dot16::ZERO,
237                })
238                .or(hhea),
239        }
240    }

242    /// Reads the metrics from a Type 1 font.
243    ///
244    /// Such a font has only a bounding box, a glyph count and a matrix. The
245    /// ascender and descender are the top and bottom of that box. Everything
246    /// else is unset.
247    pub fn from_type1(font: &Type1Font) -> Self {
248        let bbox = font.bbox();
249        let bounds = BoundingBox {
250            x_min: bbox.x_min.to_f48dot16(),
251            y_min: bbox.y_min.to_f48dot16(),
252            x_max: bbox.x_max.to_f48dot16(),
253            y_max: bbox.y_max.to_f48dot16(),
254        };
255        let raw_units_per_em = font.upem().clamp(0, u16::MAX as i32) as u16;
256        Self {
257            units_per_em: units_per_em(raw_units_per_em),
258            raw_units_per_em,
259            num_glyphs: font.num_glyphs(),
260            bounds,
261            hhea_line: Some(LineBox {
262                ascender: bounds.y_max,
263                descender: bounds.y_min,
264                line_gap: F48Dot16::ZERO,
265            }),
266            ..Default::default()
267        }
268    }
269
270    /// Reads the metrics from a standalone CFF font.
271    pub fn from_cff(font: &CffFontRef<'_>) -> Self {
272        let raw_units_per_em = font.upem().clamp(0, u16::MAX as i32) as u16;
273        let mut metrics = Self {
274            units_per_em: units_per_em(raw_units_per_em),
275            raw_units_per_em,
276            num_glyphs: font.num_glyphs(),
277            ..Default::default()
278        };
279        if let Some(metadata) = font.metadata() {
280            let bbox = metadata.bbox();
281            metrics.bounds = BoundingBox {
282                x_min: bbox.x_min.to_f48dot16(),
283                y_min: bbox.y_min.to_f48dot16(),
284                x_max: bbox.x_max.to_f48dot16(),
285                y_max: bbox.y_max.to_f48dot16(),
286            };
287            metrics.hhea_line = Some(LineBox {
288                ascender: metrics.bounds.y_max,
289                descender: metrics.bounds.y_min,
290                line_gap: F48Dot16::ZERO,
291            });
292        }
293        metrics
294    }

297/// Returns the design units per em to use for a font whose `head` holds
298/// `stated`.
299///
300/// This is `stated` if it is between 16 and 16384, and 1000 otherwise, which
301/// matches HarfBuzz.
302/// Returns a line with its ascender above the baseline and its descender
303/// below it.
304///
305/// The `hhea`, `vhea` and `OS/2` descenders are negative by the
306/// specification, but some fonts store them positive, which would put the
307/// bottom of the line above the baseline. Either end is read by its size
308/// alone. The gap is kept as stored.
309fn line_box(ascender: F48Dot16, descender: F48Dot16, line_gap: F48Dot16) -> LineBox {
310    LineBox {
311        ascender: ascender.abs(),
312        descender: -descender.abs(),
313        line_gap,
314    }
315}
316
317pub(crate) fn units_per_em(stated: u16) -> u16 {
318    if (16..=16384).contains(&stated) {
319        stated
320    } else {
321        1000
322    }

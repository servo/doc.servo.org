1//! Font model interop for Skrifa and HarfRust.
2//!
3//! This should not be used directly.
4
5use crate::model::font::{Font, Tables};
6use alloc::{boxed::Box, sync::Arc};
7use core::any::Any;
8
9/// Internal interop point for HarfRust.
10///
11/// Calling this directly will make your `Font` **UNUSABLE** with
12/// HarfRust.
13pub fn _get_or_init_shaping_data(
14    font: &Font,
15    f: impl FnOnce() -> Box<dyn Any + Send + Sync>,
16) -> &dyn Any {
17    &**font.shared().0.shaping_data.get_or_init(|| f())
18}
19
20/// Internal interop point for Skrifa.
21///
22/// Returns data that Skrifa derives from the font at its location. Every
23/// clone of an instance shares the data. The slot for the default location
24/// belongs to the font, so all default instances share it. A font without
25/// variations is always at its default location, so it has one slot. Data
26/// that is the same at every location belongs in the default location's
27/// slot, which any instance can reach through [`Font::default_instance`].
28///
29/// Calling this directly will make your `Font` **UNUSABLE** with Skrifa.
30pub fn _get_or_init_scaling_data(
31    font: &Font,
32    f: impl FnOnce() -> Box<dyn Any + Send + Sync>,
33) -> &dyn Any {
34    &**font.scaling_data().get_or_init(|| f())
35}
36
37/// Internal interop point for HarfRust.
38///
39/// Returns the shared tables of an SFNT font, for use as the owner of
40/// cached data that borrows from them. Returns `None` for other fonts.
41pub fn _table_cache_host(font: &Font) -> Option<Arc<Tables>> {
42    font.shared().tables_arc().cloned()
43}

45#[cfg(test)]
46mod tests {
47    use super::*;
48
49    #[test]
50    fn recover_shaping_data() {
51        struct ShapingData {
52            arg: i32,
53        }
54        let font = Font::new(font_test_data::AHEM, 0).unwrap();
55        let data = _get_or_init_shaping_data(&font, || Box::new(ShapingData { arg: -42 }));
56        let actual = data.downcast_ref::<ShapingData>().unwrap();
57        assert_eq!(actual.arg, -42);
58    }
59
60    #[test]
61    fn scaling_data_is_per_location() {
62        struct Data(i32);
63        let read = |data: &dyn Any| data.downcast_ref::<Data>().unwrap().0;
64        let font = Font::new(font_test_data::VAZIRMATN_VAR, 0).unwrap();
65        let bold = font
66            .instance_builder()
67            .variations([("wght", 700.0)])
68            .build();
69        let _ = _get_or_init_scaling_data(&font, || Box::new(Data(1)));
70        // A varied instance has its own, which its clones share.
71        assert_eq!(
72            read(_get_or_init_scaling_data(&bold, || Box::new(Data(2)))),
73            2
74        );
75        assert_eq!(
76            read(_get_or_init_scaling_data(&bold.clone(), || Box::new(Data(
77                3
78            )))),
79            2
80        );
81        // The default location's is the font's, reached from any instance.
82        assert_eq!(
83            read(_get_or_init_scaling_data(&bold.default_instance(), || {
84                Box::new(Data(4))
85            })),
86            1
87        );
88        // A font without variations is always at its default location.
89        let plain = Font::new(font_test_data::AHEM, 0).unwrap();
90        let _ = _get_or_init_scaling_data(&plain, || Box::new(Data(5)));
91        let asked = plain
92            .instance_builder()
93            .variations([("wght", 700.0)])
94            .build();
95        assert_eq!(
96            read(_get_or_init_scaling_data(&asked, || Box::new(Data(6)))),
97            5
98        );
99    }
100
101    #[test]
102    fn table_cache_host_shares_tables() {
103        let font = Font::new(font_test_data::AHEM, 0).unwrap();
104        let host = _table_cache_host(&font).unwrap();
105        assert!(core::ptr::eq(&*host, font.tables()));
106    }

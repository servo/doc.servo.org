182    /// Returns `true` if the font needs its TrueType instructions to draw
183    /// correct contours.
184    ///
185    /// Such fonts are often called "tricky". They are recognized by family name
186    /// or by the checksums of the `cvt`, `fpgm` and `prep` tables, and only
187    /// fonts with `glyf` outlines qualify. The result is computed once and
188    /// cached. See [`is_hint_reliant`](crate::tables::glyf::is_hint_reliant).
189    pub fn is_hint_reliant(&self) -> bool {
190        use core::sync::atomic::Ordering;
191        const UNKNOWN: u8 = 0;
192        const KNOWN: u8 = 1;
193        // Only an SFNT font can have TrueType outlines.
194        let Some(sfnt) = self.shared().sfnt() else {
195            return false;
196        };
197        let cell = &sfnt.hint_reliant;
198        match cell.load(Ordering::Relaxed) {
199            UNKNOWN => {
200                let answer = crate::tables::glyf::is_hint_reliant(&self.tables());
201                cell.store(KNOWN + answer as u8, Ordering::Relaxed);
202                answer
203            }
204            state => state > KNOWN,
205        }
206    }

581    /// Returns subfont `index` of the charstrings for `source`, parsed at
582    /// this instance's location.
583    ///
584    /// The first [`CACHED_SUBFONTS`] subfonts are parsed once and kept,
585    /// since nearly every glyph in a font shares one of them. A standalone
586    /// CFF font parses them when the font is read. An SFNT font parses each
587    /// once per location, when a glyph first needs it. A subfont past them
588    /// is parsed each time it is asked for, so a font with thousands of
589    /// subfonts costs no more memory than one with eight.
590    pub(crate) fn cff_subfont(&self, source: OutlineSource, index: u16) -> Option<Subfont> {
591        let cff = self.charstrings(source)?;
592        if let Some(subfonts) = self.shared().standalone_cff_subfonts() {
593            return match subfonts.get(usize::from(index)) {
594                Some(subfont) => *subfont,
595                None => cff.subfont(index, &[]).ok(),
596            };
597        }
598        let location = self.sfnt_location()?;
599        let cache = match source {
600            OutlineSource::Cff2 => &location.cff2_subfonts,
601            _ => &location.cff_subfonts,
602        };
603        let coords = self.normalized_coords();
604        let subfonts = cache.get_or_init(|| {
605            let count = (cff.num_subfonts() as usize).min(CACHED_SUBFONTS);
606            (0..count).map(|_| Once::new()).collect()
607        });
608        match subfonts.get(usize::from(index)) {
609            Some(subfont) => *subfont.get_or_init(|| cff.subfont(index, coords).ok()),
610            None => cff.subfont(index, coords).ok(),
611        }
612    }

614    /// Returns `true` if the font has glyphs from `source`, one of the
615    /// bitmap tables, `COLR` or `VARC`.
616    ///
617    /// Each source is probed the first time it is asked about and the answer
618    /// kept, so asking loads no table that measuring with it would not.
619    pub(crate) fn has_layered_source(&self, source: GlyphSource) -> bool {
620        use core::sync::atomic::Ordering;
621        let Some(sfnt) = self.shared().sfnt() else {
622            return false;
623        };
624        let shift = 2 * (source as u32 - 1);
625        match (sfnt.glyph_sources.load(Ordering::Relaxed) >> shift) & 3 {
626            SOURCE_ABSENT => return false,
627            SOURCE_PRESENT => return true,
628            _ => {}
629        }
630        let present = match source {
631            GlyphSource::Sbix | GlyphSource::Cbdt | GlyphSource::Ebdt => !self
632                .bitmap_strikes_with(GlyphSourceOrder::new(&[source]))
633                .is_empty(),
634            GlyphSource::Colr => !self.color_glyphs().is_empty(),
635            GlyphSource::Varc => self.varc_outlines().is_some(),
636            _ => return false,
637        };
638        let state = if present {
639            SOURCE_PRESENT
640        } else {
641            SOURCE_ABSENT
642        };
643        sfnt.glyph_sources
644            .fetch_or(state << shift, Ordering::Relaxed);
645        present
646    }

648    /// Returns `true` if the font has glyphs from any bitmap table, `COLR`
649    /// or `VARC` that `order` lists.
650    ///
651    /// Reads the cached results once and probes only the sources not yet
652    /// checked.
653    pub(crate) fn has_layered_source_in(&self, order: GlyphSourceOrder) -> bool {
654        let Some(sfnt) = self.shared().sfnt() else {
655            return false;
656        };
657        let known = sfnt
658            .glyph_sources
659            .load(core::sync::atomic::Ordering::Relaxed);
660        order.iter().any(|source| {
661            if !matches!(
662                source,
663                GlyphSource::Sbix
664                    | GlyphSource::Cbdt
665                    | GlyphSource::Ebdt
666                    | GlyphSource::Colr
667                    | GlyphSource::Varc
668            ) {
669                return false;
670            }
671            match (known >> (2 * (source as u32 - 1))) & 3 {
672                SOURCE_ABSENT => false,
673                SOURCE_PRESENT => true,
674                _ => self.has_layered_source(source),
675            }
676        })
677    }

679    /// Returns the probed state of the bitmap, color and composite sources.
680    #[cfg(test)]
681    pub(crate) fn probed_glyph_sources(&self) -> u32 {
682        self.shared().sfnt().map_or(0, |sfnt| {
683            sfnt.glyph_sources
684                .load(core::sync::atomic::Ordering::Relaxed)
685        })
686    }

815/// Builder for configuring a font instance.
816pub struct InstanceBuilder {
817    font: SharedFont,
818    coords: CoordStorage,
819    /// The blend coordinates of a Type 1 multiple master location, which
820    /// `coords` holds rounded to 2.14. The weights are computed from these.
821    type1_blend: SmallVec<[Fixed; 4]>,

824impl InstanceBuilder {
825    /// Sets the variations for the font instance from an unordered sequence of
826    /// variations in user space.
827    ///
828    /// Omitted axes will be set to their default values. Unsupported axes are
829    /// ignored. If an axis is specified multiple times, the last value is used.
830    ///
831    /// This will overwrite any previous variation settings.
832    pub fn variations<V>(mut self, variations: V) -> Self
833    where
834        V: IntoIterator,
835        V::Item: Into<Variation>,
836    {
837        self.set_variations(variations);
838        self
839    }
840
841    /// Sets the variations for the font instance from an ordered sequence
842    /// of normalized coordinates.
843    ///
844    /// If the number of provided coordinates is less than the number of axes,
845    /// the remaining axes will be set to their default values. If the number
846    /// of provided coordinates is greater than the number of axes, the extra
847    /// coordinates will be ignored.
848    ///
849    /// This will overwrite any previous variation settings.
850    pub fn normalized_coords(mut self, coords: impl IntoIterator<Item = NormalizedCoord>) -> Self {
851        self.set_coords(coords);
852        self
853    }
854
855    /// Sets the variations for the font instance from a named instance.
856    ///
857    /// If the given named instance index is invalid, then variation settings
858    /// will be reset to default.
859    ///
860    /// This will overwrite any previous variation settings.
861    pub fn named_instance(mut self, index: u32) -> Self {
862        self.set_named_instance(index);
863        self
864    }
865
866    /// Sets the variations for the font instance from a named instance, with
867    /// additional overrides.
868    ///
869    /// If the given named instance index is invalid, then it is ignored and
870    /// only overrides are applied.
871    ///
872    /// This will overwrite any previous variation settings.
873    pub fn named_instance_with_overrides<V>(mut self, index: u32, overrides: V) -> Self
874    where
875        V: IntoIterator,
876        V::Item: Into<Variation>,
877    {

879        self
880    }

882    /// Builds the font instance.
883    pub fn build(self) -> Font {
884        // An all-default location is just the font, so nothing needs owning.
885        // This is what keeps `VariedInstance::coords` non-empty.
886        if self.coords.as_slice().is_empty() {
887            return Font(Repr::Default(self.font));
888        }
889        let kind = match &self.font.0.kind {
890            KindRepr::Sfnt(_) => VariedKind::Sfnt(VariedSfnt {
891                location: SfntLocation::new(),
892                gvar_scalars: Once::new(),
893                hvar_scalars: Once::new(),
894                colr_scalars: Once::new(),
895                vvar_scalars: Once::new(),
896            }),
897            KindRepr::Type1(type1) => {
898                let Some(mm) = type1.font.multiple_master() else {
899                    return Font(Repr::Default(self.font));
900                };
901                let mut weights = SmallVec::from_elem(Fixed::ZERO, mm.num_masters());
902                mm.blend_to_weights(&self.type1_blend, &mut weights);
903                VariedKind::Type1(VariedType1 { weights })
904            }
905            // A standalone CFF font has no axes, so no coordinates reach
906            // here; it is at its default location.
907            KindRepr::StandaloneCff(_) => return Font(Repr::Default(self.font)),
908        };
909        Font(Repr::Varied(Arc::new(VariedInstance {
910            font: self.font,
911            coords: self.coords,
912            location: LocationCaches::new(),
913            kind,
914        })))
915    }

918impl InstanceBuilder {
919    fn set_variations<V>(&mut self, variations: V)
920    where
921        V: IntoIterator,
922        V::Item: Into<Variation>,
923    {
924        if let Some(mm) = self.font.multiple_master() {
925            // Design coordinates, starting from each axis's default.
926            let axes = self.font.type1_axes();
927            let mut design: SmallVec<[Fixed; 4]> =
928                axes.iter().map(|axis| axis.default_value()).collect();
929            let mut named: SmallVec<[bool; 4]> = SmallVec::from_elem(false, design.len());
930            for variation in variations {
931                let variation = variation.into();
932                for ((axis, value), named) in axes.iter().zip(design.iter_mut()).zip(&mut named) {
933                    if axis.axis_tag() == variation.tag {
934                        *value = Fixed::from_f64(variation.value as f64);
935                        *named = true;
936                    }
937                }
938            }
939            let mut blend = SmallVec::from_elem(Fixed::ZERO, design.len());
940            mm.design_to_blend(&design, &mut blend);
941            // An axis no variation names keeps the default blend. Its
942            // default design coordinate doesn't always map back to it.
943            let mut default = SmallVec::<[Fixed; 4]>::from_elem(Fixed::ZERO, design.len());
944            mm.default_blend(&mut default);
945            for ((value, default), named) in blend.iter_mut().zip(default).zip(named) {
946                if !named {
947                    *value = default;
948                }
949            }
950            self.set_type1_blend(blend);
951            return;
952        }
953        let tables = self.font.tables();
954        if let Ok(fvar) = tables.fvar() {
955            set_variations(&fvar, tables.avar().ok(), &mut self.coords, variations);
956        } else {
957            self.coords.clear();
958        }
959    }
960
961    fn set_coords(&mut self, coords: impl IntoIterator<Item = NormalizedCoord>) {
962        if let Some(mm) = self.font.multiple_master() {
963            // Blend coordinates, with the default blend for any not given.
964            let mut blend: SmallVec<[Fixed; 4]> = SmallVec::from_elem(Fixed::ZERO, mm.axes().len());
965            mm.default_blend(&mut blend);
966            for (dst, src) in blend.iter_mut().zip(coords) {
967                *dst = src.to_fixed();
968            }
969            self.set_type1_blend(blend);
970            return;
971        }
972        if let Ok(fvar) = self.font.tables().fvar() {
973            let count = fvar.axis_count() as usize;
974            self.coords.resize(count, NormalizedCoord::ZERO);
975            for (dst, src) in self.coords.as_mut_slice().iter_mut().zip(
976                coords
977                    .into_iter()
978                    .chain(core::iter::repeat(NormalizedCoord::ZERO)),
979            ) {
980                *dst = src;
981            }
982            clear_if_all_zeroes(&mut self.coords);
983        } else {
984            self.coords.clear();
985        }
986    }
987
988    /// Sets the location of a Type 1 multiple master font.
989    ///
990    /// Every blend is kept, even all zeros, which is a master rather than
991    /// the font's default.
992    ///
993    /// The weights are computed from `blend` itself. The coordinates are
994    /// those the weights give back, which rounding in the weights can move
995    /// from `blend` by the smallest step, as FreeType reports them.
996    fn set_type1_blend(&mut self, blend: SmallVec<[Fixed; 4]>) {
997        let font = self.font.clone();
998        let Some(mm) = font.multiple_master() else {
999            return;
1000        };
1001        let mut weights: SmallVec<[Fixed; 16]> = SmallVec::from_elem(Fixed::ZERO, mm.num_masters());
1002        mm.blend_to_weights(&blend, &mut weights);
1003        let mut reported: SmallVec<[Fixed; 4]> = SmallVec::from_elem(Fixed::ZERO, blend.len());
1004        mm.weights_to_blend(&weights, &mut reported);
1005        self.coords = reported.iter().map(|value| value.to_f2dot14()).collect();
1006        self.type1_blend = blend;
1007    }
1008
1009    fn set_named_instance(&mut self, index: u32) {
1010        let tables = self.font.tables();
1011        if let Ok(fvar) = tables.fvar() {
1012            set_variations(
1013                &fvar,
1014                tables.avar().ok(),
1015                &mut self.coords,
1016                named_instance_variations(&fvar, index),
1017            );
1018        } else {
1019            self.coords.clear();
1020        }
1021    }
1022
1023    fn set_named_instance_with_overrides<V>(&mut self, index: u32, overrides: V)
1024    where
1025        V: IntoIterator,
1026        V::Item: Into<Variation>,
1027    {
1028        // A Type 1 multiple master font has no named instances, so only the
1029        // overrides apply.
1030        if self.font.multiple_master().is_some() {
1031            self.set_variations(overrides);
1032            return;
1033        }
1034        let tables = self.font.tables();
1035        if let Ok(fvar) = tables.fvar() {
1036            set_variations(
1037                &fvar,
1038                tables.avar().ok(),
1039                &mut self.coords,
1040                named_instance_variations(&fvar, index)
1041                    .chain(overrides.into_iter().map(Into::into)),
1042            );
1043        } else {
1044            self.coords.clear();
1045        }
1046    }
1047}

1049// Helper to extract an iterator of Variation from a named instance index.
1050fn named_instance_variations<'a>(
1051    fvar: &'a Fvar,
1052    index: u32,
1053) -> impl Iterator<Item = Variation> + 'a {
1054    fvar.axis_instance_arrays()
1055        .ok()
1056        .and_then(|arrays| {
1057            let axes = arrays.axes();
1058            arrays.instances().get(index as usize).ok().map(|instance| {
1059                axes.iter()
1060                    .zip(instance.coordinates)
1061                    .map(|(axis, coord)| Variation::new(axis.axis_tag(), coord.get().to_f32()))
1062            })
1063        })
1064        .into_iter()
1065        .flatten()
1066}

1068/// Helper for setting variations.
1069///
1070/// Pulled out into a separate function to avoid borrow checker issues.
1071fn set_variations<V>(fvar: &Fvar, avar: Option<Avar>, coords: &mut CoordStorage, variations: V)
1072where
1073    V: IntoIterator,
1074    V::Item: Into<Variation>,
1075{
1076    coords.resize(fvar.axis_count() as usize, NormalizedCoord::ZERO);
1077    fvar.user_to_normalized(
1078        avar.as_ref(),
1079        variations
1080            .into_iter()
1081            .map(Into::into)
1082            .map(|var| (var.tag, Fixed::from_f64(var.value as _))),
1083        coords.as_mut_slice(),
1084    );
1085    clear_if_all_zeroes(coords);
1086}

1096    /// Creates a new font from the given source and font index.
1097    ///
1098    /// See [`Font::new`] for how `index` is used.
1099    fn new(source: impl Into<Source>, index: u32) -> Option<Self> {
1100        let source = source.into();
1101        let kind = if let Some(tables) = Tables::new(source.clone(), index) {
1102            Some(KindRepr::Sfnt(SfntCaches::new(Arc::new(tables), index)))
1103        } else if let Source::Blob(blob) = &source {
1104            match Format::new(blob) {
1105                Some(Format::Type1) => Type1Font::new(blob).ok().map(|font| {
1106                    let axes = font
1107                        .multiple_master()
1108                        .map(super::variation::type1_axis_records)
1109                        .unwrap_or_default();
1110                    KindRepr::Type1(Type1Data { font, axes })
1111                }),
1112                Some(Format::Cff(count)) if index < count => {
1113                    let font = Yoke::<CffFontRef<'static>, Arc<Blob>>::try_attach_to_cart(
1114                        Arc::new(blob.clone()),
1115                        |blob| CffFontRef::new_cff(blob, index, None),
1116                    )
1117                    .ok()?;
1118                    // A subfont that cannot be read fails only its own
1119                    // glyphs, not the font.
1120                    let count = font.get().num_subfonts().min(CACHED_SUBFONTS as _);
1121                    let subfonts = (0..count)
1122                        .map(|index| font.get().subfont(index, &[]).ok())
1123                        .collect();
1124                    Some(KindRepr::StandaloneCff(StandaloneCffData {
1125                        font,
1126                        subfonts,
1127                        index,
1128                        unicode_charmap: Once::new(),
1129                        cid_glyphs: Once::new(),
1130                    }))
1131                }
1132                _ => None,
1133            }
1134        } else {
1135            None
1136        };
1137        let kind = kind?;
1138        let repr = SharedFontRepr {
1139            source,
1140            kind,
1141            shaping_data: Once::new(),
1142            glyph_name_index: Once::new(),
1143            default_location: LocationCaches::new(),
1144        };
1145        Some(Self(Arc::new(repr)))
1146    }

1563/// Builds the table from each CID to its glyph, keeping the first glyph for
1564/// a CID.
1565///
1566/// `cids` yields each glyph and its CID, glyph zero first with CID zero.
1567fn cid_table<I: Iterator<Item = (u16, usize)>>(cids: impl Fn() -> I) -> Box<[u16]> {
1568    let max_cid = cids().map(|(_, cid)| cid).max().unwrap_or(0);
1569    let mut glyphs = alloc::vec![0u16; max_cid + 1];
1570    // Zero marks an empty slot. Glyph zero has CID zero, so that slot is
1571    // already right and no later glyph takes it.
1572    for (glyph, cid) in cids() {
1573        if cid != 0 && glyphs[cid] == 0 {
1574            glyphs[cid] = glyph;
1575        }
1576    }
1577    glyphs.into_boxed_slice()
1578}

1810/// Lazy atomic storage for feature variation selections.
1811///
1812/// We don't want to load the GSUB and GPOS tables unless explicitly requested.
1813struct FeatureVarsStorage {
1814    gsub: AtomicU32,
1815    gpos: AtomicU32,
1816    /// Written before the indices, so it is current once they are.
1817    has_lookup_variations: AtomicBool,
1818}
1819
1820impl Default for FeatureVarsStorage {
1821    fn default() -> Self {
1822        Self::new()
1823    }
1824}
1825
1826impl FeatureVarsStorage {
1827    // Both sentinels sit where no record index reaches: a feature variation
1828    // record is eight bytes, so indexing this high would take a font of tens
1829    // of gigabytes.
1830    /// Nothing has read the tables yet.
1831    const UNCHECKED: u32 = u32::MAX;
1832    /// The tables were read and selected nothing.
1833    const ABSENT: u32 = u32::MAX - 1;
1834
1835    fn new() -> Self {
1836        Self {
1837            gsub: AtomicU32::new(Self::UNCHECKED),
1838            gpos: AtomicU32::new(Self::UNCHECKED),
1839            has_lookup_variations: AtomicBool::new(false),
1840        }
1841    }
1842
1843    fn load(&self, font: &SharedFont, coords: &[NormalizedCoord]) -> SelectedFeatureVariations {
1844        let (gsub, gpos) = (
1845            self.gsub.load(atomic::Ordering::Acquire),
1846            self.gpos.load(atomic::Ordering::Acquire),
1847        );
1848        if gsub != Self::UNCHECKED && gpos != Self::UNCHECKED {
1849            return SelectedFeatureVariations {
1850                gsub: Self::decode(gsub),
1851                gpos: Self::decode(gpos),
1852                has_lookup_variations: self.has_lookup_variations.load(atomic::Ordering::Relaxed),
1853            };
1854        }
1855        // Two threads that race here select the same indices, so the second
1856        // store writes what the first did.
1857        let selected = SelectedFeatureVariations::new(&font.tables(), coords);
1858        self.has_lookup_variations
1859            .store(selected.has_lookup_variations, atomic::Ordering::Relaxed);
1860        self.gsub
1861            .store(Self::encode(selected.gsub), atomic::Ordering::Release);
1862        self.gpos
1863            .store(Self::encode(selected.gpos), atomic::Ordering::Release);
1864        selected
1865    }
1866
1867    /// Returns the index a stored word holds, if it holds one.
1868    fn decode(value: u32) -> Option<u32> {
1869        (value < Self::ABSENT).then_some(value)

4use crate::fine::Splat4thExt;
5use crate::peniko::{BlendMode, Compose};
6use vello_common::fearless_simd::*;
7
8pub(crate) trait ComposeExt {
9    fn compose<S: Simd>(
10        &self,
11        simd: S,
12        src_c: f32x16<S>,
13        bg_c: f32x16<S>,
14        alpha_mask: Option<f32x16<S>>,
15    ) -> f32x16<S>;
16}
17
18impl ComposeExt for BlendMode {
19    fn compose<S: Simd>(
20        &self,
21        simd: S,
22        src_c: f32x16<S>,
23        bg_c: f32x16<S>,
24        alpha_mask: Option<f32x16<S>>,
25    ) -> f32x16<S> {
26        simd.vectorize(
27            #[inline(always)]
28            || compose_inner(*self, simd, src_c, bg_c, alpha_mask),
29        )
30    }
31}
32
33#[inline(always)]
34fn compose_inner<S: Simd>(
35    blend_mode: BlendMode,
36    simd: S,
37    src_c: f32x16<S>,
38    bg_c: f32x16<S>,
39    alpha_mask: Option<f32x16<S>>,
40) -> f32x16<S> {
41    // There some non-obvious subtleties worth highlighting here.
42    // We support two kinds of blending (in this case, we focus on compositing specifically):
43    // - Isolated blending, where layers as a whole are blended together with their backdrop.
44    //   If we are currently performing this kind of blending, `alpha_mask` will always be `None`.
45    //   After all, there is no concrete shape opacity associated with a layer. Instead, we are
46    //   just compositing the RGBA values at _all_ positions of the source layer with the backdrop
47    //   layer. For example, if the backdrop contains a green rectangle and source layer is just
48    //   empty, if we perform blending with `Compose::Clear`, then _everything_ will be cleared,
49    //   because we are compositing the whole source layer with the whole backdrop, and not
50    //   just the parts of the source layer that have actually be drawn on.
51    // - Non-isolated blending, where a single path is blended with the backdrop. In this case,
52    //   `alpha_mask` _might_ be `Some` and contain the alpha values of the strips we are currently
53    //   compositing. Remember that strips always have a fixed height of 4, because of this, the
54    //   strips might cover areas that aren't actually covered by the path (and just have an alpha
55    //   value of 0, or a value between 0-254 for anti-aliased parts). Because of this, for
56    //   non-isolated blending, we need to lerp the result with the backdrop using `alpha_mask`.
57
58    let mut res = match blend_mode.compose {
59        Compose::SrcOver => SrcOver::compose(simd, src_c, bg_c),
60        Compose::Clear => Clear::compose(simd, src_c, bg_c),
61        Compose::Copy => Copy::compose(simd, src_c, bg_c),
62        Compose::DestOver => DestOver::compose(simd, src_c, bg_c),
63        Compose::Dest => Dest::compose(simd, src_c, bg_c),
64        Compose::SrcIn => SrcIn::compose(simd, src_c, bg_c),
65        Compose::DestIn => DestIn::compose(simd, src_c, bg_c),
66        Compose::SrcOut => SrcOut::compose(simd, src_c, bg_c),
67        Compose::DestOut => DestOut::compose(simd, src_c, bg_c),
68        Compose::SrcAtop => SrcAtop::compose(simd, src_c, bg_c),
69        Compose::DestAtop => DestAtop::compose(simd, src_c, bg_c),
70        Compose::Xor => Xor::compose(simd, src_c, bg_c),
71        Compose::Plus => Plus::compose(simd, src_c, bg_c),
72        // Have not been able to find a formula for this, so just fallback to Plus.
73        Compose::PlusLighter => Plus::compose(simd, src_c, bg_c),
74    };
75
76    if let Some(alpha_mask) = alpha_mask {
77        let alpha_mask_inv = 1.0 - alpha_mask;
78        res = alpha_mask * res + alpha_mask_inv * bg_c;
79    }
80
81    res
82}
83
84macro_rules! compose {
85    ($name:ident, $fa:expr, $fb:expr, $sat:expr) => {
86        struct $name;
87
88        impl $name {
89            #[inline(always)]
90            fn compose<S: Simd>(simd: S, src_c: f32x16<S>, bg_c: f32x16<S>) -> f32x16<S> {
91                let al_b = bg_c.splat_4th();
92                let al_s = src_c.splat_4th();
93
94                let fa = $fa(simd, al_s, al_b);
95                let fb = $fb(simd, al_s, al_b);
96
97                if $sat {
98                    (src_c * fa + fb * bg_c)
99                        .min(f32x16::splat(simd, 1.0))
100                        .max(f32x16::splat(simd, 0.0))
101                } else {
102                    src_c * fa + fb * bg_c
103                }
104            }
105        }
106    };
107}

109compose!(
110    Clear,
111    |simd, _, _| f32x16::splat(simd, 0.0),
112    |simd, _, _| f32x16::splat(simd, 0.0),
113    false
114);
115compose!(
116    Copy,
117    |simd, _, _| f32x16::splat(simd, 1.0),
118    |simd, _, _| f32x16::splat(simd, 0.0),
119    false
120);
121compose!(
122    SrcOver,
123    |simd, _, _| f32x16::splat(simd, 1.0),
124    |_, al_s: f32x16<S>, _| 1.0 - al_s,
125    false
126);
127compose!(
128    DestOver,
129    |_, _, al_b: f32x16<S>| 1.0 - al_b,

138);

4use crate::peniko;
5use core::slice::ChunksExact;
6use vello_common::encode::EncodedGradient;
7use vello_common::fearless_simd::*;
8
9/// An accelerated gradient painter for u8.
10///
11/// Assumes that the gradient has no undefined positions.
12#[derive(Debug)]
13pub(crate) struct GradientPainter<'a, S: Simd> {
14    gradient: &'a EncodedGradient,
15    lut: &'a [[u8; 4]],
16    t_vals: ChunksExact<'a, f32>,
17    scale_factor: f32x16<S>,
18    simd: S,
19}
20
21impl<'a, S: Simd> GradientPainter<'a, S> {
22    pub(crate) fn new(simd: S, gradient: &'a EncodedGradient, t_vals: &'a [f32]) -> Self {
23        simd.vectorize(
24            #[inline(always)]
25            || {
26                let lut = gradient.u8_lut(simd);
27                let scale_factor = f32x16::splat(simd, lut.scale_factor());
28
29                Self {
30                    gradient,
31                    scale_factor,
32                    lut: lut.lut(),
33                    t_vals: t_vals.chunks_exact(16),
34                    simd,
35                }
36            },
37        )
38    }
39}
40
41impl<S: Simd> Iterator for GradientPainter<'_, S> {
42    type Item = u8x64<S>;
43
44    #[inline(always)]
45    fn next(&mut self) -> Option<Self::Item> {
46        let extend = self.gradient.extend;
47        let pos = f32x16::from_slice(self.simd, self.t_vals.next()?);
48        let t_vals = apply_extend(pos, extend);
49        let indices = (t_vals * self.scale_factor).to_int::<u32x16<S>>();
50
51        let mut vals = [0_u8; 64];
52        for (val, idx) in vals.chunks_exact_mut(4).zip(*indices) {
53            val.copy_from_slice(&self.lut[idx as usize]);
54        }
55
56        Some(u8x64::from_slice(self.simd, &vals))
57    }
58}
59
60impl<S: Simd> crate::fine::Painter for GradientPainter<'_, S> {
61    fn paint_u8(mut self, buf: &mut [u8]) {
62        self.simd.vectorize(
63            #[inline(always)]
64            || {
65                for chunk in buf.chunks_exact_mut(64) {
66                    self.next().unwrap().store_slice(chunk);
67                }
68            },
69        );
70    }
71
72    fn paint_f32(self, _: &mut [f32]) {
73        unimplemented!()
74    }
75}

77// TODO: Maybe delete this method and use `apply_extend` from highp by splitting into two f32x8.
78#[inline(always)]
79pub(crate) fn apply_extend<S: Simd>(val: f32x16<S>, extend: peniko::Extend) -> f32x16<S> {
80    match extend {
81        peniko::Extend::Pad => val.max(0.0).min(1.0),
82        peniko::Extend::Repeat => (val - val.floor()).fract(),
83        // See <https://github.com/google/skia/blob/220738774f7a0ce4a6c7bd17519a336e5e5dea5b/src/opts/SkRasterPipeline_opts.h#L6472-L6475>
84        peniko::Extend::Reflect => ((val - 1.0) - 2.0 * ((val - 1.0) * 0.5).floor() - 1.0)
85            .abs()
86            .max(0.0)
87            .min(1.0),
88    }

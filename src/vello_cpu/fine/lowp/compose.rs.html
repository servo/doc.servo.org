10pub(crate) trait ComposeExt {
11    fn compose<S: Simd>(
12        &self,
13        simd: S,
14        src_c: u8x32<S>,
15        bg_c: u8x32<S>,
16        alpha_mask: Option<u8x32<S>>,
17    ) -> u8x32<S>;
18}
19
20impl ComposeExt for BlendMode {
21    fn compose<S: Simd>(
22        &self,
23        simd: S,
24        src_c: u8x32<S>,
25        bg_c: u8x32<S>,
26        alpha_mask: Option<u8x32<S>>,
27    ) -> u8x32<S> {
28        simd.vectorize(
29            #[inline(always)]
30            || compose_inner(*self, simd, src_c, bg_c, alpha_mask),
31        )
32    }
33}
34
35#[inline(always)]
36fn compose_inner<S: Simd>(
37    blend_mode: BlendMode,
38    simd: S,
39    src_c: u8x32<S>,
40    bg_c: u8x32<S>,
41    alpha_mask: Option<u8x32<S>>,
42) -> u8x32<S> {
43    let mut res = match blend_mode.compose {
44        Compose::SrcOver => SrcOver::compose(simd, src_c, bg_c),
45        Compose::Clear => Clear::compose(simd, src_c, bg_c),
46        Compose::Copy => Copy::compose(simd, src_c, bg_c),
47        Compose::DestOver => DestOver::compose(simd, src_c, bg_c),
48        Compose::Dest => Dest::compose(simd, src_c, bg_c),
49        Compose::SrcIn => SrcIn::compose(simd, src_c, bg_c),
50        Compose::DestIn => DestIn::compose(simd, src_c, bg_c),
51        Compose::SrcOut => SrcOut::compose(simd, src_c, bg_c),
52        Compose::DestOut => DestOut::compose(simd, src_c, bg_c),
53        Compose::SrcAtop => SrcAtop::compose(simd, src_c, bg_c),
54        Compose::DestAtop => DestAtop::compose(simd, src_c, bg_c),
55        Compose::Xor => Xor::compose(simd, src_c, bg_c),
56        Compose::Plus => Plus::compose(simd, src_c, bg_c),
57        // Have not been able to find a formula for this, so just fallback to Plus.
58        Compose::PlusLighter => Plus::compose(simd, src_c, bg_c),
59    };
60
61    if let Some(alpha_mask) = alpha_mask {
62        let alpha_mask_inv = 255 - alpha_mask;
63        let p1 = simd.widen_u8x32(alpha_mask) * simd.widen_u8x32(res);
64        let p2 = simd.widen_u8x32(alpha_mask_inv) * simd.widen_u8x32(bg_c);
65        res = simd.narrow_u16x32((p1 + p2).div_255());
66    }
67
68    res
69}
70
71macro_rules! compose {
72    ($name:ident, $fa:expr, $fb:expr, $sat:expr) => {
73        struct $name;
74
75        impl $name {
76            #[inline(always)]
77            fn compose<S: Simd>(simd: S, src_c: u8x32<S>, bg_c: u8x32<S>) -> u8x32<S> {
78                let al_b = bg_c.splat_4th();
79                let al_s = src_c.splat_4th();
80
81                let fa = $fa(simd, al_s, al_b);
82                let fb = $fb(simd, al_s, al_b);
83
84                if $sat {
85                    simd.narrow_u16x32(
86                        (simd.widen_u8x32(src_c.normalized_mul(fa))
87                            + simd.widen_u8x32(fb.normalized_mul(bg_c)))
88                        .min(u16x32::splat(simd, 255))
89                        .max(u16x32::splat(simd, 0)),
90                    )
91                } else {
92                    src_c.normalized_mul(fa) + fb.normalized_mul(bg_c)
93                }
94            }
95        }
96    };
97}

18#[derive(MallocSizeOf)]
19pub struct FragmentTree {
20    /// Fragments at the top-level of the tree.
21    ///
22    /// If the root element has `display: none`, there are zero fragments.
23    /// Otherwise, there is at least one:
24    ///
25    /// * The first fragment is generated by the root element.
26    /// * There may be additional fragments generated by positioned boxes
27    ///   that have the initial containing block.
28    pub(crate) root_fragments: Vec<Fragment>,
29
30    /// The scrollable overflow rectangle for the entire tree
31    /// <https://drafts.csswg.org/css-overflow/#scrollable>
32    scrollable_overflow: Cell<Option<PhysicalRect<Au>>>,
33
34    /// The containing block used in the layout of this fragment tree.
35    pub(crate) initial_containing_block: PhysicalRect<Au>,
36
37    /// Whether or not the viewport is sensitive to scroll input events.
38    pub viewport_scroll_sensitivity: AxesScrollSensitivity,
39}
40
41impl FragmentTree {
42    pub(crate) fn new(
43        root_fragments: Vec<Fragment>,
44        initial_containing_block: PhysicalRect<Au>,
45        viewport_scroll_sensitivity: AxesScrollSensitivity,
46    ) -> Self {
47        Self {
48            root_fragments,
49            scrollable_overflow: Cell::default(),
50            initial_containing_block,
51            viewport_scroll_sensitivity,
52        }
53    }
54
55    /// The root fragment for this fragment tree, if the root element does not have
56    /// `display: none;`. Note that positioned elements that have the initial containing
57    /// block as their containing block are also direct children of the fragment tree
58    /// root, but they are not returned by this getter.
59    pub(crate) fn root_box_fragment(&self) -> Option<Arc<BoxFragment>> {
60        self.root_fragments.iter().find_map(|root_fragment| {
61            let box_fragment = root_fragment.retrieve_box_fragment()?;
62            box_fragment
63                .base
64                .flags
65                .contains(FragmentFlags::IS_ROOT_ELEMENT)
66                .then(|| box_fragment.clone())
67        })
68    }
69
70    pub fn print(&self) {
71        let mut print_tree = PrintTree::new("Fragment Tree");
72        for fragment in &self.root_fragments {
73            fragment.print(&mut print_tree);
74        }
75    }
76
77    pub(crate) fn scrollable_overflow(&self) -> PhysicalRect<Au> {
78        if let Some(scrollable_overflow) = self.scrollable_overflow.get() {
79            return scrollable_overflow;
80        }
81        let scrollable_overflow = self.calculate_scrollable_overflow();
82        self.scrollable_overflow.set(Some(scrollable_overflow));
83        scrollable_overflow
84    }
85
86    pub(crate) fn clear_scrollable_overflow(&self) {
87        self.scrollable_overflow.set(None);
88    }
89
90    /// Calculate the scrollable overflow / scrolling area for this [`FragmentTree`] according
91    /// to <https://drafts.csswg.org/cssom-view/#scrolling-area>.
92    fn calculate_scrollable_overflow(&self) -> PhysicalRect<Au> {
93        let Some(first_root_fragment) = self.root_box_fragment() else {
94            return self.initial_containing_block;
95        };
96
97        let scrollable_overflow =
98            self.root_fragments
99                .iter()
100                .fold(euclid::Rect::default(), |overflow, fragment| {
101                    // Scrollable overflow should be accumulated in the block that
102                    // establishes the containing block for the element. Thus, fixed
103                    // positioned fragments whose containing block is the initial
104                    // containing block should not be included in overflow calculation.
105                    // See <https://www.w3.org/TR/css-overflow-3/#scrollable>.
106                    if fragment
107                        .retrieve_box_fragment()
108                        .is_some_and(|box_fragment| {
109                            box_fragment.style().get_box().position == Position::Fixed
110                        })
111                    {
112                        return overflow;
113                    }
114
115                    overflow.union(&fragment.scrollable_overflow_for_parent())
116                });
117
118        // Ensure that scrollable overflow that is unreachable is not included in the final
119        // rectangle. See <https://drafts.csswg.org/css-overflow/#scrolling-direction>.
120        first_root_fragment
121            .with_style()
122            .clip_wholly_unreachable_scrollable_overflow(
123                scrollable_overflow,
124                self.initial_containing_block,
125            )
126    }
127
128    pub(crate) fn find<T>(
129        &self,
130        mut process_func: impl FnMut(&Fragment, usize, &PhysicalRect<Au>) -> Option<T>,
131    ) -> Option<T> {
132        let info = ContainingBlockManager {
133            for_non_absolute_descendants: &self.initial_containing_block,
134            for_absolute_descendants: None,
135            for_absolute_and_fixed_descendants: &self.initial_containing_block,
136        };
137        self.root_fragments
138            .iter()
139            .find_map(|child| child.find(&info, 0, &mut process_func))
140    }
141
142    /// Find the `<body>` element's [`Fragment`], if it exists in this [`FragmentTree`].
143    pub(crate) fn body_fragment(&self) -> Option<Arc<BoxFragment>> {
144        fn find_body_from_box_fragment(
145            box_fragment: &Arc<BoxFragment>,
146        ) -> Option<Arc<BoxFragment>> {
147            if box_fragment.is_body_element_of_html_element_root() {
148                return Some(box_fragment.clone());
149            }
150
151            // The fragment for the `<body>` element is typically a child of the root (though,
152            // not if it's absolutely positioned), so we need to recurse into the children of
153            // the root to find it.
154            //
155            // Additionally, recurse into any anonymous fragments, as the `<body>` fragment may
156            // have created anonymous parents (for instance by creating an inline formatting context).
157            if box_fragment.is_root_element() || box_fragment.base.is_anonymous() {
158                find_body(&box_fragment.children)
159            } else {
160                None
161            }
162        }
163
164        fn find_body(children: &[Fragment]) -> Option<Arc<BoxFragment>> {
165            children.iter().find_map(|fragment| {
166                match fragment {
167                    Fragment::LayoutRoot(layout_root) => {
168                        find_body_from_box_fragment(&layout_root.inner_box_fragment())
169                    },
170                    Fragment::Box(box_fragment) | Fragment::Float(box_fragment) => {
171                        find_body_from_box_fragment(box_fragment)
172                    },
173                    Fragment::Positioning(positioning_fragment)
174                        if positioning_fragment.base.is_anonymous() =>
175                    {
176                        // If the `<body>` element is a `display: inline` then it might be nested inside of a
177                        // `PositioningFragment` for the purposes of putting it on the first line of the implied
178                        // inline formatting context.
179                        find_body(&positioning_fragment.children)
180                    },
181                    _ => None,
182                }
183            })
184        }
185
186        find_body(&self.root_fragments)
187    }
188}

22/// Can contain child fragments with relative coordinates, but does not contribute to painting
23/// itself. [`PositioningFragment`]s may be completely anonymous, or just non-painting Fragments
24/// generated by boxes.
25#[derive(MallocSizeOf)]
26pub(crate) struct PositioningFragment {
27    pub base: BaseFragment,
28
29    /// The style for this [`PositioningFragment`].
30    pub style: SharedStyle,
31
32    pub children: Vec<Fragment>,
33
34    /// The scrollable overflow of this anonymous fragment's children.
35    scrollable_overflow: SyncPhysicalRectAu,
36    scrollable_overflow_is_up_to_date: AtomicBool,
37
38    /// The cumulative content rect of this [`PositioningFragment`]'s parent in
39    /// coordinates relative to the initial containing block, but not taking into account
40    /// any transforms.
41    pub cumulative_parent_content_rect: SyncPhysicalRectAu,
42
43    /// Whether or not this [`PositioningFragment`] is a line box.
44    is_line_box: bool,
45
46    /// The containing spatial tree node of this [`PositioningFragment`]. This is assigned
47    /// during `StackingContextTree` construction, so isn't available before that time.
48    /// This is used to for determining final viewport size and position of this node and
49    /// will also be used in the future for hit testing.
50    pub spatial_tree_node: AtomicOptionScrollTreeNodeId,
51}
52
53impl PositioningFragment {
54    pub fn new_anonymous(
55        style: ServoArc<ComputedValues>,
56        rect: PhysicalRect<Au>,
57        children: FragmentChildren,
58        is_line_box: bool,
59    ) -> Arc<Self> {
60        let writing_mode = style.writing_mode;
61        Self::new_anonymous_with_writing_mode_override(
62            style,
63            rect,
64            children,
65            is_line_box,
66            writing_mode,
67        )
68    }
69
70    pub fn new_anonymous_with_writing_mode_override(
71        style: ServoArc<ComputedValues>,
72        rect: PhysicalRect<Au>,
73        children: FragmentChildren,
74        is_line_box: bool,
75        writing_mode: WritingMode,
76    ) -> Arc<Self> {
77        let children = children.into_physical_fragments(
78            writing_mode,
79            LogicalVec2::from_physical_size(&rect.size, writing_mode),
80        );
81        Self::new_with_base_fragment_info(
82            BaseFragmentInfo::anonymous(),
83            style,
84            rect,
85            children,
86            is_line_box,
87        )
88    }
89
90    pub fn new_empty(
91        base_fragment_info: BaseFragmentInfo,
92        rect: PhysicalRect<Au>,
93        style: ServoArc<ComputedValues>,
94    ) -> Arc<Self> {
95        Self::new_with_base_fragment_info(base_fragment_info, style, rect, Vec::new(), false)
96    }
97
98    fn new_with_base_fragment_info(
99        base_fragment_info: BaseFragmentInfo,
100        style: ServoArc<ComputedValues>,
101        rect: PhysicalRect<Au>,
102        children: Vec<Fragment>,
103        is_line_box: bool,
104    ) -> Arc<Self> {
105        Arc::new(Self {
106            base: BaseFragment::new(base_fragment_info, rect),
107            style: style.into(),
108            children,
109            scrollable_overflow: Default::default(),
110            scrollable_overflow_is_up_to_date: AtomicBool::new(false),
111            cumulative_parent_content_rect: Default::default(),
112            is_line_box,
113            spatial_tree_node: Default::default(),
114        })
115    }
116
117    #[inline]
118    pub(crate) fn set_cumulative_content_rect(&self, content_rect: &PhysicalRect<Au>) {
119        self.cumulative_parent_content_rect.set(*content_rect);
120    }
121
122    pub fn offset_by_containing_block(
123        &self,
124        rect: &PhysicalRect<Au>,
125        cumulative_content_rect_calculation: CumulativeContentRectCalculation<'_>,
126    ) -> PhysicalRect<Au> {
127        cumulative_content_rect_calculation.ensure();
128        rect.translate(self.cumulative_parent_content_rect.origin().to_vector())
129    }
130
131    /// Get the scrollable overflow for this [`PositioningFragment`] relative to its
132    /// containing block, recalculating scrollable overflow when necessary, for instance
133    /// after a style change.
134    pub(crate) fn scrollable_overflow_for_parent(&self) -> PhysicalRect<Au> {
135        if self
136            .scrollable_overflow_is_up_to_date
137            .load(Ordering::Acquire)
138        {
139            self.scrollable_overflow.get()
140        } else {
141            let rect = self.calculate_scrollable_overflow();
142            self.scrollable_overflow.set(rect);
143            self.scrollable_overflow_is_up_to_date
144                .store(true, Ordering::Release);
145            rect
146        }
147    }
148
149    /// Clear the scrollable overflow on this [`PositioningFragment`]. This is called
150    /// during damage propagation when a fragment is preserved, itself or one of its
151    /// descendants has scrollable overflow damage.
152    pub(crate) fn clear_scrollable_overflow(&self) {
153        self.scrollable_overflow_is_up_to_date
154            .store(false, Ordering::Release);
155    }
156
157    fn calculate_scrollable_overflow(&self) -> PhysicalRect<Au> {
158        self.children
159            .iter()
160            .fold(PhysicalRect::zero(), |acc, child| {
161                acc.union(
162                    &child
163                        .scrollable_overflow_for_parent()
164                        .translate(self.base.rect().origin.to_vector()),
165                )
166            })
167    }
168
169    pub(crate) fn is_line_box(&self) -> bool {
170        self.is_line_box
171    }
172
173    pub fn print(&self, tree: &mut PrintTree) {
174        tree.new_level(format!(
175            "PositioningFragment {:?}\nscrollable-overflow={:?}",
176            self.base,
177            PrintableRect(self.scrollable_overflow.get()),
178        ));
179
180        for child in &self.children {
181            child.print(tree);
182        }
183        tree.end_level();
184    }
185}

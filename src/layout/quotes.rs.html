367fn quotes_data_for_lang(lang: &str) -> QuotesData {
368    // All valid language codes are at least two bytes long.
369    if lang.len() < 2 {
370        return DEFAULT_QUOTES;
371    }
372
373    // Found an exact match for the requested lang.
374    if let Some(quotes_data) = QUOTES_MAP.get(lang) {
375        return *quotes_data;
376    }
377
378    // Try parsing lang as a Locale and canonicalizing the subtags, then see if
379    // we can match it with region or script subtags, if present, or just the
380    // primary language tag.
381    let locale = match lang.parse::<Locale>() {
382        Err(_) => return DEFAULT_QUOTES,
383        Ok(locale) => locale,
384    };
385
386    let lang = locale.id.language.to_string();
387
388    if let Some(quotes_data) = QUOTES_MAP.get(lang.as_str()) {
389        return *quotes_data;
390    }
391
392    if let Some(quotes_data) = locale
393        .id
394        .region
395        .and_then(|region| QUOTES_MAP.get(format!("{lang}-{region}").as_str()))
396    {
397        return *quotes_data;
398    }
399
400    if let Some(quotes_data) = locale
401        .id
402        .script
403        .and_then(|script| QUOTES_MAP.get(format!("{lang}-{script}").as_str()))
404    {
405        return *quotes_data;
406    }
407
408    DEFAULT_QUOTES
409}

411pub(crate) fn quotes_for_lang(lang: &str, depth: usize) -> QuotePair {
412    let quotes_data = quotes_data_for_lang(lang);
413    match depth {
414        0 => quotes_data.quotes,
415        _ => quotes_data.alternative_quotes,
416    }

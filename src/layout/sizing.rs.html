24#[derive(PartialEq)]
25pub(crate) enum IntrinsicSizingMode {
26    /// Used to refer to a min-content contribution or max-content contribution.
27    /// This is the size that a box contributes to its containing block’s min-content
28    /// or max-content size. Note this is based on the outer size of the box,
29    /// and takes into account the relevant sizing properties of the element.
30    /// <https://drafts.csswg.org/css-sizing-3/#contributions>
31    Contribution,
32    /// Used to refer to a min-content size or max-content size.
33    /// This is the size based on the contents of an element, without regard for its context.
34    /// Note this is usually based on the inner (content-box) size of the box,
35    /// and ignores the relevant sizing properties of the element.
36    /// <https://drafts.csswg.org/css-sizing-3/#intrinsic>
37    Size,
38}

46/// <https://drafts.csswg.org/css-sizing/#intrinsic-sizes>
47impl ContentSizes {
48    pub fn max(&self, other: Self) -> Self {
49        Self {
50            min_content: self.min_content.max(other.min_content),
51            max_content: self.max_content.max(other.max_content),
52        }
53    }
54
55    pub fn max_assign(&mut self, other: Self) {
56        *self = self.max(other);
57    }
58
59    pub fn union(&self, other: &Self) -> Self {
60        Self {
61            min_content: self.min_content.max(other.min_content),
62            max_content: self.max_content + other.max_content,
63        }
64    }
65
66    pub fn union_assign(&mut self, other: &Self) {
67        self.min_content.max_assign(other.min_content);
68        self.max_content += other.max_content;
69    }
70
71    pub fn map(&self, f: impl Fn(Au) -> Au) -> Self {
72        Self {
73            min_content: f(self.min_content),
74            max_content: f(self.max_content),
75        }
76    }
77}

106impl ContentSizes {
107    /// Clamps the provided amount to be between the min-content and the max-content.
108    /// This is called "shrink-to-fit" in CSS2, and "fit-content" in CSS Sizing.
109    /// <https://drafts.csswg.org/css2/visudet.html#shrink-to-fit-float>
110    /// <https://drafts.csswg.org/css-sizing/#funcdef-width-fit-content>
111    pub fn shrink_to_fit(&self, available_size: Au) -> Au {
112        // This formula is slightly different than what the spec says,
113        // to ensure that the minimum wins for a malformed ContentSize
114        // whose min_content is larger than its max_content.
115        available_size.min(self.max_content).max(self.min_content)
116    }
117}

128#[expect(clippy::too_many_arguments)]
129pub(crate) fn outer_inline(
130    base: &LayoutBoxBase,
131    layout_style: &LayoutStyle,
132    containing_block: &IndefiniteContainingBlock,
133    auto_minimum: &LogicalVec2<Au>,
134    auto_block_size_stretches_to_containing_block: bool,
135    is_replaced: bool,
136    establishes_containing_block: bool,
137    skippable_for_block_percentage_quirk: bool,
138    get_preferred_aspect_ratio: impl FnOnce(&LogicalVec2<Au>) -> Option<AspectRatio>,
139    get_inline_content_size: impl FnOnce(&ConstraintSpace) -> InlineContentSizesResult,
140    get_tentative_block_content_size: impl FnOnce(Option<AspectRatio>) -> Option<ContentSizes>,
141) -> InlineContentSizesResult {
142    let ContentBoxSizesAndPBM {
143        content_box_sizes,
144        pbm,
145        mut depends_on_block_constraints,
146        preferred_size_computes_to_auto,
147    } = layout_style.content_box_sizes_and_padding_border_margin(containing_block);
148    let margin = pbm.margin.map(|v| v.auto_is(Au::zero));
149    let pbm_sums = LogicalVec2 {
150        block: pbm.padding_border_sums.block + margin.block_sum(),
151        inline: pbm.padding_border_sums.inline + margin.inline_sum(),
152    };
153    let style = layout_style.style();
154    let is_table = layout_style.is_table();
155    // TODO: Replace `depends_on_contents` with `content_size.get().is_some()` once `LazyCell::get()`
156    // becomes stable.
157    let depends_on_contents = Cell::new(false);
158    let content_size = LazyCell::new(|| {
159        depends_on_contents.set(true);
160        let constraint_space = if establishes_containing_block {
161            let available_block_size = containing_block
162                .size
163                .block
164                .map(|v| Au::zero().max(v - pbm_sums.block));
165            let automatic_size = if preferred_size_computes_to_auto.block &&
166                auto_block_size_stretches_to_containing_block
167            {
168                depends_on_block_constraints = true;
169                Size::Stretch
170            } else {
171                Size::FitContent
172            };
173            let aspect_ratio = get_preferred_aspect_ratio(&pbm.padding_border_sums);
174            let mut block_size =
175                if let Some(block_content_size) = get_tentative_block_content_size(aspect_ratio) {
176                    SizeConstraint::Definite(content_box_sizes.block.resolve(
177                        LogicalAxis::Block,
178                        automatic_size,
179                        || auto_minimum.block,
180                        available_block_size,
181                        || block_content_size,
182                        is_table,
183                    ))
184                } else {
185                    content_box_sizes.block.resolve_extrinsic(
186                        automatic_size,
187                        auto_minimum.block,
188                        available_block_size,
189                    )
190                };
191            if skippable_for_block_percentage_quirk {
192                block_size.replace_if_indefinite_with(containing_block.size.block.into())
193            }
194            ConstraintSpace::new(block_size, style, aspect_ratio)
195        } else {
196            // Even if the size doesn't directly depend on block constraints, since this box
197            // doesn't establish a containing block, its contents can depend on its block
198            // constraints. And thus have a dependency via the intrinsic size.
199            depends_on_block_constraints = true;
200            // This assumes that there is no preferred aspect ratio, or that there is no
201            // block size constraint to be transferred so the ratio is irrelevant.
202            // We only get into here for anonymous blocks, for which the assumption holds.
203            ConstraintSpace::new(
204                containing_block.size.block.into(),
205                containing_block.style,
206                None,
207            )
208        };
209        get_inline_content_size(&constraint_space)
210    });
211    let resolve_non_initial = |inline_size, stretch_values| {
212        Some(match inline_size {
213            Size::Initial => return None,
214            Size::Numeric(numeric) => (numeric, numeric, false),
215            Size::MinContent => (
216                content_size.sizes.min_content,
217                content_size.sizes.min_content,
218                content_size.depends_on_block_constraints,
219            ),
220            Size::MaxContent => (
221                content_size.sizes.max_content,
222                content_size.sizes.max_content,
223                content_size.depends_on_block_constraints,
224            ),
225            Size::FitContent => (
226                content_size.sizes.min_content,
227                content_size.sizes.max_content,
228                content_size.depends_on_block_constraints,
229            ),
230            Size::FitContentFunction(size) => {
231                let size = content_size.sizes.shrink_to_fit(size);
232                (size, size, content_size.depends_on_block_constraints)
233            },
234            Size::Stretch => return stretch_values,
235            Size::QuirkyStretch => unreachable!("Quirky stretch can't be an inline size"),
236        })
237    };
238    let (mut preferred_min_content, preferred_max_content, preferred_depends_on_block_constraints) =
239        resolve_non_initial(content_box_sizes.inline.preferred, None)
240            .unwrap_or_else(|| resolve_non_initial(Size::FitContent, None).unwrap());
241    let (mut min_min_content, mut min_max_content, mut min_depends_on_block_constraints) =
242        resolve_non_initial(
243            content_box_sizes.inline.min,
244            Some((Au::zero(), Au::zero(), false)),
245        )
246        .unwrap_or((auto_minimum.inline, auto_minimum.inline, false));
247    let (mut max_min_content, max_max_content, max_depends_on_block_constraints) =
248        resolve_non_initial(content_box_sizes.inline.max, None)
249            .map(|(min_content, max_content, depends_on_block_constraints)| {
250                (
251                    Some(min_content),
252                    Some(max_content),
253                    depends_on_block_constraints,
254                )
255            })
256            .unwrap_or_default();
257
258    // https://drafts.csswg.org/css-sizing-3/#replaced-percentage-min-contribution
259    // > If the box is replaced, a cyclic percentage in the value of any max size property
260    // > or preferred size property (width/max-width/height/max-height), is resolved against
261    // > zero when calculating the min-content contribution in the corresponding axis.
262    //
263    // This means that e.g. the min-content contribution of `width: calc(100% + 100px)`
264    // should be 100px, but it's just zero on other browsers, so we do the same.
265    if is_replaced {
266        let has_percentage = |size: Size<LengthPercentage>| {
267            // We need a comment here to avoid breaking `./mach test-tidy`.
268            matches!(size, Size::Numeric(numeric) if numeric.has_percentage())
269        };
270        let writing_mode = containing_block.style.writing_mode;
271        if content_box_sizes.inline.preferred.is_initial() &&
272            has_percentage(style.box_size(writing_mode).inline)
273        {
274            preferred_min_content = Au::zero();
275        }
276        if content_box_sizes.inline.max.is_initial() &&
277            has_percentage(style.max_box_size(writing_mode).inline)
278        {
279            max_min_content = Some(Au::zero());
280        }
281    }
282
283    // Regardless of their sizing properties, tables are always forced to be at least
284    // as big as their min-content size, so floor the minimums.
285    if is_table {
286        min_min_content.max_assign(content_size.sizes.min_content);
287        min_max_content.max_assign(content_size.sizes.min_content);
288        min_depends_on_block_constraints |= content_size.depends_on_block_constraints;
289    }
290
291    base.outer_inline_content_sizes_depend_on_content
292        .store(depends_on_contents.get(), Ordering::Relaxed);
293    InlineContentSizesResult {
294        sizes: ContentSizes {
295            min_content: preferred_min_content
296                .clamp_between_extremums(min_min_content, max_min_content) +
297                pbm_sums.inline,
298            max_content: preferred_max_content
299                .clamp_between_extremums(min_max_content, max_max_content) +
300                pbm_sums.inline,
301        },
302        depends_on_block_constraints: depends_on_block_constraints &&
303            (preferred_depends_on_block_constraints ||
304                min_depends_on_block_constraints ||
305                max_depends_on_block_constraints),
306    }
307}

309#[derive(Clone, Copy, Debug, MallocSizeOf)]
310pub(crate) struct InlineContentSizesResult {
311    pub sizes: ContentSizes,
312    pub depends_on_block_constraints: bool,
313}

315pub(crate) trait ComputeInlineContentSizes {
316    fn compute_inline_content_sizes(
317        &self,
318        layout_context: &LayoutContext,
319        constraint_space: &ConstraintSpace,
320    ) -> InlineContentSizesResult;
321
322    /// Returns the same result as [`Self::compute_inline_content_sizes()`], but adjusted
323    /// to floor the max-content size by the min-content size.
324    /// This is being discussed in <https://github.com/w3c/csswg-drafts/issues/12076>.
325    fn compute_inline_content_sizes_with_fixup(
326        &self,
327        layout_context: &LayoutContext,
328        constraint_space: &ConstraintSpace,
329    ) -> InlineContentSizesResult {
330        let mut result = self.compute_inline_content_sizes(layout_context, constraint_space);
331        let sizes = &mut result.sizes;
332        sizes.max_content.max_assign(sizes.min_content);
333        result
334    }
335}

337/// The possible values accepted by the sizing properties.
338/// <https://drafts.csswg.org/css-sizing/#sizing-properties>
339#[derive(Clone, Debug, PartialEq)]
340pub(crate) enum Size<T> {
341    /// Represents an `auto` value for the preferred and minimum size properties,
342    /// or `none` for the maximum size properties.
343    /// <https://drafts.csswg.org/css-sizing/#valdef-width-auto>
344    /// <https://drafts.csswg.org/css-sizing/#valdef-max-width-none>
345    Initial,
346    /// <https://drafts.csswg.org/css-sizing/#valdef-width-min-content>
347    MinContent,
348    /// <https://drafts.csswg.org/css-sizing/#valdef-width-max-content>
349    MaxContent,
350    /// <https://drafts.csswg.org/css-sizing-4/#valdef-width-fit-content>
351    FitContent,
352    /// <https://drafts.csswg.org/css-sizing-3/#funcdef-width-fit-content>
353    FitContentFunction(T),
354    /// <https://drafts.csswg.org/css-sizing-4/#valdef-width-stretch>
355    Stretch,
356    /// Represents the special stretching behavior of `<html>` and `<body>`
357    /// in quirks mode. Roughly, it behaves as the maximum of `fit-content`
358    /// and `stretch`. It can only be used as a preferred block size.
359    /// <https://quirks.spec.whatwg.org/#the-html-element-fills-the-viewport-quirk>
360    /// <https://quirks.spec.whatwg.org/#the-body-element-fills-the-html-element-quirk>
361    QuirkyStretch,
362    /// Represents a numeric `<length-percentage>`, but resolved as a `T`.
363    /// <https://drafts.csswg.org/css-sizing/#valdef-width-length-percentage-0>
364    Numeric(T),
365}

383impl<T: Clone> Size<T> {
384    #[inline]
385    pub(crate) fn to_numeric(&self) -> Option<T> {
386        match self {
387            Self::Numeric(numeric) => Some(numeric).cloned(),
388            _ => None,
389        }
390    }
391
392    #[inline]
393    pub(crate) fn map<U>(&self, f: impl FnOnce(T) -> U) -> Size<U> {
394        match self {
395            Size::Initial => Size::Initial,
396            Size::MinContent => Size::MinContent,
397            Size::MaxContent => Size::MaxContent,
398            Size::FitContent => Size::FitContent,
399            Size::FitContentFunction(size) => Size::FitContentFunction(f(size.clone())),
400            Size::Stretch => Size::Stretch,
401            Size::QuirkyStretch => Size::QuirkyStretch,
402            Size::Numeric(numeric) => Size::Numeric(f(numeric.clone())),
403        }
404    }
405}

441impl Size<LengthPercentage> {
442    #[inline]
443    pub(crate) fn to_percentage(&self) -> Option<Percentage> {
444        self.to_numeric()
445            .and_then(|length_percentage| length_percentage.to_percentage())
446    }
447
448    /// Resolves percentages in a preferred size, against the provided basis.
449    /// If the basis is missing, percentages are considered cyclic.
450    /// <https://www.w3.org/TR/css-sizing-3/#preferred-size-properties>
451    /// <https://www.w3.org/TR/css-sizing-3/#cyclic-percentage-size>
452    #[inline]
453    pub(crate) fn resolve_percentages_for_preferred(&self, basis: Option<Au>) -> Size<Au> {
454        match self {
455            Size::Numeric(numeric) => numeric
456                .maybe_to_used_value(basis)
457                .map_or(Size::Initial, Size::Numeric),
458            Size::FitContentFunction(numeric) => {
459                // Under discussion in https://github.com/w3c/csswg-drafts/issues/11805
460                numeric
461                    .maybe_to_used_value(basis)
462                    .map_or(Size::FitContent, Size::FitContentFunction)
463            },
464            _ => self.map(|_| unreachable!("This shouldn't be called for keywords")),
465        }
466    }
467
468    /// Resolves percentages in a maximum size, against the provided basis.
469    /// If the basis is missing, percentages are considered cyclic.
470    /// <https://www.w3.org/TR/css-sizing-3/#preferred-size-properties>
471    /// <https://www.w3.org/TR/css-sizing-3/#cyclic-percentage-size>
472    #[inline]
473    pub(crate) fn resolve_percentages_for_max(&self, basis: Option<Au>) -> Size<Au> {
474        match self {
475            Size::Numeric(numeric) => numeric
476                .maybe_to_used_value(basis)
477                .map_or(Size::Initial, Size::Numeric),
478            Size::FitContentFunction(numeric) => {
479                // Under discussion in https://github.com/w3c/csswg-drafts/issues/11805
480                numeric
481                    .maybe_to_used_value(basis)
482                    .map_or(Size::MaxContent, Size::FitContentFunction)
483            },
484            _ => self.map(|_| unreachable!("This shouldn't be called for keywords")),
485        }
486    }
487}

501impl Size<Au> {
502    /// Resolves a preferred size into a numerical value.
503    /// <https://www.w3.org/TR/css-sizing-3/#preferred-size-properties>
504    #[inline]
505    pub(crate) fn resolve_for_preferred<F: FnOnce() -> ContentSizes>(
506        &self,
507        automatic_size: Size<Au>,
508        stretch_size: Option<Au>,
509        content_size: &LazyCell<ContentSizes, F>,
510    ) -> Au {
511        match self {
512            Self::Initial => {
513                assert!(!automatic_size.is_initial());
514                automatic_size.resolve_for_preferred(automatic_size, stretch_size, content_size)
515            },
516            Self::MinContent => content_size.min_content,
517            Self::MaxContent => content_size.max_content,
518            Self::FitContentFunction(size) => content_size.shrink_to_fit(*size),
519            Self::FitContent => {
520                content_size.shrink_to_fit(stretch_size.unwrap_or_else(|| content_size.max_content))
521            },
522            Self::Stretch => stretch_size.unwrap_or_else(|| content_size.max_content),
523            Self::QuirkyStretch => stretch_size
524                .unwrap_or_else(|| content_size.max_content)
525                .max(content_size.min_content),
526            Self::Numeric(numeric) => *numeric,
527        }
528    }
529
530    /// Resolves a minimum size into a numerical value.
531    /// <https://www.w3.org/TR/css-sizing-3/#min-size-properties>
532    #[inline]
533    pub(crate) fn resolve_for_min<F: FnOnce() -> ContentSizes>(
534        &self,
535        get_automatic_minimum_size: impl FnOnce() -> Au,
536        stretch_size: Option<Au>,
537        content_size: &LazyCell<ContentSizes, F>,
538        is_table: bool,
539    ) -> Au {
540        let result = match self {
541            Self::Initial => get_automatic_minimum_size(),
542            Self::MinContent => content_size.min_content,
543            Self::MaxContent => content_size.max_content,
544            Self::FitContentFunction(size) => content_size.shrink_to_fit(*size),
545            Self::FitContent => content_size.shrink_to_fit(stretch_size.unwrap_or_default()),
546            Self::Stretch => stretch_size.unwrap_or_default(),
547            Self::QuirkyStretch => unreachable!("Quirky stretch can't be a min size"),
548            Self::Numeric(numeric) => *numeric,
549        };
550        if is_table {
551            // In addition to the specified minimum, the inline size of a table is forced to be
552            // at least as big as its min-content size.
553            //
554            // Note that if there are collapsed columns, only the inline size of the table grid will
555            // shrink, while the size of the table wrapper (being computed here) won't be affected.
556            // However, collapsed rows should typically affect the block size of the table wrapper,
557            // so it might be wrong to use this function for that case.
558            // This is being discussed in https://github.com/w3c/csswg-drafts/issues/11408
559            result.max(content_size.min_content)
560        } else {
561            result
562        }
563    }
564
565    /// Resolves a maximum size into a numerical value.
566    /// <https://www.w3.org/TR/css-sizing-3/#max-size-properties>
567    #[inline]
568    pub(crate) fn resolve_for_max<F: FnOnce() -> ContentSizes>(
569        &self,
570        stretch_size: Option<Au>,
571        content_size: &LazyCell<ContentSizes, F>,
572    ) -> Option<Au> {
573        Some(match self {
574            Self::Initial => return None,
575            Self::MinContent => content_size.min_content,
576            Self::MaxContent => content_size.max_content,
577            Self::FitContentFunction(size) => content_size.shrink_to_fit(*size),
578            Self::FitContent => content_size.shrink_to_fit(stretch_size.unwrap_or(MAX_AU)),
579            Self::Stretch => return stretch_size,
580            Self::QuirkyStretch => unreachable!("Quirky stretch can't be a max size"),
581            Self::Numeric(numeric) => *numeric,
582        })
583    }
584
585    /// Tries to resolve an extrinsic size into a numerical value.
586    /// Extrinsic sizes are those based on the context of an element, without regard for its contents.
587    /// <https://drafts.csswg.org/css-sizing-3/#extrinsic>
588    ///
589    /// Returns `None` if either:
590    /// - The size is intrinsic. A quirky stretch doesn't count as intrinsic here.
591    /// - The size is the initial one.
592    ///   TODO: should we allow it to behave as `stretch` instead of assuming it's intrinsic?
593    /// - The provided `stretch_size` is `None` but we need its value.
594    #[inline]
595    pub(crate) fn maybe_resolve_extrinsic(&self, stretch_size: Option<Au>) -> Option<Au> {
596        match self {
597            Self::Initial |
598            Self::MinContent |
599            Self::MaxContent |
600            Self::FitContent |
601            Self::FitContentFunction(_) => None,
602            Self::Stretch | Self::QuirkyStretch => stretch_size,
603            Self::Numeric(numeric) => Some(*numeric),
604        }
605    }
606}

608/// Represents the sizing constraint that the preferred, min and max sizing properties
609/// impose on one axis.
610#[derive(Clone, Copy, Debug, MallocSizeOf, PartialEq)]
611pub(crate) enum SizeConstraint {
612    /// Represents a definite preferred size, clamped by minimum and maximum sizes (if any).
613    Definite(Au),
614    /// Represents an indefinite preferred size that allows a range of values between
615    /// the first argument (minimum size) and the second one (maximum size).
616    MinMax(Au, Option<Au>),
617}
618
619impl Default for SizeConstraint {
620    #[inline]
621    fn default() -> Self {
622        Self::MinMax(Au::default(), None)
623    }
624}
625
626impl SizeConstraint {
627    #[inline]
628    pub(crate) fn new(preferred_size: Option<Au>, min_size: Au, max_size: Option<Au>) -> Self {
629        preferred_size.map_or_else(
630            || Self::MinMax(min_size, max_size),
631            |size| Self::Definite(size.clamp_between_extremums(min_size, max_size)),
632        )
633    }
634
635    #[inline]
636    pub(crate) fn is_definite(self) -> bool {
637        matches!(self, Self::Definite(_))
638    }
639
640    #[inline]
641    pub(crate) fn to_definite(self) -> Option<Au> {
642        match self {
643            Self::Definite(size) => Some(size),
644            _ => None,
645        }
646    }
647
648    #[inline]
649    pub(crate) fn definite_or_min(self) -> Au {
650        match self {
651            Self::Definite(size) => size,
652            Self::MinMax(min, _) => min,
653        }
654    }
655
656    #[inline]
657    pub(crate) fn replace_if_indefinite_with(&mut self, replacement: Self) {
658        if !self.is_definite() {
659            *self = replacement
660        }
661    }
662}
663
664impl From<Option<Au>> for SizeConstraint {
665    fn from(size: Option<Au>) -> Self {
666        size.map(SizeConstraint::Definite).unwrap_or_default()
667    }
668}

670#[derive(Clone, Debug, Default)]
671pub(crate) struct Sizes {
672    /// <https://drafts.csswg.org/css-sizing-3/#preferred-size-properties>
673    pub preferred: Size<Au>,
674    /// <https://drafts.csswg.org/css-sizing-3/#min-size-properties>
675    pub min: Size<Au>,
676    /// <https://drafts.csswg.org/css-sizing-3/#max-size-properties>
677    pub max: Size<Au>,
678}
679
680impl Sizes {
681    #[inline]
682    pub(crate) fn new(preferred: Size<Au>, min: Size<Au>, max: Size<Au>) -> Self {
683        Self {
684            preferred,
685            min,
686            max,
687        }
688    }
689
690    /// Resolves the three sizes into a single numerical value.
691    #[inline]
692    pub(crate) fn resolve(
693        &self,
694        axis: LogicalAxis,
695        automatic_size: Size<Au>,
696        get_automatic_minimum_size: impl FnOnce() -> Au,
697        stretch_size: Option<Au>,
698        get_content_size: impl FnOnce() -> ContentSizes,
699        is_table: bool,
700    ) -> Au {
701        if is_table && axis == LogicalAxis::Block {
702            // The intrinsic block size of a table already takes sizing properties into account,
703            // but it can be a smaller amount if there are collapsed rows.
704            // Therefore, disregard sizing properties and just defer to the intrinsic size.
705            // This is being discussed in https://github.com/w3c/csswg-drafts/issues/11408
706            return get_content_size().max_content;
707        }
708        let (preferred, min, max) = self.resolve_each(
709            automatic_size,
710            get_automatic_minimum_size,
711            stretch_size,
712            get_content_size,
713            is_table,
714        );
715        preferred.clamp_between_extremums(min, max)
716    }
717
718    /// Resolves each of the three sizes into a numerical value, separately.
719    /// - The 1st returned value is the resolved preferred size.
720    /// - The 2nd returned value is the resolved minimum size.
721    /// - The 3rd returned value is the resolved maximum size. `None` means no maximum.
722    #[inline]
723    pub(crate) fn resolve_each(
724        &self,
725        automatic_size: Size<Au>,
726        get_automatic_minimum_size: impl FnOnce() -> Au,
727        stretch_size: Option<Au>,
728        get_content_size: impl FnOnce() -> ContentSizes,
729        is_table: bool,
730    ) -> (Au, Au, Option<Au>) {
731        // The provided `get_content_size` is a FnOnce but we may need its result multiple times.
732        // A LazyCell will only invoke it once if needed, and then reuse the result.
733        let content_size = LazyCell::new(get_content_size);
734        (
735            self.preferred
736                .resolve_for_preferred(automatic_size, stretch_size, &content_size),
737            self.min.resolve_for_min(
738                get_automatic_minimum_size,
739                stretch_size,
740                &content_size,
741                is_table,
742            ),
743            self.max.resolve_for_max(stretch_size, &content_size),
744        )
745    }
746
747    /// Tries to extrinsically resolve the three sizes into a single [`SizeConstraint`].
748    /// Values that are intrinsic or need `stretch_size` when it's `None` are handled as such:
749    /// - On the preferred size, they make the returned value be an indefinite [`SizeConstraint::MinMax`].
750    /// - On the min size, they are treated as `auto`, enforcing the automatic minimum size.
751    /// - On the max size, they are treated as `none`, enforcing no maximum.
752    #[inline]
753    pub(crate) fn resolve_extrinsic(
754        &self,
755        automatic_size: Size<Au>,
756        automatic_minimum_size: Au,
757        stretch_size: Option<Au>,
758    ) -> SizeConstraint {
759        let (preferred, min, max) =
760            self.resolve_each_extrinsic(automatic_size, automatic_minimum_size, stretch_size);
761        SizeConstraint::new(preferred, min, max)
762    }
763
764    /// Tries to extrinsically resolve each of the three sizes into a numerical value, separately.
765    /// This can't resolve values that are intrinsic or need `stretch_size` but it's `None`.
766    /// - The 1st returned value is the resolved preferred size. If it can't be resolved then
767    ///   the returned value is `None`. Note that this is different than treating it as `auto`.
768    ///   TODO: This needs to be discussed in <https://github.com/w3c/csswg-drafts/issues/11387>.
769    /// - The 2nd returned value is the resolved minimum size. If it can't be resolved then we
770    ///   treat it as the initial `auto`, returning the automatic minimum size.
771    /// - The 3rd returned value is the resolved maximum size. If it can't be resolved then we
772    ///   treat it as the initial `none`, returning `None`.
773    #[inline]
774    pub(crate) fn resolve_each_extrinsic(
775        &self,
776        automatic_size: Size<Au>,
777        automatic_minimum_size: Au,
778        stretch_size: Option<Au>,
779    ) -> (Option<Au>, Au, Option<Au>) {
780        (
781            if self.preferred.is_initial() {
782                automatic_size.maybe_resolve_extrinsic(stretch_size)
783            } else {
784                self.preferred.maybe_resolve_extrinsic(stretch_size)
785            },
786            self.min
787                .maybe_resolve_extrinsic(stretch_size)
788                .unwrap_or(automatic_minimum_size),
789            self.max.maybe_resolve_extrinsic(stretch_size),
790        )
791    }
792}

794struct LazySizeData<'a> {
795    sizes: &'a Sizes,
796    axis: LogicalAxis,
797    automatic_size: Size<Au>,
798    get_automatic_minimum_size: fn() -> Au,
799    stretch_size: Option<Au>,
800    is_table: bool,
801}
802
803/// Represents a size that can't be fully resolved until the intrinsic size
804/// is known. This is useful in the block axis, since the intrinsic size
805/// depends on layout, but the other inputs are known beforehand.
806pub(crate) struct LazySize<'a> {
807    result: OnceCell<Au>,
808    data: Option<LazySizeData<'a>>,
809}
810
811impl<'a> LazySize<'a> {
812    pub(crate) fn new(
813        sizes: &'a Sizes,
814        axis: LogicalAxis,
815        automatic_size: Size<Au>,
816        get_automatic_minimum_size: fn() -> Au,
817        stretch_size: Option<Au>,
818        is_table: bool,
819    ) -> Self {
820        Self {
821            result: OnceCell::new(),
822            data: Some(LazySizeData {
823                sizes,
824                axis,
825                automatic_size,
826                get_automatic_minimum_size,
827                stretch_size,
828                is_table,
829            }),
830        }
831    }
832
833    /// Creates a [`LazySize`] that will resolve to the intrinsic size.
834    /// Should be equivalent to [`LazySize::new()`] with default parameters,
835    /// but avoiding the trouble of getting a reference to a [`Sizes::default()`]
836    /// which lives long enough.
837    ///
838    /// TODO: It's not clear what this should do if/when [`LazySize::resolve()`]
839    /// is changed to accept a [`ContentSizes`] as the intrinsic size.
840    pub(crate) fn intrinsic() -> Self {
841        Self {
842            result: OnceCell::new(),
843            data: None,
844        }
845    }
846
847    /// Resolves the [`LazySize`] into [`Au`], caching the result.
848    /// The argument is a callback that computes the intrinsic size lazily.
849    ///
850    /// TODO: The intrinsic size should probably be a [`ContentSizes`] instead of [`Au`].
851    pub(crate) fn resolve(&self, get_content_size: impl FnOnce() -> Au) -> Au {
852        *self.result.get_or_init(|| {
853            let Some(ref data) = self.data else {
854                return get_content_size();
855            };
856            data.sizes.resolve(
857                data.axis,
858                data.automatic_size,
859                data.get_automatic_minimum_size,
860                data.stretch_size,
861                || get_content_size().into(),
862                data.is_table,
863            )
864        })
865    }
866}
867
868impl From<Au> for LazySize<'_> {
869    /// Creates a [`LazySize`] that will resolve to the given [`Au`],
870    /// ignoring the intrinsic size.
871    fn from(value: Au) -> Self {
872        let result = OnceCell::new();
873        result.set(value).unwrap();
874        LazySize { result, data: None }
875    }
876}

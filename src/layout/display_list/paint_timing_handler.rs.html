66/// <https://wicg.github.io/container-timing/#get-the-container-root-element>

78impl ContainerTimingRootChain {
79    /// Returns the roots that apply to the contents of the fragment with the given `base`,
80    /// given the `inherited` roots that apply to the fragment itself.
81    pub(crate) fn for_contents_of(
82        inherited: &ContainerTimingRoots,
83        base: &BaseFragment,
84    ) -> ContainerTimingRoots {
85        let flags = base.flags;
86        let ignore = flags.contains(FragmentFlags::IS_CONTAINER_TIMING_IGNORE);
87        let root = base
88            .tag
89            .filter(|_| flags.contains(FragmentFlags::IS_CONTAINER_TIMING_ROOT))
90            .map(|tag| tag.node);
91
92        match root {
93            // An element can produce nested fragments (e.g. an inline box split around a
94            // block), so don't push the same root twice.
95            Some(root) if inherited.as_ref().is_some_and(|chain| chain.root == root) => {
96                inherited.clone()
97            },
98            // A root that is also ignored only reports to itself, not to enclosing roots.
99            Some(root) => Some(Rc::new(Self {
100                root,
101                parent: if ignore { None } else { inherited.clone() },
102            })),
103            // We're at an ignore point, so don't report to any enclosing roots.
104            None if ignore => None,
105            // Otherwise, just inherit the enclosing roots.
106            None => inherited.clone(),
107        }
108    }
109
110    /// Iterate over the roots in this chain, nearest first.
111    pub(crate) fn iter(&self) -> impl Iterator<Item = OpaqueNode> + '_ {
112        std::iter::successors(Some(self), |chain| chain.parent.as_deref()).map(|chain| chain.root)
113    }
114}

263    /// <https://www.w3.org/TR/paint-timing/#paintable>
264    pub(crate) fn paintable(&self, bounds: LayoutRect, opacity: f32) -> bool {
265        // An element el is paintable when all of the following apply:
266        // > el is being rendered.
267        // > el’s used visibility is visible.
268        // Note: Above conditions are met, as we selectively call this API.
269
270        // > el and all of its ancestors' used opacity is greater than zero.
271        if opacity <= 0.0 {
272            return false;
273        }
274
275        // > el’s paintable bounding rect intersects with the scrolling area of the document.
276        !self.paintable_bounding_rect(bounds).is_empty()
277    }

288    /// <https://www.w3.org/TR/largest-contentful-paint/#sec-effective-visual-size>
289    fn effective_visual_size(
290        &self,
291        intersection_rect: LayoutRect,
292        candidate_type: LCPCandidateType<'_>,
293    ) -> Option<f32> {
294        // Step 1. Let width be intersectionRect's width, rounded up to the
295        // nearest integer.
296        // Step 2. Let height be intersectionRect's height, rounded up to the
297        // nearest integer.
298        // Step 3. Let size be width * height.
299        let mut size = intersection_rect.area();
300
301        // Step 4. Let root be document's browsing context's top-level browsing
302        // context's active document.
303        // Note: This is not needed as we already have the viewport rect.
304
305        // Step 5. Let rootWidth be root's visual viewport's width,
306        // excluding any scrollbars.
307        // Step 6. Let rootHeight be root's visual viewport's height excluding
308        // any scrollbars.
309        // Step 7. If size is equal to rootWidth times rootHeight, return null.
310        if size >= self.viewport_rect.area() {
311            return None;
312        }
313
314        // Step 8: If imageRequest is not null, run the following steps to
315        // adjust for image position and upscaling.
316        // Note: This is skipped for Text aka the case of null request from specs
317        if let LCPCandidateType::Image(record) = candidate_type {
318            // TODO Step 8.1: If imageRequest's response's content length in bytes
319            // is less than size * 0.004, then return null. (Not Implemented)
320
321            // Step 8.2: Let concreteDimensions be imageRequest's concrete object
322            // size within element.
323            // Step 8.3: Let visibleDimensions be concreteDimensions, adjusted for
324            // positioning by object-position or background-position and element's
325            // content box.
326            // Note: bounds are already adjusted for positioning and content box
327            let visible_dimensions = record
328                .bounds
329                .intersection(&record.clip_rect)
330                .unwrap_or(LayoutRect::zero());
331
332            // Step 8.4: Let clientContentRect be the smallest DOMRectReadOnly
333            // containing visibleDimensions with element's transforms applied.
334            let client_content_rect =
335                transform_f32_rectangle(visible_dimensions.to_rect(), record.transform)
336                    .unwrap_or_default();
337
338            // Step 8.5: Let intersectingClientContentRect be the intersection of
339            // clientContentRect with intersectionRect.
340            let intersecting_client_content_rect = client_content_rect
341                .intersection(&intersection_rect.to_rect())
342                .unwrap_or(Rect::zero());
343
344            // Step 8.6: Set width to intersectingClientContentRect's width,
345            // rounded up to the nearest integer.
346            // Step 8.7: Set height to intersectingClientContentRect's height,
347            // rounded up to the nearest integer.
348            // Step 8.8: Set size to width * height.
349            size = intersecting_client_content_rect.area();
350
351            // Step 8.9: Let naturalArea be imageRequest's natural width * imageRequest's natural height.
352            if let (Some(natural_width), Some(natural_height)) =
353                (record.natural_width, record.natural_height)
354            {
355                let natural_area = natural_width.to_f32_px() * natural_height.to_f32_px();
356
357                // Step 8.10: If naturalArea is 0, then return null.
358                if natural_area == 0.0 {
359                    return None;
360                }
361                // Step 8.11: Let boundingClientArea be clientContentRect's width *
362                // clientContentRect's height.
363                let bounding_client_area =
364                    client_content_rect.width() * client_content_rect.height();
365
366                // Step 8.12: Let scaleFactor be boundingClientArea / naturalArea.
367                let scale_factor = bounding_client_area / natural_area;
368
369                // Step 8.13: If scaleFactor is greater than 1, then divide size by scaleFactor.
370                if scale_factor > 1.0 {
371                    size /= scale_factor;
372                }
373            }
374        }
375
376        // Step 9: Return an effective visual size result with size set to size,
377        // width set to width, and height set to height.
378        Some(size)
379    }

381    /// <https://www.w3.org/TR/largest-contentful-paint/#compute-a-new-largest-contentful-paint-candidate>
382    #[servo_tracing::instrument(
383        name = "Compute New LCP Candidate",
384        skip_all,
385        fields(
386            image_count = painted_images.len(),
387            text_count = painted_text_nodes.len(),
388        )
389    )]
390    fn compute_new_lcp_candidate(
391        &mut self,
392        painted_images: Vec<PendingImageRecord>,
393        painted_text_nodes: HashMap<OpaqueNode, TextRecord>,
394    ) -> Option<LCPCandidate> {
395        // Step 1. Let currentSize be currentCandidate’s size if
396        // currentCandidate is not null or 0 otherwise.
397        // Step 2. Let largestSize be currentSize.
398        let mut largest_size = self
399            .lcp_candidate
400            .as_ref()
401            .map_or(0.0, |candidate| candidate.area as f32);
402
403        // Step 3. Let newCandidate be null.
404        let mut new_candidate = None;
405
406        // Step 4. For each record of paintedImages:
407        for record in painted_images {
408            // Step 4.1. Let imageElement be record’s element.
409
410            // Step 4.2. If imageElement is not exposed for paint timing, given
411            // document, continue.
412            // Note: Satisfied, as the display-list builder only visits the
413            // connected DOM tree of the fully-active document being laid out.
414
415            // Step 4.3. Let intersectionRect be the value returned by the
416            // intersection rect algorithm using imageElement as the target
417            // and viewport as the root.
418            let intersection_rect =
419                transform_f32_rectangle(record.clip_rect.to_rect(), record.transform)
420                    .unwrap_or_default()
421                    .intersection(&self.viewport_rect.to_rect())
422                    .map(|rect| rect.to_box2d())
423                    .unwrap_or_default();
424
425            // Step 4.4. Let result be the effective visual size of imageElement
426            // given intersectionRect and record's request.
427            let result =
428                self.effective_visual_size(intersection_rect, LCPCandidateType::Image(&record));
429
430            // Step 4.5. If result is null, continue.
431            let Some(result) = result else {
432                continue;
433            };
434            // Step 4.6. If result's size is less than or equal to
435            // largestSize, continue.
436            if result <= largest_size {
437                continue;
438            }
439
440            // Step 4.7. Set largestSize to result’s size.
441            largest_size = result;
442
443            // Step 4.8. Set newCandidate to be a new largest contentful paint candidate ...
444            let uuid = self.lcp_next_uuid;
445            self.lcp_next_uuid += 1;
446            new_candidate = Some(LCPCandidate::new(
447                LCPCandidateID(uuid),
448                result as usize,
449                record.url,
450                record.tag.map(|tag| tag.node),
451            ));
452        }
453
454        // Step 5. For each textNode of paintedTextNodes,
455        for (_, record) in painted_text_nodes {
456            // Step 5.1. If textNode is not exposed for paint timing, given
457            // document, continue.
458            // Note: Satisfied, as the display-list builder only visits the
459            // connected DOM tree of the fully-active document being laid out.
460
461            // Step 5.2. If textNode has alpha channel value <=0 or opacity
462            // value <=0:
463            if record.style.get_color().alpha <= 0.0 || record.style.slow_clone_opacity() <= 0.0 {
464                // Step 5.2.1. If textNode's text-shadow value is none,
465                // textNode's stroke-color value is transparent and textNode's
466                // stroke-image value is none, continue.
467                // TODO: Update when we implement the `stroke-color`/`stroke-image`
468                // properties, as of now they are default (`transparent`/`none`)
469                if record.style.get_inherited_text().text_shadow.0.is_empty() {
470                    continue;
471                }
472            }
473            // Step 5.3. Let intersectionRect be the union of the border boxes of
474            // all Text nodes in textNode’s set of owned text nodes,
475            // intersected with the visual viewport.
476            let intersection_rect = record
477                .border_boxes
478                .into_iter()
479                .reduce(|a, b| a.union(&b))
480                .unwrap_or_default()
481                .intersection(&self.viewport_rect)
482                .unwrap_or_default();
483            // Step 5.4. Let result be the effective visual size of textNode
484            // given intersectionRect and null.
485            let result = self.effective_visual_size(intersection_rect, LCPCandidateType::Text);
486
487            // Step 5.5. If result is null, continue.
488            let Some(result) = result else {
489                continue;
490            };
491            // Step 5.6. If result's size is less than or equal to
492            // largestSize, continue.
493            if result <= largest_size {
494                continue;
495            }
496
497            // Step 5.7. Set largestSize to result’s size.
498            largest_size = result;
499
500            // Step 5.8. Set newCandidate to be a new largest contentful paint candidate ...
501            let uuid = self.lcp_next_uuid;
502            self.lcp_next_uuid += 1;
503            new_candidate = Some(LCPCandidate::new(
504                LCPCandidateID(uuid),
505                result as usize,
506                None,
507                Some(record.tag.node),
508            ));
509        }
510
511        // TODO Step 6. If newCandidate is not null and currentSize is greater than 0:
512        // TODO Step 6.1. If newCandidate’s width minus currentCandidate’s
513        // width is less than or equal to 3, and newCandidate’s height minus
514        // currentCandidate’s height is less than or equal to 3, return null.
515
516        // Step 7. Return newCandidate.
517        new_candidate
518    }

553    /// Accumulate a painted text or image fragment, painted by `node`, into the Container
554    /// Timing state of each of the enclosing `container_roots`.
555    ///
556    /// Called during display list building, unlike the LCP collection which only
557    /// accumulates here and computes in [`Self::mark_paint_timing`]. Container Timing has
558    /// no per-frame winner to pick: each fragment either adds new painted area to its
559    /// container or it does not, so the work can be done as fragments are visited.
560    ///
561    /// <https://wicg.github.io/container-timing/#maybe-update-last-new-painted-area>
562    #[servo_tracing::instrument(name = "Update Container Timing", skip_all)]
563    #[expect(clippy::too_many_arguments)]
564    pub(crate) fn update_container_timing(
565        &mut self,
566        container_roots: &ContainerTimingRootChain,
567        node: OpaqueNode,
568        bounds: LayoutRect,
569        clip_rect: LayoutRect,
570        transform: FastLayoutTransform,
571        natural_width: Option<Au>,
572        natural_height: Option<Au>,
573    ) {
574        let intersection_rect = transform_f32_rectangle(clip_rect.to_rect(), transform)
575            .unwrap_or_default()
576            .intersection(&self.viewport_rect.to_rect())
577            .map(|rect| rect.to_box2d())
578            .unwrap_or_default();
579
580        // Gate on the same "is this fragment eligible to be reported at all" check LCP
581        // uses.
582        let candidate_type = PendingImageRecord {
583            tag: None,
584            bounds,
585            clip_rect,
586            transform,
587            url: None,
588            natural_width,
589            natural_height,
590        };
591        let candidate_type = match (natural_width, natural_height) {
592            (Some(_), Some(_)) => LCPCandidateType::Image(&candidate_type),
593            _ => LCPCandidateType::Text,
594        };
595        if self
596            .effective_visual_size(intersection_rect, candidate_type)
597            .is_none()
598        {
599            return;
600        }
601
602        // Report to every enclosing root, not just the nearest one.
603        for container_root in container_roots.iter() {
604            let pending = self
605                .container_timing_pending
606                .entry(container_root)
607                .or_default();
608            if !pending.node_regions.contains_key(&node) {
609                pending.touched_order.push(node);
610            }
611            pending
612                .node_regions
613                .entry(node)
614                .or_default()
615                .union(intersection_rect);
616        }
617    }
618
619    /// A node only counts the first time it is ever seen contributing to a container
620    /// (tracked by [`ContainerRecord::contributing_nodes`]).
621    fn finalize_container_timing(&mut self) {
622        for (container_root, pending) in std::mem::take(&mut self.container_timing_pending) {
623            let record = self
624                .container_timing_records
625                .entry(container_root)
626                .or_insert_with(|| ContainerRecord {
627                    contributing_nodes: FxHashSet::default(),
628                    painted_region: PaintedRegion::default(),
629                    last_painted_element: None,
630                });
631
632            let mut grew = false;
633            for node in pending.touched_order {
634                if !record.contributing_nodes.insert(node) {
635                    continue;
636                }
637                record.painted_region.merge(&pending.node_regions[&node]);
638                record.last_painted_element = Some(node);
639                grew = true;
640            }
641
642            if grew {
643                self.dirty_containers.insert(container_root);
644            }
645        }
646    }
647
648    pub(crate) fn take_container_timing_records(&mut self) -> Vec<ContainerTimingRecord> {
649        self.finalize_container_timing();
650        let mut records = Vec::with_capacity(self.dirty_containers.len());
651        for container_root in std::mem::take(&mut self.dirty_containers) {
652            let Some(record) = self.container_timing_records.get(&container_root) else {
653                continue;
654            };
655            let id = ContainerTimingID(self.container_timing_next_uuid);
656            self.container_timing_next_uuid += 1;
657            records.push(ContainerTimingRecord {
658                id,
659                size: record.painted_region.area(),
660                intersection_rect: record.painted_region.bounds(),
661                root_element: container_root,
662                last_painted_element: record.last_painted_element,
663            });
664        }
665        records
666    }

668    /// <https://www.w3.org/TR/paint-timing/#first-paint>
669    fn should_report_first_paint(&self) -> bool {
670        // Step 1. If document's set of previously reported paints contains
671        // "first-paint", then return false.
672        if self
673            .previously_reported_paints
674            .contains(PaintTimingReport::FirstPaint)
675        {
676            return false;
677        }
678        // Step 2. If document contains at least one element that is
679        // paintable, then return true.
680        // Step 3. Otherwise, return false.
681        self.is_document_paintable
682    }

684    /// <https://www.w3.org/TR/paint-timing/#first-contentful-paint>
685    fn should_report_first_contentful_paint(&self) -> bool {
686        // Step 1. If document's set of previously reported paints contains
687        // "first-contentful-paint", then return false.
688        if self
689            .previously_reported_paints
690            .contains(PaintTimingReport::FirstContentfulPaint)
691        {
692            return false;
693        }
694        // Step 2. If document contains at least one element that is both
695        // paintable and contentful, then return true.
696        // Step 3. Otherwise, return false.
697        self.is_document_paintable && self.is_document_contentful
698    }

700    /// <https://www.w3.org/TR/paint-timing/#mark-paint-timing>
701    ///
702    /// Note: Step 10 is not in the current version of the specifications.
703    /// Refer <https://github.com/w3c/paint-timing/issues/122> for details on
704    /// the issue and for the modified steps yet to be merged.
705    #[servo_tracing::instrument(name = "Mark Paint Timing", skip_all, fields(halt_lcp = halt_lcp))]
706    pub(crate) fn mark_paint_timing(
707        &mut self,
708        paint_timing_eligible: bool,
709        halt_lcp: bool,
710    ) -> PaintTimingReport {
711        // Step 1. If the document's browsing context is not paint-timing
712        // eligible, return.
713        if !paint_timing_eligible {
714            return PaintTimingReport::default();
715        }
716
717        // Step 2. Let paintTimingInfo be a new paint timing info, whose
718        // rendering update end time is the current high resolution time given
719        // document's relevant global object.
720        // Note: This is satisfied in the script thread.
721
722        // Step 3. Let paintedImages be a new ordered set.
723        // Step 4. Let paintedTextNodes be a new ordered set.
724
725        // Step 5. For each record in doc's images pending rendering list:
726        // Step 5.1. If record's request is available and ready to be painted,
727        // then run the following steps:
728        // Note: Only available images are accumulated, hence it is fulfilled.
729        // Step 5.1.1. Append record to paintedImages.
730        // Step 5.1.2. Remove record from doc's images pending rendering list.
731        let painted_images: Vec<_> = std::mem::take(&mut self.images_pending_rendering)
732            .into_iter()
733            .filter(|record| {
734                record
735                    .tag
736                    .is_none_or(|tag| self.reported_image_nodes.insert(tag.node))
737            })
738            .collect();
739
740        // Step 6. For each Element element in doc's descendants:
741        // Step 6.1. If element is contained in doc's set of elements with
742        // rendered text, continue.
743        // Step 6.2. If element's set of owned text nodes is empty, continue.
744        // Step 6.3. Append element to doc's set of elements with rendered text.
745        // Step 6.4. Append element to paintedTextNodes.
746        let painted_text_nodes: HashMap<_, _> =
747            std::mem::take(&mut self.elements_with_pending_rendered_text)
748                .into_iter()
749                .filter(|(node, _record)| self.elements_with_rendered_text.insert(*node))
750                .collect();
751
752        // Step 7. Let reportedPaints be the document’s set of previously
753        // reported paints. (Directly accessing)
754
755        // TODO Step 8. Let frameTimingInfo be document’s current frame timing info.
756        // TODO Step 9. Set document’s current frame timing info to null.
757
758        // Step 10. Let flushPaintTimings be the following steps:
759
760        // Note: A new PaintTimingReport to accumulate the paints.
761        let mut paint_timing_report = PaintTimingReport::default();
762
763        // Step 10.1. If document should report first paint, then:
764        // Note: This step is not in the current version of the specifications,
765        // are waiting to be merged at w3c/paint-timing#123.
766        if self.should_report_first_paint() {
767            // Step 10.1.1. Report paint timing given document, "first-paint",
768            // and paintTimingInfo.
769            paint_timing_report.insert(PaintTimingReport::FirstPaint);
770        }
771
772        // Step 10.2. If document should report first contentful paint, then:
773        if self.should_report_first_contentful_paint() {
774            // Step 10.2.1. Report paint timing given document,
775            // "first-contentful-paint", and paintTimingInfo.
776            paint_timing_report.insert(PaintTimingReport::FirstContentfulPaint);
777        }
778
779        // Step 10.3. Report largest contentful paint given document,
780        // paintTimingInfo, paintedImages and paintedTextNodes.
781        self.report_largest_contentful_paint(halt_lcp, painted_images, painted_text_nodes);
782
783        // Note: Append the newly reported paints aka [`PaintTimingReport`] to
784        // the document's set of previously reported paints.
785        self.previously_reported_paints |= paint_timing_report;
786
787        paint_timing_report
788    }

56pub(crate) struct ContainingBlock {
57    /// The SpatialId of the spatial node that contains the children
58    /// of this containing block.
59    scroll_node_id: ScrollTreeNodeId,
60
61    /// The size of the parent scroll frame of this containing block, used for resolving
62    /// sticky margins. If this is None, then this is a direct descendant of a reference
63    /// frame and sticky positioning isn't taken into account.
64    scroll_frame_size: Option<LayoutSize>,
65
66    /// The [`ClipId`] to use for the children of this containing block.
67    clip_id: ClipId,
68
69    /// The physical rect of this containing block.
70    rect: PhysicalRect<Au>,
71
72    /// Normally containing block offsets and display list items are positioned relative
73    /// to their parent reference frame, but cumulative containing block boundaries on
74    /// fragments need to disregard reference frames entirely. This value tracks the
75    /// accumulated offset from the origin of the parent reference frame of this
76    /// containing block.
77    accumulated_reference_frame_offset: PhysicalVec<Au>,
78
79    /// Whether current containing block established a scroll frame.
80    established_scroll_frame: bool,
81}

129impl StackingContextTree {
130    /// Create a new [DisplayList] given the dimensions of the layout and the WebRender
131    /// pipeline id.
132    pub fn new(
133        fragment_tree: &FragmentTree,
134        viewport_details: ViewportDetails,
135        pipeline_id: wr::PipelineId,
136        first_reflow: bool,
137        debug: &DiagnosticsLogging,
138    ) -> Self {
139        let scrollable_overflow = fragment_tree.scrollable_overflow();
140        let scroll_area = scrollable_overflow.union(&fragment_tree.initial_containing_block);
141        let scroll_area = LayoutSize::from_untyped(Size2D::new(
142            scroll_area.size.width.to_f32_px(),
143            scroll_area.size.height.to_f32_px(),
144        ));
145
146        let viewport_size = viewport_details.layout_size();
147        let paint_info = PaintDisplayListInfo::new(
148            viewport_details,
149            scroll_area,
150            pipeline_id,
151            // This epoch is set when the WebRender display list is built. For now use a dummy value.
152            Default::default(),
153            fragment_tree.viewport_scroll_sensitivity,
154            first_reflow,
155        );
156
157        let root_scroll_node_id = paint_info.root_scroll_node_id;
158        let cb_for_non_fixed_descendants = ContainingBlock::new(
159            fragment_tree.initial_containing_block,
160            root_scroll_node_id,
161            Some(viewport_size),
162            ClipId::INVALID,
163            PhysicalVec::zero(),
164            true,
165        );
166        let cb_for_fixed_descendants = ContainingBlock::new(
167            fragment_tree.initial_containing_block,
168            paint_info.root_reference_frame_id,
169            None,
170            ClipId::INVALID,
171            PhysicalVec::zero(),
172            false,
173        );
174
175        // We need to specify all three containing blocks here, because absolute
176        // descendants of the root cannot share the containing block we specify
177        // for fixed descendants. In this case, they need to have the spatial
178        // id of the root scroll frame, whereas fixed descendants need the
179        // spatial id of the root reference frame so that they do not scroll with
180        // page content.
181        let containing_block_info = ContainingBlockInfo {
182            for_non_absolute_descendants: &cb_for_non_fixed_descendants,
183            for_absolute_descendants: Some(&cb_for_non_fixed_descendants),
184            for_absolute_and_fixed_descendants: &cb_for_fixed_descendants,
185        };
186
187        let mut stacking_context_tree = Self {
188            // This is just a temporary value that will be replaced once we have finished
189            // building the tree.
190            root_stacking_context: StackingContext::root(root_scroll_node_id),
191            paint_info,
192            clip_store: Default::default(),
193        };
194
195        let text_decorations = Default::default();
196        let mut root_stacking_context = StackingContext::root(root_scroll_node_id);
197        if let Some(root_box_fragment) = fragment_tree.root_box_fragment() {
198            Fragment::Box(root_box_fragment).build_stacking_context_tree(
199                &mut stacking_context_tree,
200                &containing_block_info,
201                &mut root_stacking_context,
202                // The root element might be absolutely positioned and we still want
203                // to process it, if it is.
204                StackingContextBuildMode::IncludeHoisted,
205                &text_decorations,
206                &None, /* container_timing_roots */
207                false, /* participates_in_3d_rendering_context */
208            );
209        }
210
211        root_stacking_context.sort();
212        stacking_context_tree.root_stacking_context = root_stacking_context;
213
214        if debug.is_enabled(DiagnosticsLoggingOption::StackingContextTree) {
215            stacking_context_tree
216                .root_stacking_context
217                .print(&mut PrintTree::new("Stacking Context Tree"));
218        }
219
220        stacking_context_tree
221    }

286    /// Given a [`Fragment`] and a point in the viewport of the page, return the point in
287    /// the [`Fragment`]'s content rectangle in its transformed coordinate system
288    /// (untransformed CSS pixels). Note that the point may be outside the [`Fragment`]'s
289    /// boundaries.
290    ///
291    /// TODO: Currently, this only works for [`BoxFragment`], but we should extend it to
292    /// other types of [`Fragment`]s in the future.
293    pub(crate) fn offset_in_fragment(
294        &self,
295        fragment: &Fragment,
296        point_in_viewport: PhysicalPoint<Au>,
297    ) -> Option<Point2D<Au, CSSPixel>> {
298        let fragment = fragment.retrieve_box_fragment()?;
299        let spatial_tree_node = fragment.spatial_tree_node()?;
300        let transform = self
301            .paint_info
302            .scroll_tree
303            .cumulative_root_to_node_transform(spatial_tree_node)?;
304        let transformed_point = transform
305            .project_point2d(point_in_viewport.map(Au::to_f32_px).cast_unit())?
306            .map(Au::from_f32_px)
307            .cast_unit();
308
309        // Find the origin of the fragment relative to its reference frame in the same coordinate system.
310        let reference_frame_origin = self
311            .paint_info
312            .scroll_tree
313            .reference_frame_offset(spatial_tree_node)
314            .map(Au::from_f32_px);
315        let fragment_origin = fragment
316            .cumulative_content_box_rect(
317                ContainingBlockCalculation::AlreadyDoneWithStackingContextTree,
318            )
319            .origin -
320            reference_frame_origin.cast_unit();
321
322        // Use that to find the offset from the fragment origin.
323        Some(transformed_point - fragment_origin)
324    }

354/// Either a stacking context or a stacking container, per the definitions in
355/// <https://drafts.csswg.org/css-position-4/#painting-order>.
356///
357/// We use the term “real stacking context” in situations that call for a stacking context
358/// but not a stacking container. Only positioned stacking containers every get a
359/// `StackingContext`. The rest are handled inline during the `PaintTraversal`.
360#[derive(MallocSizeOf)]
361pub struct StackingContext {
362    /// The [`BoxFragment`] that established this stacking context. This is used to paint this
363    /// [`StackingContext`] and traverse its descendants for painting.
364    ///
365    /// This is `None` for the root stacking context.
366    pub(crate) fragment: StackingContextFragments,
367
368    /// The [`StackingContextType`] of this [`StackingContet`], which determines if it
369    /// a stacking context or a stacking container.
370    pub(crate) context_type: StackingContextType,
371
372    /// Child [`StackingContext`]s of this [`StackingContext`].
373    pub(crate) children: Vec<StackingContext>,
374
375    /// The offset of the containing block, used to properly paint child fragments of this
376    /// stacking context or stacking container.
377    pub(crate) containing_block_origin: PhysicalPoint<Au>,
378
379    /// The spatial id of this [`StackingContext`].
380    pub(crate) scroll_tree_node_id: ScrollTreeNodeId,
381
382    /// The clip id of this [`StackingContext`] if it has one.
383    pub(crate) clip_id: ClipId,
384
385    /// The z-index of this [`StackingContext`]. Note that `auto` is represented as 0.
386    pub(crate) z_index: i32,
387
388    /// The text decorations that apply to this [`StackingContext`] propagated via the box tree.
389    #[conditional_malloc_size_of]
390    pub(crate) text_decorations: Rc<Vec<FragmentTextDecoration>>,
391
392    /// The Container Timing roots that enclose this [`StackingContext`] in tree order,
393    /// not including its own root fragment. Stacking contexts are painted out of tree
394    /// order, so these are captured while building the stacking context tree.
395    #[ignore_malloc_size_of = "Small and shared with other stacking contexts"]
396    pub(crate) container_timing_roots: ContainerTimingRoots,
397
398    /// If this [`StackingContext`] also created a WebRender reference frame, this field
399    /// holds information about that reference frame.
400    pub(crate) reference_frame_info: Option<StackingContextReferenceFrameInfo>,
401
402    /// Whether or not this [`StackingContext`] participates in a 3D rendering context.
403    /// See <https://www.w3.org/TR/css-transforms-2/#3d-rendering-contexts>
404    pub(crate) participates_in_a_3d_rendering_context: bool,
405}

444            children: Default::default(),
445            scroll_tree_node_id: spatial_id,
446            clip_id,
447            z_index,
448            text_decorations,
449            container_timing_roots,
450            reference_frame_info,
451            participates_in_a_3d_rendering_context,
452        }
453    }
454
455    pub(crate) fn fragment(&self) -> Option<&Arc<BoxFragment>> {
456        match &self.fragment {
457            StackingContextFragments::Root => None,
458            StackingContextFragments::Fragment(box_fragment) => Some(box_fragment),
459        }
460    }
461
462    fn sort(&mut self) {
463        self.children.sort_by_key(|child| child.z_index)
464    }
465
466    fn print(&self, tree: &mut PrintTree) {
467        let fragment_string = match &self.fragment {
468            StackingContextFragments::Root => "Root".into(),
469            StackingContextFragments::Fragment(box_fragment) => format!(
470                "{:?} rect={:?}",
471                box_fragment.base.tag,
472                box_fragment.content_rect()
473            ),
474        };
475
476        tree.new_level(format!(
477            "{fragment_string} z-index={:?} spatial={:?} clip={:?}",
478            self.z_index, self.scroll_tree_node_id, self.clip_id
479        ));
480
481        for child in self.children.iter() {
482            child.print(tree);
483        }
484
485        tree.end_level();
486    }

490pub(crate) enum StackingContextBuildMode {
491    IncludeHoisted,
492    SkipHoisted,
493}

495impl Fragment {
496    #[expect(clippy::too_many_arguments)]
497    pub(crate) fn build_stacking_context_tree(
498        &self,
499        stacking_context_tree: &mut StackingContextTree,
500        containing_block_info: &ContainingBlockInfo,
501        stacking_context: &mut StackingContext,
502        mode: StackingContextBuildMode,
503        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
504        container_timing_roots: &ContainerTimingRoots,
505        participates_in_3d_rendering_context: bool,
506    ) {
507        let containing_block = containing_block_info.get_containing_block_for_fragment(self);
508        let cumulative_containing_block = containing_block
509            .rect
510            .translate(containing_block.accumulated_reference_frame_offset);
511        self.set_containing_block(&cumulative_containing_block);
512
513        if self
514            .base()
515            .is_some_and(|base| base.flags.contains(FragmentFlags::IS_COLLAPSED))
516        {
517            return;
518        }
519
520        let fragment_clone = self.clone();
521        match self {
522            Fragment::Box(fragment) | Fragment::Float(fragment) => {
523                if mode == StackingContextBuildMode::SkipHoisted &&
524                    fragment
525                        .style()
526                        .get_box()
527                        .position
528                        .is_absolutely_positioned()
529                {
530                    return;
531                }
532
533                let text_decorations = match self {
534                    Fragment::Float(..) => &Default::default(),
535                    _ => text_decorations,
536                };
537
538                fragment.build_stacking_context_tree(
539                    fragment_clone,
540                    stacking_context_tree,
541                    containing_block,
542                    containing_block_info,
543                    stacking_context,
544                    text_decorations,
545                    container_timing_roots,
546                    participates_in_3d_rendering_context,
547                );
548            },
549            Fragment::LayoutRoot(..) => {
550                // These fragments are processed at their originating
551                // `Fragment::AbsoluteOrFixedPositionedPlaceholder` position.
552            },
553            Fragment::AbsoluteOrFixedPositionedPlaceholder(fragment) => {
554                let shared_fragment = fragment.borrow();
555                let fragment_ref = match shared_fragment.fragment.as_ref() {
556                    Some(fragment_ref) => fragment_ref,
557                    None => unreachable!("Found hoisted box with missing fragment."),
558                };
559
560                fragment_ref.build_stacking_context_tree(
561                    stacking_context_tree,
562                    containing_block_info,
563                    stacking_context,
564                    StackingContextBuildMode::IncludeHoisted,
565                    &Default::default(),
566                    container_timing_roots,
567                    participates_in_3d_rendering_context,
568                );
569            },
570            Fragment::Positioning(fragment) => {
571                fragment.build_stacking_context_tree(
572                    stacking_context_tree,
573                    containing_block,
574                    containing_block_info,
575                    stacking_context,
576                    text_decorations,
577                    container_timing_roots,
578                    participates_in_3d_rendering_context,
579                );
580            },
581            Fragment::Text(_) | Fragment::Image(_) | Fragment::IFrame(_) => {},
582        }
583    }
584}

601impl BoxFragment {
602    pub(crate) fn stacking_context_type(&self) -> Option<StackingContextType> {
603        let flags = self.base.flags;
604        let style = self.style();
605        if style.establishes_stacking_context(flags) {
606            return Some(StackingContextType::StackingContext);
607        }
608
609        let box_style = &style.get_box();
610        if box_style.position != ComputedPosition::Static {
611            return Some(StackingContextType::StackingContainer);
612        }
613
614        None
615    }
616
617    #[expect(clippy::too_many_arguments)]
618    fn build_stacking_context_tree(
619        self: &Arc<Self>,
620        fragment: Fragment,
621        stacking_context_tree: &mut StackingContextTree,
622        containing_block: &ContainingBlock,
623        containing_block_info: &ContainingBlockInfo,
624        parent_stacking_context: &mut StackingContext,
625        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
626        container_timing_roots: &ContainerTimingRoots,
627        participates_in_3d_rendering_context: bool,
628    ) {
629        self.clear_stacking_context_tree_traversal_data();
630        self.build_stacking_context_tree_maybe_creating_reference_frame(
631            fragment,
632            stacking_context_tree,
633            containing_block,
634            containing_block_info,
635            parent_stacking_context,
636            text_decorations,
637            container_timing_roots,
638            participates_in_3d_rendering_context,
639        );
640    }
641
642    #[expect(clippy::too_many_arguments)]
643    fn build_stacking_context_tree_maybe_creating_reference_frame(
644        self: &Arc<Self>,
645        fragment: Fragment,
646        stacking_context_tree: &mut StackingContextTree,
647        containing_block: &ContainingBlock,
648        containing_block_info: &ContainingBlockInfo,
649        parent_stacking_context: &mut StackingContext,
650        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
651        container_timing_roots: &ContainerTimingRoots,
652        participates_in_3d_rendering_context: bool,
653    ) {
654        let reference_frame_data =
655            match self.reference_frame_data_if_necessary(&containing_block.rect) {
656                Some(reference_frame_data) => reference_frame_data,
657                None => {
658                    return self.build_stacking_context_tree_maybe_creating_stacking_context(
659                        fragment,
660                        stacking_context_tree,
661                        containing_block,
662                        containing_block_info,
663                        parent_stacking_context,
664                        text_decorations,
665                        container_timing_roots,
666                        None, /* reference_frame_info */
667                        participates_in_3d_rendering_context,
668                    );
669                },
670            };
671
672        // <https://drafts.csswg.org/css-transforms/#transform-function-lists>
673        // > If a transform function causes the current transformation matrix of an object
674        // > to be non-invertible, the object and its content do not get displayed.
675        if !reference_frame_data.transform.is_invertible() {
676            self.clear_stacking_context_tree_traversal_data_recursively();
677            return;
678        }
679
680        let style = self.style();
681        let frame_origin_for_query = self
682            .cumulative_border_box_rect(
683                ContainingBlockCalculation::AlreadyDoneWithStackingContextTree,
684            )
685            .origin
686            .to_webrender();
687        let new_spatial_id = stacking_context_tree.push_reference_frame(
688            reference_frame_data.origin.to_webrender(),
689            frame_origin_for_query,
690            containing_block.scroll_node_id,
691            style.used_transform_style(self.base.flags).to_webrender(),
692            reference_frame_data.transform,
693            reference_frame_data.kind,
694        );
695
696        // WebRender reference frames establish a new coordinate system at their
697        // origin (the border box of the fragment). We need to ensure that any
698        // coordinates we give to WebRender in this reference frame are relative
699        // to the fragment border box. We do this by adjusting the containing
700        // block origin. Note that the `for_absolute_descendants` and
701        // `for_all_absolute_and_fixed_descendants` properties are now bogus,
702        // but all fragments that establish reference frames also establish
703        // containing blocks for absolute and fixed descendants, so those
704        // properties will be replaced before recursing into children.
705        assert!(style.establishes_containing_block_for_all_descendants(self.base.flags));
706        let reference_frame_offset = reference_frame_data.origin.to_vector();
707        let adjusted_containing_block = ContainingBlock::new(
708            containing_block.rect.translate(-reference_frame_offset),
709            new_spatial_id,
710            None,
711            ClipId::INVALID,
712            containing_block.accumulated_reference_frame_offset + reference_frame_offset,
713            false,
714        );
715        let new_containing_block_info =
716            containing_block_info.new_for_non_absolute_descendants(&adjusted_containing_block);
717
718        let reference_frame_info = StackingContextReferenceFrameInfo {
719            parent_spatial_node_id: containing_block.scroll_node_id,
720            captured_clip_id: containing_block.clip_id,
721        };
722
723        self.build_stacking_context_tree_maybe_creating_stacking_context(
724            fragment,
725            stacking_context_tree,
726            &adjusted_containing_block,
727            &new_containing_block_info,
728            parent_stacking_context,
729            text_decorations,
730            container_timing_roots,
731            Some(reference_frame_info),
732            participates_in_3d_rendering_context,
733        );
734    }
735
736    #[expect(clippy::too_many_arguments)]
737    fn build_stacking_context_tree_maybe_creating_stacking_context(
738        self: &Arc<Self>,
739        fragment: Fragment,
740        stacking_context_tree: &mut StackingContextTree,
741        containing_block: &ContainingBlock,
742        containing_block_info: &ContainingBlockInfo,
743        parent_stacking_context: &mut StackingContext,
744        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
745        container_timing_roots: &ContainerTimingRoots,
746        reference_frame_info: Option<StackingContextReferenceFrameInfo>,
747        participates_in_3d_rendering_context: bool,
748    ) {
749        let with_style = &self.with_style();
750        let style = with_style.style();
751        let Some(stacking_context_type) = self.stacking_context_type() else {
752            with_style.build_stacking_context_tree_for_children(
753                stacking_context_tree,
754                containing_block,
755                containing_block_info,
756                parent_stacking_context,
757                text_decorations,
758                container_timing_roots,
759                participates_in_3d_rendering_context,
760            );
761            return;
762        };
763
764        let new_scroll_frame_size = containing_block_info
765            .for_non_absolute_descendants
766            .scroll_frame_size;
767        let spatial_id = self.build_sticky_frame_if_necessary(
768            stacking_context_tree,
769            containing_block.scroll_node_id,
770            &containing_block.rect,
771            &new_scroll_frame_size,
772            containing_block.established_scroll_frame,
773        );
774
775        let clip_id = with_style.build_clip_frame_if_necessary(
776            stacking_context_tree,
777            spatial_id.unwrap_or(containing_block.scroll_node_id),
778            containing_block.clip_id,
779            &containing_block.rect,
780        );
781
782        let clip_id = stacking_context_tree
783            .clip_store
784            .add_for_clip_path(
785                &style.get_svg().clip_path,
786                spatial_id.unwrap_or(containing_block.scroll_node_id),
787                clip_id.unwrap_or(containing_block.clip_id),
788                with_style,
789                containing_block.rect.origin,
790            )
791            .or(clip_id);
792
793        let containing_block = if clip_id.is_some() || spatial_id.is_some() {
794            if let Some(clip_id) = clip_id {
795                self.set_generated_clip_id(clip_id);
796            }
797            if let Some(spatial_id) = spatial_id {
798                self.set_generated_scroll_tree_node_id(spatial_id);
799            }
800            &ContainingBlock {
801                scroll_node_id: spatial_id.unwrap_or(containing_block.scroll_node_id),
802                clip_id: clip_id.unwrap_or(containing_block.clip_id),
803                ..*containing_block
804            }
805        } else {
806            containing_block
807        };
808
809        let box_fragment = fragment
810            .retrieve_box_fragment()
811            .expect("Should never try to make stacking context for non-BoxFragment")
812            .clone();
813
814        let mut child_stacking_context = parent_stacking_context.create_descendant(
815            stacking_context_type,
816            containing_block.rect.origin,
817            containing_block.scroll_node_id,
818            containing_block.clip_id,
819            box_fragment,
820            text_decorations.clone(),
821            container_timing_roots.clone(),
822            reference_frame_info,
823            participates_in_3d_rendering_context,
824        );
825        with_style.build_stacking_context_tree_for_children(
826            stacking_context_tree,
827            containing_block,
828            containing_block_info,
829            &mut child_stacking_context,
830            text_decorations,
831            container_timing_roots,
832            participates_in_3d_rendering_context,
833        );
834
835        let mut stolen_children = vec![];
836        if stacking_context_type != StackingContextType::StackingContext {
837            stolen_children =
838                std::mem::replace(&mut child_stacking_context.children, stolen_children);
839        } else {
840            child_stacking_context.sort();
841        }
842
843        parent_stacking_context
844            .children
845            .push(child_stacking_context);
846        parent_stacking_context
847            .children
848            .append(&mut stolen_children);
849    }
850}

852impl BoxFragmentWithStyle<'_> {
853    #[expect(clippy::too_many_arguments)]
854    fn build_stacking_context_tree_for_children(
855        &self,
856        stacking_context_tree: &mut StackingContextTree,
857        containing_block: &ContainingBlock,
858        containing_block_info: &ContainingBlockInfo,
859        stacking_context: &mut StackingContext,
860        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
861        container_timing_roots: &ContainerTimingRoots,
862        participates_in_3d_rendering_context: bool,
863    ) {
864        let style = self.style();
865        let establishes_containing_block_for_all_descendants =
866            style.establishes_containing_block_for_all_descendants(self.base.flags);
867        let establishes_containing_block_for_absolute_descendants =
868            style.establishes_containing_block_for_absolute_descendants(self.base.flags);
869
870        let mut new_scroll_node_id = containing_block.scroll_node_id;
871        self.spatial_tree_node.set(Some(new_scroll_node_id));
872
873        // We want to build the scroll frame after the background and border, because
874        // they shouldn't scroll with the rest of the box content.
875        let mut new_scroll_frame_size = containing_block_info
876            .for_non_absolute_descendants
877            .scroll_frame_size;
878
879        let mut established_scroll_frame = containing_block.established_scroll_frame;
880        // We want to skip leaving scroll frame for anonymous fragments,
881        // because anonymous fragment belong to same node as its closest non-anonymous child.
882        if established_scroll_frame && !self.base.is_anonymous() {
883            established_scroll_frame = false;
884        }
885        let mut new_clip_id = containing_block.clip_id;
886        if let Some(overflow_frame_data) = self.build_overflow_frame_if_necessary(
887            stacking_context_tree,
888            new_scroll_node_id,
889            new_clip_id,
890            &containing_block.rect,
891        ) {
892            new_clip_id = overflow_frame_data.clip_id;
893            self.set_generated_clip_id(new_clip_id);
894
895            if let Some(scroll_frame_data) = overflow_frame_data.scroll_frame_data {
896                new_scroll_node_id = scroll_frame_data.scroll_tree_node_id;
897                new_scroll_frame_size = Some(scroll_frame_data.scroll_frame_rect.size());
898                established_scroll_frame = true;
899                self.set_generated_scroll_tree_node_id(new_scroll_node_id);
900            }
901        }
902
903        let padding_rect = self
904            .padding_rect()
905            .translate(containing_block.rect.origin.to_vector());
906        let content_rect = self
907            .content_rect()
908            .translate(containing_block.rect.origin.to_vector());
909
910        let for_absolute_descendants = ContainingBlock::new(
911            padding_rect,
912            new_scroll_node_id,
913            new_scroll_frame_size,
914            new_clip_id,
915            containing_block.accumulated_reference_frame_offset,
916            established_scroll_frame,
917        );
918        let for_non_absolute_descendants = ContainingBlock::new(
919            content_rect,
920            new_scroll_node_id,
921            new_scroll_frame_size,
922            new_clip_id,
923            containing_block.accumulated_reference_frame_offset,
924            established_scroll_frame,
925        );
926
927        // Create a new `ContainingBlockInfo` for descendants depending on
928        // whether or not this fragment establishes a containing block for
929        // absolute and fixed descendants.
930        let new_containing_block_info = if establishes_containing_block_for_all_descendants {
931            containing_block_info.new_for_absolute_and_fixed_descendants(
932                &for_non_absolute_descendants,
933                &for_absolute_descendants,
934            )
935        } else if establishes_containing_block_for_absolute_descendants {
936            containing_block_info.new_for_absolute_descendants(
937                &for_non_absolute_descendants,
938                &for_absolute_descendants,
939            )
940        } else {
941            containing_block_info.new_for_non_absolute_descendants(&for_non_absolute_descendants)
942        };
943
944        // Text decorations are not propagated to atomic inline-level descendants.
945        // From https://drafts.csswg.org/css2/#lining-striking-props:
946        // > Note that text decorations are not propagated to floating and absolutely
947        // > positioned descendants, nor to the contents of atomic inline-level descendants
948        // > such as inline blocks and inline tables.
949        let text_decorations = match self.is_atomic_inline_level() ||
950            self.base
951                .flags
952                .contains(FragmentFlags::IS_OUTSIDE_LIST_ITEM_MARKER)
953        {
954            true => &Default::default(),
955            false => text_decorations,
956        };
957
958        let new_text_decoration;
959        let text_decorations = match style.get_text_decoration_line() {
960            &TextDecorationLine::NONE => text_decorations,
961            line => {
962                let mut new_vector = (**text_decorations).clone();
963                let color = style.get_color();
964                new_vector.push(FragmentTextDecoration {
965                    line: *line,
966                    color: style.get_text_decoration_color().resolve_to_absolute(color),
967                    style: style.slow_clone_text_decoration_style(),
968                    thickness: style.slow_clone_text_decoration_thickness(),
969                });
970                new_text_decoration = Rc::new(new_vector);
971                &new_text_decoration
972            },
973        };
974
975        let participates_in_3d_rendering_context = if self.base.is_anonymous() {
976            participates_in_3d_rendering_context
977        } else {
978            self.style()
979                .establishes_or_extends_3d_rendering_context(self.base.flags)
980        };
981
982        let container_timing_roots =
983            ContainerTimingRootChain::for_contents_of(container_timing_roots, &self.base);
984
985        for child in &self.children {
986            child.build_stacking_context_tree(
987                stacking_context_tree,
988                &new_containing_block_info,
989                stacking_context,
990                StackingContextBuildMode::SkipHoisted,
991                text_decorations,
992                &container_timing_roots,
993                participates_in_3d_rendering_context,
994            );
995        }
996    }

998    fn build_clip_frame_if_necessary(
999        &self,
1000        stacking_context_tree: &mut StackingContextTree,
1001        parent_scroll_node_id: ScrollTreeNodeId,
1002        parent_clip_id: ClipId,
1003        containing_block_rect: &PhysicalRect<Au>,
1004    ) -> Option<ClipId> {
1005        let style = self.style();
1006        let position = style.get_box().position;
1007        // https://drafts.csswg.org/css2/#clipping
1008        // The clip property applies only to absolutely positioned elements
1009        if !position.is_absolutely_positioned() {
1010            return None;
1011        }
1012
1013        // Only rectangles are supported for now.
1014        let clip_rect = match style.get_effects().clip {
1015            ClipRectOrAuto::Rect(rect) => rect,
1016            _ => return None,
1017        };
1018
1019        let border_rect = self.border_rect();
1020        let clip_rect = clip_rect
1021            .for_border_rect(border_rect)
1022            .translate(containing_block_rect.origin.to_vector())
1023            .to_webrender();
1024        Some(stacking_context_tree.clip_store.add(
1025            BorderRadius::zero(),
1026            clip_rect,
1027            parent_scroll_node_id,
1028            parent_clip_id,
1029        ))
1030    }

1032    fn build_overflow_frame_if_necessary(
1033        &self,
1034        stacking_context_tree: &mut StackingContextTree,
1035        parent_scroll_node_id: ScrollTreeNodeId,
1036        parent_clip_id: ClipId,
1037        containing_block_rect: &PhysicalRect<Au>,
1038    ) -> Option<OverflowFrameData> {
1039        let style = self.style();
1040        let overflow = style.effective_overflow(self.base.flags);
1041
1042        if overflow.x == ComputedOverflow::Visible && overflow.y == ComputedOverflow::Visible {
1043            return None;
1044        }
1045
1046        // Non-scrollable overflow path
1047        if overflow.x == ComputedOverflow::Clip || overflow.y == ComputedOverflow::Clip {
1048            let overflow_clip_margin = style.get_margin().overflow_clip_margin;
1049            let mut overflow_clip_rect = match overflow_clip_margin.visual_box {
1050                OverflowClipMarginBox::ContentBox => self.content_rect(),
1051                OverflowClipMarginBox::PaddingBox => self.padding_rect(),
1052                OverflowClipMarginBox::BorderBox => self.border_rect(),
1053            }
1054            .translate(containing_block_rect.origin.to_vector())
1055            .to_webrender();
1056
1057            // Adjust by the overflow clip margin.
1058            // https://drafts.csswg.org/css-overflow-3/#overflow-clip-margin
1059            let clip_margin_offset = overflow_clip_margin.offset.px();
1060            overflow_clip_rect = overflow_clip_rect.inflate(clip_margin_offset, clip_margin_offset);
1061
1062            // The clipping region only gets rounded corners if both axes have `overflow: clip`.
1063            // https://drafts.csswg.org/css-overflow-3/#corner-clipping
1064            let radii;
1065            if overflow.x == ComputedOverflow::Clip && overflow.y == ComputedOverflow::Clip {
1066                let builder = BuilderForBoxFragment::new(self, containing_block_rect.origin);
1067                let mut offsets_from_border = SideOffsets2D::new_all_same(clip_margin_offset);
1068                match overflow_clip_margin.visual_box {
1069                    OverflowClipMarginBox::ContentBox => {
1070                        offsets_from_border -= (self.border + self.padding).to_webrender();
1071                    },
1072                    OverflowClipMarginBox::PaddingBox => {
1073                        offsets_from_border -= self.border.to_webrender();
1074                    },
1075                    OverflowClipMarginBox::BorderBox => {},
1076                };
1077                radii = offset_radii(builder.border_radius(), offsets_from_border);
1078            } else if overflow.x != ComputedOverflow::Clip {
1079                let max = LayoutRect::max_rect();
1080                overflow_clip_rect.min.x = max.min.x;
1081                overflow_clip_rect.max.x = max.max.x;
1082                radii = BorderRadius::zero();
1083            } else {
1084                let max = LayoutRect::max_rect();
1085                overflow_clip_rect.min.y = max.min.y;
1086                overflow_clip_rect.max.y = max.max.y;
1087                radii = BorderRadius::zero();
1088            }
1089
1090            let clip_id = stacking_context_tree.clip_store.add(
1091                radii,
1092                overflow_clip_rect,
1093                parent_scroll_node_id,
1094                parent_clip_id,
1095            );
1096
1097            return Some(OverflowFrameData {
1098                clip_id,
1099                scroll_frame_data: None,
1100            });
1101        }
1102
1103        let scroll_frame_rect = self
1104            .padding_rect()
1105            .translate(containing_block_rect.origin.to_vector())
1106            .to_webrender();
1107
1108        let clip_id = stacking_context_tree.clip_store.add(
1109            BuilderForBoxFragment::new(self, containing_block_rect.origin).border_radius(),
1110            scroll_frame_rect,
1111            parent_scroll_node_id,
1112            parent_clip_id,
1113        );
1114
1115        let tag = self.base.tag?;
1116        let external_scroll_id = wr::ExternalScrollId(
1117            tag.to_display_list_fragment_id(),
1118            stacking_context_tree.paint_info.pipeline_id,
1119        );
1120
1121        let mut x_sensitivity: ScrollType = overflow.x.into();
1122        let mut y_sensitivity: ScrollType = overflow.y.into();
1123        let touch_action = TouchAction::from(style.get_box().touch_action);
1124        // `touch-action` only restricts direct touch manipulation; mouse wheel
1125        // (`InputEvents`) and script-driven scrolling are unaffected, so we only
1126        // strip `ScrollType::Touch` from the excluded axis.
1127        match touch_action {
1128            TouchAction::PanX => {
1129                y_sensitivity.remove(ScrollType::Touch);
1130            },
1131            TouchAction::PanY => {
1132                x_sensitivity.remove(ScrollType::Touch);
1133            },
1134            TouchAction::None => {
1135                x_sensitivity.remove(ScrollType::Touch);
1136                y_sensitivity.remove(ScrollType::Touch);
1137            },
1138            TouchAction::Auto => {},
1139        }
1140        let sensitivity = AxesScrollSensitivity {
1141            x: x_sensitivity,
1142            y: y_sensitivity,
1143        };
1144
1145        let scroll_tree_node_id = stacking_context_tree.define_scroll_frame(
1146            parent_scroll_node_id,
1147            external_scroll_id,
1148            self.scrollable_overflow().to_webrender(),
1149            scroll_frame_rect,
1150            sensitivity,
1151            touch_action,
1152        );
1153
1154        Some(OverflowFrameData {
1155            clip_id,
1156            scroll_frame_data: Some(ScrollFrameData {
1157                scroll_tree_node_id,
1158                scroll_frame_rect,
1159            }),
1160        })
1161    }
1162}

1164impl BoxFragment {
1165    fn build_sticky_frame_if_necessary(
1166        &self,
1167        stacking_context_tree: &mut StackingContextTree,
1168        parent_scroll_node_id: ScrollTreeNodeId,
1169        containing_block_rect: &PhysicalRect<Au>,
1170        scroll_frame_size: &Option<LayoutSize>,
1171        established_scroll_frame: bool,
1172    ) -> Option<ScrollTreeNodeId> {
1173        let style = self.style();
1174        if style.get_box().position != ComputedPosition::Sticky {
1175            return None;
1176        }
1177
1178        let scroll_frame_size_for_resolve = match scroll_frame_size {
1179            Some(size) => size,
1180            None => {
1181                // This is a direct descendant of a reference frame.
1182                &stacking_context_tree
1183                    .paint_info
1184                    .viewport_details
1185                    .layout_size()
1186            },
1187        };
1188
1189        // Percentages sticky positions offsets are resovled against the size of the
1190        // nearest scroll frame instead of the containing block like for other types
1191        // of positioning.
1192        let scroll_frame_height = Au::from_f32_px(scroll_frame_size_for_resolve.height);
1193        let scroll_frame_width = Au::from_f32_px(scroll_frame_size_for_resolve.width);
1194        let offsets = style.physical_box_offsets();
1195        let offsets = PhysicalSides::<AuOrAuto>::new(
1196            offsets.top.map(|v| v.to_used_value(scroll_frame_height)),
1197            offsets.right.map(|v| v.to_used_value(scroll_frame_width)),
1198            offsets.bottom.map(|v| v.to_used_value(scroll_frame_height)),
1199            offsets.left.map(|v| v.to_used_value(scroll_frame_width)),
1200        );
1201        self.set_resolved_sticky_insets(offsets);
1202
1203        if scroll_frame_size.is_none() {
1204            return None;
1205        }
1206
1207        if offsets.top.is_auto() &&
1208            offsets.right.is_auto() &&
1209            offsets.bottom.is_auto() &&
1210            offsets.left.is_auto()
1211        {
1212            return None;
1213        }
1214
1215        // https://drafts.csswg.org/css-position/#stickypos-insets
1216        // > For each side of the box, if the corresponding inset property is not `auto`, and the
1217        // > corresponding border edge of the box would be outside the corresponding edge of the
1218        // > sticky view rectangle, the box must be visually shifted (as for relative positioning)
1219        // > to be inward of that sticky view rectangle edge, insofar as it can while its position
1220        // > box remains contained within its containing block.
1221        // > The *position box* is its margin box, except that for any side for which the distance
1222        // > between its margin edge and the corresponding edge of its containing block is less
1223        // > than its corresponding margin, that distance is used in place of that margin.
1224        //
1225        // Amendments:
1226        // - Using the "margin edge" seems nonsensical, the spec must mean "border edge" instead:
1227        //   https://github.com/w3c/csswg-drafts/issues/12833
1228        // - `auto` margins need to be treated as zero:
1229        //   https://github.com/w3c/csswg-drafts/issues/12852
1230        //
1231        // We implement this by enforcing a minimum negative offset and a maximum positive offset.
1232        // The logic below is a simplified (but equivalent) version of the description above.
1233        let border_rect = self.border_rect();
1234        let computed_margin = style.physical_margin();
1235        let parent_scroll_node = stacking_context_tree
1236            .paint_info
1237            .scroll_tree
1238            .get_node(parent_scroll_node_id);
1239        let sticky_offset_boundary = match parent_scroll_node.info {
1240            SpatialTreeNodeInfo::Scroll(ref scrollable_node_info) if established_scroll_frame => {
1241                let content_rect = &scrollable_node_info.content_rect;
1242                &PhysicalRect::new(
1243                    PhysicalPoint::new(
1244                        Au::from_f32_px(content_rect.min.x),
1245                        Au::from_f32_px(content_rect.min.y),
1246                    ),
1247                    PhysicalSize::new(
1248                        Au::from_f32_px(content_rect.max.x - content_rect.min.x),
1249                        Au::from_f32_px(content_rect.max.y - content_rect.min.y),
1250                    ),
1251                )
1252            },
1253            _ => containing_block_rect,
1254        };
1255        // Signed distance between each side of the border box to the corresponding side of the
1256        // containing block. Note that |border_rect| is already in the coordinate system of the
1257        // containing block.
1258        let distance_from_border_box_to_cb = PhysicalSides::new(
1259            border_rect.min_y(),
1260            sticky_offset_boundary.width() - border_rect.max_x(),
1261            sticky_offset_boundary.height() - border_rect.max_y(),
1262            border_rect.min_x(),
1263        );
1264        // Shrinks the signed distance by the margin, producing a limit on how much we can shift
1265        // the sticky positioned box without forcing the margin to move outside of the containing
1266        // block.
1267        let offset_bound = |distance, used_margin, computed_margin: LengthPercentageOrAuto| {
1268            let used_margin = if computed_margin.is_auto() {
1269                Au::zero()
1270            } else {
1271                used_margin
1272            };
1273            Au::zero().max(distance - used_margin).to_f32_px()
1274        };
1275
1276        // This is the minimum negative offset and then the maximum positive offset. We specify
1277        // all sides, but they will have no effect if the corresponding inset property is `auto`.
1278        let vertical_offset_bounds = wr::StickyOffsetBounds::new(
1279            -offset_bound(
1280                distance_from_border_box_to_cb.top,
1281                self.margin.top,
1282                computed_margin.top,
1283            ),
1284            offset_bound(
1285                distance_from_border_box_to_cb.bottom,
1286                self.margin.bottom,
1287                computed_margin.bottom,
1288            ),
1289        );
1290        let horizontal_offset_bounds = wr::StickyOffsetBounds::new(
1291            -offset_bound(
1292                distance_from_border_box_to_cb.left,
1293                self.margin.left,
1294                computed_margin.left,
1295            ),
1296            offset_bound(
1297                distance_from_border_box_to_cb.right,
1298                self.margin.right,
1299                computed_margin.right,
1300            ),
1301        );
1302
1303        let frame_rect = border_rect
1304            .translate(containing_block_rect.origin.to_vector())
1305            .to_webrender();
1306
1307        // These are the "margins" between the scrollport and |frame_rect|. They are not the same
1308        // as CSS margins.
1309        let margins = SideOffsets2D::new(
1310            offsets.top.non_auto().map(|v| v.to_f32_px()),
1311            offsets.right.non_auto().map(|v| v.to_f32_px()),
1312            offsets.bottom.non_auto().map(|v| v.to_f32_px()),
1313            offsets.left.non_auto().map(|v| v.to_f32_px()),
1314        );
1315
1316        let sticky_node_id = stacking_context_tree.define_sticky_frame(
1317            parent_scroll_node_id,
1318            frame_rect,
1319            margins,
1320            vertical_offset_bounds,
1321            horizontal_offset_bounds,
1322        );
1323
1324        Some(sticky_node_id)
1325    }

1327    /// Optionally returns the data for building a reference frame, without yet building it.
1328    fn reference_frame_data_if_necessary(
1329        &self,
1330        containing_block_rect: &PhysicalRect<Au>,
1331    ) -> Option<ReferenceFrameData> {
1332        if !self
1333            .style()
1334            .has_effective_transform_or_perspective(self.base.flags)
1335        {
1336            return None;
1337        }
1338
1339        let relative_border_rect = self.border_rect();
1340        let border_rect = relative_border_rect.translate(containing_block_rect.origin.to_vector());
1341        let transform = self.calculate_transform_matrix(&border_rect);
1342        let perspective = self.calculate_perspective_matrix(&border_rect);
1343        let (reference_frame_transform, reference_frame_kind) = match (transform, perspective) {
1344            (None, Some(perspective)) => (
1345                perspective,
1346                wr::ReferenceFrameKind::Perspective {
1347                    scrolling_relative_to: None,
1348                },
1349            ),
1350            (Some(transform), None) => (
1351                transform,
1352                wr::ReferenceFrameKind::Transform {
1353                    is_2d_scale_translation: false,
1354                    should_snap: false,
1355                    paired_with_perspective: false,
1356                },
1357            ),
1358            (Some(transform), Some(perspective)) => (
1359                perspective.then(&transform),
1360                wr::ReferenceFrameKind::Perspective {
1361                    scrolling_relative_to: None,
1362                },
1363            ),
1364            (None, None) => unreachable!(),
1365        };
1366
1367        Some(ReferenceFrameData {
1368            origin: border_rect.origin,
1369            transform: reference_frame_transform,
1370            kind: reference_frame_kind,
1371        })
1372    }

1374    /// Returns the 4D matrix representing this fragment's transform.
1375    pub fn calculate_transform_matrix(
1376        &self,
1377        border_rect: &Rect<Au, CSSPixel>,
1378    ) -> Option<LayoutTransform> {
1379        let style = self.style();
1380        let list = &style.get_box().transform;
1381        let length_rect = au_rect_to_length_rect(border_rect);
1382        // https://drafts.csswg.org/css-transforms-2/#individual-transforms
1383        let rotate = match style.get_rotate() {
1384            GenericRotate::Rotate(angle) => (0., 0., 1., *angle),
1385            GenericRotate::Rotate3D(x, y, z, angle) => {
1386                // These are the raw, unormalized values from CSS, but euclid expects
1387                // rotation input to be normalized, so we must do that first.
1388                get_normalized_vector_and_angle(*x, *y, *z, *angle)
1389            },
1390            GenericRotate::None => (0., 0., 1., Angle::zero()),
1391        };
1392        let scale = match style.get_scale() {
1393            GenericScale::Scale(sx, sy, sz) => (*sx, *sy, *sz),
1394            GenericScale::None => (1., 1., 1.),
1395        };
1396        let translation = match style.get_translate() {
1397            GenericTranslate::Translate(x, y, z) => LayoutTransform::translation(
1398                x.resolve(length_rect.size.width).px(),
1399                y.resolve(length_rect.size.height).px(),
1400                z.px(),
1401            ),
1402            GenericTranslate::None => LayoutTransform::identity(),
1403        };
1404
1405        let angle = euclid::Angle::radians(rotate.3.radians());
1406        let transform_base = list
1407            .to_transform_3d_matrix(Some(&length_rect.to_untyped()))
1408            .ok()?;
1409        let transform = LayoutTransform::from_untyped(&transform_base.0)
1410            .then_rotate(rotate.0, rotate.1, rotate.2, angle)
1411            .then_scale(scale.0, scale.1, scale.2)
1412            .then(&translation);
1413
1414        let transform_origin = &style.get_box().transform_origin;
1415        let transform_origin_x = transform_origin
1416            .horizontal
1417            .to_used_value(border_rect.size.width)
1418            .to_f32_px();
1419        let transform_origin_y = transform_origin
1420            .vertical
1421            .to_used_value(border_rect.size.height)
1422            .to_f32_px();
1423        let transform_origin_z = transform_origin.depth.px();
1424
1425        Some(transform.change_basis(transform_origin_x, transform_origin_y, transform_origin_z))
1426    }

1428    /// Returns the 4D matrix representing this fragment's perspective.
1429    pub fn calculate_perspective_matrix(
1430        &self,
1431        border_rect: &Rect<Au, CSSPixel>,
1432    ) -> Option<LayoutTransform> {
1433        let style = self.style();
1434        match style.get_box().perspective {
1435            Perspective::Length(length) => {
1436                let perspective_origin = &style.get_box().perspective_origin;
1437                let perspective_origin = LayoutPoint::new(
1438                    perspective_origin
1439                        .horizontal
1440                        .percentage_relative_to(border_rect.size.width.into())
1441                        .px(),
1442                    perspective_origin
1443                        .vertical
1444                        .percentage_relative_to(border_rect.size.height.into())
1445                        .px(),
1446                );
1447
1448                let perspective_matrix = LayoutTransform::from_untyped(
1449                    &transform::create_perspective_matrix(length.px()),
1450                );
1451
1452                Some(perspective_matrix.change_basis(
1453                    perspective_origin.x,
1454                    perspective_origin.y,
1455                    0.0,
1456                ))
1457            },
1458            Perspective::None => None,
1459        }
1460    }

1462    fn clear_stacking_context_tree_traversal_data_recursively(&self) {
1463        fn clear_stacking_context_tree_traversal_data_on_fragments(fragments: &[Fragment]) {
1464            for fragment in fragments.iter() {
1465                match fragment {
1466                    Fragment::LayoutRoot(layout_root_fragment) => layout_root_fragment
1467                        .inner_box_fragment()
1468                        .clear_stacking_context_tree_traversal_data_recursively(),
1469                    Fragment::Box(box_fragment) | Fragment::Float(box_fragment) => {
1470                        box_fragment.clear_stacking_context_tree_traversal_data_recursively();
1471                    },
1472                    Fragment::Positioning(positioning_fragment) => {
1473                        clear_stacking_context_tree_traversal_data_on_fragments(
1474                            &positioning_fragment.children,
1475                        );
1476                    },
1477                    _ => {},
1478                }
1479            }
1480        }
1481
1482        self.spatial_tree_node.set(None);
1483        self.clear_stacking_context_tree_traversal_data();
1484        clear_stacking_context_tree_traversal_data_on_fragments(&self.children);
1485    }

1488impl PositioningFragment {
1489    #[expect(clippy::too_many_arguments)]
1490    fn build_stacking_context_tree(
1491        &self,
1492        stacking_context_tree: &mut StackingContextTree,
1493        containing_block: &ContainingBlock,
1494        containing_block_info: &ContainingBlockInfo,
1495        stacking_context: &mut StackingContext,
1496        text_decorations: &Rc<Vec<FragmentTextDecoration>>,
1497        container_timing_roots: &ContainerTimingRoots,
1498        participates_in_3d_rendering_context: bool,
1499    ) {
1500        let rect = self
1501            .base
1502            .rect()
1503            .translate(containing_block.rect.origin.to_vector());
1504        let new_containing_block = containing_block.new_replacing_rect(&rect);
1505        let new_containing_block_info =
1506            containing_block_info.new_for_non_absolute_descendants(&new_containing_block);
1507
1508        for child in &self.children {
1509            child.build_stacking_context_tree(
1510                stacking_context_tree,
1511                &new_containing_block_info,
1512                stacking_context,
1513                StackingContextBuildMode::SkipHoisted,
1514                text_decorations,
1515                container_timing_roots,
1516                participates_in_3d_rendering_context,
1517            );
1518        }
1519    }
1520}

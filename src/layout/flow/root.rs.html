32#[derive(MallocSizeOf)]
33pub struct BoxTree {
34    /// Contains typically exactly one block-level box, which was generated by the root element.
35    /// There may be zero if that element has `display: none`.
36    root: BlockFormattingContext,
37    /// Whether or not the viewport should be sensitive to scrolling input events in two axes
38    pub(crate) viewport_overflow: AxesOverflow,
39    /// The style to use for the initial containing block.
40    pub(crate) initial_containing_block_style: ServoArc<ComputedValues>,
41}
42
43impl BoxTree {
44    #[servo_tracing::instrument(name = "Box Tree Construction", skip_all)]
45    pub(crate) fn construct(context: &LayoutContext, root_element: ServoLayoutNode<'_>) -> Self {
46        let style_for_writing_mode =
47            Self::writing_mode_source_style(&context.style_context, root_element);
48        let boxes = construct_for_root_element(context, root_element, &style_for_writing_mode);
49
50        // Zero box for `:root { display: none }`, one for the root element otherwise.
51        assert!(boxes.len() <= 1);
52
53        let default_style = context
54            .style_context
55            .stylist
56            .device()
57            .default_computed_values_arc();
58        let initial_containing_block_style = BoxTree::style_with_writing_mode_source_style(
59            default_style.clone(),
60            &style_for_writing_mode,
61        );
62
63        let viewport_overflow = Self::viewport_overflow(root_element, boxes.first());
64        let contents = BlockContainer::BlockLevelBoxes(boxes);
65        let contains_floats = contents.contains_floats();
66        Self {
67            root: BlockFormattingContext {
68                contents,
69                contains_floats,
70            },
71            // From https://www.w3.org/TR/css-overflow-3/#overflow-propagation:
72            // > If visible is applied to the viewport, it must be interpreted as auto.
73            // > If clip is applied to the viewport, it must be interpreted as hidden.
74            viewport_overflow: viewport_overflow.to_scrollable(),
75            initial_containing_block_style,
76        }
77    }
78
79    /// From <https://drafts.csswg.org/css-writing-modes/#principal-flow>:
80    ///
81    /// > The principal writing mode of the document is determined by the used
82    /// > writing-mode, direction, and text-orientation values of the root element. This
83    /// > writing mode is used, for example, to determine the direction of scrolling and
84    /// > the default page progression direction.
85    /// >
86    /// > As a special case for handling HTML documents, if the root element has a body
87    /// > child element [HTML] whose display value is not none, the used value of the of
88    /// > writing-mode and direction properties on root element are taken from the
89    /// > computed writing-mode and direction of the first such child element instead of
90    /// > from the root element’s own values. The UA may also propagate the value of
91    /// > text-orientation in this manner. Note that this does not affect the computed
92    /// > values of writing-mode, direction, or text-orientation of the root element
93    /// > itself.
94    pub(crate) fn writing_mode_source_style(
95        context: &SharedStyleContext,
96        root_element: ServoLayoutNode<'_>,
97    ) -> ServoArc<ComputedValues> {
98        // See the similar check in viewport_overflow. If there's no root box, then there
99        // will never be a box for the body either.
100        let root_element_style = root_element.style(context);
101
102        // https://drafts.csswg.org/css-display-3/#transformations
103        // “'display' of 'contents' computes to 'block' on the root element”
104        debug_assert_ne!(*root_element_style.get_display(), StyleDisplay::Contents);
105
106        if *root_element_style.get_display() == StyleDisplay::None {
107            return root_element_style;
108        }
109
110        // Unlike what the spec implies, we stop iterating when we find the first <body>,
111        // even if it's not suitable because it lacks a box. This matches other browsers.
112        // See https://github.com/w3c/csswg-drafts/issues/12644
113        let Some(body) = root_element.flat_tree_children().find(|child| {
114            child
115                .as_element()
116                .is_some_and(|element| element.is_body_element_of_html_element_root())
117        }) else {
118            return root_element_style;
119        };
120
121        // We only propagate from the <body> if it generates a box. The spec only checks for
122        // `display: none`, but Blink and WebKit don't propagate for `display: contents` either.
123        // Gecko propagates from the <body> regardless of its `display`.
124        // See https://github.com/w3c/csswg-drafts/issues/12643
125        let body_style = body.style(context);
126        let body_display = body_style.get_display();
127        if *body_display == StyleDisplay::Contents || *body_display == StyleDisplay::None {
128            return root_element.style(context);
129        }
130        body_style.clone()
131    }

133    pub(crate) fn style_with_writing_mode_source_style(
134        base_style: ServoArc<ComputedValues>,
135        writing_mode_source_style: &ServoArc<ComputedValues>,
136    ) -> ServoArc<ComputedValues> {
137        if base_style.writing_mode == writing_mode_source_style.writing_mode &&
138            base_style.get_direction() == writing_mode_source_style.get_direction() &&
139            base_style.get_text_orientation() == writing_mode_source_style.get_text_orientation()
140        {
141            return base_style;
142        }
143
144        let mut new_style = (*base_style).clone();
145        let inherited_box = new_style.mutate_inherited_box();
146        inherited_box.writing_mode = writing_mode_source_style.get_inherited_box().writing_mode;
147        inherited_box.direction = writing_mode_source_style.get_inherited_box().direction;
148        inherited_box.text_orientation = writing_mode_source_style
149            .get_inherited_box()
150            .text_orientation;
151        new_style.writing_mode = writing_mode_source_style.writing_mode;
152        ServoArc::new(new_style)
153    }
154
155    fn viewport_overflow(
156        root_element: ServoLayoutNode<'_>,
157        root_box: Option<&ArcRefCell<BlockLevelBox>>,
158    ) -> AxesOverflow {
159        // From <https://www.w3.org/TR/css-overflow-3/#overflow-propagation>:
160        // > UAs must apply the overflow-* values set on the root element to the viewport when the
161        // > root element’s display value is not none. However, when the root element is an [HTML]
162        // > html element (including XML syntax for HTML) whose overflow value is visible (in both
163        // > axes), and that element has as a child a body element whose display value is also not
164        // > none, user agents must instead apply the overflow-* values of the first such child
165        // > element to the viewport. The element from which the value is propagated must then have a
166        // > used overflow value of visible.
167
168        // If there is no root box, the root element has `display: none`, so don't propagate.
169        // The spec isn't very clear about what value to use, but the initial value seems fine.
170        // See https://github.com/w3c/csswg-drafts/issues/12649
171        let Some(root_box) = root_box else {
172            return AxesOverflow::default();
173        };
174
175        let propagate_from_body = || {
176            // Unlike what the spec implies, we stop iterating when we find the first <body>,
177            // even if it's not suitable because it lacks a box. This matches other browsers.
178            // See https://github.com/w3c/csswg-drafts/issues/12644
179            let body = root_element.flat_tree_children().find(|child| {
180                child
181                    .as_element()
182                    .is_some_and(|element| element.is_body_element_of_html_element_root())
183            })?;
184
185            // We only propagate from the <body> if it generates a box. The spec only checks for
186            // `display: none`, but other browsers don't propagate for `display: contents` either.
187            // See https://github.com/w3c/csswg-drafts/issues/12643
188            let body_layout_data = body.inner_layout_data()?;
189            let mut body_box = body_layout_data.self_box.borrow_mut();
190            body_box.as_mut()?.with_base_mut(|base| {
191                base.base_fragment_info
192                    .flags
193                    .insert(FragmentFlags::PROPAGATED_OVERFLOW_TO_VIEWPORT);
194                AxesOverflow::from(&*base.style)
195            })
196        };
197
198        root_box.borrow_mut().with_base_mut(|base| {
199            let root_overflow = AxesOverflow::from(&*base.style);
200            if root_overflow.x == Overflow::Visible &&
201                root_overflow.y == Overflow::Visible &&
202                let Some(body_overflow) = propagate_from_body()
203            {
204                return body_overflow;
205            }
206            base.base_fragment_info
207                .flags
208                .insert(FragmentFlags::PROPAGATED_OVERFLOW_TO_VIEWPORT);
209            root_overflow
210        })
211    }
212}

214fn construct_for_root_element(
215    context: &LayoutContext,
216    root_element: ServoLayoutNode<'_>,
217    style_for_writing_mode: &ServoArc<ComputedValues>,
218) -> Vec<ArcRefCell<BlockLevelBox>> {
219    let root_style = BoxTree::style_with_writing_mode_source_style(
220        root_element.style(&context.style_context),
221        style_for_writing_mode,
222    );
223    let info = NodeAndStyleInfo::new(root_element, root_style);
224    let box_style = info.style.get_box();
225
226    let display_inside = match Display::from(box_style.display) {
227        Display::None => return Vec::new(),
228        Display::Contents => {
229            // Unreachable because the style crate adjusts the computed values:
230            // https://drafts.csswg.org/css-display-3/#transformations
231            // “'display' of 'contents' computes to 'block' on the root element”
232            unreachable!()
233        },
234        // The root element is blockified, ignore DisplayOutside
235        Display::GeneratingBox(display_generating_box) => display_generating_box.display_inside(),
236    };
237
238    let contents = Contents::for_element(root_element, context);
239
240    let propagated_data = PropagatedBoxTreeData::default();
241    let root_box = if box_style.position.is_absolutely_positioned() {
242        BlockLevelBox::OutOfFlowAbsolutelyPositionedBox(ArcRefCell::new(
243            AbsolutelyPositionedBox::construct(context, &info, display_inside, contents),
244        ))
245    } else if box_style.float.is_floating() {
246        BlockLevelBox::OutOfFlowFloatBox(FloatBox::construct(
247            context,
248            &info,
249            display_inside,
250            contents,
251            propagated_data,
252        ))
253    } else {
254        BlockLevelBox::Independent(IndependentFormattingContext::construct(
255            context,
256            &info,
257            display_inside,
258            contents,
259            propagated_data,
260        ))
261    };
262
263    let root_box = ArcRefCell::new(root_box);
264    root_element
265        .box_slot()
266        .set(LayoutBox::BlockLevel(root_box.clone()));
267    vec![root_box]
268}

270impl BoxTree {
271    #[servo_tracing::instrument(name = "Fragment Tree Construction", skip_all)]
272    pub(crate) fn layout(
273        &self,
274        layout_context: &LayoutContext,
275        viewport: UntypedSize2D<Au>,
276    ) -> FragmentTree {
277        let physical_containing_block: Rect<Au, CSSPixel> =
278            PhysicalSize::from_untyped(viewport).into();
279
280        let initial_containing_block = DefiniteContainingBlock {
281            size: LogicalVec2::from_physical_size(
282                &physical_containing_block.size,
283                self.initial_containing_block_style.writing_mode,
284            ),
285            style: &self.initial_containing_block_style,
286        };
287
288        let mut positioning_context = PositioningContext::default();
289        let independent_layout = self.root.layout(
290            layout_context,
291            &mut positioning_context,
292            &(&initial_containing_block).into(),
293            &initial_containing_block.size.block.into(),
294            None,
295        );
296
297        let mut root_fragments = independent_layout.fragments;
298
299        // Zero box for `:root { display: none }`, one for the root element otherwise.
300        assert!(root_fragments.len() <= 1);
301
302        // There may be more fragments at the top-level
303        // (for positioned boxes whose containing is the initial containing block)
304        // but only if there was one fragment for the root element.
305        positioning_context.layout_initial_containing_block_children(
306            layout_context,
307            &initial_containing_block,
308            &mut root_fragments,
309        );
310
311        let viewport_scroll_sensitivity = AxesScrollSensitivity {
312            x: self.viewport_overflow.x.into(),
313            y: self.viewport_overflow.y.into(),
314        };
315
316        FragmentTree::new(
317            root_fragments,
318            physical_containing_block,
319            viewport_scroll_sensitivity,
320        )
321    }

60/// The state used when laying out a collection of [`LineItem`]s into a line. This state is stored
61/// per-inline container. For instance, when laying out the conents of a `<span>` a fresh
62/// [`LineItemLayoutInlineContainerState`] is pushed onto [`LineItemLayout`]'s stack of states.
63pub(super) struct LineItemLayoutInlineContainerState {
64    /// If this inline container is not the root inline container, the identifier of the [`super::InlineBox`]
65    /// that is currently being laid out.
66    pub identifier: Option<InlineBoxIdentifier>,
67
68    /// The fragments and their associated logical rectangles
69    fragments_and_logical_rect: Vec<FragmentAndLogicalRect>,
70
71    /// The current inline advance of the layout in the coordinates of this inline box.
72    pub inline_advance: Au,
73
74    /// Flags which track various features during layout.
75    flags: LineLayoutInlineContainerFlags,
76
77    /// The offset of the parent, relative to the start position of the line, including any block
78    /// and leading padding, border, and margin (in the writing mode of the containing block).
79    pub parent_offset: LogicalVec2<Au>,
80
81    /// The block offset of the parent's baseline relative to the block start of the line. This
82    /// is often the same as [`Self::parent_offset`], but can be different for the root
83    /// element.
84    pub baseline_offset: Au,
85
86    /// The adjustment to this inline container's final position due to relative positioning.
87    /// This is used to find the offset from the line start for floats, which are positioned
88    /// relative to the inline formatting context itself.
89    relative_offset: LogicalVec2<Au>,
90
91    /// If this inline box establishes a containing block for positioned elements, this
92    /// is a fresh positioning context to contain them. Otherwise, this holds the starting
93    /// offset in the *parent* positioning context so that static positions can be updated
94    /// at the end of layout.
95    pub positioning_context_or_start_offset_in_parent:
96        Either<PositioningContext, PositioningContextLength>,
97}

137/// The second phase of [`super::InlineFormattingContext`] layout: once items are gathered
138/// for a line, we must lay them out and create fragments for them, properly positioning them

201    /// Start and end inline boxes in tree order, so that it reflects the given inline box.
202    fn prepare_layout_for_inline_box(&mut self, new_inline_box: Option<InlineBoxIdentifier>) {
203        // Optimize the case where we are moving to the root of the inline box stack.
204        let Some(new_inline_box) = new_inline_box else {
205            while !self.state_stack.is_empty() {
206                self.end_inline_box();
207            }
208            return;
209        };
210
211        // Otherwise, follow the path given to us by our collection of inline boxes, so we know which
212        // inline boxes to start and end.
213        let path = self
214            .layout
215            .ifc
216            .inline_boxes
217            .get_path(self.current_state.identifier, new_inline_box);
218        for token in path {
219            match token {
220                InlineBoxTreePathToken::Start(ref identifier) => self.start_inline_box(identifier),
221                InlineBoxTreePathToken::End(_) => self.end_inline_box(),
222            }
223        }
224    }

226    /// If the inline formatting context that this line is being laid out for had
227    /// right-to-left content, reorder the line contents according to their pre-calculated
228    /// BiDi levels.
229    ///
230    /// Returns an iterator over the line contents.
231    fn reorder_line_items_for_bidi(
232        &self,
233        mut line_items: Vec<LineItem>,
234    ) -> impl Iterator<Item = LineItem> + use<> {
235        let iterator = |line_items: Vec<LineItem>| {
236            // `BidiInfo::reorder_visual` will reorder the contents of the line so that they
237            // are in the correct order as if one was looking at the line from left-to-right.
238            // During this layout we do not lay out from left to right. Instead we lay out
239            // from inline-start to inline-end. If the overall line contents have been flipped
240            // for BiDi, flip them again so that they are in line start-to-end order rather
241            // than left-to-right order.
242            if self.containing_block().style.writing_mode.is_bidi_ltr() {
243                Either::Left(line_items.into_iter())
244            } else {
245                Either::Right(line_items.into_iter().rev())
246            }
247        };
248
249        if !self.layout.ifc.has_right_to_left_content {
250            // Even if the actual content of the inline formatting context does not
251            // contain internal right-to-left text, the overall direction of the inline
252            // formatting context might be right-to-left. In that case we still want to
253            // return a reverse iterator.
254            return iterator(line_items);
255        }
256
257        let mut last_level = Level::ltr();
258        let levels: Vec<_> = line_items
259            .iter()
260            .map(|item| {
261                let level = match item {
262                    LineItem::TextRun(_, text_run) => text_run.info.font_info.bidi_level,
263                    // TODO: This level needs either to be last_level, or if there were
264                    // unicode characters inserted for the inline box, we need to get the
265                    // level from them.
266                    LineItem::InlineStartBoxPaddingBorderMargin(_) => last_level,
267                    LineItem::InlineEndBoxPaddingBorderMargin(_) => last_level,
268                    LineItem::Atomic(_, atomic) => atomic.bidi_level,
269                    LineItem::AbsolutelyPositioned(..) => last_level,
270                    LineItem::Float(..) => {
271                        // At this point the float is already positioned, so it doesn't really matter what
272                        // position it's fragment has in the order of line items.
273                        last_level
274                    },
275                    LineItem::BlockLevel(..) => last_level,
276                    LineItem::Tab { bidi_level, .. } => *bidi_level,
277                };
278                last_level = level;
279                level
280            })
281            .collect();
282
283        sort_by_indices_in_place(&mut line_items, BidiInfo::reorder_visual(&levels));
284        iterator(line_items)
285    }

287    pub(super) fn layout(&mut self, line_items: Vec<LineItem>) -> Vec<FragmentAndLogicalRect> {
288        let line_item_iterator = self.reorder_line_items_for_bidi(line_items);
289        for item in line_item_iterator.into_iter().by_ref() {
290            // When preparing to lay out a new line item, start and end inline boxes, so that the current
291            // inline box state reflects the item's parent. Items in the line are not necessarily in tree
292            // order due to BiDi and other reordering so the inline box of the item could potentially be
293            // any in the inline formatting context.
294            self.prepare_layout_for_inline_box(item.inline_box_identifier());
295
296            match item {
297                LineItem::InlineStartBoxPaddingBorderMargin(state) => {
298                    if state.writing_direction_reversed {
299                        self.current_state
300                            .flags
301                            .insert(LineLayoutInlineContainerFlags::HAD_INLINE_END_PBM);
302                    }
303                },
304                LineItem::InlineEndBoxPaddingBorderMargin(state) => {
305                    if !state.writing_direction_reversed {
306                        self.current_state
307                            .flags
308                            .insert(LineLayoutInlineContainerFlags::HAD_INLINE_END_PBM);
309                    }
310                },
311                LineItem::TextRun(_, text_run) => self.layout_text_run(text_run),
312                LineItem::Atomic(_, atomic) => self.layout_atomic(atomic),
313                LineItem::AbsolutelyPositioned(_, absolute) => self.layout_absolute(absolute),
314                LineItem::Float(_, float) => self.layout_float(float),
315                LineItem::BlockLevel(_, block_level) => self.layout_block_level(block_level),
316                LineItem::Tab { advance, .. } => self.layout_tab(advance),
317            }
318        }
319
320        // Move back to the root of the inline box tree, so that all boxes are ended.
321        self.prepare_layout_for_inline_box(None);
322
323        std::mem::take(&mut self.current_state.fragments_and_logical_rect)
324    }

326    fn current_positioning_context_mut(&mut self) -> &mut PositioningContext {
327        if let Either::Left(ref mut positioning_context) = self
328            .current_state
329            .positioning_context_or_start_offset_in_parent
330        {
331            return positioning_context;
332        }
333        self.state_stack
334            .iter_mut()
335            .rev()
336            .find_map(
337                |state| match state.positioning_context_or_start_offset_in_parent {
338                    Either::Left(ref mut positioning_context) => Some(positioning_context),
339                    Either::Right(_) => None,
340                },
341            )
342            .unwrap_or(self.layout.positioning_context)
343    }

345    fn start_inline_box(&mut self, identifier: &InlineBoxIdentifier) {
346        let inline_box_state =
347            &*self.layout.inline_box_states[identifier.index_in_inline_boxes as usize];
348        let inline_box = self.layout.ifc.inline_boxes.get(identifier);
349        let inline_box = &*(inline_box.borrow());
350
351        let space_above_baseline = inline_box_state.calculate_space_above_baseline();
352        let block_start_offset =
353            self.calculate_inline_box_block_start(inline_box_state, space_above_baseline);
354
355        let leading_edge_on_line = inline_box_state.leading_edge_on_current_line.take();
356        let flags = if leading_edge_on_line {
357            LineLayoutInlineContainerFlags::HAD_INLINE_START_PBM
358        } else {
359            LineLayoutInlineContainerFlags::empty()
360        };
361
362        let apply_leading_edge_pbm =
363            !self.for_block_level && (inline_box_state.should_clone_pbm() || leading_edge_on_line);
364        let leading_pbm = if apply_leading_edge_pbm {
365            inline_box_state.pbm_inline_start()
366        } else {
367            Au::zero()
368        };
369
370        let positioning_context_or_start_offset_in_parent =
371            match PositioningContext::new_for_layout_box_base(&inline_box.base) {
372                Some(positioning_context) => Either::Left(positioning_context),
373                None => Either::Right(self.current_positioning_context_mut().len()),
374            };
375
376        let containing_block = self.layout.containing_block();
377        let style = &inline_box.base.style;
378        let relative_offset = if style.get_box().position == Position::Relative {
379            relative_adjustement(style, containing_block)
380        } else {
381            LogicalVec2::zero()
382        };
383
384        let parent_offset = LogicalVec2 {
385            inline: self.current_state.inline_advance +
386                self.current_state.parent_offset.inline +
387                leading_pbm,
388            block: block_start_offset,
389        };
390
391        self.cumulative_relative_offset += relative_offset;
392        let outer_state = std::mem::replace(
393            &mut self.current_state,
394            LineItemLayoutInlineContainerState::new(
395                Some(*identifier),
396                parent_offset,
397                relative_offset,
398                block_start_offset + space_above_baseline,
399                positioning_context_or_start_offset_in_parent,
400                flags,
401            ),
402        );
403
404        self.state_stack.push(outer_state);
405    }

407    fn end_inline_box(&mut self) {
408        let outer_state = self.state_stack.pop().expect("Ended unknown inline box");
409        let inner_state = std::mem::replace(&mut self.current_state, outer_state);
410        self.cumulative_relative_offset -= inner_state.relative_offset;
411
412        let identifier = inner_state.identifier.expect("Ended unknown inline box");
413        let inline_box_state =
414            &*self.layout.inline_box_states[identifier.index_in_inline_boxes as usize];
415        let inline_box = self.layout.ifc.inline_boxes.get(&identifier);
416        let inline_box = &*(inline_box.borrow());
417
418        let containing_block = self.layout.containing_block();
419        let containing_block_writing_mode = containing_block.style.writing_mode;
420
421        let mut padding = inline_box_state.pbm.padding;
422        let mut border = inline_box_state.pbm.border;
423        let mut margin = inline_box_state.pbm.margin.auto_is(Au::zero);
424        // PBM must not be cloned onto lines that exist only to support a block-level box.
425        // See https://github.com/w3c/csswg-drafts/issues/14104
426        if self.for_block_level {
427            padding = LogicalSides::zero();
428            border = LogicalSides::zero();
429            margin = LogicalSides::zero();
430        } else if !inline_box_state.should_clone_pbm() {
431            let had_start = inner_state
432                .flags
433                .contains(LineLayoutInlineContainerFlags::HAD_INLINE_START_PBM);
434            let had_end = inner_state
435                .flags
436                .contains(LineLayoutInlineContainerFlags::HAD_INLINE_END_PBM);
437
438            if !had_start {
439                padding.inline_start = Au::zero();
440                border.inline_start = Au::zero();
441                margin.inline_start = Au::zero();
442            }
443            if !had_end {
444                padding.inline_end = Au::zero();
445                border.inline_end = Au::zero();
446                margin.inline_end = Au::zero();
447            }
448        }
449        let pbm_sums = padding + border + margin;
450
451        // Make `content_rect` relative to the parent Fragment.
452        let content_rect = LogicalRect {
453            start_corner: LogicalVec2 {
454                inline: self.current_state.inline_advance + pbm_sums.inline_start,
455                block: inner_state.parent_offset.block - self.current_state.parent_offset.block,
456            } + inner_state.relative_offset,
457            size: LogicalVec2 {
458                inline: inner_state.inline_advance,
459                block: if self.is_phantom_line {
460                    Au::zero()
461                } else {
462                    inline_box_state.base.font_metrics.line_gap
463                },
464            },
465        };
466
467        let mut fragment = BoxFragment::new_with_writing_mode_override(
468            inline_box.base.base_fragment_info,
469            inline_box.base.style.clone(),
470            FragmentChildren::Logical(inner_state.fragments_and_logical_rect),
471            // Previously all the fragment's children were positioned relative to the linebox,
472            // but they need to be made relative to this fragment.
473            content_rect.as_physical_deprecated(containing_block),
474            PhysicalBoxSides::from_logical(padding, border, margin, containing_block_writing_mode),
475            None, /* specific_layout_info */
476            containing_block_writing_mode,
477        );
478        if self.for_block_level {
479            fragment.base.flags.insert(FragmentFlags::DO_NOT_PAINT);
480        }
481
482        match inner_state.positioning_context_or_start_offset_in_parent {
483            Either::Left(mut positioning_context) => {
484                positioning_context
485                    .layout_collected_children(self.layout.layout_context, &mut fragment);
486                positioning_context.adjust_static_position_of_hoisted_fragments(
487                    &content_rect,
488                    containing_block_writing_mode,
489                    PositioningContextLength::zero(),
490                );
491                self.current_positioning_context_mut()
492                    .append(positioning_context);
493            },
494            Either::Right(start_offset) => {
495                self.current_positioning_context_mut()
496                    .adjust_static_position_of_hoisted_fragments(
497                        &content_rect,
498                        containing_block_writing_mode,
499                        start_offset,
500                    );
501            },
502        }
503
504        self.current_state.inline_advance += inner_state.inline_advance + pbm_sums.inline_sum();
505
506        let fragment = Fragment::Box(Arc::new(fragment));
507        inline_box.base.add_fragment(fragment.clone());
508        self.current_state
509            .fragments_and_logical_rect
510            .push(FragmentAndLogicalRect::new(fragment, content_rect));
511    }

513    fn calculate_inline_box_block_start(
514        &self,
515        inline_box_state: &InlineBoxContainerState,
516        space_above_baseline: Au,
517    ) -> Au {
518        if self.is_phantom_line {
519            return Au::zero();
520        };
521        let font_metrics = &inline_box_state.base.font_metrics;
522        let style = &inline_box_state.base.style;
523        let line_gap = font_metrics.line_gap;
524
525        // The baseline offset that we have in `Self::baseline_offset` is relative to the line
526        // baseline, so we need to make it relative to the line block start.
527        match inline_box_state.base.style.get_baseline_shift() {
528            BaselineShift::Keyword(BaselineShiftKeyword::Top) => {
529                let line_height = line_height(style, font_metrics, &inline_box_state.base.flags);
530                (line_height - line_gap).scale_by(0.5)
531            },
532            BaselineShift::Keyword(BaselineShiftKeyword::Center) => {
533                (self.line_metrics.block_size - line_gap).scale_by(0.5)
534            },
535            BaselineShift::Keyword(BaselineShiftKeyword::Bottom) => {
536                let line_height = line_height(style, font_metrics, &inline_box_state.base.flags);
537                let half_leading = (line_height - line_gap).scale_by(0.5);
538                self.line_metrics.block_size - line_height + half_leading
539            },
540            _ => {
541                self.line_metrics.baseline_block_offset + inline_box_state.base.baseline_offset -
542                    space_above_baseline
543            },
544        }
545    }

547    fn layout_text_run(&mut self, text_item: TextRunLineItem) {
548        if text_item.text.is_empty() && !text_item.is_empty_for_text_cursor {
549            return;
550        }
551
552        let mut number_of_justification_opportunities = 0;
553        let mut inline_advance = text_item
554            .text
555            .iter()
556            .map(|shaped_text_slice| {
557                number_of_justification_opportunities += shaped_text_slice.total_word_separators();
558                shaped_text_slice.total_advance()
559            })
560            .sum();
561
562        if !self.justification_adjustment.is_zero() {
563            inline_advance += self
564                .justification_adjustment
565                .scale_by(number_of_justification_opportunities as f32);
566        }
567
568        // The block start of the TextRun is often zero (meaning it has the same font metrics as the
569        // inline box's strut), but for children of the inline formatting context root or for
570        // fallback fonts that use baseline relative alignment, it might be different.
571        let font_metrics = text_item.info.font_info.font.metrics();
572        let start_corner = LogicalVec2 {
573            inline: self.current_state.inline_advance,
574            block: self.current_state.baseline_offset -
575                font_metrics.ascent -
576                self.current_state.parent_offset.block,
577        };
578        let content_rect = LogicalRect {
579            start_corner,
580            size: LogicalVec2 {
581                block: font_metrics.line_gap,
582                inline: inline_advance,
583            },
584        };
585
586        let font_key = text_item.info.font_info.font.key(
587            self.layout.layout_context.painter_id,
588            &self.layout.layout_context.font_context,
589        );
590
591        self.current_state.inline_advance += inline_advance;
592        self.current_state
593            .fragments_and_logical_rect
594            .push(FragmentAndLogicalRect::new(
595                Fragment::Text(Arc::new(TextFragment {
596                    base: BaseFragment::new(text_item.base_fragment_info, PhysicalRect::zero()),
597                    run_data: text_item.text_fragment_run_data,
598                    font_metrics: font_metrics.clone(),
599                    font_key,
600                    glyphs: text_item.text,
601                    justification_adjustment: self.justification_adjustment,
602                    character_range_in_dom_node: text_item.character_range_in_dom_node,
603                    is_empty_for_text_cursor: text_item.is_empty_for_text_cursor,
604                })),
605                content_rect,
606            ));
607    }
608
609    fn layout_atomic(&mut self, atomic: AtomicLineItem) {
610        // The initial `start_corner` of the Fragment is only the PaddingBorderMargin sum start
611        // offset, which is the sum of the start component of the padding, border, and margin.
612        // This needs to be added to the calculated block and inline positions.
613        // Make the final result relative to the parent box.
614        let containing_block = self.containing_block();
615        let ifc_writing_mode = containing_block.style.writing_mode;
616        let content_rect = {
617            let atomic_fragment = &atomic.fragment;
618            let block_start = atomic.calculate_block_start(&self.line_metrics);
619            let padding_border_margin_sides = LogicalSides::from_physical(
620                &atomic_fragment.padding_border_margin(),
621                ifc_writing_mode,
622            );
623
624            let mut atomic_offset = LogicalVec2 {
625                inline: self.current_state.inline_advance +
626                    padding_border_margin_sides.inline_start,
627                block: block_start - self.current_state.parent_offset.block +
628                    padding_border_margin_sides.block_start,
629            };
630
631            let style = atomic_fragment.style();
632            if style.get_box().position == Position::Relative {
633                atomic_offset += relative_adjustement(&style, containing_block);
634            }
635
636            // Reconstruct a logical rectangle relative to the inline box container that will be used
637            // after the inline box is processed to find a final physical rectangle.
638            LogicalRect {
639                start_corner: atomic_offset,
640                size: LogicalVec2::from_physical_size(
641                    &atomic_fragment.content_rect().size,
642                    ifc_writing_mode,
643                ),
644            }
645        };
646
647        let mut positioning_context = atomic.positioning_context;
648        positioning_context.adjust_static_position_of_hoisted_fragments(
649            &content_rect,
650            containing_block.style.writing_mode,
651            PositioningContextLength::zero(),
652        );
653        self.current_positioning_context_mut()
654            .append(positioning_context);
655
656        self.current_state.inline_advance += atomic.size.inline;
657
658        self.current_state
659            .fragments_and_logical_rect
660            .push(FragmentAndLogicalRect::new(
661                Fragment::Box(atomic.fragment),
662                content_rect,
663            ));
664    }
665
666    fn layout_absolute(&mut self, absolute: AbsolutelyPositionedLineItem) {
667        let absolutely_positioned_box = (*absolute.absolutely_positioned_box).borrow();
668        let style = absolutely_positioned_box.context.style();
669
670        // From https://drafts.csswg.org/css2/#abs-non-replaced-width
671        // > The static-position containing block is the containing block of a
672        // > hypothetical box that would have been the first box of the element if its
673        // > specified position value had been static and its specified float had been
674        // > none. (Note that due to the rules in section 9.7 this hypothetical
675        // > calculation might require also assuming a different computed value for
676        // > display.)
677        //
678        // This box is different based on the original `display` value of the
679        // absolutely positioned element. If it's `inline` it would be placed inline
680        // at the top of the line, but if it's block it would be placed in a new
681        // block position after the linebox established by this line.
682        let block_position = self.layout.placement_state.current_margin.solve() -
683            self.current_state.parent_offset.block;
684        let initial_start_corner =
685            if style.get_box().original_display.outside() == DisplayOutside::Inline {
686                // Top of the line at the current inline position.
687                LogicalVec2 {
688                    inline: self.current_state.inline_advance,
689                    block: block_position,
690                }
691            } else {
692                // After the bottom of the line at the start of the inline formatting context.
693                // Note that phantom lines are treated as being zero-height for this purpose.
694                // <https://drafts.csswg.org/css-inline-3/#invisible-line-boxes>
695                LogicalVec2 {
696                    inline: -self.current_state.parent_offset.inline,
697                    block: if absolute.preceding_line_content_would_produce_phantom_line {
698                        block_position
699                    } else {
700                        block_position + self.line_metrics.block_size
701                    },
702                }
703            };
704
705        // Since alignment of absolutes in inlines is currently always `start`, the size of
706        // of the static position rectangle does not matter.
707        let containing_block = self.containing_block();
708        let containing_block_writing_mode = containing_block.style.writing_mode;
709        let static_position_rect = LogicalRect {
710            start_corner: initial_start_corner,
711            size: LogicalVec2::zero(),
712        };
713
714        let hoisted_box = AbsolutelyPositionedBox::to_hoisted(
715            absolute.absolutely_positioned_box.clone(),
716            StaticPositionRect::new(static_position_rect, containing_block_writing_mode),
717            LogicalVec2 {
718                inline: AlignFlags::START,
719                block: AlignFlags::START,
720            },
721            containing_block_writing_mode,
722        );
723
724        let hoisted_fragment = hoisted_box.fragment.clone();
725        self.current_positioning_context_mut().push(hoisted_box);
726        self.current_state
727            .fragments_and_logical_rect
728            .push(FragmentAndLogicalRect::new(
729                Fragment::AbsoluteOrFixedPositionedPlaceholder(hoisted_fragment),
730                LogicalRect::zero(),
731            ));
732    }
733
734    fn layout_float(&mut self, mut float: FloatLineItem) {
735        // The `BoxFragment` for this float is positioned relative to the IFC, so we need
736        // to move it to be positioned relative to our parent InlineBox line item. Float
737        // fragments are children of these InlineBoxes and not children of the inline
738        // formatting context, so that they are parented properly for StackingContext
739        // properties such as opacity & filters.
740        let offset = LogicalVec2 {
741            inline: Au::zero(),
742            block: -self.line_metrics.block_offset,
743        } - self.current_state.parent_offset -
744            self.cumulative_relative_offset;
745        float.fragment.logical_rect.start_corner += offset;
746
747        self.current_state
748            .fragments_and_logical_rect
749            .push(float.fragment.map_fragment(Fragment::Float));
750    }
751
752    fn layout_block_level(&mut self, mut block_level: FragmentAndLogicalRect) {
753        let content_rect = &mut block_level.logical_rect;
754        // Block-level boxes are always placed at the logical origin of the line.
755        content_rect.start_corner.inline -= self.current_state.parent_offset.inline;
756        content_rect.start_corner.block -= self.line_metrics.block_offset;
757        self.current_state
758            .fragments_and_logical_rect
759            .push(block_level);
760    }
761
762    #[inline]
763    fn containing_block(&self) -> &ContainingBlock<'_> {
764        self.layout.containing_block()
765    }
766
767    fn layout_tab(&mut self, advance: Au) {
768        self.current_state.inline_advance += advance;
769    }
770}

835    /// benefit of placing a text cursor on an otherwise empty editable line.

838
839impl TextRunLineItem {
840    /// Trim the removable white space at the end of this [`TextRunLineItem`].
841    pub(crate) fn trim_removable_white_space_at_end(&mut self) {
842        while let Some(last) = self.text.last_mut() {
843            if last.all_removable() {
844                self.text.pop();
845                continue;
846            }
847            *last = last.without_removable_white_space();
848            break;
849        }
850    }
851
852    fn trim_whitespace_at_start(&mut self, whitespace_trimmed: &mut Au) -> bool {
853        let index_of_first_non_whitespace = self
854            .text
855            .iter()
856            .position(|slice| !slice.all_removable())
857            .unwrap_or(self.text.len());
858
859        *whitespace_trimmed += self
860            .text
861            .drain(0..index_of_first_non_whitespace)
862            .map(|glyph| glyph.total_advance())
863            .sum();
864
865        // Only keep going if we only encountered whitespace.
866        self.text.is_empty()
867    }
868
869    /// Returns a [`TrailingWhiteSpace`] with the measurement of hanging
870    /// and removable white space and boolean saying whether or not there
871    /// was preceding content.
872    pub(crate) fn trailing_white_space(&self) -> (TrailingWhiteSpace<Au>, bool) {
873        let mut hangable = Au::zero();
874        let mut removable = Au::zero();
875        let mut preceded_by_content = false;
876
877        for slice in self.text.iter().rev() {
878            if hangable.is_zero() {
879                removable += slice.removable_advance();
880            } else {
881                hangable += slice.removable_advance();
882            }
883            hangable += slice.hangable_advance();
884
885            if slice.has_non_hangable_non_removable_content() {
886                preceded_by_content = true;
887                break;
888            }
889        }
890
891        (
892            TrailingWhiteSpace {
893                hangable,
894                removable,
895            },
896            preceded_by_content,
897        )
898    }
899}

903    pub size: LogicalVec2<Au>,
904    pub positioning_context: PositioningContext,
905
906    /// The block offset of this items' baseline relative to the baseline of the line.
907    /// This will be zero for boxes with `vertical-align: top` and `vertical-align:
908    /// bottom` since their baselines are calculated late in layout.
909    pub baseline_offset_in_parent: Au,
910
911    /// The offset of the baseline inside this item.
912    pub baseline_offset_in_item: Au,
913
914    /// The BiDi level of this [`AtomicLineItem`] to enable reordering.
915    pub bidi_level: Level,
916}
917
918impl AtomicLineItem {
919    /// Given the metrics for a line, our vertical alignment, and our block size, find a block start
920    /// position relative to the top of the line.
921    fn calculate_block_start(&self, line_metrics: &LineMetrics) -> Au {
922        match self.fragment.style().get_baseline_shift() {
923            BaselineShift::Keyword(BaselineShiftKeyword::Top) => Au::zero(),
924            BaselineShift::Keyword(BaselineShiftKeyword::Center) => {
925                (line_metrics.block_size - self.size.block).scale_by(0.5)
926            },
927            BaselineShift::Keyword(BaselineShiftKeyword::Bottom) => {
928                line_metrics.block_size - self.size.block
929            },
930
931            // This covers all baseline-relative vertical alignment.
932            _ => {
933                let baseline = line_metrics.baseline_block_offset + self.baseline_offset_in_parent;
934                baseline - self.baseline_offset_in_item
935            },
936        }
937    }
938}

960/// Sort a mutable slice by the given indices array in place, reording the slice so that final
961/// value of `slice[x]` is `slice[indices[x]]`.
962fn sort_by_indices_in_place<T>(data: &mut [T], mut indices: Vec<usize>) {
963    for idx in 0..data.len() {
964        if indices[idx] == idx {
965            continue;
966        }
967
968        let mut current_idx = idx;
969        loop {
970            let target_idx = indices[current_idx];
971            indices[current_idx] = current_idx;
972            if indices[target_idx] == target_idx {
973                break;
974            }
975            data.swap(current_idx, target_idx);
976            current_idx = target_idx;
977        }
978    }

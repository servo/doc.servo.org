1//! Implements actual generic array abstraction for all supported types from `typenum` (1 to 255)
2
3use typenum::*;
4
5macro_rules! impl_generic_array {
6    ($($type: ty),*) => {
7        $(
8            impl private::Sealed for $type {}
9            impl Capacity for $type {
10                type Array = [u8; Self::USIZE];
11            }
12
13            impl private::Sealed for <$type as Capacity>::Array {}
14            impl ArraySlice for <$type as Capacity>::Array {
15                const CAPACITY: usize = <$type as Unsigned>::USIZE;
16
17                #[inline]
18                fn as_slice(&self) -> &[u8] {
19                    self
20                }
21
22                #[inline]
23                unsafe fn as_mut_slice(&mut self) -> &mut [u8] {
24                    self
25                }
26
27                #[inline]
28                fn zeroed() -> Self {
29                    [0; Self::CAPACITY]
30                }
31            }
32        )*
33    }
34}
35
36/// Private module to hide access to sealed trait
37mod private {
38    /// Trait impossible to be implemented outside of this crate, seals other traits
39    pub trait Sealed {}
40}
41
42/// Implements needed types for all types of arrays (bigger than 32 don't have the default traits)
43#[doc(hidden)]
44pub trait ArraySlice: private::Sealed {
45    /// Capacity represented by type
46    const CAPACITY: usize;
47
48    /// Returns slice of the entire array
49    fn as_slice(&self) -> &[u8];
50    /// Returns mutable slice of the entire array
51    unsafe fn as_mut_slice(&mut self) -> &mut [u8];
52    /// Returns array filled with zeroes
53    fn zeroed() -> Self;
54}
55
56/// Converts between `typenum` types and its corresponding array
57#[doc(hidden)]
58pub trait Capacity: Unsigned + private::Sealed {
59    /// Array with specified capacity
60    type Array: ArraySlice + Copy;
61}

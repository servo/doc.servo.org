1//! Compensating for the engine characteristics (rounding).
2//!
3//! Implements 4 instructions.
4//!
5//! See <https://learn.microsoft.com/en-us/typography/opentype/spec/tt_instructions#compensating-for-the-engine-characteristics>
6
7use super::{Engine, OpResult};
8
9impl Engine<'_> {
10    /// Round value.
11    ///
12    /// ROUND\[ab\] (0x68 - 0x6B)
13    ///
14    /// Pops: n1
15    /// Pushes: n2
16    ///
17    /// Rounds a value according to the state variable round_state while
18    /// compensating for the engine. n1 is popped off the stack and,
19    /// depending on the engine characteristics, is increased or decreased
20    /// by a set amount. The number obtained is then rounded and pushed
21    /// back onto the stack as n2.
22    ///
23    /// See <https://learn.microsoft.com/en-us/typography/opentype/spec/tt_instructions#round-value>
24    /// and <https://gitlab.freedesktop.org/freetype/freetype/-/blob/57617782464411201ce7bbc93b086c1b4d7d84a5/src/truetype/ttinterp.c#L3143>
25    pub(super) fn op_round(&mut self) -> OpResult {
26        // ClearType rounds to its own finer horizontal grid, which the
27        // interpreter approximates by not rounding at all.
28        let native_cleartype_x = self.native_cleartype_x();
29        let n1 = self.value_stack.pop_f26dot6()?;
30        let n2 = if native_cleartype_x {
31            n1
32        } else {
33            self.graphics.round(n1)
34        };
35        self.value_stack.push(n2.to_bits())
36    }
37}

47    // A glyph program that has opted out of backward compatibility measures
48    // and moves along the x axis lands between pixels, so ROUND leaves the
49    // value alone.
50    #[test]
51    fn round_is_suppressed_for_native_cleartype() {
52        let mut mock = MockEngine::new();
53        let mut engine = mock.engine();
54        engine.op_rtg().unwrap();
55        engine.program.reset(Program::Glyph);
56        engine.graphics.backward_compatibility = false;
57        engine.cleartype = ClearTypeState {
58            mode: ClearType::Enhanced,
59            ..Default::default()
60        };
61        for value in [32, 50, -32] {
62            engine.value_stack.push(value).unwrap();
63            engine.op_round().unwrap();
64            assert_eq!(engine.value_stack.pop().unwrap(), value);
65        }
66        // The same program still rounds while backward compatibility is on.
67        engine.graphics.backward_compatibility = true;
68        engine.value_stack.push(32).unwrap();
69        engine.op_round().unwrap();
70        assert_eq!(engine.value_stack.pop().unwrap(), 64);
71        // And a font that needs hinting to produce correct contours keeps the
72        // classic behavior.
73        engine.graphics.backward_compatibility = false;
74        engine.cleartype.is_hint_reliant = true;
75        engine.value_stack.push(32).unwrap();
76        engine.op_round().unwrap();
77        assert_eq!(engine.value_stack.pop().unwrap(), 64);
78    }

80    #[test]
81    fn round() {
82        let mut mock = MockEngine::new();
83        let mut engine = mock.engine();
84        use RoundMode::*;
85        let cases = [
86            (Grid, &[(0, 0), (32, 64), (-32, -64), (64, 64), (50, 64)]),
87            (
88                HalfGrid,
89                &[(0, 32), (32, 32), (-32, -32), (64, 96), (50, 32)],
90            ),
91            (
92                DoubleGrid,
93                &[(0, 0), (32, 32), (-32, -32), (64, 64), (50, 64)],
94            ),
95            (DownToGrid, &[(0, 0), (32, 0), (-32, 0), (64, 64), (50, 0)]),
96            (
97                UpToGrid,
98                &[(0, 0), (32, 64), (-32, -64), (64, 64), (50, 64)],
99            ),
100            (Off, &[(0, 0), (32, 32), (-32, -32), (64, 64), (50, 50)]),
101        ];
102        for (mode, values) in cases {
103            match mode {
104                Grid => engine.op_rtg().unwrap(),
105                HalfGrid => engine.op_rthg().unwrap(),
106                DoubleGrid => engine.op_rtdg().unwrap(),
107                DownToGrid => engine.op_rdtg().unwrap(),
108                UpToGrid => engine.op_rutg().unwrap(),
109                Off => engine.op_roff().unwrap(),
110                _ => unreachable!(),
111            }
112            for (input, expected) in values {
113                engine.value_stack.push(*input).unwrap();
114                engine.op_round().unwrap();
115                let result = engine.value_stack.pop().unwrap();
116                assert_eq!(*expected, result);
117            }
118        }
119    }

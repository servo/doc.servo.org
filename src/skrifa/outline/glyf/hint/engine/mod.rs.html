43/// TrueType bytecode interpreter.
44pub struct Engine<'a> {
45    program: ProgramState<'a>,
46    graphics: GraphicsState<'a>,
47    definitions: DefinitionState<'a>,
48    cvt: Cvt<'a>,
49    storage: Storage<'a>,
50    value_stack: ValueStack<'a>,
51    work_budget: WorkBudget,
52    axis_count: u16,
53    coords: &'a [F2Dot14],
54    cleartype: ClearTypeState,

57impl<'a> Engine<'a> {
58    #[allow(clippy::too_many_arguments)]
59    pub fn new(
60        cvt_len: usize,
61        program: ProgramState<'a>,
62        graphics: RetainedGraphicsState,
63        definitions: DefinitionState<'a>,
64        cvt: impl Into<Cvt<'a>>,
65        storage: impl Into<Storage<'a>>,
66        value_stack: ValueStack<'a>,
67        twilight: Zone<'a>,
68        glyph: Zone<'a>,
69        axis_count: u16,
70        coords: &'a [F2Dot14],
71        is_composite: bool,
72        cleartype: ClearTypeState,
73    ) -> Self {
74        let point_count = if glyph.points.is_empty() {
75            None
76        } else {
77            Some(glyph.points.len())
78        };
79        let graphics = GraphicsState {
80            retained: graphics,
81            zones: [twilight, glyph],
82            is_composite,
83            ..Default::default()
84        };
85        Self {
86            program,
87            graphics,
88            definitions,
89            cvt: cvt.into(),
90            storage: storage.into(),
91            value_stack,
92            work_budget: WorkBudget::new(cvt_len, point_count),
93            axis_count,
94            coords,
95            cleartype,
96        }
97    }

99    /// Returns true while a glyph program moves points along the horizontal
100    /// axis of native ClearType.
101    ///
102    /// ClearType has sixteen horizontal positions per pixel, and the
103    /// interpreter has one. The interpreter approximates the finer grid by
104    /// leaving the affected distances unrounded. This is computed on demand,
105    /// so changes to the projection and freedom vectors or to `INSTCTRL`
106    /// take effect immediately.
107    #[inline]
108    fn native_cleartype_x(&self) -> bool {
109        self.cleartype.is_native_enhanced()
110            && self.program.initial == Program::Glyph
111            && self.graphics.target.is_smooth()
112            && !self.graphics.backward_compatibility
113            && self.graphics.proj_vector.y == 0
114            && self.graphics.freedom_vector.y == 0
115    }
116
117    pub fn backward_compatibility(&self) -> bool {
118        self.graphics.backward_compatibility
119    }
120
121    pub fn retained_graphics_state(&self) -> &RetainedGraphicsState {
122        &self.graphics.retained
123    }
124}

126/// Tracks budgets for control flow to limit execution time.
127struct WorkBudget {
128    /// Maximum number of times we can do backward jumps or
129    /// loop calls.
130    loop_limit: usize,
131    /// Current number of backward jumps executed.
132    backward_jumps: usize,
133    /// Current number of loop call iterations executed.
134    loop_calls: usize,
135    /// Counts number of instructions skipped when handling if/else conditions.
136    skipped: usize,
137}
138
139impl WorkBudget {
140    fn new(cvt_len: usize, point_count: Option<usize>) -> Self {
141        // Compute limits for loop calls and backward jumps.
142        // See <https://gitlab.freedesktop.org/freetype/freetype/-/blob/57617782464411201ce7bbc93b086c1b4d7d84a5/src/truetype/ttinterp.c#L6955>
143        let loop_limit = if let Some(point_count) = point_count {
144            (point_count * 10).max(50) + (cvt_len / 10).max(50)
145        } else {
146            300 + 22 * cvt_len
147        };
148        // FreeType has two variables for neg_jump_counter_max and
149        // loopcall_counter_max but sets them to the same value so
150        // we'll just use a single limit.
151        Self {
152            loop_limit,
153            backward_jumps: 0,
154            loop_calls: 0,
155            skipped: 0,
156        }
157    }
158
159    fn reset(&mut self) {
160        self.backward_jumps = 0;
161        self.loop_calls = 0;
162        self.skipped = 0;
163    }
164
165    fn doing_backward_jump(&mut self) -> Result<(), HintErrorKind> {
166        self.backward_jumps += 1;
167        if self.backward_jumps > self.loop_limit {
168            Err(HintErrorKind::ExceededExecutionBudget)
169        } else {
170            Ok(())
171        }
172    }
173
174    fn doing_loop_call(&mut self, count: usize) -> Result<(), HintErrorKind> {
175        self.loop_calls += count;
176        if self.loop_calls > self.loop_limit {
177            Err(HintErrorKind::ExceededExecutionBudget)
178        } else {
179            Ok(())
180        }
181    }
182
183    fn skipping_instruction(&mut self) -> Result<(), HintErrorKind> {
184        self.skipped += 1;
185        if self.skipped > MAX_RUN_INSTRUCTIONS {
186            Err(HintErrorKind::ExceededExecutionBudget)
187        } else {
188            Ok(())
189        }
190    }
191}

294        /// Helper to push values to the stack, invoke a callback and check

296        pub(super) fn test_exec(
297            &mut self,
298            push: &[i32],
299            expected_result: impl Into<i32>,
300            mut f: impl FnMut(&mut Engine),
301        ) {
302            for &val in push {
303                self.value_stack.push(val).unwrap();
304            }
305            f(self);

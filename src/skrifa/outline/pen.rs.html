9/// Style for path conversion.
10///
11/// The order to process points in a glyf point stream is ambiguous when the
12/// first point is off-curve. Major implementations differ. Which one would
13/// you like to match?
14///
15/// **If you add a new one make sure to update the fuzzer.**
16#[derive(Debug, Default, Copy, Clone)]
17pub enum PathStyle {
18    /// If the first point is quadratic off-curve, check if the last is on-curve.
19    /// If it is, start there. If it isn't, start at the implied midpoint
20    /// between first and last.
21    #[default]
22    FreeType,
23    /// If the first point is quadratic off-curve, check if the second is on-curve.
24    /// If it is, start there. If it isn't, start at the implied midpoint
25    /// between first and second.
26    ///
27    /// Matches hb-draw's interpretation of a point stream.
28    HarfBuzz,
29}
30
31impl From<PathStyle> for PathContourStart {
32    fn from(style: PathStyle) -> Self {
33        match style {
34            PathStyle::FreeType => Self::ScanBackward,
35            PathStyle::HarfBuzz => Self::ScanForward,
36        }
37    }
38}

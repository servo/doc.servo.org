27/// We enable "best effort" mode by default but allow it to be disabled with
28/// a feature for testing.
29const SHAPER_MODE: ShaperMode = if cfg!(feature = "autohint_shaping") {
30    ShaperMode::BestEffort
31} else {
32    ShaperMode::Nominal
33};

35/// Set of derived glyph styles that are used for automatic hinting.
36///
37/// These are invariant per font so can be precomputed and reused for multiple
38/// instances when requesting automatic hinting with [`Engine::Auto`](super::super::hint::Engine::Auto).
39/// A [`ScalerFont`](crate::ScalerFont) stores them with its font and shares
40/// them automatically.
41#[derive(Clone, Debug)]
42pub struct GlyphStyles(Arc<GlyphStyleMap>);
43
44impl GlyphStyles {
45    /// Precomputes the full set of glyph styles for the given outlines.
46    pub fn new(outlines: &OutlineGlyphCollection) -> Self {
47        if let Some(font) = outlines.font() {
48            // The style table is sized by the glyph count, which a font can
49            // overstate; it never needs more entries than there are outlines.
50            Self::from_context(&FontRefContext::with_outlines(font, outlines, &[]))
51        } else {
52            Self(Default::default())
53        }
54    }
55
56    /// Precomputes the full set of glyph styles for the font behind
57    /// `context`.
58    pub(crate) fn from_context(context: &dyn AutohintContext) -> Self {
59        let shaper = Shaper::new(context, SHAPER_MODE);
60        Self(Arc::new(GlyphStyleMap::new(context.glyph_count(), &shaper)))
61    }
62
63    /// Returns the style for the given glyph.
64    pub fn get(&self, gid: GlyphId) -> Option<GlyphStyle> {
65        self.0.style(gid)
66    }
67}

69/// The data the automatic hinter derives from a font at one location.
70///
71/// It does not depend on the hinting target.
72pub(crate) struct LocatedState {
73    styles: GlyphStyles,
74    /// Measured from the outlines at the location. Each style is measured
75    /// when first drawn, or all at once without `std`.
76    metrics: UnscaledStyleMetricsSet,
77    is_fixed_width: bool,
78    style: Style,
79}
80
81impl LocatedState {
82    #[cfg_attr(not(feature = "std"), allow(unused_variables))]
83    fn new(context: &dyn AutohintContext, styles: GlyphStyles, lazy_metrics: bool) -> Self {
84        #[cfg(feature = "std")]
85        let metrics = if lazy_metrics {
86            UnscaledStyleMetricsSet::lazy(&styles.0)
87        } else {
88            UnscaledStyleMetricsSet::precomputed(context, SHAPER_MODE, &styles.0)
89        };
90        #[cfg(not(feature = "std"))]
91        let metrics = UnscaledStyleMetricsSet::precomputed(context, SHAPER_MODE, &styles.0);
92        Self {
93            styles,
94            metrics,
95            is_fixed_width: context.is_fixed_pitch(),
96            style: context.style(),
97        }
98    }
99
100    /// Returns true if both states share one allocation of glyph styles.
101    #[cfg(test)]
102    pub(crate) fn shares_styles_with(&self, other: &Self) -> bool {
103        Arc::ptr_eq(&self.styles.0, &other.styles.0)
104    }
105
106    /// Returns the state of `font` at its location, measured from
107    /// `outlines`.
108    ///
109    /// The state is computed once and stored with the font, so every scaler
110    /// that draws the font from the same sources shares one copy. One copy is
111    /// kept for each combination of base source and `VARC` that a scaler
112    /// draws from.
113    ///
114    /// The metrics are measured from outlines at the location, so a variable
115    /// font keeps one copy per location. A font without variations has only
116    /// its default location, so it keeps a single copy. The glyph styles are
117    /// the same at every location because style coverage reads the
118    /// substitutions for all locations. A font at a non-default location takes
119    /// them from the default location's copy instead of computing its own.
120    pub(crate) fn of_outlines(font: &Font, outlines: OutlineGlyphs<'_>) -> Arc<Self> {
121        let key = SourceKey::of(&outlines);
122        let build = || {
123            let context = super::context::FontContext::new(font, outlines);
124            let styles = if font.normalized_coords().is_empty() {
125                GlyphStyles::from_context(&context)
126            } else {
127                let default = font.default_instance();
128                let outlines = default.outline_glyphs_with(key.order());
129                Self::of_outlines(&default, outlines).styles.clone()
130            };
131            Arc::new(Self::new(&context, styles, true))
132        };
133        let data =
134            _font_interop::_get_or_init_scaling_data(font, || Box::new(ScalingData::default()));
135        match data.downcast_ref::<ScalingData>() {
136            Some(data) => data.autohint[key.0].get_or_init(build).clone(),
137            // Another version of this crate filled the slot, so this state
138            // is this scaler's alone. That rebuilds the styles and metrics,
139            // the bulk of a first draw, for every scaler of the font. Only a
140            // build linking two versions of the crate gets here.
141            None => build(),
142        }
143    }
144}

146/// The outline sources a [`LocatedState`] is measured from: the base source
147/// and whether `VARC` composites are drawn over it.
148#[derive(Copy, Clone)]
149struct SourceKey(usize);
150
151impl SourceKey {
152    /// The base sources, in the order of the keys.
153    const BASES: [Option<GlyphSource>; 5] = [
154        None,
155        Some(GlyphSource::Glyf),
156        Some(GlyphSource::Cff2),
157        Some(GlyphSource::Cff),
158        Some(GlyphSource::Type1),
159    ];
160    const COUNT: usize = Self::BASES.len() * 2;
161
162    fn of(outlines: &OutlineGlyphs<'_>) -> Self {
163        let base = match outlines.underlying_source() {
164            Some(OutlineSource::Glyf) => 1,
165            Some(OutlineSource::Cff2) => 2,
166            Some(OutlineSource::Cff) => 3,
167            Some(OutlineSource::Type1) => 4,
168            Some(OutlineSource::Varc) | None => 0,
169        };
170        let varc = outlines.source() == Some(OutlineSource::Varc);
171        Self(base * 2 + varc as usize)
172    }
173
174    /// Returns an order that resolves to the same sources.
175    fn order(self) -> GlyphSourceOrder {
176        let varc = (self.0 % 2 == 1).then_some(GlyphSource::Varc);
177        let sources: alloc::vec::Vec<_> = varc.into_iter().chain(Self::BASES[self.0 / 2]).collect();
178        GlyphSourceOrder::new(&sources)
179    }
180}

189#[cfg(feature = "std")]
190type Once<T> = std::sync::OnceLock<T>;
191
192/// A cell written once, for builds without `std`.
193#[cfg(not(feature = "std"))]
194struct Once<T>(once_cell::race::OnceBox<T>);
195
196#[cfg(not(feature = "std"))]
197impl<T> Default for Once<T> {
198    fn default() -> Self {
199        Self(once_cell::race::OnceBox::new())
200    }
201}
202
203#[cfg(not(feature = "std"))]
204impl<T> Once<T> {
205    fn get_or_init(&self, f: impl FnOnce() -> T) -> &T {
206        self.0.get_or_init(|| Box::new(f()))
207    }
208}

210#[derive(Clone)]
211pub(crate) struct Instance {
212    state: Arc<LocatedState>,
213    target: Target,
214}
215
216impl Instance {
217    /// Returns the state used by this instance.
218    ///
219    /// All instances for one font at one location share this state.
220    #[cfg(test)]
221    pub fn state(&self) -> &Arc<LocatedState> {
222        &self.state
223    }
224
225    pub fn new(
226        context: &dyn AutohintContext,
227        target: Target,
228        styles: Option<GlyphStyles>,
229        lazy_metrics: bool,
230    ) -> Self {
231        let styles = styles.unwrap_or_else(|| GlyphStyles::from_context(context));
232        Self {
233            state: Arc::new(LocatedState::new(context, styles, lazy_metrics)),
234            target,
235        }
236    }
237
238    /// Creates an instance that hints for `target` using precomputed state.
239    pub fn from_state(state: Arc<LocatedState>, target: Target) -> Self {
240        Self { state, target }
241    }
242
243    pub fn draw(
244        &self,
245        context: &dyn AutohintContext,
246        size: Size,
247        glyph_id: GlyphId,
248        path_style: PathStyle,
249        pen: &mut impl OutlinePen,
250    ) -> Result<AdjustedMetrics, DrawError> {
251        let state = &*self.state;
252        let style = state
253            .styles
254            .0
255            .style(glyph_id)
256            .ok_or(DrawError::GlyphNotFound(glyph_id))?;
257        let metrics = state
258            .metrics
259            .get(
260                |compute| compute(context),
261                SHAPER_MODE,
262                &state.styles.0,
263                glyph_id,
264            )
265            .ok_or(DrawError::GlyphNotFound(glyph_id))?;
266        let mut outline = Outline::default();
267        let has_overlaps = context
268            .fill_outline(glyph_id, &mut outline, super::QuirksMode::default())
269            .ok_or(DrawError::GlyphNotFound(glyph_id))??;
270        let units_per_em = outline.units_per_em;
271        let scale = Scale::new(
272            size.ppem().unwrap_or(units_per_em as f32),
273            units_per_em,
274            state.style,
275            self.target,
276            metrics.style_class().script.group,
277        );
278        let hinted_metrics = super::hint::hint_outline(&mut outline, &metrics, &scale, Some(style));
279        let h_advance = outline.advance;
280        let mut pp1x = 0;
281        let mut pp2x = fixed_mul(h_advance, hinted_metrics.x_scale);
282        let is_light = self.target.is_light() || self.target.preserve_linear_metrics();
283        // <https://gitlab.freedesktop.org/freetype/freetype/-/blob/57617782464411201ce7bbc93b086c1b4d7d84a5/src/autofit/afloader.c#L422>
284        if !is_light {
285            if let (true, Some(edge_metrics)) = (
286                !scale.flags.contains(ScaleFlags::NO_ADVANCE),
287                hinted_metrics.edge_metrics,
288            ) {
289                let old_rsb = pp2x - edge_metrics.right_opos;
290                let old_lsb = edge_metrics.left_opos;
291                let new_lsb = edge_metrics.left_pos;
292                let mut pp1x_uh = new_lsb - old_lsb;
293                let mut pp2x_uh = edge_metrics.right_pos + old_rsb;
294                if old_lsb < 24 {
295                    pp1x_uh -= 8;
296                }
297                if old_rsb < 24 {
298                    pp2x_uh += 8;
299                }
300                pp1x = pix_round(pp1x_uh);
301                pp2x = pix_round(pp2x_uh);
302                if pp1x >= new_lsb && old_lsb > 0 {
303                    pp1x -= 64;
304                }
305                if pp2x <= edge_metrics.right_pos && old_rsb > 0 {
306                    pp2x += 64;
307                }
308            } else {
309                pp1x = pix_round(pp1x);
310                pp2x = pix_round(pp2x);
311            }
312        } else {
313            pp1x = pix_round(pp1x);
314            pp2x = pix_round(pp2x);
315        }
316        if pp1x != 0 {
317            for point in &mut outline.points {
318                point.x -= pp1x;
319            }
320        }
321        let advance = if !is_light
322            && (state.is_fixed_width || (metrics.digits_have_same_width && style.is_digit()))
323        {
324            fixed_mul(h_advance, scale.x_scale)
325        } else if h_advance != 0 {
326            pp2x - pp1x
327        } else {
328            0
329        };
330        outline.to_path(path_style, pen)?;
331        Ok(AdjustedMetrics {
332            has_overlaps,
333            lsb: None,
334            advance_width: Some(F26Dot6::from_bits(pix_round(advance)).to_f32()),
335        })
336    }

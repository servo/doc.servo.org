10/// Errors that may occur when drawing glyphs.
11#[derive(Clone, Debug)]
12pub enum DrawError {
13    /// No viable sources were available.
14    NoSources,
15    /// The requested glyph was not present in the font.
16    GlyphNotFound(GlyphId),
17    /// Exceeded memory limits when loading a glyph.
18    InsufficientMemory,
19    /// Exceeded a recursion limit when loading a glyph.
20    RecursionLimitExceeded(GlyphId),
21    /// Glyph outline contains too many points.
22    TooManyPoints(GlyphId),
23    /// Error occurred during hinting.
24    HintingFailed(HintError),
25    /// An anchor point had invalid indices.
26    InvalidAnchorPoint(GlyphId, u16),
27    /// Error occurred while loading a PostScript (CFF/CFF2) glyph.
28    PostScript(CffError),
29    /// The font data does not make sense: absent where it was required, too
30    /// short, or self-inconsistent.
31    ///
32    /// Carries nothing. A table that is missing, one that is truncated and one
33    /// whose offsets do not resolve are the same answer to "can this glyph be
34    /// drawn", and there is nothing a caller can do differently between them.
35    Malformed,
36    /// HarfBuzz style drawing with hints is not supported
37    // Error rather than silently returning unhinted per f2f discussion.
38    HarfBuzzHintingUnsupported,
39}
40
41impl DrawError {
42    /// Converts an error from loading `glyph` through read-fonts.
43    ///
44    /// read-fonts errors do not include the glyph, since the caller already
45    /// has it. This adds the glyph to the error.
46    pub(crate) fn from_outline_error(error: read_fonts::model::Error, glyph: GlyphId) -> Self {
47        use read_fonts::model::Error;
48        match error {
49            Error::MissingResource => Self::GlyphNotFound(glyph),
50            Error::LimitExceeded => Self::RecursionLimitExceeded(glyph),
51            Error::InsufficientMemory => Self::InsufficientMemory,
52            // Only a charstring fails this way; a TrueType hinter reports its
53            // own error.
54            Error::InvalidOperation => Self::PostScript(error),
55            _ => Self::Malformed,
56        }
57    }
58}
59
60impl From<HintError> for DrawError {
61    fn from(value: HintError) -> Self {
62        Self::HintingFailed(value)
63    }
64}
65
66impl From<CffError> for DrawError {
67    fn from(value: CffError) -> Self {
68        Self::PostScript(value)
69    }
70}

72impl fmt::Display for DrawError {
73    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
74        match self {
75            Self::NoSources => write!(f, "No glyph sources are available for the given font"),
76            Self::GlyphNotFound(gid) => write!(f, "Glyph {gid} was not found in the given font"),
77            Self::InsufficientMemory => write!(f, "exceeded memory limits"),
78            Self::RecursionLimitExceeded(gid) => write!(
79                f,
80                "Recursion limit ({}) exceeded when loading composite component {gid}",
81                read_fonts::limits::MAX_RECURSION_DEPTH,
82            ),
83            Self::TooManyPoints(gid) => write!(f, "Glyph {gid} contains more than 64k points"),
84            Self::HintingFailed(e) => write!(f, "{e}"),
85            Self::InvalidAnchorPoint(gid, index) => write!(
86                f,
87                "Invalid anchor point index ({index}) for composite glyph {gid}",
88            ),
89            Self::PostScript(e) => write!(f, "{e}"),
90            Self::Malformed => write!(f, "font data was absent or malformed"),
91            Self::HarfBuzzHintingUnsupported => write!(
92                f,
93                "HarfBuzz style paths with hinting is not (yet?) supported"
94            ),
95        }
96    }
97}

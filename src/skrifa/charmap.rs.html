1//! Mapping of characters (codepoints, not graphemes) to nominal glyph identifiers.
2//!
3//! If you have never run into character to glyph mapping before
4//! [Glyph IDs and the 'cmap' table](https://rsheeter.github.io/font101/#glyph-ids-and-the-cmap-table)
5//! might be informative.
6//!
7//! The functionality in this module provides a 1-to-1 mapping from Unicode
8//! characters (or [Unicode variation sequences](http://unicode.org/faq/vs.html)) to
9//! nominal or "default" internal glyph identifiers for a given font.
10//! This is a necessary first step, but generally insufficient for proper layout of
11//! [complex text](https://en.wikipedia.org/wiki/Complex_text_layout) or even
12//! simple text containing diacritics and ligatures.
13//!
14//! Comprehensive mapping of characters to positioned glyphs requires a process called
15//! shaping. For more detail, see: [Why do I need a shaping engine?](https://harfbuzz.github.io/why-do-i-need-a-shaping-engine.html)
16

28/// Mapping of characters to nominal glyph identifiers.
29///
30/// The mappings are derived from the [cmap](https://learn.microsoft.com/en-us/typography/opentype/spec/cmap)
31/// table. Depending on the font, the returned mapping may have entries that point to virtual/phantom glyph
32/// ids beyond the glyph count of the `MAXP` or `maxp` table, which are only used during the shaping process,
33/// for example.
34/// An optional `DMAP` table (ISO/IEC 14496-22:2026) is consulted first, with
35/// unmapped characters and variation sequences falling back to `cmap`.
36///
37/// ## Obtaining a Charmap
38///
39/// Typically a Charmap is acquired by calling [charmap](crate::MetadataProvider::charmap) on a [FontRef].
40///
41/// ## Selection strategy
42///
43/// Fonts may contain multiple subtables in various formats supporting different encodings. The selection
44/// strategy implemented here is designed to choose mappings that capture the broadest available Unicode
45/// coverage:
46///
47/// * Unicode characters: a symbol mapping subtable is selected if available. Otherwise, subtables supporting
48///   the Unicode full repertoire or Basic Multilingual Plane (BMP) are preferred, in that order. Formats
49///   [4](https://learn.microsoft.com/en-us/typography/opentype/spec/cmap#format-4-segment-mapping-to-delta-values)
50///   and [12](https://learn.microsoft.com/en-us/typography/opentype/spec/cmap#format-12-segmented-coverage) are
51///   supported.
52///
53/// * Unicode variation sequences: these are provided by a format
54///   [14](https://learn.microsoft.com/en-us/typography/opentype/spec/cmap#format-14-unicode-variation-sequences)
55///   or format 15 (ISO/IEC 14496-22:2026) subtable, preferring format 15.
56///
57#[derive(Clone, Default)]
58pub struct Charmap<'a> {
59    cmap: TableCharmap<'a>,
60    dmap: TableCharmap<'a>,
61    cmap_limits: CmapIterLimits,
62}
63
64#[derive(Clone, Default)]
65struct TableCharmap<'a> {
66    codepoint_subtable: Option<CodepointSubtable<'a>>,
67    variant_subtable: Option<UvsSubtable<'a>>,
68}
69
70impl<'a> Charmap<'a> {
71    /// Creates a new character map from the given font.
72    pub fn new(font: &FontRef<'a>) -> Self {
73        Self {
74            cmap: TableCharmap::new(font.cmap().ok()),
75            dmap: TableCharmap::new(font.dmap().ok().map(|table| table.as_cmap())),
76            cmap_limits: CmapIterLimits::default_for_font(font),
77        }
78    }
79
80    /// Returns true if a suitable Unicode character mapping is available.
81    pub fn has_map(&self) -> bool {
82        self.dmap.codepoint_subtable.is_some() || self.cmap.codepoint_subtable.is_some()
83    }
84
85    /// Returns true if a symbol mapping was selected.
86    pub fn is_symbol(&self) -> bool {
87        [&self.dmap, &self.cmap].iter().any(|table| {
88            table
89                .codepoint_subtable
90                .as_ref()
91                .is_some_and(|x| x.is_symbol)
92        })
93    }
94
95    /// Returns true if a Unicode variation sequence mapping is available.
96    pub fn has_variant_map(&self) -> bool {
97        self.dmap.variant_subtable.is_some() || self.cmap.variant_subtable.is_some()
98    }
99
100    /// Maps a character to a nominal glyph identifier.
101    ///
102    /// Returns `None` if a mapping does not exist.
103    pub fn map(&self, ch: impl Into<u32>) -> Option<GlyphId> {
104        let ch = ch.into();
105        self.dmap
106            .codepoint_subtable
107            .as_ref()
108            .and_then(|table| table.map(ch))
109            .or_else(|| self.cmap.codepoint_subtable.as_ref()?.map(ch))
110    }
111
112    /// Returns an iterator over all mappings of codepoint to nominal glyph
113    /// identifiers in the character map.
114    pub fn mappings(&self) -> Mappings<'a> {
115        Mappings {
116            dmap: self.dmap.mappings(self.cmap_limits).peekable(),
117            cmap: self.cmap.mappings(self.cmap_limits).peekable(),
118        }
119    }
120
121    /// Maps a character and variation selector to a nominal glyph identifier.
122    ///
123    /// Returns `None` if a mapping does not exist.
124    pub fn map_variant(&self, ch: impl Into<u32>, selector: impl Into<u32>) -> Option<MapVariant> {
125        let (ch, selector) = (ch.into(), selector.into());
126        self.dmap
127            .variant_subtable
128            .as_ref()
129            .and_then(|table| table.map_variant(ch, selector))
130            .or_else(|| {
131                self.cmap
132                    .variant_subtable
133                    .as_ref()?
134                    .map_variant(ch, selector)
135            })
136    }
137
138    /// Returns an iterator over all mappings of character and variation
139    /// selector to nominal glyph identifier in the character map.
140    pub fn variant_mappings(&self) -> VariantMappings<'a> {
141        VariantMappings {
142            dmap: self
143                .dmap
144                .variant_subtable
145                .clone()
146                .map(|table| table.iter_with_limits(self.cmap_limits)),
147            cmap: self
148                .cmap
149                .variant_subtable
150                .clone()
151                .map(|table| table.iter_with_limits(self.cmap_limits)),
152            dmap_subtable: self.dmap.variant_subtable.clone(),
153        }
154    }
155}
156
157impl<'a> TableCharmap<'a> {
158    fn new(cmap: Option<Cmap<'a>>) -> Self {
159        let Some(cmap) = cmap else {
160            return Self::default();
161        };
162        let selection = MappingSelection::new(&cmap);
163        Self {
164            codepoint_subtable: selection
165                .codepoint_subtable
166                .map(|subtable| CodepointSubtable {
167                    subtable,
168                    is_symbol: selection.mapping_index.codepoint_subtable_is_symbol,
169                }),
170            variant_subtable: selection.variant_subtable,
171        }
172    }
173
174    fn mappings(&self, limits: CmapIterLimits) -> SubtableMappings<'a> {
175        self.codepoint_subtable
176            .as_ref()
177            .map(|subtable| {
178                SubtableMappings(match &subtable.subtable {
179                    SupportedSubtable::Format4(cmap4) => MappingsInner::Format4(cmap4.iter()),
180                    SupportedSubtable::Format12(cmap12) => {
181                        MappingsInner::Format12(cmap12.iter_with_limits(limits))
182                    }
183                    SupportedSubtable::Format13(cmap13) => {
184                        MappingsInner::Format13(cmap13.iter_with_limits(limits))
185                    }
186                })
187            })
188            .unwrap_or(SubtableMappings(MappingsInner::None))
189    }
190}

192/// Cacheable indices of selected mapping tables for materializing a character
193/// map.
194///
195/// Since [`Charmap`] carries a lifetime, it is difficult to store in a cache.
196/// This type serves as an acceleration structure that allows for construction
197/// of a character map while skipping the search for the most suitable Unicode
198/// mappings.
199#[derive(Copy, Clone, Default, Debug)]
200pub struct MappingIndex {
201    cmap: TableMappingIndex,
202    dmap: TableMappingIndex,
203    cmap_limits: CmapIterLimits,
204}
205
206#[derive(Copy, Clone, Default, Debug)]
207struct TableMappingIndex {
208    /// Index of Unicode or symbol mapping subtable.
209    codepoint_subtable: Option<u16>,
210    /// True if the above is a symbol mapping.
211    codepoint_subtable_is_symbol: bool,
212    /// Index of Unicode variation selector subtable.
213    variant_subtable: Option<u16>,
214}
215
216impl MappingIndex {
217    /// Finds the indices of the most suitable Unicode mapping tables in the
218    /// given font.
219    pub fn new(font: &FontRef) -> Self {
220        let index = |table: Option<Cmap<'_>>| {
221            table
222                .map(|table| MappingSelection::new(&table).mapping_index)
223                .unwrap_or_default()
224        };
225        Self {
226            cmap: index(font.cmap().ok()),
227            dmap: index(font.dmap().ok().map(|table| table.as_cmap())),
228            cmap_limits: CmapIterLimits::default_for_font(font),
229        }
230    }
231
232    /// Creates a new character map for the given font using the tables referenced by
233    /// the precomputed indices.
234    ///
235    /// The font should be the same as the one used to construct this object.
236    pub fn charmap<'a>(&self, font: &FontRef<'a>) -> Charmap<'a> {
237        Charmap {
238            cmap: self.cmap.charmap(font.cmap().ok()),
239            dmap: self
240                .dmap
241                .charmap(font.dmap().ok().map(|table| table.as_cmap())),
242            cmap_limits: self.cmap_limits,
243        }
244    }
245}
246
247impl TableMappingIndex {
248    fn charmap<'a>(&self, cmap: Option<Cmap<'a>>) -> TableCharmap<'a> {
249        let Some(cmap) = cmap else {
250            return TableCharmap::default();
251        };
252        let records = cmap.encoding_records();
253        let data = cmap.offset_data();
254        TableCharmap {
255            codepoint_subtable: self
256                .codepoint_subtable
257                .and_then(|index| get_subtable(data, records, index))
258                .and_then(SupportedSubtable::new)
259                .map(|subtable| CodepointSubtable {
260                    subtable,
261                    is_symbol: self.codepoint_subtable_is_symbol,
262                }),
263            variant_subtable: self
264                .variant_subtable
265                .and_then(|index| get_subtable(data, records, index))
266                .and_then(|subtable| match subtable {
267                    CmapSubtable::Format14(table) => Some(UvsSubtable::from(table)),
268                    CmapSubtable::Format15(table) => Some(UvsSubtable::from(table)),
269                    _ => None,
270                }),
271        }
272    }
273}

275/// Iterator over all mappings of character to nominal glyph identifier
276/// in a character map.
277///
278/// This is created with the [`Charmap::mappings`] method.
279#[derive(Clone)]
280pub struct Mappings<'a> {
281    dmap: core::iter::Peekable<SubtableMappings<'a>>,
282    cmap: core::iter::Peekable<SubtableMappings<'a>>,
283}
284
285impl Iterator for Mappings<'_> {
286    type Item = (u32, GlyphId);
287
288    fn next(&mut self) -> Option<Self::Item> {
289        match (self.dmap.peek(), self.cmap.peek()) {
290            (Some(delta), Some(base)) if delta.0 == base.0 => {
291                self.cmap.next();
292                self.dmap.next()
293            }
294            (Some(delta), Some(base)) if delta.0 < base.0 => self.dmap.next(),
295            (Some(_), None) => self.dmap.next(),
296            _ => self.cmap.next(),
297        }
298    }
299}
300
301#[derive(Clone)]
302struct SubtableMappings<'a>(MappingsInner<'a>);
303
304impl Iterator for SubtableMappings<'_> {
305    type Item = (u32, GlyphId);
306
307    fn next(&mut self) -> Option<Self::Item> {
308        loop {
309            let item = match &mut self.0 {
310                MappingsInner::None => None,
311                MappingsInner::Format4(iter) => iter.next(),
312                MappingsInner::Format12(iter) => iter.next(),
313                MappingsInner::Format13(iter) => iter.next(),
314            }?;
315            if item.1 != GlyphId::NOTDEF {
316                return Some(item);
317            }
318        }
319    }
320}

322#[derive(Clone)]
323enum MappingsInner<'a> {
324    None,
325    Format4(Cmap4Iter<'a>),
326    Format12(Cmap12Iter<'a>),
327    Format13(Cmap13Iter<'a>),
328}

330/// Iterator over all mappings of character and variation selector to
331/// nominal glyph identifier in a character map.
332///
333/// This is created with the [`Charmap::variant_mappings`] method.
334#[derive(Clone)]
335pub struct VariantMappings<'a> {
336    dmap: Option<UvsSubtableIter<'a>>,
337    cmap: Option<UvsSubtableIter<'a>>,
338    dmap_subtable: Option<UvsSubtable<'a>>,
339}
340
341impl Iterator for VariantMappings<'_> {
342    type Item = (u32, u32, MapVariant);
343
344    fn next(&mut self) -> Option<Self::Item> {
345        if let Some(item) = self.dmap.as_mut().and_then(Iterator::next) {
346            return Some(item);
347        }
348        self.cmap.as_mut()?.find(|(ch, selector, _)| {
349            self.dmap_subtable
350                .as_ref()
351                .and_then(|table| table.map_variant(*ch, *selector))
352                .is_none()
353        })
354    }
355}

357fn get_subtable<'a>(
358    data: FontData<'a>,
359    records: &[EncodingRecord],
360    index: u16,
361) -> Option<CmapSubtable<'a>> {
362    records
363        .get(index as usize)
364        .and_then(|record| record.subtable(data).ok())
365}

367#[derive(Clone)]
368struct CodepointSubtable<'a> {
369    subtable: SupportedSubtable<'a>,
370    /// True if the subtable is a symbol mapping.
371    is_symbol: bool,
372}
373
374impl CodepointSubtable<'_> {
375    fn map(&self, codepoint: u32) -> Option<GlyphId> {
376        self.map_impl(codepoint).or_else(|| {
377            if self.is_symbol && codepoint <= 0x00FF {
378                // From HarfBuzz:
379                // For symbol-encoded OpenType fonts, we duplicate the
380                // U+F000..F0FF range at U+0000..U+00FF.  That's what
381                // Windows seems to do, and that's hinted about at:
382                // https://docs.microsoft.com/en-us/typography/opentype/spec/recom
383                // under "Non-Standard (Symbol) Fonts".
384                // See <https://github.com/harfbuzz/harfbuzz/blob/453ded05392af38bba9f89587edce465e86ffa6b/src/hb-ot-cmap-table.hh#L1595>
385                self.map_impl(codepoint + 0xF000)
386            } else {
387                None
388            }
389        })
390    }
391
392    fn map_impl(&self, codepoint: u32) -> Option<GlyphId> {
393        let gid = match &self.subtable {
394            SupportedSubtable::Format4(subtable) => subtable.map_codepoint(codepoint),
395            SupportedSubtable::Format12(subtable) => subtable.map_codepoint(codepoint),
396            SupportedSubtable::Format13(subtable) => subtable.map_codepoint(codepoint),
397        }?;
398        (gid != GlyphId::NOTDEF).then_some(gid)
399    }
400}
401
402#[derive(Clone)]
403enum SupportedSubtable<'a> {
404    Format4(Cmap4<'a>),
405    Format12(Cmap12<'a>),
406    Format13(Cmap13<'a>),
407}
408
409impl<'a> SupportedSubtable<'a> {
410    fn new(subtable: CmapSubtable<'a>) -> Option<Self> {
411        Some(match subtable {
412            CmapSubtable::Format4(cmap4) => Self::Format4(cmap4),
413            CmapSubtable::Format12(cmap12) => Self::Format12(cmap12),
414            CmapSubtable::Format13(cmap13) => Self::Format13(cmap13),
415            _ => return None,
416        })
417    }
418
419    fn from_cmap_record(cmap: &Cmap<'a>, record: &cmap::EncodingRecord) -> Option<Self> {
420        Self::new(record.subtable(cmap.offset_data()).ok()?)
421    }
422}

424/// The mapping kind of a cmap subtable.
425///
426/// The ordering is significant and determines the priority of subtable
427/// selection (greater is better).
428#[derive(Copy, Clone, PartialEq, PartialOrd)]
429enum MappingKind {
430    None = 0,
431    UnicodeBmp = 1,
432    UnicodeFull = 2,
433    Symbol = 3,
434}
435
436/// The result of searching the cmap table for the "best" available
437/// subtables.
438///
439/// For `codepoint_subtable`, best means either symbol (which is preferred)
440/// or a Unicode subtable with the greatest coverage.
441///
442/// For `variant_subtable`, best means format 15, falling back to format 14.
443struct MappingSelection<'a> {
444    /// The mapping index accelerator that holds indices of the following
445    /// subtables.
446    mapping_index: TableMappingIndex,
447    /// Either a symbol subtable or the Unicode subtable with the
448    /// greatest coverage.
449    codepoint_subtable: Option<SupportedSubtable<'a>>,
450    /// Subtable that supports mapping Unicode variation sequences.
451    variant_subtable: Option<UvsSubtable<'a>>,
452}
453
454impl<'a> MappingSelection<'a> {
455    fn new(cmap: &Cmap<'a>) -> Self {
456        const ENCODING_MS_SYMBOL: u16 = 0;
457        const ENCODING_MS_UNICODE_CS: u16 = 1;
458        const ENCODING_APPLE_ID_UNICODE_32: u16 = 4;
459        const ENCODING_APPLE_ID_VARIANT_SELECTOR: u16 = 5;
460        const ENCODING_UNICODE_FULL_REPERTOIRE: u16 = 6;
461        const ENCODING_MS_ID_UCS_4: u16 = 10;
462        let mut mapping_index = TableMappingIndex::default();
463        let mut mapping_kind = MappingKind::None;
464        let mut codepoint_subtable = None;
465        let mut variant_subtable = None;
466        if let Some((index, subtable)) = cmap.uvs_subtable() {
467            mapping_index.variant_subtable = Some(index);
468            variant_subtable = Some(subtable);
469        }
470        let mut maybe_choose_subtable = |kind, index, subtable| {
471            if kind > mapping_kind {
472                mapping_kind = kind;
473                mapping_index.codepoint_subtable_is_symbol = kind == MappingKind::Symbol;
474                mapping_index.codepoint_subtable = Some(index as u16);
475                codepoint_subtable = Some(subtable);
476            }
477        };
478        // This generally follows the same strategy as FreeType, searching the encoding
479        // records in reverse and prioritizing UCS-4 subtables over UCS-2.
480        // See <https://gitlab.freedesktop.org/freetype/freetype/-/blob/ac5babe87629107c43f627e2cd17c6cf4f2ecd43/src/base/ftobjs.c#L1370>
481        // The exception is that we prefer a symbol subtable over all others which matches the behavior
482        // of HarfBuzz.
483        // See <https://github.com/harfbuzz/harfbuzz/blob/453ded05392af38bba9f89587edce465e86ffa6b/src/hb-ot-cmap-table.hh#L1818>
484        for (i, record) in cmap.encoding_records().iter().enumerate().rev() {
485            match (record.platform_id(), record.encoding_id()) {
486                // Unicode variation sequences are chosen by Cmap::uvs_subtable.
487                (PlatformId::Unicode, ENCODING_APPLE_ID_VARIANT_SELECTOR) => {}
488                (PlatformId::Windows, ENCODING_MS_SYMBOL) => {
489                    // Symbol
490                    if let Some(subtable) = SupportedSubtable::from_cmap_record(cmap, record) {
491                        maybe_choose_subtable(MappingKind::Symbol, i, subtable);
492                    }
493                }
494                (PlatformId::Windows, ENCODING_MS_ID_UCS_4)
495                | (PlatformId::Unicode, ENCODING_APPLE_ID_UNICODE_32)
496                | (PlatformId::Unicode, ENCODING_UNICODE_FULL_REPERTOIRE) => {
497                    // Unicode full repertoire
498                    if let Some(subtable) = SupportedSubtable::from_cmap_record(cmap, record) {
499                        maybe_choose_subtable(MappingKind::UnicodeFull, i, subtable);
500                    }
501                }
502                (PlatformId::ISO, _)
503                | (PlatformId::Unicode, _)
504                | (PlatformId::Windows, ENCODING_MS_UNICODE_CS) => {
505                    // Unicode BMP only
506                    if let Some(subtable) = SupportedSubtable::from_cmap_record(cmap, record) {
507                        maybe_choose_subtable(MappingKind::UnicodeBmp, i, subtable);
508                    }
509                }
510                _ => {}
511            }
512        }
513        Self {
514            mapping_index,
515            codepoint_subtable,
516            variant_subtable,
517        }
518    }
519}

527    #[test]
528    fn format15_variants_and_cached_selection() {
529        use font_test_data::cmap::{font_with_cmaps, format12, format14, format15, table};
530        let nominal = format12(&[(65, 1)]);
531        let old = format14(0xfe0f, &[], &[(65, 2), (67, 3)]);
532        let new = format15(0xfe0f, &[65], &[(66, 70000), (68, 0xffffff)]);
533        for records in [
534            vec![
535                (3, 10, nominal.as_slice()),
536                (0, 5, old.as_slice()),
537                (0, 5, new.as_slice()),
538            ],
539            vec![
540                (3, 10, nominal.as_slice()),
541                (0, 5, new.as_slice()),
542                (0, 5, old.as_slice()),
543            ],
544        ] {
545            let base = table(&records);
546            let data = font_with_cmaps(Some(&base), None);
547            let font = FontRef::new(&data).unwrap();
548            for charmap in [font.charmap(), MappingIndex::new(&font).charmap(&font)] {
549                assert!(charmap.has_variant_map());
550                let expected = [
551                    (65, 0xfe0f, MapVariant::UseDefault),

553                    (68, 0xfe0f, MapVariant::Variant(GlyphId::new(0xffffff))),
554                ];
555                assert_eq!(charmap.variant_mappings().collect::<Vec<_>>(), expected);
556                for (ch, selector, mapping) in expected {
557                    assert_eq!(charmap.map_variant(ch, selector), Some(mapping));
558                }
559                assert_eq!(charmap.map_variant(67u32, 0xfe0fu32), None);
560            }
561        }

564    #[test]
565    fn dmap_variation_precedence_across_formats() {
566        use font_test_data::cmap::{font_with_cmaps, format14, format15, table};
567        let old = format14(0xfe0f, &[], &[(65, 1), (66, 2)]);
568        let new = format15(0xfe0f, &[], &[(65, 70000), (67, 80000)]);
569        for (base, delta, expected) in [
570            (&old, &new, vec![(65u32, 70000), (66, 2), (67, 80000)]),
571            (&new, &old, vec![(65u32, 1), (66, 2), (67, 80000)]),
572        ] {
573            let base = table(&[(0, 5, base)]);
574            let delta = table(&[(0, 5, delta)]);
575            for cmap in [Some(base.as_slice()), None] {
576                let data = font_with_cmaps(cmap, Some(&delta));
577                let font = FontRef::new(&data).unwrap();
578                for charmap in [font.charmap(), MappingIndex::new(&font).charmap(&font)] {
579                    let mut seen = Vec::new();
580                    for (ch, selector, mapping) in charmap.variant_mappings() {
581                        assert_eq!(charmap.map_variant(ch, selector), Some(mapping));
582                        assert!(!seen.contains(&(ch, selector)));
583                        seen.push((ch, selector));
584                    }
585                    if cmap.is_some() {
586                        assert_eq!(seen.len(), expected.len());

695    #[test]
696    fn dmap_and_cmap15_agree_across_public_charmaps_at_boundaries() {
697        use font_test_data::cmap::{font_with_cmaps, format12, format15, table};
698        use read_fonts::model::Font;
699
700        let selector = 0xe0100;
701        let base = table(&[
702            (
703                3,
704                10,
705                &format12(&[(0x20000, 1), (0x20001, 2), (0x20002, 2), (0x10ffff, 4)]),
706            ),
707            (
708                0,
709                5,
710                &format15(
711                    selector,
712                    &[],
713                    &[
714                        (0x20000, 3),
715                        (0x20001, 4),
716                        (0x20002, 65536),
717                        (0x10ffff, 65535),
718                    ],
719                ),
720            ),
721        ]);
722        let delta = table(&[
723            (
724                3,
725                10,
726                &format12(&[(0x20000, 3), (0x20001, 4), (0x20002, 0), (0x10ffff, 5)]),
727            ),
728            (
729                0,
730                5,
731                &format15(selector, &[0x20000], &[(0x20001, 0xffffff)]),
732            ),
733        ]);
734        let bytes = font_with_cmaps(Some(&base), Some(&delta));
735        let font = FontRef::new(&bytes).unwrap();
736        let cached = Font::new(bytes.clone(), 0).unwrap();
737        let index = MappingIndex::new(&font);
738        let nominal = [(0x20000, 3), (0x20001, 4), (0x20002, 2), (0x10ffff, 5)]
739            .map(|(ch, gid)| (ch, GlyphId::new(gid)));
740        let variants = [
741            (0x20000, 3),
742            (0x20001, 0xffffff),
743            (0x20002, 65536),
744            (0x10ffff, 65535),
745        ]
746        .map(|(ch, gid)| (ch, selector, GlyphId::new(gid)));
747        for charmap in [font.charmap(), index.charmap(&font)] {
748            let mut mappings: Vec<_> = charmap.mappings().collect();
749            mappings.sort_unstable();
750            assert_eq!(mappings, nominal);
751            let mut resolved: Vec<_> = charmap
752                .variant_mappings()
753                .map(|(ch, vs, variant)| {
754                    let glyph = match variant {
755                        MapVariant::UseDefault => charmap.map(ch).unwrap(),
756                        MapVariant::Variant(glyph) => glyph,
757                    };
758                    (ch, vs, glyph)
759                })
760                .collect();
761            resolved.sort_unstable();
762            assert_eq!(resolved, variants);
763            for (ch, glyph) in nominal {
764                assert_eq!(charmap.map(ch), Some(glyph));
765            }
766            for (ch, vs, glyph) in variants {
767                let variant = charmap.map_variant(ch, vs).unwrap();
768                assert_eq!(
769                    match variant {
770                        MapVariant::UseDefault => charmap.map(ch).unwrap(),
771                        MapVariant::Variant(glyph) => glyph,
772                    },
773                    glyph
774                );
775            }
776            assert_eq!(charmap.map_variant(0x20000u32, selector + 1), None);
777            assert_eq!(charmap.map(0x20003u32), None);
778        }
779        for instance in [
780            &cached,
781            &cached.default_instance(),
782            &cached.instance_builder().build(),
783        ] {
784            let charmap = instance.charmap();
785            let mut mappings: Vec<_> = charmap.unicodes().collect();
786            mappings.sort_unstable();
787            assert_eq!(mappings, nominal);
788            let mut resolved: Vec<_> = charmap.unicode_variants().collect();
789            resolved.sort_unstable();
790            assert_eq!(resolved, variants);
791            for (ch, glyph) in nominal {
792                assert_eq!(charmap.map_unicode(ch), Some(glyph));
793            }
794            for (ch, vs, glyph) in variants {
795                assert_eq!(charmap.map_unicode_variant(ch, vs), Some(glyph));
796            }
797        }
798    }

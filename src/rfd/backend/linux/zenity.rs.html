9#[derive(Debug)]
10pub enum ZenityError {
11    Io(std::io::Error),
12    FromUtf8Error(std::string::FromUtf8Error),
13}
14
15impl Error for ZenityError {}
16
17impl Display for ZenityError {
18    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
19        match self {
20            ZenityError::Io(io) => write!(f, "{io}"),
21            ZenityError::FromUtf8Error(err) => err.fmt(f),
22        }
23    }
24}
25
26impl From<std::io::Error> for ZenityError {
27    fn from(value: std::io::Error) -> Self {
28        Self::Io(value)
29    }
30}
31
32impl From<std::string::FromUtf8Error> for ZenityError {
33    fn from(value: std::string::FromUtf8Error) -> Self {
34        Self::FromUtf8Error(value)
35    }
36}

38pub type ZenityResult<T> = Result<T, ZenityError>;

40fn command() -> Command {
41    let mut cmd = Command::new("zenity");
42    cmd.arg("--no-markup");
43    cmd
44}

46fn add_filters(command: &mut Command, filters: &[Filter]) {
47    for f in filters.iter() {
48        command.arg("--file-filter");
49
50        let extensions: Vec<_> = f
51            .extensions
52            .iter()
53            .map(|ext| format!("*.{}", ext))
54            .collect();
55
56        command.arg(format!("{} | {}", f.name, extensions.join(" ")));
57    }
58}
59
60fn add_filename(command: &mut Command, file_name: &Option<String>) {
61    if let Some(name) = file_name.as_ref() {
62        command.arg("--filename");
63        command.arg(name);
64    }
65}

67async fn run(mut command: Command) -> ZenityResult<Option<String>> {
68    let res = {
69        let (tx, rx) = crate::oneshot::channel();
70        std::thread::spawn(move || {
71            tx.send(command.output()).ok();
72        });
73        rx.await.map_err(std::io::Error::other)??
74    };
75    let buffer = String::from_utf8(res.stdout)?;
76
77    Ok((res.status.success() || !buffer.is_empty()).then_some(buffer))
78}

80pub async fn pick_file(dialog: &FileDialog) -> ZenityResult<Option<PathBuf>> {
81    let mut command = command();
82    command.arg("--file-selection");
83
84    add_filters(&mut command, &dialog.filters);
85    add_filename(&mut command, &dialog.file_name);
86
87    run(command).await.map(|res| {
88        res.map(|buffer| {
89            let trimed = buffer.trim();
90            trimed.into()
91        })
92    })
93}

157pub async fn message(
158    level: &MessageLevel,
159    btns: &MessageButtons,
160    title: &str,
161    description: &str,
162) -> ZenityResult<MessageDialogResult> {
163    let cmd = match level {
164        MessageLevel::Info => "--info",
165        MessageLevel::Warning => "--warning",
166        MessageLevel::Error => "--error",
167    };
168
169    let ok_label = match btns {
170        MessageButtons::Ok => None,
171        MessageButtons::OkCustom(ok) => Some(ok),
172        _ => None,
173    };
174
175    let mut command = command();
176    command.args([cmd, "--title", title, "--text", description]);
177
178    if let Some(ok) = ok_label {
179        command.args(["--ok-label", ok]);
180    }
181
182    run(command).await.map(|res| match res {
183        Some(_) => MessageDialogResult::Ok,
184        None => MessageDialogResult::Cancel,
185    })
186}

188pub async fn question(
189    btns: &MessageButtons,
190    title: &str,
191    description: &str,
192) -> ZenityResult<MessageDialogResult> {
193    let mut command = command();
194    command.args(["--question", "--title", title, "--text", description]);
195
196    match btns {
197        MessageButtons::OkCancel => {
198            command.args(["--ok-label", "Ok"]);
199            command.args(["--cancel-label", "Cancel"]);
200        }
201        MessageButtons::OkCancelCustom(ok, cancel) => {
202            command.args(["--ok-label", ok.as_str()]);
203            command.args(["--cancel-label", cancel.as_str()]);
204        }
205        MessageButtons::YesNoCancel => {
206            command.args(["--extra-button", "No"]);
207            command.args(["--cancel-label", "Cancel"]);
208        }
209        MessageButtons::YesNoCancelCustom(yes, no, cancel) => {
210            command.args(["--ok-label", yes.as_str()]);
211            command.args(["--cancel-label", cancel.as_str()]);
212            command.args(["--extra-button", no.as_str()]);
213        }
214        _ => {}
215    }
216
217    run(command).await.map(|res| match btns {
218        MessageButtons::OkCancel => match res {
219            Some(_) => MessageDialogResult::Ok,
220            None => MessageDialogResult::Cancel,
221        },
222        MessageButtons::YesNo => match res {
223            Some(_) => MessageDialogResult::Yes,
224            None => MessageDialogResult::No,
225        },
226        MessageButtons::OkCancelCustom(ok, cancel) => match res {
227            Some(_) => MessageDialogResult::Custom(ok.clone()),
228            None => MessageDialogResult::Custom(cancel.clone()),
229        },
230        MessageButtons::YesNoCancel => match res {
231            Some(output) if output.is_empty() => MessageDialogResult::Yes,
232            Some(_) => MessageDialogResult::No,
233            None => MessageDialogResult::Cancel,
234        },
235        MessageButtons::YesNoCancelCustom(yes, no, cancel) => match res {
236            Some(output) if output.is_empty() => MessageDialogResult::Custom(yes.clone()),
237            Some(_) => MessageDialogResult::Custom(no.clone()),
238            None => MessageDialogResult::Custom(cancel.clone()),
239        },
240        _ => MessageDialogResult::Cancel,
241    })
242}

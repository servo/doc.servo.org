25#[allow(clippy::too_many_arguments)]
26pub(crate) fn prepare_devtools_request(
27    request_id: String,
28    url: ServoUrl,
29    method: Method,
30    headers: HeaderMap,
31    body: Option<Vec<u8>>,
32    pipeline_id: PipelineId,
33    connect_time: Duration,
34    send_time: Duration,
35    destination: Destination,
36    is_xhr: bool,
37    browsing_context_id: BrowsingContextId,
38) -> ChromeToDevtoolsControlMsg {
39    let started_date_time = SystemTime::now();
40    let request = DevtoolsHttpRequest {
41        url,
42        method,
43        headers,
44        body: body.map(Bytes::from),
45        pipeline_id,
46        started_date_time,
47        time_stamp: started_date_time
48            .duration_since(UNIX_EPOCH)
49            .unwrap_or_default()
50            .as_secs() as i64,
51        connect_time,
52        send_time,
53        destination,
54        is_xhr,
55        browsing_context_id,
56    };
57    let net_event = NetworkEvent::HttpRequestUpdate(request);
58
59    ChromeToDevtoolsControlMsg::NetworkEvent(request_id, net_event)
60}

62pub(crate) fn send_request_to_devtools(
63    msg: ChromeToDevtoolsControlMsg,
64    devtools_chan: &Sender<DevtoolsControlMsg>,
65) {
66    if matches!(msg, ChromeToDevtoolsControlMsg::NetworkEvent(_, ref network_event) if !network_event.forward_to_devtools())
67    {
68        return;
69    }
70    if let Err(e) = devtools_chan.send(DevtoolsControlMsg::FromChrome(msg)) {
71        error!("DevTools send failed: {e}");
72    }
73}

75pub(crate) fn send_response_to_devtools(
76    request: &Request,
77    context: &FetchContext,
78    response: &Response,
79    body_data: Option<Vec<u8>>,
80) {
81    let meta = match response.metadata() {
82        Ok(FetchMetadata::Unfiltered(m)) => m,
83        Ok(FetchMetadata::Filtered { unsafe_, .. }) => unsafe_,
84        Err(_) => {
85            log::warn!("No metadata available, skipping devtools response.");
86            return;
87        },
88    };
89    send_response_values_to_devtools(
90        meta.headers.map(Serde::into_inner),
91        meta.status,
92        body_data,
93        response.cache_state,
94        request,
95        context.devtools_chan.clone(),
96    );
97}

99#[allow(clippy::too_many_arguments)]
100pub(crate) fn send_response_values_to_devtools(
101    headers: Option<HeaderMap>,
102    status: HttpStatus,
103    body: Option<Vec<u8>>,
104    cache_state: CacheState,
105    request: &Request,
106    devtools_chan: Option<Sender<DevtoolsControlMsg>>,
107) {
108    if let (Some(devtools_chan), Some(pipeline_id), Some(webview_id)) = (
109        devtools_chan,
110        request.pipeline_id,
111        request.target_webview_id,
112    ) {
113        let browsing_context_id = webview_id.into();
114        let from_cache = matches!(cache_state, CacheState::Local | CacheState::Validated);
115
116        let devtoolsresponse = DevtoolsHttpResponse {
117            headers,
118            status,
119            body: body.map(Bytes::from),
120            from_cache,
121            pipeline_id,
122            browsing_context_id,
123        };
124        let net_event_response = NetworkEvent::HttpResponse(devtoolsresponse);
125
126        let msg =
127            ChromeToDevtoolsControlMsg::NetworkEvent(request.id.0.to_string(), net_event_response);
128
129        let _ = devtools_chan.send(DevtoolsControlMsg::FromChrome(msg));
130    }
131}

133pub(crate) fn send_security_info_to_devtools(
134    request: &Request,
135    context: &FetchContext,
136    response: &Response,
137) {
138    let meta = match response.metadata() {
139        Ok(FetchMetadata::Unfiltered(m)) => m,
140        Ok(FetchMetadata::Filtered { unsafe_, .. }) => unsafe_,
141        Err(_) => {
142            log::warn!("No metadata available, skipping devtools security info.");
143            return;
144        },
145    };
146
147    if let (Some(devtools_chan), Some(security_info), Some(webview_id)) = (
148        context.devtools_chan.clone(),
149        meta.tls_security_info,
150        request.target_webview_id,
151    ) {
152        let update = NetworkEvent::SecurityInfo(SecurityInfoUpdate {
153            browsing_context_id: webview_id.into(),
154            security_info: Some(security_info),
155        });
156
157        let msg = ChromeToDevtoolsControlMsg::NetworkEvent(request.id.0.to_string(), update);
158
159        let _ = devtools_chan.send(DevtoolsControlMsg::FromChrome(msg));
160    }
161}

163pub(crate) fn send_early_httprequest_to_devtools(request: &Request, context: &FetchContext) {
164    // Do not forward data requests to devtools
165    if request.url().scheme() == "data" {
166        return;
167    }
168    if let (Some(devtools_chan), Some(browsing_context_id), Some(pipeline_id)) = (
169        context.devtools_chan.as_ref(),
170        request.target_webview_id.map(|id| id.into()),
171        request.pipeline_id,
172    ) {
173        // Build the partial DevtoolsHttpRequest
174        let devtools_request = DevtoolsHttpRequest {
175            url: request.current_url(),
176            method: request.method.clone(),
177            headers: request.headers.clone(),
178            body: None,
179            pipeline_id,
180            started_date_time: SystemTime::now(),
181            time_stamp: 0,
182            connect_time: Duration::from_millis(0),
183            send_time: Duration::from_millis(0),
184            destination: request.destination,
185            is_xhr: false,
186            browsing_context_id,
187        };
188
189        let msg = ChromeToDevtoolsControlMsg::NetworkEvent(
190            request.id.0.to_string(),
191            NetworkEvent::HttpRequest(devtools_request),
192        );
193
194        send_request_to_devtools(msg, devtools_chan);
195    }

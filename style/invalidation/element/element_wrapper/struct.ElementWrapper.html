pub struct ElementWrapper<'a, 'b, E>
where
    E: TElement,
{
    element: E,
    snapshots: &'a Snapshots<'b>,
}
Expand description

A simple wrapper over an element and a snapshot, that allows us to selector-match against a past state of the element.
